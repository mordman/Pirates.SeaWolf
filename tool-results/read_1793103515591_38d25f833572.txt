   196→   196→  private wireEvents() {
   197→   197→    const distToPlayer = (p: THREE.Vector3) => p.distanceTo(this.player.position);
   198→   198→    this.bus.on(GameEvents.CANNON_FIRED, (p) => {
   199→   199→      const d = (p as { pos: THREE.Vector3; power: number });
   200→   200→      this.audio.cannon(d.power, distToPlayer(d.pos));
   201→   201→      if (distToPlayer(d.pos) < 60) this.rig.addShake(0.35 * d.power);
   202→   202→    });
   203→   203→    this.bus.on(GameEvents.PROJECTILE_SPLASH, (p) => {
   204→   204→      this.audio.splash(distToPlayer((p as { pos: THREE.Vector3 }).pos));
   205→   205→    });
   206→   206→    this.bus.on(GameEvents.PROJECTILE_HIT, (p) => {
   207→   207→      const d = (p as { pos: THREE.Vector3; target: Ship });
   208→   208→      this.audio.woodHit(distToPlayer(d.pos));
   209→   209→      if (d.target === this.player) this.rig.addShake(0.5);
   210→   210→      // ИИ реагирует на агрессию игрока
   211→   211→      const ai = this.aiShips.find(a => a.ship === d.target);
   212→   212→      if (ai) ai.onPlayerAggression(this.aiShips);
   213→   213→    });
   214→   214→    this.bus.on(GameEvents.MESSAGE, (p) => {
   215→   215→      this.pushMessage((p as { text: string }).text);
   216→   216→    });
   217→   217→  }

   339→   339→  private handleInput(dt: number) {
   340→   340→    const inp = this.input;
   341→   341→    if (!this.started || this.dead) { inp.endFrame(); return; }
   342→   342→    const p = this.player;
   343→   343→
   344→   344→    // паруса
   345→   345→    if (inp.pressed('KeyW')) { p.sailTarget = Math.min(3, p.sailTarget + 1); if (p.anchored) this.pushMessage('Сначала поднять якорь (F)'); }
   346→   346→    if (inp.pressed('KeyS')) {
   347→   347→      if (p.sailTarget > 0) p.sailTarget = Math.max(0, p.sailTarget - 1);
   348→   348→    }
   349→   349→    // задний ход на вёслах: держать S при убранных парусах
   350→   350→    p.reverseThrust = inp.isDown('KeyS') && p.sailTarget === 0 && !p.anchored;
   351→   351→    if (p.reverseThrust) this.lastSOffset += dt; else this.lastSOffset = 0;
   352→   352→
   353→   353→    // руль: A — влево, D — вправо (положительный руль = поворот вправо)
   354→   354→    const steer = (inp.isDown('KeyD') ? 1 : 0) - (inp.isDown('KeyA') ? 1 : 0);
   355→   355→    if (this.docked) p.rudderTarget = 0;
   356→   356→    else p.rudderTarget = steer; // D — право (штирборт), A — лево (порт);
   357→   357→    // изометрическая камера следует за курсом, поэтому на экране
   358→   358→    // D всегда поворачивает вправо, A — влево
   359→   359→
   360→   360→    // якорь
   361→   361→    if (inp.pressed('KeyF')) {
   362→   362→      if (this.docked) { this.undock(); }
   363→   363→      else {
   364→   364→        p.anchored = !p.anchored;
   365→   365→        this.audio.bell();
   366→   366→        this.pushMessage(p.anchored ? 'Якорь отдан' : 'Якорь поднят');
   367→   367→      }
   368→   368→    }
   369→   369→
   370→   370→    // камера
   371→   371→    if (inp.pressed('KeyC')) {
   372→   372→      this.rig.cycleMode();
   373→   373→      this.pushMessage(`Камера: ${CAMERA_LABEL[this.rig.mode]}`);
   374→   374→    }
   375→   375→
   376→   376→    // боеприпасы
   377→   377→    const ammoKeys: [string, AmmoType][] = [['Digit1', 'ball'], ['Digit2', 'chain'], ['Digit3', 'grape'], ['Digit4', 'fire']];
   378→   378→    for (const [key, kind] of ammoKeys) {
   379→   379→      if (inp.pressed(key)) {
   380→   380→        this.combat.ammo = kind;
   381→   381→        this.pushMessage(`Боеприпас: ${AMMO_INFO[kind].name}`);
   382→   382→      }
   383→   383→    }
   384→   384→
   385→   385→    // прицеливание: наведение мышью
   386→   386→    if (this.combat.aim.active) {
   387→   387→      this.combat.aim.azimuth = clamp(this.combat.aim.azimuth - inp.mouseDX * 0.0022, -0.62, 0.62);
   388→   388→      this.combat.aim.elevation = clamp(this.combat.aim.elevation + inp.mouseDY * 0.0016, 0.01, 0.24);
   389→   389→    }
   390→   390→
   391→   391→    // Q/E — борта; E ещё и взаимодействие
   392→   392→    const boardTarget = this.canBoard();
   393→   393→    const canDockHere = this.canDock();
   394→   394→    this.boardingTarget = boardTarget;
   395→   395→    if (inp.pressed('KeyQ')) {
   396→   396→      this.combat.toggleAim(-1);
   397→   397→    }
   398→   398→    if (inp.pressed('KeyE')) {
   399→   399→      if (boardTarget) {
   400→   400→        this.takePrize(boardTarget);
   401→   401→      } else if (canDockHere && !this.combat.aim.active) {
   402→   402→        this.dock();
   403→   403→      } else {
   404→   404→        this.combat.toggleAim(1);
   405→   405→      }
   406→   406→    }
   407→   407→
   408→   408→    // огонь
   409→   409→    if (this.combat.aim.active) {
   410→   410→      if (inp.lmbJust) this.combat.fireOne();
   411→   411→      if (inp.pressed('Space')) this.combat.fireBroadside();
   412→   412→    }

   418→   418→  // ---------- столкновения кораблей ----------
   419→   419→  private shipCollisions() {
   420→   420→    const all = [this.player, ...this.aiShips.map(a => a.ship)];
   421→   421→    for (let i = 0; i < all.length; i++) {
   422→   422→      for (let j = i + 1; j < all.length; j++) {
   423→   423→        const a = all[i], b = all[j];
   424→   424→        if (a.sunk || b.sunk) continue;
   425→   425→        const minD = (a.preset.length + b.preset.length) * 0.30;
   426→   426→        const d = a.position.distanceTo(b.position);
   427→   427→        if (d < minD && d > 0.01) {
   428→   428→          const n = new THREE.Vector3().subVectors(a.position, b.position).setY(0).normalize();
   429→   429→          const push = (minD - d) * 0.5;
   430→   430→          a.position.addScaledVector(n, push);
   431→   431→          b.position.addScaledVector(n, -push);
   432→   432→          const relV = new THREE.Vector3().subVectors(a.velocity, b.velocity).dot(n);
   433→   433→          if (relV < -1.2) {
   434→   434→            const dmg = Math.min(22, -relV * 2.4);
   435→   435→            a.applyDamage({ kind: 'ram', amount: dmg });
   436→   436→            b.applyDamage({ kind: 'ram', amount: dmg });
   437→   437→            const mid = new THREE.Vector3().addVectors(a.position, b.position).multiplyScalar(0.5);
   438→   438→            this.particles.splash(mid.x, 0.5, mid.z, 1.2);
   439→   439→            this.audio.woodHit(0);
   440→   440→            this.rig.addShake(0.5);
   441→   441→            if (a === this.player || b === this.player) {
   442→   442→              this.pushMessage('Столкновение! Корпус повреждён');
   443→   443→            }
   444→   444→            // военный реагирует на таран
   445→   445→            for (const ai of this.aiShips) {
   446→   446→              if (ai.ship === a || ai.ship === b) ai.onPlayerAggression(this.aiShips);
   447→   447→            }
   448→   448→          }
   449→   449→          a.velocity.multiplyScalar(0.85);
   450→   450→          b.velocity.multiplyScalar(0.85);
   451→   451→        }
   452→   452→      }
   453→   453→    }
   454→   454→  }

   528→   528→  // ---------- погода ----------
   529→   529→  private updateWeather(dt: number) {
   530→   530→    this.windTimer -= dt;
   531→   531→    if (this.windTimer < 0) {
   532→   532→      this.windTimer = 18 + Math.random() * 22;
   533→   533→      this.windTargetSpeed = 3.5 + Math.random() * 9.5;
   534→   534→    }
   535→   535→    this.gustT += dt;
   536→   536→    const gust = Math.sin(this.gustT * 0.5) * 0.9 + Math.sin(this.gustT * 1.7 + 2) * 0.5;
   537→   537→    this.wind.speed += (this.windTargetSpeed + gust - this.wind.speed) * Math.min(1, dt * 0.3);
   538→   538→    this.wind.from += dt * 0.008 * Math.sin(this.time * 0.037 + 1.3);
   539→   539→    SEA.amplitude = clamp(0.45 + (this.wind.speed - 3) * 0.09, 0.4, 1.5);
   540→   540→  }

   542→   542→  // ---------- кадр ----------
   543→   543→  private frame = () => {
   544→   544→    this.rafId = requestAnimationFrame(this.frame);
   545→   545→    const now = performance.now();
   546→   546→    let dt = (now - this.lastFrame) / 1000;
   547→   547→    this.lastFrame = now;
   548→   548→    this.fpsSmooth += (1 / Math.max(dt, 1e-4) - this.fpsSmooth) * 0.05; // истинный FPS
   549→   549→    dt = Math.min(dt, 0.1);
   550→   550→
   551→   551→    this.handleInput(dt);
   552→   552→
   553→   553→    // физика фиксированным шагом
   554→   554→    this.accum += dt;
   555→   555→    const STEP = 1 / 60;
   556→   556→    let steps = 0;
   557→   557→    while (this.accum >= STEP && steps < 5) {
   558→   558→      this.time += STEP;
   559→   559→      this.updateWeather(STEP);
   560→   560→      this.player.fixedStep(STEP, this.time);
   561→   561→      for (const ai of this.aiShips) ai.ship.fixedStep(STEP, this.time);
   562→   562→      for (const ai of this.aiShips) {
   563→   563→        ai.update(STEP, this.time, this.player, this.combat, (t) => this.pushMessage(t));
   564→   564→      }
   565→   565→      this.shipCollisions();
   566→   566→      this.accum -= STEP;
   567→   567→      steps++;
   568→   568→    }
   569→   569→
   570→   570→    // телеметрия для отладки/E2E: window.__sea (ватерлиния: y ≈ waterY)
   571→   571→    const dbg = window as unknown as { __sea: Record<string, number> };
   572→   572→    dbg.__sea = {
   573→   573→      fps: this.fpsSmooth,
   574→   574→      x: this.player.position.x, y: this.player.position.y, z: this.player.position.z,
   575→   575→      waterY: waterHeightAt(this.player.position.x, this.player.position.z, this.time),
   576→   576→      roll: this.player.euler.z, pitch: this.player.euler.x,
   577→   577→      heading: this.player.heading,
   578→   578→      rudder: this.player.rudder, angY: this.player.angular.y,
   579→   579→      speed: this.player.speedKnots(),
   580→   580→    };
   581→   581→
   582→   582→    // действия команды (R)
   583→   583→    if (this.player.pumpAction && !this.docked) {
   584→   584→      const msg = this.player.crewAction(dt);
   585→   585→      if (msg && Math.random() < dt * 2) this.pushMessage(msg);
   586→   586→    }
   587→   587→
   588→   588→    // sinking игрока
   589→   589→    if (this.player.sinking && !this.dead && this.player.sinkT > 5) {
   590→   590→      this.dead = true;
   591→   591→      this.pushMessage('Корабль затонул...');
   592→   592→    }
   593→   593→
   594→   594→    // floating origin
   595→   595→    const playerScene = this._v1.copy(this.player.position).add(this.shift);
   596→   596→    if (Math.hypot(playerScene.x, playerScene.z) > 2200) {
   597→   597→      this.shift.x -= Math.round(playerScene.x / 50) * 50;
   598→   598→      this.shift.z -= Math.round(playerScene.z / 50) * 50;
   599→   599→      this.worldGroup.position.copy(this.shift);
   600→   600→      this.pushMessage('Смена квадрата моря');
   601→   601→    }
   602→   602→    this.negShift.copy(this.shift).negate();
   603→   603→
   604→   604→    // визуал кораблей
   605→   605→    this.player.updateVisual(this.time, dt, this.particles);
   606→   606→    for (const ai of this.aiShips) ai.ship.updateVisual(this.time, dt, this.particles);
   607→   607→
   608→   608→    // системы
   609→   609→    this.combat.update(dt, this.time, waterHeightAt);
   610→   610→    this.particles.update(dt);
   611→   611→    this.world.update(this.time, dt, this.player.position, waterHeightAt);
   612→   612→
   613→   613→    // небо и свет
   614→   614→    playerScene.copy(this.player.position).add(this.shift);
   615→   615→    const { dayF } = this.sky.update(dt, this.time, this.rig.camera.position, playerScene, 0.42);
   616→   616→    const env = this.sky.environmentColors(dayF, 0);
   617→   617→    this.water.setEnvironment(this.sky.sunDir, env.sun, env.sky, env.horizon, env.fog, 0.00035);
   618→   618→    (this.scene.fog as THREE.FogExp2).color.copy(env.fog);
   619→   619→    (this.scene.fog as THREE.FogExp2).density = 0.00030 + (1 - dayF) * 0.00006;
   620→   620→
   621→   621→    // фонарь игрока ночью
   622→   622→    if (this.player.visual.lantern) {
   623→   623→      this.player.visual.lantern.intensity = dayF < 0.35 ? 5 : 0;
   624→   624→    }
   625→   625→
   626→   626→    // камера
   627→   627→    const shipInfo = {
   628→   628→      scenePos: playerScene,
   629→   629→      heading: this.player.heading,
   630→   630→      roll: this.player.euler.z,
   631→   631→      pitch: this.player.euler.x,
   632→   632→      speedFrac: clamp(this.player.speed() / this.player.preset.maxSpeed, 0, 1),
   633→   633→    };
   634→   634→    const aimDir = this.combat.aim.active ? this.combat.aimDirection(this._v2) : null;
   635→   635→    this.rig.update(dt, {
   636→   636→      active: this.combat.aim.active,
   637→   637→      side: this.combat.aim.side,
   638→   638→      dir: aimDir ?? this._v2.set(1, 0, 0),
   639→   639→    }, shipInfo, {
   640→   640→      mouseDX: this.input.mouseDX, mouseDY: this.input.mouseDY,
   641→   641→      wheel: this.input.wheelDelta, rmb: this.input.rmb,
   642→   642→    });
   643→   643→
   644→   644→    // вода
   645→   645→    this.water.update(this.time, this.rig.camera.position, this.negShift);
   646→   646→
   647→   647→    // звук
   648→   648→    this.audio.update(dt, {
   649→   649→      windSpeed: this.wind.speed,
   650→   650→      speed: this.player.speed(),
   651→   651→      rollRate: this.player.angular.z,
   652→   652→      nearLand: this.nearLand(),
   653→   653→      dayF,
   654→   654→    });
   655→   655→
   656→   656→    // сообщения
   657→   657→    for (let i = this.msgTtl.length - 1; i >= 0; i--) {
   658→   658→      this.msgTtl[i] -= dt;
   659→   659→      if (this.msgTtl[i] <= 0) { this.messages.splice(i, 1); this.msgTtl.splice(i, 1); }
   660→   660→    }
   661→   661→
   662→   662→    this.updateHud();
   663→   663→    this.renderer.render(this.scene, this.rig.camera);
   664→   664→    this.input.endFrame();
   665→   665→  };
