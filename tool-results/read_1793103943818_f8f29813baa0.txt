    30→      <div className={`h-full transition-all duration-300 ${color}`} style={{ width: `${Math.max(0, Math.min(1, pct)) * 100}%` }} />

    35→function WindCompass({ hud }: { hud: HudState }) {
    36→  const rel = hud.windRelDeg;
    37→  return (
    38→    <div className="flex items-center gap-3 p-3">
    39→      <svg width="84" height="84" viewBox="0 0 100 100" className="shrink-0">
    40→        <circle cx="50" cy="50" r="46" fill="#0d0906" stroke="#7c5a2a" strokeWidth="2" />
    41→        <circle cx="50" cy="50" r="35" fill="none" stroke="#3d2e18" strokeWidth="1" />
    42→        {['С', 'В', 'Ю', 'З'].map((d, i) => {
    43→          const a = (i * 90 - 90) * (Math.PI / 180);
    44→          return (
    45→            <text key={d} x={50 + Math.cos(a) * 40} y={50 + Math.sin(a) * 40 + 4} textAnchor="middle"
    46→              fill={d === 'С' ? '#e8c87a' : '#8a734d'} fontSize="11" fontWeight="bold">{d}</text>

    57→      <div className="grid grid-cols-[auto_auto] gap-x-3 gap-y-1 text-xs">
    58→        <span className="text-stone-400">Ветер</span>
    59→        <span className="text-sky-200 font-semibold">{hud.windSpeed.toFixed(1)} м/с</span>

   145→      grad.addColorStop(0.75, '#c9b26a');

   148→      ctx.beginPath(); ctx.arc(px(isl.x), pz(isl.z), r, 0, Math.PI * 2); ctx.fill();

   150→      ctx.font = 'bold 11px serif';
   151→      ctx.textAlign = 'center';
   152→      ctx.fillText(isl.name, px(isl.x), pz(isl.z) + r + 14);
