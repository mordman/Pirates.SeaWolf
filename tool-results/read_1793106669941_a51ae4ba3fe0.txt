   167→  /** Применить новые настройки на лету (вызывается экраном настроек) */
   168→  applySettings(s: GameSettings) {

   170→    this.settings = s;

   172→    this.renderer.setPixelRatio(Math.min(window.devicePixelRatio, g.pixelRatio));
   173→    if (prev.graphics.shadows !== g.shadows) this.setShadows(g.shadows);
   174→    if (prev.graphics.shadowRes !== g.shadowRes) this.sky.setShadowRes(g.shadowRes);
   175→    if (prev.graphics.waterDetail !== g.waterDetail || prev.graphics.drawDistance !== g.drawDistance) {
   176→      this.water.configure(g.waterDetail, g.drawDistance);
   177→    }
   178→    this.water.setFoamQuality(g.foamQuality);
   179→    this.rig.camera.far = g.drawDistance * 2 + 4000;
   180→    this.rig.camera.updateProjectionMatrix();
   181→    this.sky.setDomeScale(g.drawDistance / 6500 * 1.12);
   182→    (this.scene.fog as THREE.FogExp2).density = this.fogBase();
   183→    this.audio.setVolume(s.audio.volume);

   187→  private setShadows(on: boolean) {
   188→    this.renderer.shadowMap.enabled = on;
   189→    this.renderer.shadowMap.needsUpdate = true;
   190→    // переключение теней требует перекомпиляции материалов
   191→    this.scene.traverse((o) => {
   192→      const mesh = o as THREE.Mesh;
   193→      if (mesh.material) {
   194→        const mats = Array.isArray(mesh.material) ? mesh.material : [mesh.material];
   195→        for (const m of mats) (m as THREE.Material).needsUpdate = true;
   196→      }
   197→    });

   260→  private wireEvents() {
   261→    const distToPlayer = (p: THREE.Vector3) => p.distanceTo(this.player.position);
   262→    this.bus.on(GameEvents.CANNON_FIRED, (p) => {
   263→      const d = (p as { pos: THREE.Vector3; power: number });
   264→      this.audio.cannon(d.power, distToPlayer(d.pos));
   265→      if (distToPlayer(d.pos) < 60) this.rig.addShake(0.35 * d.power);
   266→    });
   267→    this.bus.on(GameEvents.PROJECTILE_SPLASH, (p) => {
   268→      this.audio.splash(distToPlayer((p as { pos: THREE.Vector3 }).pos));
   269→    });
   270→    this.bus.on(GameEvents.PROJECTILE_HIT, (p) => {
   271→      const d = (p as { pos: THREE.Vector3; target: Ship });
   272→      this.audio.woodHit(distToPlayer(d.pos));
   273→      if (d.target === this.player) this.rig.addShake(0.5);
   274→      // ИИ реагирует на агрессию игрока
   275→      const ai = this.aiShips.find(a => a.ship === d.target);
   276→      if (ai) ai.onPlayerAggression(this.aiShips);
   277→    });
   278→    this.bus.on(GameEvents.MESSAGE, (p) => {
   279→      this.pushMessage((p as { text: string }).text);
   280→    });
   281→  }

   321→  get repairCost(): number { return Math.ceil((this.player.hullMax - this.player.hullHP) * 2.2); }

   323→  portRepair() {
   324→    const cost = this.repairCost;
   325→    if (cost <= 0) return 'Корпус уже в порядке';
   326→    if (this.player.gold < cost) return 'Не хватает золота';
   327→    this.player.gold -= cost;
   328→    this.player.hullHP = this.player.hullMax;
   329→    this.player.sailHP = this.player.preset.sailHP;
   330→    this.player.water = 0;
   331→    this.player.leakRate = 0;
   332→    this.pushMessage(`Полный ремонт выполнен (-${cost} зол.)`);
   333→    return null;
   334→  }

   354→    if (this.player.gold < 30) return 'Не хватает золота';

   360→  portSellCargo(): string | null {
   361→    let total = 0;
   362→    for (const k of Object.keys(this.player.cargo)) {
   363→      total += (this.player.cargo[k] || 0) * CARGO_PRICES[k];
   364→      this.player.cargo[k] = 0;
   365→    }
   366→    if (total <= 0) return 'Трюм пуст';
   367→    this.player.gold += total;
   368→    this.pushMessage(`Груз продан: +${total} зол.`);
   369→    return null;

   372→  // ---------- абордаж/приз ----------
   373→  canBoard(): AIShip | null {
   374→    if (this.docked || this.player.sinking) return null;
   375→    for (const ai of this.aiShips) {
   376→      if (ai.state !== 'surrender' || ai.ship.sinking || ai.ship.sunk) continue;
   377→      const d = ai.ship.position.distanceTo(this.player.position);
   378→      if (d < 55 && this.player.speedKnots() < 5) return ai;
   379→    }
   380→    return null;
   381→  }

   383→  takePrize(ai: AIShip) {
   384→    const s = ai.ship;
   385→    const gold = Math.round(s.gold * (0.85 + Math.random() * 0.5));
   386→    this.player.gold += gold;
   387→    const loot: string[] = [];
   388→    for (const k of Object.keys(s.cargo)) {
   389→      const amt = s.cargo[k] || 0;
   390→      if (amt > 0) {
   391→        this.player.cargo[k] = (this.player.cargo[k] || 0) + amt;
   392→        loot.push(`${CARGO_NAMES[k]} ×${amt}`);
   393→      }
   394→    }
   395→    this.player.ammo.ball += 8; this.player.ammo.grape += 4;
   396→    if (this.player.crew < this.player.preset.maxCrew) this.player.crew = Math.min(this.player.preset.maxCrew, this.player.crew + 2);
   397→    ai.makeDerelict();
   398→    this.audio.bell();
   399→    this.pushMessage(`Приз взят! +${gold} зол., ${loot.join(', ') || 'груза нет'}. Судно брошено.`);
   400→  }

   402→  // ---------- управление ----------
   403→  private handleInput(dt: number) {
   404→    const inp = this.input;
   405→    if (!this.started || this.dead || this.uiOpen) { inp.endFrame(); return; }
   406→    const p = this.player;
   407→    const c = this.settings.controls;
   408→
   409→    // паруса
   410→    if (inp.pressed(c.sailsUp)) { p.sailTarget = Math.min(3, p.sailTarget + 1); if (p.anchored) this.pushMessage('Сначала поднять якорь (F)'); }
   411→    if (inp.pressed(c.sailsDown)) {
   412→      if (p.sailTarget > 0) p.sailTarget = Math.max(0, p.sailTarget - 1);
   413→    }
   414→    // задний ход на вёслах: держать «убрать паруса» при убранных парусах
   415→    p.reverseThrust = inp.isDown(c.sailsDown) && p.sailTarget === 0 && !p.anchored;
   416→    if (p.reverseThrust) this.lastSOffset += dt; else this.lastSOffset = 0;
   417→
   418→    // руль: A — влево, D — вправо (положительный руль = поворот вправо,
   419→    // ползунок HUD при D уходит вправо — и корабль поворачивает вправо)
   420→    const steer = (inp.isDown(c.steerRight) ? 1 : 0) - (inp.isDown(c.steerLeft) ? 1 : 0);
   421→    if (this.docked) p.rudderTarget = 0;
   422→    else p.rudderTarget = steer; // D — штирборт (право), A — порт (лево)
   423→
   424→    // якорь
   425→    if (inp.pressed(c.anchor)) {
   426→      if (this.docked) { this.undock(); }
   427→      else {
   428→        p.anchored = !p.anchored;
   429→        this.audio.bell();
   430→        this.pushMessage(p.anchored ? 'Якорь отдан' : 'Якорь поднят');
   431→      }
   432→    }
   433→
   434→    // камера
   435→    if (inp.pressed(c.camera)) {
   436→      this.rig.cycleMode();
   437→      this.pushMessage(`Камера: ${CAMERA_LABEL[this.rig.mode]}`);
   438→    }
   439→
   440→    // боеприпасы
   441→    const ammoKeys: [string, AmmoType][] = [[c.ammo[0], 'ball'], [c.ammo[1], 'chain'], [c.ammo[2], 'grape'], [c.ammo[3], 'fire']];
   442→    for (const [key, kind] of ammoKeys) {
   443→      if (inp.pressed(key)) {
   444→        this.combat.ammo = kind;
   445→        this.pushMessage(`Боеприпас: ${AMMO_INFO[kind].name}`);
   446→      }
   447→    }
   448→
   449→    // прицеливание: наведение мышью
   450→    if (this.combat.aim.active) {
   451→      this.combat.aim.azimuth = clamp(this.combat.aim.azimuth - inp.mouseDX * 0.0022, -0.62, 0.62);
   452→      this.combat.aim.elevation = clamp(this.combat.aim.elevation + inp.mouseDY * 0.0016, 0.01, 0.24);
   453→    }
   454→
   455→    // борта; «правый борт» ещё и взаимодействие
   456→    const boardTarget = this.canBoard();
   457→    const canDockHere = this.canDock();
   458→    this.boardingTarget = boardTarget;
   459→    if (inp.pressed(c.aimPort)) {
   460→      this.combat.toggleAim(-1);
   461→    }
   462→    if (inp.pressed(c.aimStarboard)) {
   463→      if (boardTarget) {
   464→        this.takePrize(boardTarget);
   465→      } else if (canDockHere && !this.combat.aim.active) {
   466→        this.dock();
   467→      } else {
   468→        this.combat.toggleAim(1);
   469→      }
   470→    }
   471→
   472→    // огонь
   473→    if (this.combat.aim.active) {
   474→      if (inp.lmbJust) this.combat.fireOne();
   475→      if (inp.pressed(c.fire)) this.combat.fireBroadside();
   476→    }
   477→
   478→    // ремонт/откачка
   479→    p.pumpAction = inp.isDown(c.repair);
   480→  }

   482→  // ---------- столкновения кораблей ----------
   483→  private shipCollisions() {
   484→    const all = [this.player, ...this.aiShips.map(a => a.ship)];
   485→    for (let i = 0; i < all.length; i++) {
   486→      for (let j = i + 1; j < all.length; j++) {
   487→        const a = all[i], b = all[j];
   488→        if (a.sunk || b.sunk) continue;
   489→        const minD = (a.preset.length + b.preset.length) * 0.30;
   490→        const d = a.position.distanceTo(b.position);
   491→        if (d < minD && d > 0.01) {
   492→          const n = new THREE.Vector3().subVectors(a.position, b.position).setY(0).normalize();
   493→          const push = (minD - d) * 0.5;
   494→          a.position.addScaledVector(n, push);
   495→          b.position.addScaledVector(n, -push);
   496→          const relV = new THREE.Vector3().subVectors(a.velocity, b.velocity).dot(n);
   497→          if (relV < -1.2) {
   498→            const dmg = Math.min(22, -relV * 2.4);
   499→            a.applyDamage({ kind: 'ram', amount: dmg });
   500→            b.applyDamage({ kind: 'ram', amount: dmg });
   501→            const mid = new THREE.Vector3().addVectors(a.position, b.position).multiplyScalar(0.5);
   502→            this.particles.splash(mid.x, 0.5, mid.z, 1.2);
   503→            this.audio.woodHit(0);
   504→            this.rig.addShake(0.5);
   505→            if (a === this.player || b === this.player) {
   506→              this.pushMessage('Столкновение! Корпус повреждён');
   507→            }
   508→            // военный реагирует на таран
   509→            for (const ai of this.aiShips) {
   510→              if (ai.ship === a || ai.ship === b) ai.onPlayerAggression(this.aiShips);
   511→            }
   512→          }
   513→          a.velocity.multiplyScalar(0.85);
   514→          b.velocity.multiplyScalar(0.85);
   515→        }
   516→      }
   517→    }
   518→  }

   536→  private updateHud() {
   537→    const p = this.player;
   538→    const h = this.hud;
   539→    const hdgDeg = ((-p.heading * 180 / Math.PI) % 360 + 360) % 360;
   540→    const windDeg = ((-this.wind.from * 180 / Math.PI) % 360 + 360) % 360;
   541→    const rel = ((windDeg - hdgDeg) % 360 + 360) % 360;
   542→    h.started = this.started;
   543→    h.dead = this.dead;
   544→    h.speed = p.speedKnots();
   545→    h.headingDeg = Math.round(hdgDeg);
   546→    h.windFromDeg = Math.round(windDeg);
   547→    h.windSpeed = this.wind.speed;
   548→    h.windRelDeg = Math.round(rel);
   549→    h.sailLevel = p.sailTarget;
   550→    h.rudder = p.rudder;
   551→    h.hullPct = clamp(p.hullHP / p.hullMax, 0, 1);
   552→    h.sailPct = clamp(p.sailHP / p.preset.sailHP, 0, 1);
   553→    h.crew = Math.round(p.crew);
   554→    h.maxCrew = p.preset.maxCrew;
   555→    h.waterPct = clamp(p.water / p.holdCap, 0, 1);
   556→    h.onFire = p.onFire;
   557→    h.anchored = p.anchored;
   558→    h.grounded = p.grounded;
   559→    h.sinking = p.sinking;
   560→    h.ammoType = this.combat.ammo;
   561→    h.ammoCounts = { ...p.ammo };
   562→    h.planks = Math.floor(p.planks);
   563→    h.gold = Math.floor(p.gold);
   564→    h.cargo = { ...p.cargo };
   565→    h.aiming = this.combat.aim.active;
   566→    h.aimSide = this.combat.aim.side === 1 ? 'правый борт' : 'левый борт';
   567→    h.aimRange = Math.round(this.combat.aim.range);
   568→    h.gunsReadyPct = this.combat.readiness();
   569→    h.targetName = this.combat.aim.target ? this.combat.aim.target.preset.name : null;
   570→    h.cameraLabel = CAMERA_LABEL[this.rig.mode];
   571→    // часы: 0.0 = полночь
   572→    const hours = (this.sky.timeOfDay * 24 + 6) % 24;
   573→    const hh = Math.floor(hours);
   574→    const mm = Math.floor((hours - hh) * 60);
   575→    h.clock = `${String(hh).padStart(2, '0')}:${String(mm).padStart(2, '0')}`;
   576→    h.seaState = SEA.amplitude;
   577→    h.repairAction = p.pumpAction ? (p.onFire ? 'Тушение пожара...' : p.water > 0.05 ? 'Откачка воды...' : p.planks > 0 && p.hullHP < p.hullMax ? 'Ремонт...' : null) : null;
   578→    h.canDock = this.canDock();
   579→    h.canBoard = !!this.boardingTarget;
   580→    h.boardedName = this.boardingTarget?.ship.preset.name ?? null;
   581→    h.docked = this.docked;
   582→    h.messages = [...this.messages];
   583→    h.mapIslands = this.world.islands.map(i => ({ x: i.cx, z: i.cz, r: i.r, name: i.name, discovered: this.world.discovered.has(i.name) }));
   584→    h.mapShips = this.aiShips.filter(a => !a.ship.sunk).map(a => ({
   585→      x: a.ship.position.x, z: a.ship.position.z,
   586→      kind: a.ship.preset.id, surrendered: a.ship.surrendered,
   587→    }));
   588→    h.mapPlayer = { x: p.position.x, z: p.position.z, h: p.heading };
   589→    h.fps = Math.round(this.fpsSmooth);
   590→  }

   592→  // ---------- погода ----------
   593→  private updateWeather(dt: number) {
   594→    this.windTimer -= dt;
   595→    if (this.windTimer < 0) {
   596→      this.windTimer = 18 + Math.random() * 22;
   597→      this.windTargetSpeed = 3.5 + Math.random() * 9.5;
   598→    }
   599→    this.gustT += dt;
   600→    const gust = Math.sin(this.gustT * 0.5) * 0.9 + Math.sin(this.gustT * 1.7 + 2) * 0.5;
   601→    this.wind.speed += (this.windTargetSpeed + gust - this.wind.speed) * Math.min(1, dt * 0.3);
   602→    this.wind.from += dt * 0.008 * Math.sin(this.time * 0.037 + 1.3);
   603→    // амплитуда волн: 0.4 штиль .. 1.3 шторм (при 1.5 гребни накрывали палубу)
   604→    SEA.amplitude = clamp(0.45 + (this.wind.speed - 3) * 0.085, 0.4, 1.3);
   605→  }

   607→  // ---------- кадр ----------
   608→  private frame = () => {
   609→    this.rafId = requestAnimationFrame(this.frame);
   610→    const now = performance.now();
   611→    let dt = (now - this.lastFrame) / 1000;
   612→    this.lastFrame = now;
   613→    this.fpsSmooth += (1 / Math.max(dt, 1e-4) - this.fpsSmooth) * 0.05; // истинный FPS
   614→    dt = Math.min(dt, 0.1);
   615→
   616→    this.handleInput(dt);
   617→
   618→    // физика фиксированным шагом
   619→    this.accum += dt;
   620→    const STEP = 1 / 60;
   621→    let steps = 0;
   622→    while (!this.paused && this.accum >= STEP && steps < 5) {
   623→      this.time += STEP;
   624→      this.updateWeather(STEP);
   625→      this.player.fixedStep(STEP, this.time);
   626→      for (const ai of this.aiShips) ai.ship.fixedStep(STEP, this.time);
   627→      for (const ai of this.aiShips) {
   628→        ai.update(STEP, this.time, this.player, this.combat, (t) => this.pushMessage(t));
   629→      }
   630→      this.shipCollisions();
   631→      this.accum -= STEP;
   632→      steps++;
   633→    }
   634→
   635→    // телеметрия для отладки/E2E: window.__sea
   636→    const dbg = window as unknown as { __sea: Record<string, number> };
   637→    const sailsV = this.player.visual.sails.sails;
   638→    let furlSum = 0, furlMin = 1, furlMax = 0;
   639→    for (const sv of sailsV) {
   640→      furlSum += sv.furl;
   641→      furlMin = Math.min(furlMin, sv.furl);
   642→      furlMax = Math.max(furlMax, sv.furl);
   643→    }
   644→    // минимальный надводный борт: высота палубы у планширя минус вода
   645→    // в носу/корме/миделе (учитываем крен/дифферент кватернионом)
   646→    let freeboard = Infinity;
   647→    const deckLocal = this.player.preset.freeboard * 0.82 - 1.0;
   648→    for (const lz of [-this.player.preset.length * 0.38, 0, this.player.preset.length * 0.38]) {
   649→      this._v3.set(0, deckLocal, lz).applyQuaternion(this.player.quaternion).add(this.player.position);
   650→      freeboard = Math.min(freeboard, this._v3.y - waterHeightAt(this._v3.x, this._v3.z, this.time));
   651→    }
   652→    dbg.__sea = {
   653→      fps: this.fpsSmooth,
   654→      x: this.player.position.x, y: this.player.position.y, z: this.player.position.z,
   655→      waterY: waterHeightAt(this.player.position.x, this.player.position.z, this.time),
   656→      roll: this.player.euler.z, pitch: this.player.euler.x,
   657→      heading: this.player.heading,
   658→      rudder: this.player.rudder, angY: this.player.angular.y,
   659→      speed: this.player.speedKnots(),
   660→      sailLevel: this.player.sailTarget, sailFrac: this.player.sailFrac,
   661→      furlAvg: furlSum / Math.max(1, sailsV.length), furlMin, furlMax,
   662→      sailSet: sailsV.filter(sv => sv.furl < 0.5).length,
   663→      freeboard, seaAmp: SEA.amplitude, wind: this.wind.speed,
   664→    };
   665→
   666→    // действия команды (R)
   667→    if (this.player.pumpAction && !this.docked) {
   668→      const msg = this.player.crewAction(dt);
   669→      if (msg && Math.random() < dt * 2) this.pushMessage(msg);
   670→    }
   671→
   672→    // sinking игрока
   673→    if (this.player.sinking && !this.dead && this.player.sinkT > 5) {
   674→      this.dead = true;
   675→      this.pushMessage('Корабль затонул...');
   676→    }
   677→
   678→    // floating origin
   679→    const playerScene = this._v1.copy(this.player.position).add(this.shift);
   680→    if (Math.hypot(playerScene.x, playerScene.z) > 2200) {
   681→      this.shift.x -= Math.round(playerScene.x / 50) * 50;
   682→      this.shift.z -= Math.round(playerScene.z / 50) * 50;
   683→      this.worldGroup.position.copy(this.shift);
   684→      this.pushMessage('Смена квадрата моря');
   685→    }
   686→    this.negShift.copy(this.shift).negate();
   687→
   688→    // визуал кораблей
   689→    this.player.updateVisual(this.time, dt, this.particles);
   690→    for (const ai of this.aiShips) ai.ship.updateVisual(this.time, dt, this.particles);
   691→
   692→    // системы
   693→    this.combat.update(dt, this.time, waterHeightAt);
   694→    this.particles.update(dt);
   695→    this.world.update(this.time, dt, this.player.position, waterHeightAt);
   696→
   697→    // небо и свет
   698→    playerScene.copy(this.player.position).add(this.shift);
   699→    const { dayF } = this.sky.update(dt, this.time, this.rig.camera.position, playerScene, 0.42);
   700→    const env = this.sky.environmentColors(dayF, 0);
   701→    const fogD = this.fogBase() * (1 + (1 - dayF) * 0.18);
   702→    this.water.setEnvironment(this.sky.sunDir, env.sun, env.sky, env.horizon, env.fog, fogD);
   703→    (this.scene.fog as THREE.FogExp2).color.copy(env.fog);
   704→    (this.scene.fog as THREE.FogExp2).density = fogD;
   705→
   706→    // фонарь игрока ночью
   707→    if (this.player.visual.lantern) {
   708→      this.player.visual.lantern.intensity = dayF < 0.35 ? 5 : 0;
   709→    }
   710→
   711→    // камера
   712→    const shipInfo = {
   713→      scenePos: playerScene,
   714→      heading: this.player.heading,
   715→      roll: this.player.euler.z,
   716→      pitch: this.player.euler.x,
   717→      speedFrac: clamp(this.player.speed() / this.player.preset.maxSpeed, 0, 1),
   718→    };
   719→    const aimDir = this.combat.aim.active ? this.combat.aimDirection(this._v2) : null;
   720→    this.rig.update(dt, {
   721→      active: this.combat.aim.active,
   722→      side: this.combat.aim.side,
   723→      dir: aimDir ?? this._v2.set(1, 0, 0),
   724→    }, shipInfo, {
   725→      mouseDX: this.input.mouseDX, mouseDY: this.input.mouseDY,
   726→      wheel: this.input.wheelDelta, rmb: this.input.rmb,
   727→    });
   728→
   729→    // вода
   730→    this.water.update(this.time, this.rig.camera.position, this.negShift);
   731→
   732→    // звук
   733→    this.audio.update(dt, {
   734→      windSpeed: this.wind.speed,
   735→      speed: this.player.speed(),
   736→      rollRate: this.player.angular.z,
   737→      nearLand: this.nearLand(),
   738→      dayF,
   739→    });
   740→
   741→    // сообщения
   742→    for (let i = this.msgTtl.length - 1; i >= 0; i--) {
   743→      this.msgTtl[i] -= dt;
   744→      if (this.msgTtl[i] <= 0) { this.messages.splice(i, 1); this.msgTtl.splice(i, 1); }
   745→    }
   746→
   747→    this.updateHud();
   748→    this.renderer.render(this.scene, this.rig.camera);
   749→    this.input.endFrame();
   750→  };

   756→  private nearLand(): boolean {
   757→    for (const isl of this.world.islands) {

   759→    }
   760→    return false;
   761→  }

   782→    this.dead = false;
   783→    this.pushMessage(`Новый корабль у гавани. Половина золота ушла на дно...`);
