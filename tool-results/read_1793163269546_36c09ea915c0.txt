   101→   101→  constructor(presetId: string, isPlayer: boolean, wind: WindState,
   102→   102→    terrainHeight: (x: number, z: number) => number, particles: Particles) {
   103→   103→    this.preset = PRESETS[presetId];
   104→   104→    this.isPlayer = isPlayer;
   105→   105→    this.wind = wind;
   106→   106→    this.terrainHeight = terrainHeight;
   107→   107→    this.particles = particles;
   108→   108→    this.visual = buildShip(this.preset);
   109→   109→    this.hullHP = this.preset.hullHP;
   110→   110→    this.hullMax = this.preset.hullHP;
   111→   111→    this.sailHP = this.preset.sailHP;
   112→   112→    this.crew = Math.round(this.preset.maxCrew * 0.8);
   113→   113→    this.holdCap = this.preset.cargo * 0.8;
   114→   114→    this.mass = this.preset.mass;
   115→   115→
   116→   116→    const { length: L, beam: B, draft: D, freeboard: F } = this.preset;
   117→   117→    // моменты инерции (связные оси: x — крен? нет: x=право, y=верх, z=нос)
   118→   118→    // крен — вокруг z, дифферент — вокруг x, рыскание — вокруг y
   119→   119→    this.inertia.set(
   120→   120→      this.mass * (B * B + F * F) / 12 * 1.5,   // pitch (x)
   121→   121→      this.mass * (L * L + B * B) / 12 * 1.2,   // yaw (y)
   122→   122→      this.mass * (B * B + F * F) / 12 * 1.5,   // roll (z)
   123→   123→    );
   124→   124→
   125→   125→    // зонды плавучести: 7 шпангоутов × 3 точки (ниже ватерлинии).
   126→   126→    // Жёстче прежнего (нейтральная глубина 0.38 осадки) — корабль
   127→   127→    // быстрее следует за гребнем волны, не проваливается в него.
   128→   128→    const NZ = 7;
   129→   129→    let n = 0;
   130→   130→    for (let i = 0; i < NZ; i++) {
   131→   131→      const t = i / (NZ - 1);
   132→   132→      const z = THREE.MathUtils.lerp(-L * 0.44, L * 0.44, t);
   133→   133→      const prof = this.halfWidthAt(z);
   134→   134→      const offs = [0, -prof * 0.75, prof * 0.75];
   135→   135→      for (const x of offs) {
   136→   136→        this.probes.push({ off: new THREE.Vector3(x, -D * 0.38, z), k: 0, c: 0 });
   137→   137→        n++;
   138→   138→      }
   139→   139→    }
   140→   140→    const neutral = D * 0.38;
   141→   141→    const kTotal = this.mass * G / neutral;
   142→   142→    for (const p of this.probes) {
   143→   143→      p.k = kTotal / n;
   144→   144→      p.c = 2 * Math.sqrt(p.k * (this.mass / n)) * 0.5; // демпфер качки
   145→   145→    }
   146→   146→    // Резервные зонды по скулам борта чуть выше спокойной ватерлинии:
   147→   147→    // моделируют запас плавучести корпуса до планширя — гребень, накрывший
   148→   148→    // палубу, встречает сильное сопротивление и кренящий момент вправку,
   149→   149→    // «полностью по борт» корабль больше не погружается.
   150→   150→    let rn = 0;
   151→   151→    for (let i = 0; i < NZ; i++) {
   152→   152→      const t = i / (NZ - 1);
   153→   153→      const z = THREE.MathUtils.lerp(-L * 0.44, L * 0.44, t);
   154→   154→      const prof = this.halfWidthAt(z) * 0.85;
   155→   155→      for (const x of [-prof, prof]) {
   156→   156→        this.reserve.push({ off: new THREE.Vector3(x, F * 0.34, z), k: 0, c: 0 });
   157→   157→        rn++;
   158→   158→      }
   159→   159→    }
   160→   160→    // жёстче прежнего: полный вес запаса уже на 15 см погружения борта —
   161→   161→    // волна не может накрыть палубу, корпус «всплывает» на гребень
   162→   162→    const rkTotal = this.mass * G / 0.15;
   163→   163→    for (const p of this.reserve) {
   164→   164→      p.k = rkTotal / rn;
   165→   165→      p.c = 2 * Math.sqrt(p.k * (this.mass / rn)) * 1.3;
   166→   166→    }
   167→   167→
   168→   168→    this.foam = new WaterlineFoam(this.visual.outline);
   169→   169→    this.wake.setWaterHeight(waterHeightAt);
   170→   170→    if (isPlayer) enableLantern(this.visual);
   171→   171→  }
   172→   172→
   173→   173→  halfWidthAt(z: number): number {
   174→   174→    const t = clamp((z + this.preset.length / 2) / this.preset.length, 0, 1);
   175→   175→    let w: number;
   176→   176→    if (t < 0.45) w = THREE.MathUtils.lerp(0.58, 1.0, t / 0.45);
   177→   177→    else w = THREE.MathUtils.lerp(1.0, 0.0, Math.pow(clamp((t - 0.45) / 0.55, 0, 1), 0.75));
   178→   178→    return Math.max(0.3, w * this.preset.beam * 0.5);
   179→   179→  }
   180→   180→
   181→   181→  // ---------- геометрия ориентации ----------
   182→   182→  get forward(): THREE.Vector3 {
   183→   183→    const e = new THREE.Euler().setFromQuaternion(this.quaternion, 'YXZ');
   184→   184→    return new THREE.Vector3(Math.sin(e.y), 0, Math.cos(e.y)).normalize();
   185→   185→  }
   186→   186→  get heading(): number {
   187→   187→    const e = new THREE.Euler().setFromQuaternion(this.quaternion, 'YXZ');
   188→   188→    return e.y;
   189→   189→  }
   190→   190→  get euler(): THREE.Euler { return new THREE.Euler().setFromQuaternion(this.quaternion, 'YXZ'); }
   191→   191→  speed(): number { return Math.hypot(this.velocity.x, this.velocity.z); }
   192→   192→  speedKnots(): number { return this.speed() * 1.9438; }
   193→   193→
   194→   194→  /** Угол между направлением ОТКУДА дует (кажущийся) ветер и курсом, 0..PI */
   195→   195→  apparentWindAngle(): number {
   196→   196→    const wf = this.tmpV.set(Math.cos(this.wind.from), 0, Math.sin(this.wind.from));
   197→   197→    // кажущийся ветер = истинный - скорость корабля
   198→   198→    const av = this.tmpV2.set(-wf.x * this.wind.speed - this.velocity.x, 0, -wf.z * this.wind.speed - this.velocity.z);
   199→   199→    // вектор ОТКУДА дует кажущийся ветер:
   200→   200→    const aFromX = -av.x, aFromZ = -av.z;
   201→   201→    const len = Math.hypot(aFromX, aFromZ) || 1;
   202→   202→    const f = this.forward;
   203→   203→    const dot = (aFromX / len) * f.x + (aFromZ / len) * f.z;
   204→   204→    return Math.acos(clamp(dot, -1, 1));
   205→   205→  }
   206→   206→
   207→   207→  /** Полярная эффективность парусов по углу (рад): лавировка/бейдевинд/галфвинд/фордевинд */
   208→   208→  static polar(awa: number): number {
   209→   209→    const d = awa * 180 / Math.PI;
   210→   210→    if (d < 25) return 0.02;                       // левентик — почти нет хода
   211→   211→    if (d < 60) return THREE.MathUtils.lerp(0.08, 0.7, (d - 25) / 35);
   212→   212→    if (d < 110) return THREE.MathUtils.lerp(0.7, 1.0, (d - 60) / 50);  // бейдевинд-галфвинд
   213→   213→    if (d < 150) return THREE.MathUtils.lerp(1.0, 0.82, (d - 110) / 40);
   214→   214→    return THREE.MathUtils.lerp(0.82, 0.68, (d - 150) / 30);            // фордевинд
   215→   215→  }
   216→   216→
   217→   217→  // ---------- фиксированный шаг физики ----------
   218→   218→  fixedStep(dt: number, time: number) {
   219→   219→    if (this.sunk) return;
   220→   220→    this.force.set(0, 0, 0);
   221→   221→    this.torque.set(0, 0, 0);
   222→   222→
   223→   223→    const f = this._f.copy(this.forward);
   224→   224→    // штирборт (правый борт): forward × up = (−f.z, 0, f.x)
   225→   225→    const right = this._r.set(-f.z, 0, f.x).normalize();
   226→   226→    const e = this.euler;
   227→   227→    const spd = this.speed();
   228→   228→
   229→   229→    const m = this.mass + this.water * 1000; // вода в трюме утяжеляет
   230→   230→
   231→   231→    // --- гравитация ---
   232→   232→    this.force.y -= m * G;
   233→   233→
   234→   234→    // --- плавучесть по зондам (τ = r × F, F = (0, fl, 0)) ---
   235→   235→    for (const p of this.probes) {
   236→   236→      const wp = this._t1.copy(p.off).applyQuaternion(this.quaternion).add(this.position);
   237→   237→      const wy = waterHeightAt(wp.x, wp.z, time);
   238→   238→      const depth = wy - wp.y;
   239→   239→      if (depth > 0) {
   240→   240→        const rx = wp.x - this.position.x, rz = wp.z - this.position.z;
   241→   241→        // вертикальная скорость точки: v = ω × r → vy = ωz*rx − ωx*rz
   242→   242→        const vy = this.velocity.y + (this.angular.z * rx - this.angular.x * rz);
   243→   243→        let fl = p.k * Math.min(depth, 6) - p.c * vy;
   244→   244→        if (fl < 0) fl = 0;
   245→   245→        this.force.y += fl;
   246→   246→        this.torque.x += -rz * fl;
   247→   247→        this.torque.z += rx * fl;
   248→   248→      }
   249→   249→    }
   250→   250→    // --- запас плавучести бортов (пока не тонем) ---
   251→   251→    if (!this.sinking) {
   252→   252→      for (const p of this.reserve) {
   253→   253→        const wp = this._t1.copy(p.off).applyQuaternion(this.quaternion).add(this.position);
   254→   254→        const wy = waterHeightAt(wp.x, wp.z, time);
   255→   255→        const depth = wy - wp.y;
   256→   256→        if (depth > 0) {
   257→   257→          const rx = wp.x - this.position.x, rz = wp.z - this.position.z;
   258→   258→          const vy = this.velocity.y + (this.angular.z * rx - this.angular.x * rz);
   259→   259→          let fl = p.k * Math.min(depth, 2.5) - p.c * vy;
   260→   260→          if (fl < 0) fl = 0;
   261→   261→          this.force.y += fl;
   262→   262→          this.torque.x += -rz * fl;
   263→   263→          this.torque.z += rx * fl;
   264→   264→        }
   265→   265→      }
   266→   266→    }
   267→   267→
   268→   268→    // --- ветер на парусах ---
   269→   269→    this.sailFrac += (this.sailTarget / 3 - this.sailFrac) * Math.min(1, dt * 0.7);
   270→   270→    const awa = this.apparentWindAngle();
   271→   271→    const polar = Ship.polar(awa);
   272→   272→    const sailEffMul = this.sailHP / this.preset.sailHP;
   273→   273→    const damaged = sailEffMul < 0.5 ? 0.45 : sailEffMul < 0.85 ? 0.8 : 1.0;
   274→   274→    const windPower = Math.pow(this.wind.speed / 8.0, 1.6);
   275→   275→    let thrust = 0;
   276→   276→    if (!this.anchored && !this.sinking && this.mastsBroken < 2) {
   277→   277→      thrust = this.sailFrac * polar * windPower * this.preset.maxSpeed * 7.0 * damaged * (this.sailHP > 0 ? 1 : 0);
   278→   278→    }
   279→   279→    if (this.reverseThrust && !this.anchored) thrust = -this.preset.maxSpeed * 0.45; // задний ход на вёслах
   280→   280→    // сила тяги вдоль forward
   281→   281→    this.force.x += f.x * thrust * m * 0.06;
   282→   282→    this.force.z += f.z * thrust * m * 0.06;
   283→   283→    // крен от ветра: боковая составляющая ветра на парусах
   284→   284→    const wf = new THREE.Vector3(Math.cos(this.wind.from), 0, Math.sin(this.wind.from));
   285→   285→    const windTravel = wf.clone().multiplyScalar(-this.wind.speed);
   286→   286→    const lateral = windTravel.dot(right);
   287→   287→    const heelF = lateral * this.sailFrac * windPower * m * 0.05;
   288→   288→    this.torque.z += heelF;
   289→   289→    // ... и небольшое боковое сносовое усилие
   290→   290→    this.force.x += right.x * lateral * this.sailFrac * m * 0.004;
   291→   291→    this.force.z += right.z * lateral * this.sailFrac * m * 0.004;
   292→   292→
   293→   293→    // --- руль ---
   294→   294→    this.rudder += (this.rudderTarget - this.rudder) * Math.min(1, dt * 1.8);
   295→   295→    const rudderAngle = this.rudder * 0.6; // рад, макс ~34°
   296→   296→    const yawAuth = clamp(spd / 4.0, 0, 1.15); // авторитеты руля от скорости
   297→   297→    // положительный руль (D) = поворот ВПРАВО (штирборт):
   298→   298→    // вращение вокруг −Y, т.к. +Y = поворот влево (к порту)
   299→   299→    this.torque.y -= rudderAngle * yawAuth * m * 12.0; // ~6°/с на полном ходу
   300→   300→    // крен на повороте (отцентробежный)
   301→   301→    this.torque.z += this.angular.y * spd * m * 0.35;
   302→   302→    // лёгкое зарывание носа на скорости (слабое: сильное топило борт
   303→   303→    // в волну на 3+ метра — см. жалобы «корабль погружается по борт»)
   304→   304→    this.torque.x += spd * spd * m * 0.0012;
   305→   305→
   306→   306→    // --- сопротивление (по связным осям) ---
   307→   307→    const vLocal = this.velocity.clone();
   308→   308→    const fwdSpd = vLocal.dot(f);
   309→   309→    const latSpd = vLocal.dot(right);
   310→   310→    const fwdDrag = -Math.sign(fwdSpd) * (18 * fwdSpd + 55 * fwdSpd * Math.abs(fwdSpd)) * m * 0.001;
   311→   311→    const latDrag = -latSpd * (3.2) * m * 0.1;
   312→   312→    this.force.x += f.x * fwdDrag + right.x * latDrag;
   313→   313→    this.force.z += f.z * fwdDrag + right.z * latDrag;
   314→   314→    this.force.y -= this.velocity.y * m * 0.12; // вертикальное демпфирование
   315→   315→    if (this.anchored) {
   316→   316→      this.force.x -= this.velocity.x * m * 1.2;
   317→   317→      this.force.z -= this.velocity.z * m * 1.2;
   318→   318→    }
   319→   319→    if (this.grounded) {
   320→   320→      this.force.x -= this.velocity.x * m * 2.5;
   321→   321→      this.force.z -= this.velocity.z * m * 2.5;
   322→   322→    }
   323→   323→
   324→   324→    // --- угловое демпфирование по связным осям ---
   325→   325→    const invQ = this.tmpQ.copy(this.quaternion).invert();
   326→   326→    const ab = this._t3.copy(this.angular).applyQuaternion(invQ); // связные
   327→   327→    ab.x *= Math.exp(-1.6 * dt);  // дифферент
   328→   328→    ab.y *= Math.exp(-0.9 * dt);  // рыскание
   329→   329→    ab.z *= Math.exp(-1.1 * dt);  // крен
   330→   330→    this.angular.copy(ab.applyQuaternion(this.quaternion));
   331→   331→
   332→   332→    // --- ограничение крена/дифферента (мягкие стопы) ---
   333→   333→    // дифферент жёстко: 0.14 рад (8°) — нос не может нырнуть под гребень;
   334→   334→    // крен 0.36 рад (20°) — подветренный борт остаётся над водой
   335→   335→    const roll = e.z, pitch = e.x;
   336→   336→    const rollLim = 0.36, pitchLim = 0.14;
   337→   337→    if (Math.abs(roll) > rollLim) this.torque.z -= (roll - Math.sign(roll) * rollLim) * m * 9.0;
   338→   338→    if (Math.abs(pitch) > pitchLim) this.torque.x -= (pitch - Math.sign(pitch) * pitchLim) * m * 12.0;
   339→   339→    if (this.sinking) {
   340→   340→      this.sinkT += dt;
   341→   341→      this.force.y -= m * G * Math.min(0.9, this.sinkT / 9); // неотвратимое погружение
   342→   342→      this.torque.x += m * 0.25 * Math.sign(roll || 1);
   343→   343→      this.torque.z += m * 0.4 * Math.sign(roll || 1);
   344→   344→    }
   345→   345→
   346→   346→    // --- интеграция ---
   347→   347→    this.velocity.addScaledVector(this.force, dt / m);
   348→   348→    this.position.addScaledVector(this.velocity, dt);
   349→   349→    // α = I⁻¹ τ в связных осях
   350→   350→    const tb = this._t3.copy(this.torque).applyQuaternion(invQ);
   351→   351→    tb.x /= this.inertia.x; tb.y /= this.inertia.y; tb.z /= this.inertia.z;
   352→   352→    this.angular.addScaledVector(tb.applyQuaternion(this.quaternion), dt);
   353→   353→    // ограничим угловые скорости
   354→   354→    const wLen = this.angular.length();
   355→   355→    if (wLen > 1.1) this.angular.multiplyScalar(1.1 / wLen);
   356→   356→    // q̇ = 0.5 · ω⊗q → q_new = q + 0.5·dt·(ωx,ωy,ωz,0)⊗q, потом нормализация
   357→   357→    this.wq.set(this.angular.x, this.angular.y, this.angular.z, 0);
   358→   358→    this.dq.copy(this.wq).multiply(this.quaternion); // dq = ω ⊗ q
   359→   359→    const qs = 0.5 * dt;
   360→   360→    this.quaternion.set(
   361→   361→      this.quaternion.x + this.dq.x * qs,
   362→   362→      this.quaternion.y + this.dq.y * qs,
   363→   363→      this.quaternion.z + this.dq.z * qs,
   364→   364→      this.quaternion.w + this.dq.w * qs
   365→   365→    ).normalize();
   366→   366→
   367→   367→    // --- посадка на мель / столкновение с землёй ---
   368→   368→    this.checkGrounding(dt);
   369→   369→
   370→   370→    // --- течение воды в трюме ---
   371→   371→    if (this.leakRate > 0) {
   372→   372→      this.water = Math.min(this.holdCap, this.water + this.leakRate * dt);
   373→   373→    }
   374→   374→    if (this.onFire) {
   375→   375→      this.fireT += dt;
   376→   376→      this.hullHP -= dt * 1.6;
   377→   377→      if (this.fireT > 26) { this.onFire = false; this.fireT = 0; } // выгорит
   378→   378→      if (Math.random() < dt * 1.2) {
   379→   379→        const p = this.particles;
   380→   380→        const wp = this.position.clone();
   381→   381→        p.fire(wp.x + (Math.random() - 0.5) * 4, this.waterlineY() + 1 + Math.random() * 2, wp.z + (Math.random() - 0.5) * 10);
   382→   382→        p.smoke(wp.x + (Math.random() - 0.5) * 4, this.waterlineY() + 3, wp.z + (Math.random() - 0.5) * 10, this.wind ? Math.cos(this.wind.from + Math.PI) * 2 : 0, 2.2, this.wind ? Math.sin(this.wind.from + Math.PI) * 2 : 0, 3, 3.5);
   383→   383→      }
   384→   384→    }
   385→   385→    // порог затопления
   386→   386→    if (!this.sinking && this.water > this.holdCap * 0.62) this.startSinking();
   387→   387→    if (this.hullHP <= 0 && !this.sinking) this.startSinking();
   388→   388→
   389→   389→    // --- сдался? ---
   390→   390→    if (!this.isPlayer && !this.surrendered && !this.sinking) {
   391→   391→      if (this.hullHP < this.hullMax * 0.34 || this.crew < this.preset.maxCrew * 0.25) {
   392→   392→        this.surrendered = true;
   393→   393→        this.sailTarget = 0;
   394→   394→        this.onSurrender?.(this);
   395→   395→      }
   396→   396→    }
   397→   397→
   398→   398→    if (this.sinking && this.sinkT > 14 && !this.sunk) {
   399→   399→      this.sunk = true;
   400→   400→      this.visual.group.visible = false;
   401→   401→      this.onSunk?.(this);
   402→   402→    }
   403→   403→  }
   404→   404→
   405→   405→  private checkGrounding(dt: number) {
   406→   406→    if (this.sinking || this.sunk) return;
   407→   407→    const f = this.forward;
   408→   408→    const bow = this.tmpV.copy(this.position).addScaledVector(f, this.preset.length * 0.42);
   409→   409→    const stern = this.tmpV2.copy(this.position).addScaledVector(f, -this.preset.length * 0.42);
   410→   410→    let hit = false;
   411→   411→    for (const p of [bow, stern]) {
   412→   412→      const th = this.terrainHeight(p.x, p.z);
   413→   413→      if (th > this.position.y - this.preset.draft * 0.6) {
   414→   414→        hit = true;
   415→   415→        // нормаль склона
   416→   416→        const e = 2;
   417→   417→        const nx = this.terrainHeight(p.x - e, p.z) - this.terrainHeight(p.x + e, p.z);
   418→   418→        const nz = this.terrainHeight(p.x, p.z - e) - this.terrainHeight(p.x, p.z + e);
   419→   419→        const n = new THREE.Vector3(nx, 2 * e, nz).normalize();
   420→   420→        this.force.addScaledVector(n, this.mass * 22);
   421→   421→        this.velocity.multiplyScalar(0.94);
   422→   422→      }
   423→   423→    }
   424→   424→    if (hit) {
   425→   425→      if (!this.grounded && this.speed() > 3.5) {
   426→   426→        this.applyDamage({ kind: 'ground', amount: 5 + this.speed() });
   427→   427→      }
   428→   428→      this.grounded = true;
   429→   429→    } else this.grounded = false;
   430→   430→  }
   431→   431→
   432→   432→  waterlineY(): number {
   433→   433→    return this.position.y;
   434→   434→  }
   435→   435→
   436→   436→  // ---------- повреждения ----------
   437→   437→  applyDamage(d: DamageInput) {
   438→   438→    if (this.sunk || this.sinking) return;
   439→   439→    const rng = Math.random;
   440→   440→    switch (d.kind) {
   441→   441→      case 'ball': {
   442→   442→        this.hullHP -= d.amount;
   443→   443→        if (d.hitPoint && d.hitPoint.y < 0.45 + this.position.y) this.addLeak(0.06 + rng() * 0.05);
   444→   444→        if (rng() < 0.3) this.crew = Math.max(0, this.crew - 1);
   445→   445→        if (rng() < 0.18) this.breakMast();
   446→   446→        break;
   447→   447→      }
   448→   448→      case 'chain': {
   449→   449→        this.sailHP = Math.max(0, this.sailHP - d.amount);
   450→   450→        if (rng() < 0.25) this.crew = Math.max(0, this.crew - 1);
   451→   451→        break;
   452→   452→      }
   453→   453→      case 'grape': {
   454→   454→        this.crew = Math.max(0, this.crew - Math.round(d.amount));
   455→   455→        this.hullHP -= d.amount * 0.4;
   456→   456→        break;
   457→   457→      }
   458→   458→      case 'fire': {
   459→   459→        this.hullHP -= d.amount;
   460→   460→        if (rng() < 0.35 && !this.onFire) { this.onFire = true; this.fireT = 0; }
   461→   461→        if (rng() < 0.08) { this.addLeak(0.03); }
   462→   462→        break;
   463→   463→      }
   464→   464→      case 'ram': {
   465→   465→        this.hullHP -= d.amount;
   466→   466→        if (rng() < 0.5) this.addLeak(0.04);
   467→   467→        break;
   468→   468→      }
   469→   469→      case 'ground': this.hullHP -= d.amount * 0.5; break;
   470→   470→    }
   471→   471→    if (this.hullHP <= 0 && !this.sinking) this.startSinking();
   472→   472→    if (!this.isPlayer && !this.surrendered && this.hullHP < this.hullMax * 0.34) {
   473→   473→      this.surrendered = true; this.sailTarget = 0; this.onSurrender?.(this);
   474→   474→    }
   475→   475→  }
   476→   476→
   477→   477→  addLeak(rate: number) {
   478→   478→    this.leakRate = Math.min(0.5, this.leakRate + rate);
   479→   479→    if (this.leakRate < 0) this.leakRate = 0;
   480→   480→  }
   481→   481→
   482→   482→  breakMast() {
   483→   483→    if (this.mastsBroken >= this.visual.masts.length) return;
   484→   484→    this.mastsBroken++;
   485→   485→    this.mastAnim = 0;
   486→   486→    this.sailHP = Math.max(0, this.sailHP - 30);
   487→   487→    if (this.sailTarget > 0) this.sailTarget = Math.max(0, this.sailTarget - 1);
   488→   488→  }
   489→   489→
   490→   490→  startSinking() {
   491→   491→    if (this.sinking) return;
   492→   492→    this.sinking = true;
   493→   493→    this.sinkT = 0;
   494→   494→    this.onFire = false;
   495→   495→  }
   496→   496→
   497→   497→  /** Действие команды (R): пожар -> течь -> ремонт */
   498→   498→  crewAction(dt: number): string | null {
   499→   499→    if (this.onFire) { this.fireT += dt * 3; if (this.fireT > 4) { this.onFire = false; this.fireT = 0; return 'Пожар потушен'; } return 'Тушение пожара...'; }
   500→   500→    if (this.water > 0.01) {
   501→   501→      const pump = 0.05 + this.crew * 0.004;
   502→   502→      this.water = Math.max(0, this.water - pump * dt);
   503→   503→      this.leakRate = Math.max(0, this.leakRate - 0.004 * dt);
   504→   504→      return this.water > 0.05 ? 'Откачка воды...' : 'Трюм осушён';
   505→   505→    }
   506→   506→    if (this.hullHP < this.hullMax && this.planks > 0) {
   507→   507→      this.hullHP = Math.min(this.hullMax, this.hullHP + 0.9 * dt);
   508→   508→      this.planks -= dt * 0.15;
   509→   509→      return 'Ремонт корпуса...';
   510→   510→    }
   511→   511→    return null;
   512→   512→  }

   514→   514→  // ---------- визуал ----------
   515→   515→  updateVisual(time: number, dt: number, particles: Particles) {
   516→   516→    const v = this.visual;
   517→   517→    v.group.position.copy(this.position);
   518→   518→    v.group.quaternion.copy(this.quaternion);
   519→   519→
   520→   520→    // паруса: количество поставленных парусов соответствует ФАКТИЧЕСКОЙ
   521→   521→    // СКОРОСТИ — экипаж ставит паруса по мере разгона и убирает по мере
   522→   522→    // остановки; уровень W/S задаёт максимум. Рея брасопится по ветру,
   523→   523→    // «пузо» — по кажущемуся ветру
   524→   524→    const awa = this.apparentWindAngle();
   525→   525→    const polar = Ship.polar(awa);
   526→   526→    const windClamp = clamp(this.wind.speed / 9, 0.3, 1.4);
   527→   527→    // скорость, достижимая под полным парусным вооружением в этот ветер
   528→   528→    // (корень из тяги — сопротивление квадратично): ~0.85·maxSpeed
   529→   529→    const windPow = Math.pow(clamp(this.wind.speed / 8.0, 0.05, 1.6), 1.6);
   530→   530→    const attainable = this.preset.maxSpeed * 0.85 * Math.sqrt(clamp(polar * windPow, 0, 1.5));
   531→   531→    // в левентик/штиль паруса остаются на мачтах и трепещут (не убираются
   532→   532→    // механически) — убираются только когда есть чем идти, но нет хода
   533→   533→    const spdSetFrac = attainable > 1.5 ? clamp(this.speed() / attainable, 0, 1) : 1;
   534→   534→    const setTotal = Math.min(this.sailFrac, spdSetFrac);
   535→   535→    // локальный вектор «откуда дует» (x — борт, z — нос): ψ = φ + курс
   536→   536→    const psi = this.wind.from + this.heading;
   537→   537→    const wlX = Math.cos(psi); // >0 — ветер с порта (левый борт)
   538→   538→    // цель брасопки: рея поперёк ветра (нормаль паруса по ветру), ±55°
   539→   539→    let bt = Math.atan2(wlX, Math.sin(psi)) - Math.PI;
   540→   540→    bt = Math.atan2(Math.sin(bt), Math.cos(bt)); // кратчайшая дуга
   541→   541→    const braceTarget = clamp(bt, -0.96, 0.96);
   542→   542→    const bellyBase = (0.35 + 0.95 * windClamp) * (0.15 + 0.85 * polar);
   543→   543→    const flutterBase = 0.012 * windClamp * (1 + (1 - polar) * 2.2);
   544→   544→    for (const s of v.sails.sails) {
   545→   545→      // доля постановки этого паруса при текущей скорости хода
   546→   546→      const setFrac = smoothstep(s.setAt - 0.38, s.setAt, setTotal);
   547→   547→      s.furl += ((1 - setFrac) - s.furl) * Math.min(1, dt * 1.1);
   548→   548→      const sf = 1 - s.furl;
   549→   549→      s.mat.uniforms.uTime.value = time;
   550→   550→      s.mat.uniforms.uFurl.value = s.furl;
   551→   551→      s.mat.uniforms.uDamage.value = 1 - this.sailHP / this.preset.sailHP;
   552→   552→      s.mat.uniforms.uFlutter.value = flutterBase * (0.25 + 0.75 * sf);
   553→   553→      if (s.square && s.brace) {
   554→   554→        // реи разворачиваются медленно — как будто работают брашпили
   555→   555→        s.braceAngle += (braceTarget - s.braceAngle) * Math.min(1, dt * 0.5);
   556→   556→        s.brace.rotation.y = s.braceAngle;
   557→   557→        s.mat.uniforms.uBelly.value = bellyBase * sf;
   558→   558→      } else {
   559→   559→        // кливер: пузо в подветренную сторону (локальный Z → борт)
   560→   560→        s.mat.uniforms.uBelly.value = bellyBase * sf * (-wlX);
   561→   561→      }
   562→   562→    }
   563→   563→
   564→   564→    // флаг — по ветру
   565→   565→    const wf = Math.atan2(-Math.cos(this.wind.from), -Math.sin(this.wind.from));
   566→   566→    v.flag.rotation.y = wf - this.heading + Math.PI;
   567→   567→    v.flagMat.uniforms.uTime.value = time;
   568→   568→
   569→   569→    // штурвал крутится от руля: D (руль вправо) — по часовой от рулевого
   570→   570→    v.wheel.rotation.z = this.rudder * 2.2;
   571→   571→
   572→   572→    // падение мачты
   573→   573→    if (this.mastAnim >= 0 && this.mastAnim < 1 && this.mastsBroken > 0) {
   574→   574→      this.mastAnim = Math.min(1, this.mastAnim + dt * 0.5);
   575→   575→      const idx = Math.min(this.visual.masts.length - 1, this.mastsBroken - 1);
   576→   576→      const mast = this.visual.masts[idx];
   577→   577→      mast.rotation.x = this.mastAnim * 1.15; // валится вперёд
   578→   578→      mast.position.y = -this.mastAnim * 2;
   579→   579→    }
   580→   580→
   581→   581→    // пенный воротник по ватерлинии
   582→   582→    if (!this.sunk) {
   583→   583→      const wy = waterHeightAt(this.position.x, this.position.z, time);
   584→   584→      this.foam.mesh.position.set(this.position.x, wy + 0.28, this.position.z);
   585→   585→      this.foam.mesh.rotation.y = this.heading;
   586→   586→      const spdFrac = clamp(this.speed() / this.preset.maxSpeed, 0, 1);
   587→   587→      this.foam.update(time, spdFrac, this.euler.z);
   588→   588→    } else {
   589→   589→      this.foam.mesh.visible = false;
   590→   590→    }
   591→   591→
   592→   592→    // кильватерный след
   593→   593→    const sternW = this.tmpV.set(0, 0, -this.preset.length * 0.48)
   594→   594→      .applyQuaternion(this.quaternion).add(this.position);
   595→   595→    const spdFrac = clamp(this.speed() / this.preset.maxSpeed, 0, 1);
   596→   596→    if (!this.sunk && !this.grounded) {
   597→   597→      this.wake.push(sternW, spdFrac, dt, time);
   598→   598→    }
   599→   599→    this.wake.update(time);
   600→   600→
   601→   601→    // носовой бурун: брызги при ударе носа о волну
   602→   602→    if (!this.sunk) {
   603→   603→      const bowW = this.tmpV2.set(0, -this.preset.draft * 0.3, this.preset.length * 0.46)
   604→   604→        .applyQuaternion(this.quaternion).add(this.position);
   605→   605→      const waveY = waterHeightAt(bowW.x, bowW.z, time);
   606→   606→      const relV = this.velocity.y - (waveY - this.lastStern.y) * 0; // грубая оценка
   607→   607→      const impact = (waveY - bowW.y) * (0.3 + spdFrac);
   608→   608→      if (impact > 0.55 && Math.random() < 0.6) {
   609→   609→        particles.splash(bowW.x, waveY, bowW.z, 0.5 + spdFrac * 0.9);
   610→   610→      }
   611→   611→      if (spdFrac > 0.35 && Math.random() < spdFrac * 0.5) {
   612→   612→        particles.spawn(bowW.x + (Math.random() - 0.5) * 2, waveY + 0.1, bowW.z + (Math.random() - 0.5) * 2,
   613→   613→          (Math.random() - 0.5) * 2, 1.5 + Math.random() * 2.5 * spdFrac, (Math.random() - 0.5) * 2,
   614→   614→          0.5, 0.3 + Math.random() * 0.35, 0, 0.4, 10);
   615→   615→      }
   616→   616→      // крен: всплески у борта
   617→   617→      const roll = this.euler.z;
   618→   618→      this.rollSmooth += (roll - this.rollSmooth) * Math.min(1, dt * 3);
   619→   619→      if (Math.abs(this.rollSmooth) > 0.28 && Math.random() < 0.3) {
   620→   620→        const side = Math.sign(this.rollSmooth) * this.preset.beam * 0.5;
   621→   621→        const px = this.position.x + Math.cos(this.heading) * side;
   622→   622→        const pz = this.position.z - Math.sin(this.heading) * side;
   623→   623→        particles.splash(px, waterHeightAt(px, pz, time), pz, 0.5);
   624→   624→      }
   625→   625→    }
   626→   626→  }
   627→   627→
   628→   628→  /** Сдвиг мира (floating origin) */
   629→   629→  shiftWorld(delta: THREE.Vector3) {
   630→   630→    this.position.add(delta);
   631→   631→    this.wake.shiftWorld(delta);
   632→   632→  }

   634→   634→  dispose() {
   635→   635→    this.visual.group.traverse((o) => {
   636→   636→      const mesh = o as THREE.Mesh;
   637→   637→      if (mesh.geometry) mesh.geometry.dispose();
   638→   638→    });
   639→   639→  }
