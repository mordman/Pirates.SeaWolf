    29→function Bar({ pct, color, bg = 'bg-stone-800' }: { pct: number; color: string; bg?: string }) {
    30→  return (
    31→    <div className={`h-2 w-full overflow-hidden rounded-sm ${bg}`}>
    32→      <div className={`h-full transition-all duration-300 ${color}`} style={{ width: `${Math.max(0, Math.min(1, pct)) * 100}%` }} />

    34→  );

    37→function WindCompass({ hud }: { hud: HudState }) {
    38→  const rel = hud.windRelDeg;
    39→  return (
    40→    <div className="flex items-center gap-3 p-3">
    41→      <svg width="84" height="84" viewBox="0 0 100 100" className="shrink-0">
    42→        <circle cx="50" cy="50" r="46" fill="#0d0906" stroke="#7c5a2a" strokeWidth="2" />
    43→        <circle cx="50" cy="50" r="35" fill="none" stroke="#3d2e18" strokeWidth="1" />
    44→        {['С', 'В', 'Ю', 'З'].map((d, i) => {
    45→          const a = (i * 90 - 90) * (Math.PI / 180);
    46→          return (
    47→            <text key={d} x={50 + Math.cos(a) * 40} y={50 + Math.sin(a) * 40 + 4} textAnchor="middle"
    48→              fill={d === 'С' ? '#e8c87a' : '#8a734d'} fontSize="11" fontWeight="bold">{d}</text>
    49→          );
    50→        })}

    52→        <polygon points="50,26 56,62 50,55 44,62" fill="#d8c9a8" />
    53→        {/* стрелка ветра: откуда дует относительно курса */}
    54→        <g transform={`rotate(${rel} 50 50)`}>
    55→          <line x1="50" y1="18" x2="50" y2="34" stroke="#7dd3fc" strokeWidth="3" strokeLinecap="round" />
    56→          <polygon points="50,10 45,22 55,22" fill="#7dd3fc" />
    57→        </g>
    58→      </svg>
    59→      <div className="grid grid-cols-[auto_auto] gap-x-3 gap-y-1 text-xs">
    60→        <span className="text-stone-400">Ветер</span>
    61→        <span className="text-sky-200 font-semibold">{hud.windSpeed.toFixed(1)} м/с</span>
    62→        <span className="text-stone-400">Курс</span>
    63→        <span className="text-amber-100 font-semibold">{COMPASS[Math.round(hud.headingDeg / 22.5) % 16]} {hud.headingDeg}°</span>

    73→export default function Home() {
    74→  const canvasRef = useRef<HTMLCanvasElement>(null);
    75→  const gameRef = useRef<Game | null>(null);
    76→  const mapCanvasRef = useRef<HTMLCanvasElement>(null);
    77→  const [hud, setHud] = useState<HudState | null>(null);
    78→  const [mapOpen, setMapOpen] = useState(false);
    79→  const [helpOpen, setHelpOpen] = useState(false);
    80→  const [settingsOpen, setSettingsOpen] = useState(false);
    81→  const [settings, setSettings] = useState<GameSettings>(() =>
    82→    typeof window === 'undefined' ? structuredClone(DEFAULT_SETTINGS) : loadSettings());
    83→  const [portMsg, setPortMsg] = useState<string | null>(null);
    84→  const [ready, setReady] = useState(false);
    85→
    86→  useEffect(() => {
    87→    let disposed = false;
    88→    let game: Game | null = null;
    89→    (async () => {
    90→      const { Game: GameClass } = await import('@/game/game');
    91→      if (disposed || !canvasRef.current) return;
    92→      game = new GameClass(canvasRef.current);
    93→      gameRef.current = game;
    94→      game.run();
    95→      setReady(true);
    96→      const iv = setInterval(() => setHud({ ...game!.hud }), 110);
    97→      return () => clearInterval(iv);
    98→    })();
    99→    return () => {
   100→      disposed = true;
   101→      game?.dispose();
   102→      gameRef.current = null;
   103→    };
   104→  }, []);
   105→
   106→  // Tab / H / Esc — клавиши из настроек; при открытом экране настроек — пауза
   107→  useEffect(() => {
   108→    const onKey = (e: KeyboardEvent) => {
   109→      const c = settings.controls;
   110→      if (e.code === 'Escape') {
   111→        setSettingsOpen(false); setMapOpen(false); setHelpOpen(false);
   112→        const gg = gameRef.current; if (gg) { gg.uiOpen = false; gg.input.keys.clear(); }
   113→        return;
   114→      }
   115→      if (settingsOpen) return;
   116→      if (e.code === c.map) { e.preventDefault(); setMapOpen(v => !v); }
   117→      if (e.code === c.help) setHelpOpen(v => !v);
   118→    };
   119→    window.addEventListener('keydown', onKey);
   120→    return () => window.removeEventListener('keydown', onKey);
   121→  }, [settings, settingsOpen]);
   122→
   123→  const openSettings = useCallback(() => {
   124→    const gg = gameRef.current;
   125→    if (gg) gg.uiOpen = true;
   126→    setSettingsOpen(true);
   127→  }, []);
   128→
   129→  const closeSettings = useCallback(() => {
   130→    const gg = gameRef.current;
   131→    if (gg) { gg.uiOpen = false; gg.input.keys.clear(); }
   132→    setSettingsOpen(false);
   133→  }, []);
   134→
   135→  const applySettings = useCallback((s: GameSettings) => {
   136→    setSettings(s);
   137→    saveSettings(s);
   138→    gameRef.current?.applySettings(s);
   139→  }, []);
   140→
   141→  // отрисовка карты
   142→  useEffect(() => {
   143→    if (!mapOpen || !hud || !mapCanvasRef.current) return;
   144→    const cv = mapCanvasRef.current;
   145→    const ctx = cv.getContext('2d')!;
   146→    const W = cv.width, H = cv.height;
   147→    ctx.clearRect(0, 0, W, H);
   148→    ctx.fillStyle = '#0e2a33';
   149→    ctx.fillRect(0, 0, W, H);
   150→    // сетка
   151→    ctx.strokeStyle = 'rgba(255,255,255,0.05)';
   152→    for (let x = 0; x < W; x += 60) { ctx.beginPath(); ctx.moveTo(x, 0); ctx.lineTo(x, H); ctx.stroke(); }
   153→    for (let y = 0; y < H; y += 60) { ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(W, y); ctx.stroke(); }
   154→    const scale = 0.055; // мир -> пиксели
   155→    const cx = W / 2, cy = H / 2;
   156→    const px = (x: number) => cx + x * scale;
   157→    const pz = (z: number) => cy + z * scale;
   158→    // острова
   159→    for (const isl of hud.mapIslands) {
   160→      const r = Math.max(4, isl.r * scale);
   161→      if (!isl.discovered) {
   162→        ctx.strokeStyle = 'rgba(220,200,150,0.25)';
   163→        ctx.setLineDash([4, 4]);
   164→        ctx.beginPath(); ctx.arc(px(isl.x), pz(isl.z), r, 0, Math.PI * 2); ctx.stroke();
   165→        ctx.setLineDash([]);
   166→        ctx.fillStyle = 'rgba(220,200,150,0.35)';
   167→        ctx.font = '10px serif';
   168→        ctx.textAlign = 'center';
   169→        ctx.fillText('?', px(isl.x), pz(isl.z) + 3);
   170→        continue;
   171→      }
   172→      const grad = ctx.createRadialGradient(px(isl.x), pz(isl.z), r * 0.2, px(isl.x), pz(isl.z), r);
   173→      grad.addColorStop(0, '#4a7a3a');
   174→      grad.addColorStop(0.75, '#c9b26a');
   175→      grad.addColorStop(1, '#1d5d6e');
   176→      ctx.fillStyle = grad;
   177→      ctx.beginPath(); ctx.arc(px(isl.x), pz(isl.z), r, 0, Math.PI * 2); ctx.fill();
   178→      ctx.fillStyle = '#f0e2c0';
   179→      ctx.font = 'bold 11px serif';
   180→      ctx.textAlign = 'center';
   181→      ctx.fillText(isl.name, px(isl.x), pz(isl.z) + r + 14);
   182→    }
   183→    // корабли ИИ
   184→    for (const s of hud.mapShips) {
   185→      ctx.fillStyle = s.surrendered ? '#e8e4d8' : s.kind === 'brig' ? '#c23b3b' : '#d8a03c';
   186→      ctx.beginPath(); ctx.arc(px(s.x), pz(s.z), 4, 0, Math.PI * 2); ctx.fill();
   187→    }
   188→    // игрок
   189→    const p = hud.mapPlayer;
   190→    ctx.save();
   191→    ctx.translate(px(p.x), pz(p.z));
   192→    ctx.rotate(-p.h);
   193→    ctx.fillStyle = '#7dd3fc';
   194→    ctx.beginPath();
   195→    ctx.moveTo(0, -9); ctx.lineTo(6, 7); ctx.lineTo(0, 4); ctx.lineTo(-6, 7);
   196→    ctx.closePath(); ctx.fill();
   197→    ctx.restore();
   198→    // роза ветров
   199→    ctx.fillStyle = '#e8c87a';
   200→    ctx.font = 'bold 12px serif';
   201→    ctx.textAlign = 'left';
   202→    ctx.fillText(`Ветер: ${hud.windFromDeg}°, ${hud.windSpeed.toFixed(1)} м/с`, 12, H - 14);
   203→  }, [mapOpen, hud]);
   204→

   207→    if (!g) return;

   209→    setPortMsg(typeof res === 'string' ? res : null);
   210→  }, []);
   211→
   212→  const g = gameRef.current;
   213→
   214→  return (
   215→    <main className="fixed inset-0 overflow-hidden bg-black select-none">
   216→      <canvas ref={canvasRef} className="absolute inset-0 h-full w-full" />

   219→        <div className="absolute inset-0 flex items-center justify-center bg-[#0a0704]">
   220→          <div className="text-center">
   221→            <div className="mb-3 text-3xl font-serif text-amber-300">Морские волки</div>
   222→            <div className="text-stone-400">Грузим пушки, канаты и ром...</div>
   223→          </div>
   224→        </div>
   225→      )}
   226→
   227→      {hud && !hud.started && ready && (
   228→        <div className="absolute inset-0 z-40 flex items-center justify-center bg-black/70 backdrop-blur-sm">
   229→          <div className="max-w-lg rounded-xl border-2 border-amber-800/70 bg-[#1a120a] p-8 text-center shadow-2xl">
   230→            <h1 className="mb-1 font-serif text-4xl font-bold text-amber-300">Морские волки</h1>
   231→            <p className="mb-5 text-sm italic text-stone-400">Пиратская песочница на бесконечном море</p>
   232→            <div className="mb-6 grid grid-cols-2 gap-x-6 gap-y-1.5 text-left text-xs text-stone-300">
   233→              <span><b className="text-amber-200">{keyLabel(settings.controls.sailsUp)} / {keyLabel(settings.controls.sailsDown)}</b> — паруса 0–3</span>
   234→              <span><b className="text-amber-200">{keyLabel(settings.controls.steerLeft)} / {keyLabel(settings.controls.steerRight)}</b> — руль</span>
   235→              <span><b className="text-amber-200">{keyLabel(settings.controls.aimPort)} / {keyLabel(settings.controls.aimStarboard)}</b> — прицел борта</span>
   236→              <span><b className="text-amber-200">ЛКМ / {keyLabel(settings.controls.fire)}</b> — выстрел / залп</span>
   237→              <span><b className="text-amber-200">{keyLabel(settings.controls.ammo[0])}–{keyLabel(settings.controls.ammo[3])}</b> — боеприпасы</span>
   238→              <span><b className="text-amber-200">{keyLabel(settings.controls.camera)}</b> — камера (изометрия и др.)</span>
   239→              <span><b className="text-amber-200">Колесо</b> — зум</span>
   240→              <span><b className="text-amber-200">{keyLabel(settings.controls.anchor)}</b> — якорь</span>
   241→              <span><b className="text-amber-200">{keyLabel(settings.controls.repair)}</b> — ремонт / откачка</span>
   242→              <span><b className="text-amber-200">{keyLabel(settings.controls.map)}</b> — карта · <b className="text-amber-200">{keyLabel(settings.controls.help)}</b> — помощь</span>
   243→            </div>
   244→            <button
   245→              onClick={() => { gameRef.current?.start(); }}
   246→              className="pointer-events-auto rounded-lg border border-amber-600 bg-gradient-to-b from-amber-600 to-amber-800 px-8 py-3 font-serif text-lg font-bold text-amber-50 shadow-lg transition hover:from-amber-500 hover:to-amber-700 active:scale-95"

   258→            <Panel className="p-3">
   259→              <div className="mb-2 flex items-baseline justify-between">
   260→                <span className="font-serif text-sm font-bold text-amber-200">Шлюп «Бриз»</span>
   261→                <span className="text-[10px] text-stone-500">{hud.fps} FPS</span>

   263→              <div className="space-y-1.5 text-[11px]">

   267→                <div className="flex justify-between"><span className="text-stone-400">Паруса</span>

   285→          {/* ---- ветер/курс (право-верх) + настройки ---- */}
   286→          <div className="absolute right-3 top-3 flex flex-col items-end gap-2">
   287→            <Panel><WindCompass hud={hud} /></Panel>
   288→            <button onClick={openSettings}
   289→              className="pointer-events-auto rounded-lg border border-amber-900/60 bg-[#160f08]/80 px-3 py-1.5 text-xs font-semibold text-amber-200 shadow-lg backdrop-blur-sm transition hover:border-amber-500 hover:text-amber-100">
   290→              ⚙ Настройки
   291→            </button>
   292→          </div>
   293→
   294→          {/* ---- сообщения ---- */}
   295→          <div className="absolute left-1/2 top-3 w-96 -translate-x-1/2 space-y-1">

   298→                {m.text}
   299→              </div>

   302→
   303→          {/* ---- нижняя центральная панель ---- */}
   304→          <div className="absolute bottom-3 left-1/2 -translate-x-1/2">

   309→                  Паруса ({keyLabel(settings.controls.sailsUp)}/{keyLabel(settings.controls.sailsDown)})
