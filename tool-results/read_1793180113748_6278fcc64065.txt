    82→// ---------- шейдер паруса ----------
    83→function sailMaterial(baseColor: number): THREE.ShaderMaterial {
    84→  return new THREE.ShaderMaterial({
    85→    uniforms: {
    86→      uBelly: { value: 0.5 },
    87→      uFlutter: { value: 0.3 },
    88→      uTime: { value: 0 },
    89→      uDamage: { value: 0 },
    90→      uColor: { value: new THREE.Color(baseColor) },
    91→      uSunDir: { value: new THREE.Vector3(0.4, 0.8, 0.3) },
    92→      uFurl: { value: 0 },
    93→    },
    94→    vertexShader: /* glsl */`
    95→      uniform float uBelly, uTime, uFurl, uFlutter;
    96→      varying vec2 vUv;
    97→      varying vec3 vN;
    98→      varying float vDmgU;
    99→      uniform float uDamage;
   100→      void main() {
   101→        vUv = uv;
   102→        vDmgU = uDamage;
   103→        vec3 p = position;
   104→        // «пузо» паруса: синус по ширине и высоте, кромки закреплены.
   105→        // асимметрия по высоте: максимум ниже середины — наполненное
   106→        // полотно не выглядит плоским мешком. uBelly задан на CPU с
   107→        // учётом ветра и степени постановки
   108→        float belly = uBelly * sin(uv.x * 3.14159) * sin(uv.y * 3.14159)
   109→                    * (0.72 + 0.28 * (1.0 - uv.y));
   110→        // трепет свободной кромки: две волны, сильнее при слабом пузе (флюктер)
   111→        float flut = uFlutter * (
   112→          sin(uv.x * 9.0 + uTime * 7.3) * 0.6 +
   113→          sin(uv.x * 17.0 - uTime * 11.7) * 0.4
   114→        ) * sin(uv.y * 3.14159) * (1.0 - uv.y * 0.75);
   115→        float dz = belly + flut;
   116→        p.z += dz;
   117→        // нормаль аналитически (по пуза) — с полным наклоном, чтобы
   118→        // выпуклость читалась и в освещении, а не только в силуэте
   119→        float dzu = uBelly * cos(uv.x * 3.14159) * 3.14159 * sin(uv.y * 3.14159);
   120→        float dzv = uBelly * sin(uv.x * 3.14159) * 3.14159 * cos(uv.y * 3.14159);
   121→        vN = normalize(vec3(-dzu * 0.8, -dzv * 0.8, 1.0));
   122→        // рифление: при uFurl парус собирается к рее/штагу
   123→        p.y *= mix(1.0, 0.10, uFurl);
   124→        p.x *= mix(1.0, 0.55, uFurl);
   125→        p.z *= mix(1.0, 0.08, uFurl);
   126→        gl_Position = projectionMatrix * modelViewMatrix * vec4(p, 1.0);
   127→      }
   128→    `,
   129→    fragmentShader: /* glsl */`
   130→      varying vec2 vUv;
   131→      varying vec3 vN;
   132→      varying float vDmgU;
   133→      uniform vec3 uColor;
   134→      uniform vec3 uSunDir;
   135→      float hash12(vec2 p){ vec3 p3 = fract(vec3(p.xyx)*0.1031); p3+=dot(p3,p3.yzx+33.33); return fract((p3.x+p3.y)*p3.z); }
   136→      float vnoise(vec2 p){ vec2 i=floor(p), f=fract(p); vec2 u=f*f*(3.0-2.0*f);
   137→        return mix(mix(hash12(i),hash12(i+vec2(1,0)),u.x), mix(hash12(i+vec2(0,1)),hash12(i+vec2(1,1)),u.x), u.y); }
   138→      void main() {
   139→        vec3 col = uColor;
   140→        // швы (горизонтальные полотнища)
   141→        float seams = smoothstep(0.035, 0.0, abs(fract(vUv.y * 5.0) - 0.5) - 0.46);
   142→        col *= 1.0 - seams * 0.16;
   143→        // заплатки и рваные края от повреждений
   144→        float dmg = vDmgU;
   145→        float patches = step(0.55, vnoise(vUv * 4.7 + 7.3)) * dmg;
   146→        col = mix(col, col * 0.55, patches);
   147→        float holes = step(0.82 - dmg * 0.15, vnoise(vUv * 7.1 + 2.9)) * dmg * 1.4;
   148→        // простое освещение
   149→        float ndl = max(dot(normalize(vN), normalize(uSunDir)), 0.0) * 0.5 + 0.55;
   150→        col *= ndl;
   151→        float a = 1.0 - clamp(holes, 0.0, 0.9);
   152→        if (a < 0.55) discard; // дыры — без полупрозрачности (нет проблем сортировки)
   153→        gl_FragColor = vec4(col, 1.0);
   154→      }
   155→    `,
   156→    side: THREE.DoubleSide,
   157→  });
   158→}

   188→  const g = new THREE.PlaneGeometry(2.0, 1.2, 12, 5);

   192→    vertexShader: /* glsl */`
   193→      uniform float uTime;
   194→      varying vec2 vUv;
   195→      void main() {
   196→        vUv = uv;
   197→        vec3 p = position;
   198→        float w = sin(p.x * 3.0 - uTime * 7.0) * 0.12 * p.x + sin(p.x * 6.0 - uTime * 11.0) * 0.05 * p.x;
   199→        p.z += w; p.y += sin(p.x * 2.0 - uTime * 5.0) * 0.04 * p.x;
   200→        gl_Position = projectionMatrix * modelViewMatrix * vec4(p, 1.0);
   201→      }
   202→    `,
   203→    fragmentShader: /* glsl */`
   204→      uniform sampler2D uTex;
   205→      varying vec2 vUv;
   206→      void main() { gl_FragColor = texture2D(uTex, vUv); }
   207→    `,
   208→    side: THREE.DoubleSide,
   209→  });
   210→  const mesh = new THREE.Mesh(g, mat);
   211→  mesh.frustumCulled = false;
   212→  return { mesh, mat };

   215→// ---------- корпус по шпангоутам ----------
   216→export function hullProfile(t: number, P: ShipPreset) {
   217→  // t: 0 корма -> 1 нос. Полуширина, осадка, высота фальшборта
   218→  const B = P.beam / 2;
   219→  let w: number;
   220→  if (t < 0.45) w = THREE.MathUtils.lerp(0.58, 1.0, smooth01(t / 0.45));
   221→  else w = THREE.MathUtils.lerp(1.0, 0.0, Math.pow(smooth01((t - 0.45) / 0.55), 0.75));
   222→  const draft = P.draft * (0.72 + 0.28 * Math.sin(Math.PI * Math.min(1, t * 1.15)));
   223→  const bulwark = P.freeboard * (0.82 + 0.35 * Math.pow(Math.max(0, t - 0.75) / 0.25, 1.4) + 0.5 * Math.pow(1 - t, 2.2) * 0.6);
   224→  return { w: w * B, draft, bulwark };
   225→}
   226→function smooth01(x: number) { const t = Math.min(1, Math.max(0, x)); return t * t * (3 - 2 * t); }

   228→function buildHull(P: ShipPreset): { geometry: THREE.BufferGeometry; outline: THREE.Vector2[] } {
   229→  const S = 26, R = 8;
   230→  const L = P.length;
   231→  const pos: number[] = [];
   232→  const col: number[] = [];
   233→  const cHull = new THREE.Color(P.colors.hull);
   234→  const cBelow = new THREE.Color(0x33402a); // подводная часть — обрастание
   235→  const cBoot = new THREE.Color(0xd8d2c2); // пояс ватерлинии
   236→  const cWale = new THREE.Color(P.colors.trim);
   237→  const tmp = new THREE.Color();
   238→
   239→  const stations: { z: number; w: number; draft: number; bulwark: number }[] = [];
   240→  for (let s = 0; s <= S; s++) {
   241→    const t = s / S;
   242→    const p = hullProfile(t, P);
   243→    stations.push({ z: THREE.MathUtils.lerp(-L / 2, L / 2, t), w: p.w, draft: p.draft, bulwark: p.bulwark });
   244→  }
   245→
   246→  const pushPoint = (x: number, y: number, z: number) => {
   247→    pos.push(x, y, z);
   248→    // раскраска по высоте
   249→    if (y < -0.15) tmp.copy(cHull).lerp(cBelow, Math.min(1, (-y + 0.15) / P.draft));
   250→    else if (y < 0.42) tmp.copy(cBoot);
   251→    else if (y > 0.9 && y < 1.35) tmp.copy(cWale);
   252→    else tmp.copy(cHull);
   253→    // лёгкая полосатость обшивки
   254→    const stripe = 1.0 - 0.06 * (Math.sin(y * 9.0) * 0.5 + 0.5);
   255→    col.push(tmp.r * stripe, tmp.g * stripe, tmp.b * stripe);
   256→  };
   257→
   258→  for (const st of stations) {
   259→    for (let r = 0; r <= R; r++) {
   260→      const u = r / R;
   261→      const y = THREE.MathUtils.lerp(-st.draft, st.bulwark, u);
   262→      const flare = Math.pow(u, 0.6);
   263→      const halfW = st.w * (0.16 + 0.84 * flare);
   264→      pushPoint(halfW, y, st.z);
   265→      pushPoint(-halfW, y, st.z);
   266→    }
   267→  }
   268→  const idx: number[] = [];
   269→  const rowLen = (R + 1) * 2;
   270→  for (let s = 0; s < S; s++) {
   271→    for (let r = 0; r < R; r++) {
   272→      const a = s * rowLen + r * 2, b = a + 1, c = a + 2, d = a + 3;
   273→      const e = a + rowLen, f = e + 1, g = e + 2, h = e + 3;
   274→      idx.push(a, c, e, c, g, e);
   275→      idx.push(b, e + 1, d, b, f, d); // зеркальная сторона (обратный порядок)
   276→    }
   277→  }
   278→  // транцевая корма: веер
   279→  const st0 = stations[0];
   280→  const centerIdx = pos.length / 3;
   281→  pushPoint(0, -st0.draft * 0.55, st0.z - 0.01);
   282→  for (let r = 0; r <= R; r++) {
   283→    const i0 = r * 2;
   284→    if (r < R) {
   285→      idx.push(centerIdx, i0 + 1, i0); // ориентация наружу (-Z)
   286→    }
   287→  }
   288→  // днище: замыкаем корпус снизу (r=0 — нижняя кромка шпангоутов),
   289→  // нормали вниз — при крене/под водой не «просвечивает»
   290→  for (let s = 0; s < S; s++) {
   291→    const a = s * rowLen + 1;        // левый борт, низ
   292→    const b = s * rowLen;            // правый борт, низ
   293→    const c = (s + 1) * rowLen + 1;
   294→    const d = (s + 1) * rowLen;
   295→    idx.push(a, b, c, b, d, c);
   296→  }
   297→  const g = new THREE.BufferGeometry();
   298→  g.setAttribute('position', new THREE.Float32BufferAttribute(pos, 3));
   299→  g.setAttribute('color', new THREE.Float32BufferAttribute(col, 3));
   300→  g.setIndex(idx);
   301→  g.computeVertexNormals();
   302→
   303→  // контур ватерлинии
   304→  const outline: THREE.Vector2[] = [];
   305→  for (let s = 0; s <= S; s += 2) {
   306→    const st = stations[s];
   307→    const u = (st.draft) / (st.draft + st.bulwark); // u при y=0
   308→    const flare = Math.pow(u, 0.6);
   309→    const halfW = Math.max(0.25, st.w * (0.16 + 0.84 * flare));
   310→    outline.push(new THREE.Vector2(halfW, st.z));
   311→  }
   312→  for (let s = S; s >= 0; s -= 2) {
   313→    const st = stations[s];
   314→    const u = st.draft / (st.draft + st.bulwark);
   315→    const flare = Math.pow(u, 0.6);
   316→    const halfW = Math.max(0.25, st.w * (0.16 + 0.84 * flare));
   317→    outline.push(new THREE.Vector2(-halfW, st.z));
   318→  }
   319→  return { geometry: g, outline };
   320→}

   322→function buildDeck(P: ShipPreset): THREE.BufferGeometry {
   323→  // палуба: полоса по длине чуть ниже фальшборта, вплотную к обшивке.
   324→  // ВАЖНО: порядок вершин (a,b,c)(b,d,c) даёт нормали ВВЕРХ —
   325→  // иначе палуба отсекается backface-куллингом и «просвечивает».
   326→  // UV: u — поперёк (0..1 = полная ширина), v — вдоль (тайл = ширина),
   327→  // доски идут вдоль корпуса, у краёв — тёмный водосток (в текстуре).
   328→  const S = 20, L = P.length;
   329→  const pos: number[] = [], uv: number[] = [], idx: number[] = [];
   330→  for (let s = 0; s <= S; s++) {
   331→    const t = s / S;
   332→    const p = hullProfile(t, P);
   333→    const z = THREE.MathUtils.lerp(-L / 2, L / 2, t);
   334→    const deckY = p.bulwark - 1.0;
   335→    // полуширина корпуса на высоте палубы — та же формула, что в buildHull,
   336→    // чтобы палуба доходила до борта без щели
   337→    const u = Math.min(1, Math.max(0, (deckY + p.draft) / (p.draft + p.bulwark)));
   338→    const flare = Math.pow(u, 0.6);
   339→    const w = Math.max(0.15, p.w * (0.16 + 0.84 * flare) - 0.05);
   340→    pos.push(w, deckY, z, -w, deckY, z);
   341→    const vv = z / P.beam;
   342→    uv.push(0.5 + w / P.beam, vv, 0.5 - w / P.beam, vv);
   343→  }
   344→  for (let s = 0; s < S; s++) {
   345→    const a = s * 2, b = a + 1, c = a + 2, d = a + 3;
   346→    idx.push(a, b, c, b, d, c);
   347→  }
   348→  const g = new THREE.BufferGeometry();
   349→  g.setAttribute('position', new THREE.Float32BufferAttribute(pos, 3));
   350→  g.setAttribute('uv', new THREE.Float32BufferAttribute(uv, 2));
   351→  g.setIndex(idx);
   352→  g.computeVertexNormals();
   353→  return g;
   354→}

   356→/** Текстура палубных досок: конопаченые швы, разбежка стыков, гвозди,
   357→ *  волокна, потёртости; тёмные водосточные планки по краям (u≈0 и u≈1) */
   358→function makeDeckTexture(baseColor: number, seed: number): THREE.Texture {
   359→  const S = 512;
   360→  const c = document.createElement('canvas');
   361→  c.width = S; c.height = S;
   362→  const x = c.getContext('2d')!;
   363→  const base = new THREE.Color(baseColor);
   364→  let rs = seed * 2654435761 % 2147483647;
   365→  const rnd = () => { rs = (rs * 16807) % 2147483647; return rs / 2147483647; };
   366→  const N = 30, pw = S / N;
   367→  for (let i = 0; i < N; i++) {
   368→    const v = 0.84 + rnd() * 0.26;
   369→    x.fillStyle = `rgb(${Math.round(base.r * 255 * v)},${Math.round(base.g * 255 * v)},${Math.round(base.b * 255 * v)})`;
   370→    x.fillRect(i * pw, 0, pw + 1, S);
   371→    // волокна дерева
   372→    x.strokeStyle = 'rgba(30,18,8,0.10)';
   373→    x.lineWidth = 1;
   374→    for (let k = 0; k < 3; k++) {
   375→      const gx = i * pw + pw * (0.2 + 0.6 * rnd());
   376→      x.beginPath();
   377→      x.moveTo(gx, 0);
   378→      for (let y = 0; y <= S; y += 32) x.lineTo(gx + Math.sin(y * 0.045 + i * 7 + k * 3) * 2.2, y);
   379→      x.stroke();
   380→    }
   381→    // стыки досок с разбежкой + гвозди
   382→    const b1 = (i * 137 + 61) % S, b2 = (b1 + Math.round(S * 0.55)) % S;
   383→    for (const by of [b1, b2]) {
   384→      x.fillStyle = 'rgba(22,14,7,0.6)';
   385→      x.fillRect(i * pw, by, pw + 1, 2);
   386→      x.fillStyle = 'rgba(32,32,36,0.85)';
   387→      x.beginPath(); x.arc(i * pw + pw * 0.3, by + 5, 1.4, 0, 7); x.fill();
   388→      x.beginPath(); x.arc(i * pw + pw * 0.7, by - 3, 1.4, 0, 7); x.fill();
   389→    }
   390→    // конопаченый шов между досками
   391→    x.fillStyle = 'rgba(20,13,7,0.72)';
   392→    x.fillRect(i * pw, 0, 1.6, S);
   393→  }
   394→  // водосточный жёлоб по краям палубы (тёмные продольные планки)
   395→  for (const [x0, wW] of [[0, S * 0.075], [S * 0.925, S * 0.075]] as [number, number][]) {
   396→    const gr = x.createLinearGradient(x0, 0, x0 + wW, 0);
   397→    gr.addColorStop(0, 'rgba(24,15,7,0.5)');
   398→    gr.addColorStop(1, 'rgba(24,15,7,0.12)');
   399→    x.fillStyle = gr;
   400→    x.fillRect(x0, 0, wW, S);
   401→  }
   402→  // потёртости и разводы
   403→  for (let k = 0; k < 26; k++) {
   404→    const rx = rnd() * S, ry = rnd() * S, rr = 18 + rnd() * 55;
   405→    const rg = x.createRadialGradient(rx, ry, 0, rx, ry, rr);
   406→    rg.addColorStop(0, 'rgba(28,18,8,0.09)');
   407→    rg.addColorStop(1, 'rgba(0,0,0,0)');
   408→    x.fillStyle = rg;
   409→    x.fillRect(rx - rr, ry - rr, rr * 2, rr * 2);
   410→  }
   411→  const t = new THREE.CanvasTexture(c);
   412→  t.wrapS = t.wrapT = THREE.RepeatWrapping;
   413→  t.colorSpace = THREE.SRGBColorSpace;
   414→  t.anisotropy = 4;
   415→  return t;
   416→}

   418→/** Планширь: планка, закрывающая верхнюю кромку фальшборта */
   419→function buildCapRail(P: ShipPreset): THREE.BufferGeometry {
   420→  const S = 26, L = P.length;
   421→  const pos: number[] = [], col: number[] = [], idx: number[] = [];
   422→  const cRail = new THREE.Color(P.colors.trim).multiplyScalar(1.25);
   423→  const cRail2 = new THREE.Color(P.colors.trim);
   424→  const push = (px: number, py: number, z: number, c: THREE.Color) => {
   425→    pos.push(px, py, z); col.push(c.r, c.g, c.b);
   426→  };
   427→  for (let s = 0; s <= S; s++) {
   428→    const t = s / S;
   429→    const p = hullProfile(t, P);
   430→    const z = THREE.MathUtils.lerp(-L / 2, L / 2, t);
   431→    const y = p.bulwark;
   432→    // внутренняя и внешняя кромки планки (лёгкий свес наружу)
   433→    push(p.w - 0.26, y + 0.07, z, cRail);
   434→    push(p.w + 0.13, y + 0.015, z, cRail2);
   435→    push(-p.w + 0.26, y + 0.07, z, cRail);
   436→    push(-p.w - 0.13, y + 0.015, z, cRail2);
   437→  }
   438→  for (let s = 0; s < S; s++) {
   439→    const a = s * 4;
   440→    // правый борт
   441→    idx.push(a, a + 4, a + 1, a + 1, a + 4, a + 5);
   442→    // левый борт
   443→    idx.push(a + 2, a + 3, a + 6, a + 3, a + 7, a + 6);
   444→  }
   445→  const g = new THREE.BufferGeometry();
   446→  g.setAttribute('position', new THREE.Float32BufferAttribute(pos, 3));
   447→  g.setAttribute('color', new THREE.Float32BufferAttribute(col, 3));
   448→  g.setIndex(idx);
   449→  g.computeVertexNormals();
   450→  return g;
   451→}

   453→/** Запечь геометрию в общие массивы (для слияния реквизита в один меш) */
   454→function bakeInto(target: { pos: number[]; col: number[] }, geo: THREE.BufferGeometry,
   455→  color: THREE.Color, x: number, y: number, z: number, euler?: THREE.Euler) {
   456→  const g = geo.index ? geo.toNonIndexed() : geo;
   457→  const p = g.attributes.position as THREE.BufferAttribute;
   458→  const v = new THREE.Vector3();
   459→  for (let i = 0; i < p.count; i++) {
   460→    v.fromBufferAttribute(p, i);
   461→    if (euler) v.applyEuler(euler);
   462→    target.pos.push(v.x + x, v.y + y, v.z + z);
   463→    target.col.push(color.r, color.g, color.b);
   464→  }
   465→}

   467→/** Палубный реквизит: решётчатый люк, бочки, канаты, кнехты, колокол,
   468→ *  ядра, сундук, ведро — всё сливается в ОДИН меш с vertex colors */
   469→function buildDeckProps(P: ShipPreset): THREE.BufferGeometry {
   470→  const L = P.length;
   471→  const t: { pos: number[]; col: number[] } = { pos: [], col: [] };
   472→  const deckYAt = (tt: number) => hullProfile(tt, P).bulwark - 1.0;
   473→  const cWood = new THREE.Color(0x7a5c38);
   474→  const cWoodD = new THREE.Color(0x4a3420);
   475→  const cDark = new THREE.Color(P.colors.trim);
   476→  const cIron = new THREE.Color(0x24262b);
   477→  const cBrass = new THREE.Color(0xb8912f);
   478→  const cRope = new THREE.Color(0xa08656);
   479→  const cBarrel = new THREE.Color(0x8a6a42);
   480→
   481→  // --- решётчатый люк (гратинг) с коомингом ---
   482→  const hx = L * 0.08, hz = L * 0.05; // пол-размера (между мачтами)
   483→  const hy = deckYAt(0.48);
   484→  // кооминг (рамка)
   485→  bakeInto(t, new THREE.BoxGeometry(hx * 2 + 0.24, 0.22, 0.12), cWoodD, 0, hy + 0.11, hz);
   486→  bakeInto(t, new THREE.BoxGeometry(hx * 2 + 0.24, 0.22, 0.12), cWoodD, 0, hy + 0.11, -hz);
   487→  bakeInto(t, new THREE.BoxGeometry(0.12, 0.22, hz * 2), cWoodD, hx, hy + 0.11, 0);
   488→  bakeInto(t, new THREE.BoxGeometry(0.12, 0.22, hz * 2), cWoodD, -hx, hy + 0.11, 0);
   489→  // продольные планки
   490→  const nL = 9;
   491→  for (let i = 0; i < nL; i++) {
   492→    const px = -hx + 0.12 + (i / (nL - 1)) * (hx * 2 - 0.24);
   493→    bakeInto(t, new THREE.BoxGeometry(0.09, 0.07, hz * 2 - 0.1), cWood, px, hy + 0.24, 0);
   494→  }
   495→  // поперечные планки
   496→  const nT = 5;
   497→  for (let i = 0; i < nT; i++) {
   498→    const pz = -hz + 0.1 + (i / (nT - 1)) * (hz * 2 - 0.2);
   499→    bakeInto(t, new THREE.BoxGeometry(hx * 2 - 0.16, 0.07, 0.09), cWood, 0, hy + 0.30, pz);
   500→  }
   501→
   502→  // --- бочки (2 стоя, 1 лёжа) ---
   503→  const barrelG = new THREE.CylinderGeometry(0.34, 0.34, 0.8, 10);
   504→  const hoopG = new THREE.TorusGeometry(0.345, 0.022, 5, 12);
   505→  const hoopE = new THREE.Euler(Math.PI / 2, 0, 0);
   506→  const mkBarrel = (x: number, z: number, lying = false) => {
   507→    const y = deckYAt(0.5 + z / L) + (lying ? 0.34 : 0.4);
   508→    const e = lying ? new THREE.Euler(0, lying ? 0.7 : 0, Math.PI / 2) : undefined;
   509→    bakeInto(t, barrelG, cBarrel, x, y, z, e);
   510→    if (lying) {
   511→      bakeInto(t, hoopG, cIron, x, y - 0.25, z, e);
   512→      bakeInto(t, hoopG, cIron, x, y + 0.25, z, e);
   513→    } else {
   514→      bakeInto(t, hoopG, cIron, x, y - 0.28, z, hoopE);
   515→      bakeInto(t, hoopG, cIron, x, y + 0.28, z, hoopE);
   516→    }
   517→  };
   518→  mkBarrel(P.beam * 0.24, L * 0.10);
   519→  mkBarrel(-P.beam * 0.26, L * 0.04);
   520→  mkBarrel(P.beam * 0.22, -L * 0.12, true);
   521→  // --- бухты канатов ---
   522→  const coilG = new THREE.TorusGeometry(0.30, 0.10, 6, 14);
   523→  const coilE = new THREE.Euler(Math.PI / 2, 0, 0);
   524→  bakeInto(t, coilG, cRope, P.beam * 0.20, deckYAt(0.75) + 0.10, L * 0.30, coilE);
   525→  bakeInto(t, coilG, cRope, -P.beam * 0.18, deckYAt(0.62) + 0.10, L * 0.18, coilE);
   526→  bakeInto(t, coilG, cRope, P.beam * 0.34, deckYAt(0.36) + 0.10, -L * 0.15, coilE);
   527→
   528→  // --- кнехты (битенги) с поперечиной у носа ---
   529→  const bittG = new THREE.BoxGeometry(0.15, 0.6, 0.15);
   530→  for (const side of [-1, 1]) {
   531→    bakeInto(t, bittG, cWoodD, side * 0.42, deckYAt(0.85) + 0.3, L * 0.36);
   532→    bakeInto(t, bittG, cWoodD, side * 0.42, deckYAt(0.85) + 0.3, L * 0.36 - 0.5);
   533→    bakeInto(t, new THREE.BoxGeometry(0.13, 0.13, 0.75), cWoodD, side * 0.42, deckYAt(0.85) + 0.55, L * 0.36 - 0.25);
   534→  }
   535→
   536→  // --- корабельный колокол у штурвала ---
   537→  const bellZ = THREE.MathUtils.lerp(-L / 2, L / 2, 0.2) + 1.6;
   538→  bakeInto(t, new THREE.BoxGeometry(0.12, 0.8, 0.12), cWoodD, 0.8, deckYAt(0.22) + 0.4, bellZ);
   539→  bakeInto(t, new THREE.CylinderGeometry(0.05, 0.13, 0.2, 8), cBrass, 0.8, deckYAt(0.22) + 0.85, bellZ);
   540→
   541→  // --- пирамиды ядер у мачты ---
   542→  const ballG = new THREE.SphereGeometry(0.105, 7, 6);
   543→  const pile = (x: number, z: number) => {
   544→    const y0 = deckYAt(0.5 + z / L);
   545→    for (const [dx, dz] of [[-0.115, 0], [0, 0], [0.115, 0], [-0.058, 0.1], [0.058, 0.1], [0, 0.05]]) {
   546→      bakeInto(t, ballG, cIron, x + dx, y0 + 0.105, z + dz);
   547→    }
   548→  };
   549→  pile(0.85, -L * 0.16);
   550→  pile(-P.beam * 0.20, L * 0.09);
   551→
   552→  // --- сундук с инструментом ---
   553→  bakeInto(t, new THREE.BoxGeometry(0.72, 0.42, 0.5), cWoodD, -P.beam * 0.26, deckYAt(0.42) + 0.21, -L * 0.2);
   554→  bakeInto(t, new THREE.BoxGeometry(0.78, 0.1, 0.56), cDark, -P.beam * 0.26, deckYAt(0.42) + 0.46, -L * 0.2);
   555→
   556→  // --- ведро ---
   557→  bakeInto(t, new THREE.CylinderGeometry(0.17, 0.13, 0.3, 8, 1, true), cWood, -P.beam * 0.2, deckYAt(0.72) + 0.15, L * 0.24);
   558→
   559→  const g = new THREE.BufferGeometry();
   560→  g.setAttribute('position', new THREE.Float32BufferAttribute(t.pos, 3));
   561→  g.setAttribute('color', new THREE.Float32BufferAttribute(t.col, 3));
   562→  g.computeVertexNormals();
   563→  return g;
   564→}

   566→// ---------- такелаж (линии) ----------
   567→function buildRigging(P: ShipPreset, masts: { z: number; h: number }[], bowspritTip: THREE.Vector3): THREE.LineSegments {
   568→  const pts: number[] = [];
   569→  const L = P.length;
   570→  const addLine = (a: THREE.Vector3, b: THREE.Vector3) => { pts.push(a.x, a.y, a.z, b.x, b.y, b.z); };
   571→  for (const m of masts) {
   572→    const top = new THREE.Vector3(0, m.h, m.z);
   573→    // штаги к носу и корме
   574→    addLine(top, bowspritTip);
   575→    addLine(top, new THREE.Vector3(0, 1.2, -L / 2 + 0.4));
   576→    // ванты к бортам
   577→    for (const side of [-1, 1]) {
   578→      for (let k = 0; k < 3; k++) {
   579→        const zz = m.z - 1.5 + k * 1.5;
   580→        addLine(new THREE.Vector3(side * 2.6, 1.6, zz), new THREE.Vector3(0, m.h * (0.55 + k * 0.15), m.z));
   581→      }
   582→    }
   583→  }
   584→  // бакштаги к корме от грот-мачты
   585→  if (masts.length >= 2) {
   586→    const main = masts[masts.length - 1 > 1 ? masts.length - 2 : 0];
   587→    addLine(new THREE.Vector3(0, main.h * 0.8, main.z), new THREE.Vector3(2.4, 1.4, -L / 2 + 0.3));
   588→    addLine(new THREE.Vector3(0, main.h * 0.8, main.z), new THREE.Vector3(-2.4, 1.4, -L / 2 + 0.3));
   589→  }
   590→  const g = new THREE.BufferGeometry();
   591→  g.setAttribute('position', new THREE.Float32BufferAttribute(pts, 3));
   592→  const mat = new THREE.LineBasicMaterial({ color: 0x241c12, transparent: true, opacity: 0.85 });
   593→  return new THREE.LineSegments(g, mat);
   594→}

   596→// ---------- полная сборка ----------
   597→export function buildShip(P: ShipPreset): ShipVisual {
   598→  const group = new THREE.Group();
   599→  const L = P.length;
   600→
   601→  // корпус
   602→  const { geometry: hullGeo, outline } = buildHull(P);
   603→  const hullMat = new THREE.MeshStandardMaterial({ vertexColors: true, roughness: 0.82, metalness: 0.05 });
   604→  const hull = new THREE.Mesh(hullGeo, hullMat);
   605→  hull.castShadow = true;
   606→  group.add(hull);
   607→
   608→  // палуба: текстура досок с конопачением (DoubleSide — страховка от куллинга)
   609→  const deckMat = new THREE.MeshStandardMaterial({
   610→    map: makeDeckTexture(P.colors.deck, P.colors.deck + P.length),
   611→    roughness: 0.93, side: THREE.DoubleSide,
   612→  });
   613→  const deck = new THREE.Mesh(buildDeck(P), deckMat);
   614→  group.add(deck);
   615→
   616→  // планширь по верхней кромке фальшборта
   617→  const capRail = new THREE.Mesh(buildCapRail(P),
   618→    new THREE.MeshStandardMaterial({ vertexColors: true, roughness: 0.7, metalness: 0.05 }));
   619→  group.add(capRail);
   620→
   621→  const woodMat = new THREE.MeshStandardMaterial({ color: 0x6e5433, roughness: 0.8 });
   622→  const darkMat = new THREE.MeshStandardMaterial({ color: P.colors.trim, roughness: 0.75 });
   623→
   624→  // бушприт
   625→  const bowsprit = new THREE.Mesh(new THREE.CylinderGeometry(0.09, 0.16, L * 0.32, 6), woodMat);
   626→  bowsprit.rotation.x = Math.PI / 2 - 0.18;
   627→  bowsprit.position.set(0, P.freeboard * 0.75, L / 2 + L * 0.12);
   628→  group.add(bowsprit);
   629→  const bowspritTip = new THREE.Vector3(0, P.freeboard * 0.75 + L * 0.32 * 0.18, L / 2 + L * 0.12 + L * 0.32 * Math.cos(0.18) * 0.98);
   630→
   631→  // мачты
   632→  const mastDefs: { z: number; h: number }[] = [];
   633→  const H = P.freeboard + L * 0.78;
   634→  if (P.masts === 2) {
   635→    mastDefs.push({ z: L * 0.16, h: H * 0.88 });
   636→    mastDefs.push({ z: -L * 0.10, h: H });
   637→  } else {
   638→    mastDefs.push({ z: L * 0.18, h: H * 0.8 });
   639→    mastDefs.push({ z: -L * 0.05, h: H });
   640→    mastDefs.push({ z: -L * 0.28, h: H * 0.62 });
   641→  }
   642→  const masts: THREE.Object3D[] = [];
   643→  for (const md of mastDefs) {
   644→    const mast = new THREE.Group();
   645→    const pole = new THREE.Mesh(new THREE.CylinderGeometry(0.11, 0.24, md.h, 7), woodMat);
   646→    pole.position.y = md.h / 2 - P.draft * 0.3;
   647→    pole.castShadow = true;
   648→    mast.add(pole);
   649→    // марс (платформа)
   650→    const top = new THREE.Mesh(new THREE.CylinderGeometry(0.75, 0.6, 0.28, 8), darkMat);
   651→    top.position.y = md.h * 0.72 - P.draft * 0.3;
   652→    mast.add(top);
   653→    mast.position.set(0, -P.draft * 0.3 > 0 ? 0 : 0, md.z);
   654→    mast.userData.height = md.h;
   655→    masts.push(mast);
   656→    group.add(mast);
   657→  }
   658→
   659→  // паруса: каждый со своим материалом (своё рифление), квадратные —
   660→  // на поворотной группе «рея+парус» для брасопки по ветру
   661→  const sails: SailInfo[] = [];
   662→  const addSquareSail = (mastIndex: number, level: number, width: number, height: number, setAt: number) => {
   663→    const md = mastDefs[mastIndex];
   664→    const g = new THREE.PlaneGeometry(width, height, 10, 7);
   665→    g.translate(0, -height / 2, 0); // верхняя кромка у реи
   666→    const mat = sailMaterial(P.colors.sail);
   667→    const mesh = new THREE.Mesh(g, mat);
   668→    const y = md.h * level - P.draft * 0.3;
   669→    // поворотная группа: ось вращения — мачта (брасопка рея по ветру)
   670→    // рея+парус — потомок МАЧТЫ: сломанная мачта валится вместе с ними
   671→    const pivot = new THREE.Group();
   672→    pivot.position.set(0, y + 0.05, 0);
   673→    // рея
   674→    const yard = new THREE.Mesh(new THREE.CylinderGeometry(0.07, 0.07, width * 1.04, 5), woodMat);
   675→    yard.rotation.z = Math.PI / 2;
   676→    pivot.add(yard);
   677→    mesh.position.set(0, 0, -0.25);
   678→    pivot.add(mesh);
   679→    masts[mastIndex].add(pivot);
   680→    sails.push({ mesh, mat, brace: pivot, setAt, furl: 1, square: true, braceAngle: 0, mastIndex });
   681→  };
   682→  if (P.masts === 2) {
   683→    addSquareSail(0, 0.66, L * 0.3, L * 0.16, 0.62);   // фок
   684→    addSquareSail(1, 0.72, L * 0.36, L * 0.2, 0.45);   // грот нижний — ставится первым
   685→    addSquareSail(1, 0.45, L * 0.26, L * 0.14, 0.92);  // грот верхний (брамсель)
   686→  } else {
   687→    addSquareSail(0, 0.62, L * 0.24, L * 0.13, 0.60);
   688→    addSquareSail(1, 0.7, L * 0.34, L * 0.18, 0.45);
   689→    addSquareSail(1, 0.44, L * 0.24, L * 0.13, 0.92);
   690→    addSquareSail(2, 0.5, L * 0.18, L * 0.11, 0.78);
   691→  }
   692→
   693→  // кливера (треугольные) от бушприта к фок-мачте: сетка с uv —
   694→  // есть «пузо» и трепет; при рифлении собираются к голове
   695→  const addJib = (tip: THREE.Vector3, foot: THREE.Vector3, headY: number, mastZ: number, setAt: number) => {
   696→    const head = new THREE.Vector3(0, headY, mastZ - 0.3);
   697→    // локальная система: X = −мирZ (продольная), Y = мирY, Z = мирX (пузо вбок)
   698→    const a = { x: -(tip.z - head.z), y: tip.y - head.y };
   699→    const b = { x: -(foot.z - head.z), y: foot.y - head.y };
   700→    const NU = 8, NV = 5;
   701→    const pos: number[] = [], uvs: number[] = [], idx: number[] = [];
   702→    for (let i = 0; i <= NU; i++) {
   703→      const u = i / NU;
   704→      // точка на штаге (к галсовому углу) и на шкаторине
   705→      const lx = a.x * u, ly = a.y * u;
   706→      const rx = b.x * u, ry = b.y * u;
   707→      for (let j = 0; j <= NV; j++) {
   708→        const v = j / NV;
   709→        pos.push(THREE.MathUtils.lerp(lx, rx, v), THREE.MathUtils.lerp(ly, ry, v), 0);
   710→        uvs.push(u, v);
   711→      }
   712→    }
   713→    const row = NV + 1;
   714→    for (let i = 0; i < NU; i++) {
   715→      for (let j = 0; j < NV; j++) {
   716→        const p0 = i * row + j, p1 = p0 + 1, p2 = p0 + row, p3 = p2 + 1;
   717→        idx.push(p0, p1, p2, p1, p3, p2);
   718→      }
   719→    }
   720→    const g = new THREE.BufferGeometry();
   721→    g.setAttribute('position', new THREE.Float32BufferAttribute(pos, 3));
   722→    g.setAttribute('uv', new THREE.Float32BufferAttribute(uvs, 2));
   723→    g.setIndex(idx);
   724→    g.computeVertexNormals();
   725→    const mat = sailMaterial(P.colors.sail);
   726→    const mesh = new THREE.Mesh(g, mat);
   727→    mesh.position.set(head.x, head.y, head.z - mastZ);
   728→    mesh.rotation.y = Math.PI / 2; // локальный Z (пузо) -> мир X (борт)
   729→    mesh.frustumCulled = false;
   730→    masts[0].add(mesh); // потомок фок-мачты — падает вместе с ней
   731→    sails.push({ mesh, mat, setAt, furl: 1, square: false, braceAngle: 0, mastIndex: 0 });
   732→  };
   733→  const foreTop = mastDefs[0];
   734→  addJib(bowspritTip, new THREE.Vector3(0, P.freeboard * 0.72, L / 2 - 1.0), foreTop.h * 0.62 - P.draft * 0.3, foreTop.z, 0.72);
   735→  addJib(new THREE.Vector3(bowspritTip.x * 0.55, bowspritTip.y - (bowspritTip.y - P.freeboard * 0.72) * 0.45, bowspritTip.z - L * 0.1),
   736→    new THREE.Vector3(0, P.freeboard * 0.72, L / 2 - 2.0), foreTop.h * 0.78 - P.draft * 0.3, foreTop.z, 1.0);
   737→
   738→  // флаг на топе грот-мачты (потомок мачты — валится вместе с ней)
   739→  const mainMastIdx = 1; // грот-мачта и у 2-, и у 3-мачтовых
   740→  const mainMast = mastDefs[mainMastIdx];
   741→  const { mesh: flag, mat: flagMat } = flagMesh(P.flagKind);
   742→  flag.position.set(0.05, mainMast.h - P.draft * 0.3 - 0.4, 0);
   743→  masts[mainMastIdx].add(flag);
   744→
   745→  // пушки по бортам
   746→  const gunsPort: THREE.Object3D[] = [];
   747→  const gunsStarboard: THREE.Object3D[] = [];
   748→  const gunZ0 = -L * 0.22, gunZ1 = L * 0.24;
   749→  for (let i = 0; i < P.gunsPerSide; i++) {
   750→    const z = THREE.MathUtils.lerp(gunZ0, gunZ1, P.gunsPerSide === 1 ? 0.5 : i / (P.gunsPerSide - 1));
   751→    const p = hullProfile((z + L / 2) / L, P);
   752→    const deckY = p.bulwark - 0.72;
   753→    for (const side of [-1, 1]) {
   754→      const gun = new THREE.Group();
   755→      const barrel = new THREE.Mesh(new THREE.CylinderGeometry(0.11, 0.15, 1.15, 8),
   756→        new THREE.MeshStandardMaterial({ color: 0x1f1f22, roughness: 0.5, metalness: 0.75 }));
   757→      barrel.rotation.z = Math.PI / 2;
   758→      barrel.position.set(side * 0.62, 0.08, 0);
   759→      gun.add(barrel);
   760→      const carriage = new THREE.Mesh(new THREE.BoxGeometry(0.5, 0.34, 0.55), darkMat);
   761→      carriage.position.set(0, 0.0, 0);
   762→      gun.add(carriage);
   763→      gun.position.set(side * (p.w * 0.9 - 0.15), deckY, z);
   764→      gun.userData.side = side;
   765→      group.add(gun);
   766→      // локальный −X = штирборт (правый борт), локальный +X = порт (левый)
   767→      (side < 0 ? gunsStarboard : gunsPort).push(gun);
   768→    }
   769→  }
   770→
   771→  // штурвал: пьедестал (не вращается) + колесо, начало координат
   772→  // вращающейся группы — в ступице, чтобы крутился вокруг СВОЕЙ оси
   773→  const wheelAsm = new THREE.Group();
   774→  const wheel = new THREE.Group();
   775→  const wheelR = 0.55;
   776→  const hubY = 0.98; // высота оси над палубой
   777→  const wheelTorus = new THREE.Mesh(new THREE.TorusGeometry(wheelR, 0.045, 6, 18), darkMat);
   778→  wheel.add(wheelTorus);
   779→  for (let i = 0; i < 4; i++) {
   780→    const ang = (i / 4) * Math.PI;
   781→    const spoke = new THREE.Mesh(new THREE.CylinderGeometry(0.028, 0.028, wheelR * 2.3, 4), darkMat);
   782→    spoke.rotation.z = ang;
   783→    wheel.add(spoke);
   784→    // рукоятки наружу, к рулевому (в сторону кормы, -Z)
   785→    const handle = new THREE.Mesh(new THREE.CylinderGeometry(0.035, 0.035, 0.26, 5), woodMat);
   786→    handle.rotation.x = Math.PI / 2;
   787→    handle.position.set(-Math.sin(ang) * (wheelR + 0.02), Math.cos(ang) * (wheelR + 0.02), -0.14);
   788→    wheel.add(handle);
   789→  }
   790→  const wheelHub = new THREE.Mesh(new THREE.CylinderGeometry(0.1, 0.1, 0.34, 8), woodMat);
   791→  wheelHub.rotation.x = Math.PI / 2;
   792→  wheel.add(wheelHub);
   793→  wheel.position.set(0, hubY, 0);
   794→  wheelAsm.add(wheel);
   795→  // стойка, подкосы и ось — вне вращающейся группы
   796→  const post = new THREE.Mesh(new THREE.BoxGeometry(0.14, hubY, 0.12), darkMat);
   797→  post.position.set(0, hubY / 2, 0);
   798→  wheelAsm.add(post);
   799→  for (const sgn of [-1, 1]) {
   800→    const brace = new THREE.Mesh(new THREE.BoxGeometry(0.08, hubY * 1.2, 0.08), darkMat);
   801→    brace.position.set(sgn * 0.24, hubY * 0.5, 0);
   802→    brace.rotation.z = sgn * 0.55;
   803→    wheelAsm.add(brace);
   804→  }
   805→  const wheelAxle = new THREE.Mesh(new THREE.CylinderGeometry(0.04, 0.04, 0.55, 6), darkMat);
   806→  wheelAxle.rotation.x = Math.PI / 2;
   807→  wheelAxle.position.set(0, hubY, 0.12);
   808→  wheelAsm.add(wheelAxle);
   809→  const wheelT = 0.2;
   810→  wheelAsm.position.set(0, hullProfile(wheelT, P).bulwark - 1.0, THREE.MathUtils.lerp(-L / 2, L / 2, wheelT));
   811→  group.add(wheelAsm);
   812→
   813→  // палубный реквизит: гратинг-люк, бочки, канаты, кнехты, колокол, ядра,
   814→  // сундук, ведро — одним мешом
   815→  const props = new THREE.Mesh(buildDeckProps(P),
   816→    new THREE.MeshStandardMaterial({ vertexColors: true, roughness: 0.85, side: THREE.DoubleSide }));
   817→  props.castShadow = true;
   818→  group.add(props);
   819→
   820→  // шпиль, фонарь, якоря — строго на палубе
   821→  const deckYAt = (t: number) => hullProfile(t, P).bulwark - 1.0;
   822→  const capstan = new THREE.Mesh(new THREE.CylinderGeometry(0.4, 0.5, 0.7, 8), woodMat);
   823→  capstan.position.set(0, deckYAt(0.36) + 0.35, -L * 0.16);
   824→  group.add(capstan);
   825→  const sternLantern = new THREE.Mesh(new THREE.SphereGeometry(0.22, 8, 8),
   826→    new THREE.MeshStandardMaterial({ color: 0xffd88a, emissive: 0xffaa33, emissiveIntensity: 1.6 }));
   827→  sternLantern.position.set(0, P.freeboard * 1.15, -L / 2 - 0.15);
   828→  group.add(sternLantern);
   829→  let lantern: THREE.PointLight | undefined;
   830→  // фигура на носу
   831→  const figure = new THREE.Mesh(new THREE.ConeGeometry(0.22, 0.8, 6),
   832→    new THREE.MeshStandardMaterial({ color: 0xc9a24a, roughness: 0.4, metalness: 0.6 }));
   833→  figure.position.set(0, P.freeboard * 1.12, L / 2 - 0.2);
   834→  figure.rotation.x = -0.5;
   835→  group.add(figure);
   836→  // якоря
   837→  for (const side of [-1, 1]) {
   838→    const anchor = new THREE.Mesh(new THREE.BoxGeometry(0.16, 0.9, 0.3), darkMat);
   839→    anchor.position.set(side * P.beam * 0.42, P.freeboard * 0.3, L / 2 - L * 0.06);
   840→    anchor.rotation.z = side * 0.35;
   841→    group.add(anchor);
   842→  }
   843→
   844→  // такелаж
   845→  group.add(buildRigging(P, mastDefs, bowspritTip));
   846→
   847→  const rig: SailRig = { sails };
   848→
   849→  return {
   850→    group, sails: rig, gunsPort, gunsStarboard, flag, flagMat,
   851→    dims: { length: L, beam: P.beam, draft: P.draft }, outline,
   852→    masts, lantern, wheel,
   853→  };
   854→}

   856→/** Включить фонарь (ночь) — только для игрока, ради перфа */
   857→export function enableLantern(vis: ShipVisual) {
   858→  if (vis.lantern) return;
   859→  vis.lantern = new THREE.PointLight(0xffa644, 6, 26, 1.6);
   860→  vis.lantern.position.set(0, vis.dims.length * 0.06, -vis.dims.length / 2 - 0.4);
   861→  vis.group.add(vis.lantern);
   862→}
