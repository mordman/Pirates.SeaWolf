   204→  /** Применить новые настройки на лету (вызывается экраном настроек) */
   205→  applySettings(s: GameSettings) {
   206→    const prev = this.settings;
   207→    this.settings = s;
   208→    const g = s.graphics;
   209→    this.renderer.setPixelRatio(Math.min(window.devicePixelRatio, g.pixelRatio));
   210→    if (prev.graphics.shadows !== g.shadows) this.setShadows(g.shadows);
   211→    if (prev.graphics.shadowRes !== g.shadowRes) this.sky.setShadowRes(g.shadowRes);
   212→    if (prev.graphics.waterDetail !== g.waterDetail || prev.graphics.drawDistance !== g.drawDistance
   213→      || prev.graphics.waterChunkSize !== g.waterChunkSize || prev.graphics.waterLodStep !== g.waterLodStep) {
   214→      this.water.configure(g.waterDetail, g.waterChunkSize, g.drawDistance, g.waterLodStep);
   215→    }
   216→    // дистанция мишени на полигоне — переставить цель без сброса повреждений
   217→    if (prev.gameplay.provingDistance !== s.gameplay.provingDistance
   218→      && this.proving && this.pgTarget) {
   219→      const t = this.pgTarget.ship;
   220→      t.position.copy(this.pgTargetPos());
   221→      t.position.y = 0;
   222→      this.pushMessage(`Мишень переставлена: ${s.gameplay.provingDistance} м`);
   223→    }
   224→    this.water.setFoamQuality(g.foamQuality);
   225→    this.rig.camera.far = g.drawDistance * 2 + 4000;
   226→    this.rig.camera.updateProjectionMatrix();
   227→    this.sky.setDomeScale(g.drawDistance / 6500 * 1.12);
   228→    (this.scene.fog as THREE.FogExp2).density = this.fogBase();
   229→    this.audio.setVolume(s.audio.volume);
   230→    this.input.setPreventCodes(this.preventCodes());
   231→  }

   238→      const mesh = o as THREE.Mesh;
   239→      if (mesh.material) {
   240→        const mats = Array.isArray(mesh.material) ? mesh.material : [mesh.material];

   242→      }
   243→    });

   334→  private wireEvents() {
   335→    const distToPlayer = (p: THREE.Vector3) => p.distanceTo(this.player.position);
   336→    this.bus.on(GameEvents.CANNON_FIRED, (p) => {
   337→      const d = (p as { pos: THREE.Vector3; power: number });
   338→      this.audio.cannon(d.power, distToPlayer(d.pos));
   339→      if (distToPlayer(d.pos) < 60) this.rig.addShake(0.35 * d.power);
   340→    });
   341→    this.bus.on(GameEvents.PROJECTILE_SPLASH, (p) => {
   342→      this.audio.splash(distToPlayer((p as { pos: THREE.Vector3 }).pos));
   343→    });
   344→    this.bus.on(GameEvents.PROJECTILE_HIT, (p) => {
   345→      const d = (p as { pos: THREE.Vector3; target: Ship });
   346→      this.audio.woodHit(distToPlayer(d.pos));
   347→      if (d.target === this.player) this.rig.addShake(0.5);
   348→      // ИИ реагирует на агрессию игрока
   349→      const ai = this.aiShips.find(a => a.ship === d.target);
   350→      if (ai) ai.onPlayerAggression(this.aiShips);
   351→    });
   352→    this.bus.on(GameEvents.MESSAGE, (p) => {
   353→      this.pushMessage((p as { text: string }).text);
   354→    });
   355→    this.bus.on(GameEvents.MAST_BROKEN, (p) => {
   356→      const d = (p as { target: Ship; pos: THREE.Vector3 });
   357→      this.audio.woodHit(distToPlayer(d.pos));
   358→      this.rig.addShake(0.35);
   359→      const who = d.target === this.player ? 'Наш корабль' : d.target.preset.name;
   360→      this.pushMessage(`${who}: мачта переломлена!`);
   361→    });
   362→  }

   371→  private spawnProvingTarget(modelId: string) {
   372→    this.removeProvingTarget();
   373→    const terrain = (x: number, z: number) => heightAt(x, z);
   374→    const ship = new Ship(modelId, false, this.wind, terrain, this.particles);
   375→    ship.position.copy(this.pgTargetPos());
   376→    ship.position.y = 0;
   377→    ship.quaternion.setFromEuler(new THREE.Euler(0, Math.PI / 2, 0)); // параллельно игроку
   378→    ship.noSurrender = true;
   379→    this.worldGroup.add(ship.visual.group);
   380→    this.worldGroup.add(ship.foam.mesh);
   381→    this.worldGroup.add(ship.wake.mesh);
   382→    const ai = new AIShip(ship, [], 'anchored');
   383→    ai.returnFire = this.pgReturnFire;
   384→    this.pgTarget = ai;
   385→    this.combat.others.push(ship);
   386→  }

   388→  private removeProvingTarget() {
   389→    if (!this.pgTarget) return;
   390→    const s = this.pgTarget.ship;
   391→    const idx = this.combat.others.indexOf(s);
   392→    if (idx >= 0) this.combat.others.splice(idx, 1);
   393→    this.worldGroup.remove(s.visual.group);
   394→    this.worldGroup.remove(s.foam.mesh);
   395→    this.worldGroup.remove(s.wake.mesh);
   396→    s.dispose();
   397→    this.pgTarget = null;
   398→  }

   403→    this.proving = true;

   405→    // штиль: ровный ветер и лёгкая зыбь (блокируется в updateWeather)

   407→    this.wind.speed = 4.2;
   408→    SEA.amplitude = 0.42;
   409→    const p = this.player;
   410→    p.clearDamage();
   411→    p.position.copy(this.PG_ANCHOR); p.position.y = 0;
   412→    p.quaternion.setFromEuler(new THREE.Euler(0, Math.PI / 2, 0)); // курс на восток
   413→    p.anchored = false;
   414→    p.ammo = { ball: 250, chain: 120, grape: 150, fire: 60 };

   432→  resetProvingTarget() {

   580→  // ---------- управление ----------
   581→  private handleInput(dt: number) {
   582→    const inp = this.input;
   583→    if (!this.started || this.dead || this.uiOpen) { inp.endFrame(); return; }
   584→    const p = this.player;
   585→    const c = this.settings.controls;
   586→
   587→    // паруса
   588→    if (inp.pressed(c.sailsUp)) { p.sailTarget = Math.min(3, p.sailTarget + 1); if (p.anchored) this.pushMessage('Сначала поднять якорь (F)'); }
   589→    if (inp.pressed(c.sailsDown)) {
   590→      if (p.sailTarget > 0) p.sailTarget = Math.max(0, p.sailTarget - 1);
   591→    }
   592→    // задний ход на вёслах: держать «убрать паруса» при убранных парусах
   593→    p.reverseThrust = inp.isDown(c.sailsDown) && p.sailTarget === 0 && !p.anchored;
   594→    if (p.reverseThrust) this.lastSOffset += dt; else this.lastSOffset = 0;
   595→
   596→    // руль: A — влево, D — вправо (положительный руль = поворот вправо,
   597→    // ползунок HUD при D уходит вправо — и корабль поворачивает вправо).
   598→    // invertRudder — настройка для тех, кто привык к обратной схеме
   599→    let steer = (inp.isDown(c.steerRight) ? 1 : 0) - (inp.isDown(c.steerLeft) ? 1 : 0);
   600→    if (c.invertRudder) steer = -steer;
   601→    if (this.docked) p.rudderTarget = 0;
   602→    else p.rudderTarget = steer; // D — штирборт (право), A — порт (лево)
   603→
   604→    // якорь
   605→    if (inp.pressed(c.anchor)) {
   606→      if (this.docked) { this.undock(); }
   607→      else {
   608→        p.anchored = !p.anchored;
   609→        this.audio.bell();
   610→        this.pushMessage(p.anchored ? 'Якорь отдан' : 'Якорь поднят');
   611→      }
   612→    }
   613→
   614→    // камера
   615→    if (inp.pressed(c.camera)) {
   616→      this.rig.cycleMode();
   617→      this.pushMessage(`Камера: ${CAMERA_LABEL[this.rig.mode]}`);
   618→    }
   619→
   620→    // боеприпасы
   621→    const ammoKeys: [string, AmmoType][] = [[c.ammo[0], 'ball'], [c.ammo[1], 'chain'], [c.ammo[2], 'grape'], [c.ammo[3], 'fire']];
   622→    for (const [key, kind] of ammoKeys) {
   623→      if (inp.pressed(key)) {
   624→        this.combat.ammo = kind;
   625→        this.pushMessage(`Боеприпас: ${AMMO_INFO[kind].name}`);
   626→      }
   627→    }
   628→
   629→    // прицеливание: наведение мышью
   630→    if (this.combat.aim.active) {
   631→      this.combat.aim.azimuth = clamp(this.combat.aim.azimuth - inp.mouseDX * 0.0022, -0.62, 0.62);
   632→      this.combat.aim.elevation = clamp(this.combat.aim.elevation + inp.mouseDY * 0.0016, 0.01, 0.24);
   633→    }
   634→
   635→    // борта; «правый борт» ещё и взаимодействие
   636→    const boardTarget = this.canBoard();
   637→    const canDockHere = this.canDock();
   638→    this.boardingTarget = boardTarget;
   639→    if (inp.pressed(c.aimPort)) {
   640→      this.combat.toggleAim(-1);
   641→    }
   642→    if (inp.pressed(c.aimStarboard)) {
   643→      if (boardTarget) {
   644→        this.takePrize(boardTarget);
   645→      } else if (canDockHere && !this.combat.aim.active) {
   646→        this.dock();
   647→      } else {
   648→        this.combat.toggleAim(1);
   649→      }
   650→    }
   651→
   652→    // огонь
   653→    if (this.combat.aim.active) {
   654→      if (inp.lmbJust) this.combat.fireOne();
   655→      if (inp.pressed(c.fire)) this.combat.fireBroadside();
   656→    }
   657→
   658→    // ремонт/откачка
   659→    p.pumpAction = inp.isDown(c.repair);
   660→  }

   662→  // ---------- столкновения кораблей ----------
   663→  private shipCollisions() {
   664→    const all = [this.player, ...this.aiShips.map(a => a.ship)];
   665→    if (this.pgTarget && !this.pgTarget.ship.sunk) all.push(this.pgTarget.ship);
   666→    for (let i = 0; i < all.length; i++) {
   667→      for (let j = i + 1; j < all.length; j++) {
   668→        const a = all[i], b = all[j];
   669→        if (a.sunk || b.sunk) continue;
   670→        const minD = (a.preset.length + b.preset.length) * 0.30;
   671→        const d = a.position.distanceTo(b.position);
   672→        if (d < minD && d > 0.01) {
   673→          const n = new THREE.Vector3().subVectors(a.position, b.position).setY(0).normalize();
   674→          const push = (minD - d) * 0.5;
   675→          a.position.addScaledVector(n, push);
   676→          b.position.addScaledVector(n, -push);
   677→          const relV = new THREE.Vector3().subVectors(a.velocity, b.velocity).dot(n);
   678→          if (relV < -1.2) {
   679→            const dmg = Math.min(22, -relV * 2.4);
   680→            a.applyDamage({ kind: 'ram', amount: dmg });
   681→            b.applyDamage({ kind: 'ram', amount: dmg });
   682→            const mid = new THREE.Vector3().addVectors(a.position, b.position).multiplyScalar(0.5);
   683→            this.particles.splash(mid.x, 0.5, mid.z, 1.2);
   684→            this.audio.woodHit(0);
   685→            this.rig.addShake(0.5);
   686→            if (a === this.player || b === this.player) {
   687→              this.pushMessage('Столкновение! Корпус повреждён');
   688→            }
   689→            // военный реагирует на таран
   690→            for (const ai of this.aiShips) {
   691→              if (ai.ship === a || ai.ship === b) ai.onPlayerAggression(this.aiShips);
   692→            }
   693→          }
   694→          a.velocity.multiplyScalar(0.85);
   695→          b.velocity.multiplyScalar(0.85);
   696→        }
   697→      }
   698→    }
   699→  }

   798→  // ---------- погода ----------
   799→  private updateWeather(dt: number) {
   800→    // полигон: чистое море, ровный слабый ветер — всегда
   801→    if (this.proving) {
   802→      this.windTargetSpeed = 4.2;
   803→      this.wind.speed += (4.2 - this.wind.speed) * Math.min(1, dt * 0.6);
   804→      SEA.amplitude = 0.42;
   805→      return;
   806→    }
   807→    this.windTimer -= dt;
   808→    if (this.windTimer < 0) {
   809→      this.windTimer = 18 + Math.random() * 22;
   810→      this.windTargetSpeed = 3.5 + Math.random() * 9.5;
   811→    }
   812→    this.gustT += dt;
   813→    const gust = Math.sin(this.gustT * 0.5) * 0.9 + Math.sin(this.gustT * 1.7 + 2) * 0.5;
   814→    this.wind.speed += (this.windTargetSpeed + gust - this.wind.speed) * Math.min(1, dt * 0.3);
   815→    this.wind.from += dt * 0.008 * Math.sin(this.time * 0.037 + 1.3);
   816→    // амплитуда волн: 0.4 штиль .. waveHeight шторм (из настроек графики;
   817→    // 1.15 по умолчанию — гребни не накрывают палубу, борт не тонет)
   818→    const waveCap = this.settings.graphics.waveHeight;
   819→    SEA.amplitude = clamp(0.45 + (this.wind.speed - 3) * 0.085, 0.4, waveCap);
   820→  }

   822→  // ---------- кадр ----------
   823→  private frame = () => {
   824→    this.rafId = requestAnimationFrame(this.frame);
   825→    const now = performance.now();
   826→    let dt = (now - this.lastFrame) / 1000;
   827→    this.lastFrame = now;
   828→    this.fpsSmooth += (1 / Math.max(dt, 1e-4) - this.fpsSmooth) * 0.05; // истинный FPS
   829→    dt = Math.min(dt, 0.1);
   830→
   831→    this.handleInput(dt);
   832→
   833→    // физика фиксированным шагом
   834→    this.accum += dt;
   835→    const STEP = 1 / 60;
   836→    let steps = 0;
   837→    while (!this.paused && this.accum >= STEP && steps < 5) {
   838→      this.time += STEP;
   839→      this.updateWeather(STEP);
   840→      this.player.fixedStep(STEP, this.time);
   841→      for (const ai of this.aiShips) ai.ship.fixedStep(STEP, this.time);
   842→      if (this.pgTarget) this.pgTarget.ship.fixedStep(STEP, this.time);
   843→      for (const ai of this.aiShips) {
   844→        ai.update(STEP, this.time, this.player, this.combat, (t) => this.pushMessage(t));
   845→      }
   846→      if (this.pgTarget) this.pgTarget.update(STEP, this.time, this.player, this.combat, (t) => this.pushMessage(t));
   847→      this.shipCollisions();
   848→      this.accum -= STEP;
   849→      steps++;
   850→    }
   851→
   852→    // телеметрия для отладки/E2E: window.__sea
   853→    const dbg = window as unknown as { __sea: Record<string, unknown> };
   854→    const sailsV = this.player.visual.sails.sails;
   855→    let furlSum = 0, furlMin = 1, furlMax = 0;
   856→    for (const sv of sailsV) {
   857→      furlSum += sv.furl;
   858→      furlMin = Math.min(furlMin, sv.furl);
   859→      furlMax = Math.max(furlMax, sv.furl);
   860→    }
   861→    // минимальный надводный борт: высота палубы у планширя минус вода
   862→    // по носу/корме/миделю И ПО КРАЯМ ПАЛУБЫ (крен топит борт) —
   863→    // учитываем крен/дифферент кватернионом
   864→    let freeboard = Infinity;
   865→    const deckLocal = this.player.preset.freeboard * 0.82 - 1.0;
   866→    for (const lz of [-this.player.preset.length * 0.38, 0, this.player.preset.length * 0.38]) {
   867→      for (const lx of [0, -this.player.halfWidthAt(lz), this.player.halfWidthAt(lz)]) {
   868→        this._v3.set(lx, deckLocal, lz).applyQuaternion(this.player.quaternion).add(this.player.position);
   869→        freeboard = Math.min(freeboard, this._v3.y - waterHeightAt(this._v3.x, this._v3.z, this.time));
   870→      }
   871→    }
   872→    // доля постановки парусов (0..1): уровень W/S, ограниченный фактическим ходом
   873→    const p = this.player;
   874→    const windPow = Math.pow(clamp(this.wind.speed / 8.0, 0.05, 1.6), 1.6);
   875→    const attainable = p.preset.maxSpeed * 0.85 * Math.sqrt(clamp(Ship.polar(p.apparentWindAngle()) * windPow, 0, 1.5));
   876→    const setTot = Math.min(p.sailFrac, attainable > 1.5 ? clamp(p.speed() / attainable, 0, 1) : 1);
   877→    const tgt = this.proving && this.pgTarget ? this.pgTarget.ship : null;
   878→    dbg.__sea = {
   879→      fps: this.fpsSmooth,
   880→      x: p.position.x, y: p.position.y, z: p.position.z,
   881→      waterY: waterHeightAt(p.position.x, p.position.z, this.time),
   882→      roll: p.euler.z, pitch: p.euler.x,
   883→      heading: p.heading,
   884→      rudder: p.rudder, angY: p.angular.y,
   885→      speed: p.speedKnots(),
   886→      sailLevel: p.sailTarget, sailFrac: p.sailFrac,
   887→      setTot,
   888→      furlAvg: furlSum / Math.max(1, sailsV.length), furlMin, furlMax,
   889→      sailSet: sailsV.filter(sv => sv.furl < 0.5).length,
   890→      freeboard, seaAmp: SEA.amplitude, wind: this.wind.speed,
   891→      proving: this.proving ? 1 : 0,
   892→      tgtDist: tgt ? +p.position.distanceTo(tgt.position).toFixed(1) : -1,
   893→      tgtHull: tgt ? +clamp(tgt.hullHP / tgt.hullMax, 0, 1).toFixed(3) : -1,
   894→      tgtWater: tgt ? +clamp(tgt.water / tgt.holdCap, 0, 1).toFixed(3) : -1,
   895→      tgtMastsBroken: tgt ? tgt.mastFall.filter(f => f >= 0).length : -1,
   896→      tgtHoles: tgt ? tgt.hullHoles.length : -1,
   897→      tgtCrew: tgt ? tgt.crew : -1,
   898→      tgtSinking: tgt && tgt.sinking ? 1 : 0,
   899→      tgtSunk: tgt && tgt.sunk ? 1 : 0,
   900→      tgtShown: tgt ? tgt.crew3d.shown : -1,
   901→      tgtCrewDead: tgt ? tgt.crew3d.dead : -1,
   902→      gibsAlive: this.gibs.alive,
   903→      crewShown: p.crew3d.shown,
   904→      crewDead: p.crew3d.dead,
   905→      // сетка воды (e2e/диагностика «квадратов»): чанк + LOD-кольца
   906→      water: (() => {
   907→        const w = this.water.getInfo();
   908→        return {
   909→          chunk: w.chunkSize, segs: w.segs, step: +w.step.toFixed(3),
   910→          rings: w.rings.map(r => ({ i: Math.round(r.inner), o: Math.round(r.outer), c: +r.cell.toFixed(2), q: r.quads })),
   911→          farEff: Math.round(w.farEff),
   912→          pgDist: this.settings.gameplay.provingDistance,
   913→        };
   914→      })(),
   915→    };
   916→
   917→    // действия команды (R)
   918→    if (this.player.pumpAction && !this.docked) {
   919→      const msg = this.player.crewAction(dt);
   920→      if (msg && Math.random() < dt * 2) this.pushMessage(msg);
   921→    }
   922→
   923→    // sinking игрока
   924→    if (this.player.sinking && !this.dead && this.player.sinkT > 5) {
   925→      this.dead = true;
   926→      this.pushMessage('Корабль затонул...');
   927→    }
   928→
   929→    // floating origin
   930→    const playerScene = this._v1.copy(this.player.position).add(this.shift);
   931→    if (Math.hypot(playerScene.x, playerScene.z) > 2200) {
   932→      this.shift.x -= Math.round(playerScene.x / 50) * 50;
   933→      this.shift.z -= Math.round(playerScene.z / 50) * 50;
   934→      this.worldGroup.position.copy(this.shift);
   935→      this.pushMessage('Смена квадрата моря');
   936→    }
   937→    this.negShift.copy(this.shift).negate();
   938→
   939→    // визуал кораблей
   940→    this.player.updateVisual(this.time, dt, this.particles);
   941→    for (const ai of this.aiShips) ai.ship.updateVisual(this.time, dt, this.particles);
   942→    if (this.pgTarget) this.pgTarget.ship.updateVisual(this.time, dt, this.particles);
   943→
   944→    // мелкие детали кораблей — только когда камера рядом (LOD)
   945→    const camPos = this.rig.camera.position;
   946→    const lodD = this.settings.graphics.detailLod;
   947→    this._v3.copy(this.player.position).add(this.shift);
   948→    this.setDetailLod(this.player, camPos, lodD);
   949→    for (const ai of this.aiShips) this.setDetailLod(ai.ship, camPos, lodD);
   950→    if (this.pgTarget) this.setDetailLod(this.pgTarget.ship, camPos, lodD);
   951→
   952→    // системы
   953→    this.combat.update(dt, this.time, waterHeightAt);
   954→    this.particles.update(dt);
   955→    this.gibs.update(dt);
   956→    this.world.update(this.time, dt, this.player.position, waterHeightAt);
   957→
   958→    // небо и свет
   959→    playerScene.copy(this.player.position).add(this.shift);
   960→    const { dayF } = this.sky.update(dt, this.time, this.rig.camera.position, playerScene, 0.42);
   961→    const env = this.sky.environmentColors(dayF, 0);
   962→    const fogD = this.fogBase() * (1 + (1 - dayF) * 0.18);
   963→    this.water.setEnvironment(this.sky.sunDir, env.sun, env.sky, env.horizon, env.fog, fogD);
   964→    (this.scene.fog as THREE.FogExp2).color.copy(env.fog);
   965→    (this.scene.fog as THREE.FogExp2).density = fogD;
   966→    // реальное солнце — в материалы парусов (иначе «пузо» не заливается
   967→    // светом и паруса читаются плоскими при любом ветре)
   968→    {
   969→      const sd = this.sky.sunDir;
   970→      for (const ship of [this.player, ...this.aiShips.map(a => a.ship)]) {
   971→        for (const sv of ship.visual.sails.sails) {
   972→          (sv.mat.uniforms.uSunDir.value as THREE.Vector3).copy(sd);
   973→        }
   974→      }
   975→    }
   976→
   977→    // фонарь игрока ночью
   978→    if (this.player.visual.lantern) {
   979→      this.player.visual.lantern.intensity = dayF < 0.35 ? 5 : 0;
   980→    }
   981→
   982→    // камера
   983→    const shipInfo = {
   984→      scenePos: playerScene,
   985→      heading: this.player.heading,
   986→      roll: this.player.euler.z,
   987→      pitch: this.player.euler.x,
   988→      speedFrac: clamp(this.player.speed() / this.player.preset.maxSpeed, 0, 1),
   989→    };
   990→    const aimDir = this.combat.aim.active ? this.combat.aimDirection(this._v2) : null;
   991→    this.rig.update(dt, {
   992→      active: this.combat.aim.active,
   993→      side: this.combat.aim.side,
   994→      dir: aimDir ?? this._v2.set(1, 0, 0),
   995→    }, shipInfo, {
   996→      mouseDX: this.input.mouseDX, mouseDY: this.input.mouseDY,
   997→      wheel: this.input.wheelDelta, rmb: this.input.rmb,
   998→    });
   999→
  1000→    // вода
  1001→    this.water.update(this.time, this.rig.camera.position, this.negShift);
  1002→
  1003→    // звук
  1004→    this.audio.update(dt, {
  1005→      windSpeed: this.wind.speed,
  1006→      speed: this.player.speed(),
  1007→      rollRate: this.player.angular.z,
  1008→      nearLand: this.nearLand(),
  1009→      dayF,
  1010→    });
  1011→
  1012→    // сообщения
  1013→    for (let i = this.msgTtl.length - 1; i >= 0; i--) {
  1014→      this.msgTtl[i] -= dt;
  1015→      if (this.msgTtl[i] <= 0) { this.messages.splice(i, 1); this.msgTtl.splice(i, 1); }
  1016→    }
  1017→
  1018→    this.updateHud();
  1019→    this.renderer.render(this.scene, this.rig.camera);
  1020→    this.input.endFrame();
  1021→  };

  1028→  private setDetailLod(ship: Ship, camPos: THREE.Vector3, maxDist: number) {
  1029→    const d = ship.visual.detail;
  1030→    if (!d) return;
  1031→    this._v3.copy(ship.position).add(this.shift);
  1032→    d.visible = camPos.distanceTo(this._v3) < maxDist;
  1033→  }

  1035→  private nearLand(): boolean {
  1036→    for (const isl of this.world.islands) {
  1037→      if (Math.hypot(this.player.position.x - isl.cx, this.player.position.z - isl.cz) < isl.r + 260) return true;
  1038→    }
  1039→    return false;
  1040→  }

  1042→  respawn() {
  1043→    if (this.proving) this.exitProving();
  1044→    const p = this.player;
  1045→    p.hullHP = p.hullMax;
  1046→    p.sailHP = p.preset.sailHP;
  1047→    p.crew = Math.round(p.preset.maxCrew * 0.8);
  1048→    p.water = 0; p.leakRate = 0; p.onFire = false; p.fireT = 0;
  1049→    p.sinking = false; p.sinkT = 0; p.sunk = false;
  1050→    p.mastsBroken = 0; p.mastAnim = -1;
  1051→    for (let i = 0; i < p.mastHP.length; i++) { p.mastHP[i] = p.mastMaxHP; p.mastFall[i] = -1; }
  1052→    for (const m of p.visual.masts) { m.rotation.set(0, 0, 0); m.position.y = 0; }
  1053→    for (const h of p.hullHoles) { p.visual.group.remove(h.mesh); h.mesh.geometry.dispose(); }
  1054→    p.hullHoles = [];
  1055→    p.leakBias = 0;
  1056→    p.visual.group.visible = true;
  1057→    p.velocity.set(0, 0, 0);
  1058→    p.angular.set(0, 0, 0);
  1059→    p.position.copy(this.spawnPos);
  1060→    p.position.y = 0;
  1061→    p.quaternion.setFromEuler(new THREE.Euler(0, Math.atan2(
  1062→      this.spawnPos.x - this.world.islands[0].cx, this.spawnPos.z - this.world.islands[0].cz), 0));
  1063→    p.gold = Math.floor(p.gold * 0.5);
  1064→    p.cargo = { sugar: 0, rum: 0, timber: 0 };
  1065→    p.sailTarget = 0;
  1066→    this.dead = false;
  1067→    this.pushMessage(`Новый корабль у гавани. Половина золота ушла на дно...`);
  1068→  }
