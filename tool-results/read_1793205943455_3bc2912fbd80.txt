    29→export class Ship {

    42→  private reserve: Probe[] = [];   // запас плавучести надводного борта
    43→  private force = new THREE.Vector3();
    44→  private torque = new THREE.Vector3();
    45→  private _f = new THREE.Vector3();
    46→  private _r = new THREE.Vector3();

   109→  constructor(presetId: string, isPlayer: boolean, wind: WindState,
   110→    terrainHeight: (x: number, z: number) => number, particles: Particles) {
   111→    this.preset = PRESETS[presetId];
   112→    this.isPlayer = isPlayer;
   113→    this.wind = wind;
   114→    this.terrainHeight = terrainHeight;
   115→    this.particles = particles;
   116→    this.visual = buildShip(this.preset);
   117→    this.hullHP = this.preset.hullHP;
   118→    this.hullMax = this.preset.hullHP;
   119→    this.sailHP = this.preset.sailHP;
   120→    this.crew = Math.round(this.preset.maxCrew * 0.8);
   121→    this.holdCap = this.preset.cargo * 0.8;
   122→    this.mass = this.preset.mass;
   123→
   124→    const { length: L, beam: B, draft: D, freeboard: F } = this.preset;
   125→    // моменты инерции (связные оси: x — крен? нет: x=право, y=верх, z=нос)
   126→    // крен — вокруг z, дифферент — вокруг x, рыскание — вокруг y
   127→    this.inertia.set(
   128→      this.mass * (B * B + F * F) / 12 * 1.5,   // pitch (x)
   129→      this.mass * (L * L + B * B) / 12 * 1.2,   // yaw (y)
   130→      this.mass * (B * B + F * F) / 12 * 1.5,   // roll (z)
   131→    );
   132→
   133→    // зонды плавучести: 7 шпангоутов × 3 точки (ниже ватерлинии).
   134→    // Жёстче прежнего (нейтральная глубина 0.38 осадки) — корабль
   135→    // быстрее следует за гребнем волны, не проваливается в него.
   136→    const NZ = 7;
   137→    let n = 0;
   138→    for (let i = 0; i < NZ; i++) {
   139→      const t = i / (NZ - 1);
   140→      const z = THREE.MathUtils.lerp(-L * 0.44, L * 0.44, t);
   141→      const prof = this.halfWidthAt(z);
   142→      const offs = [0, -prof * 0.75, prof * 0.75];
   143→      for (const x of offs) {
   144→        this.probes.push({ off: new THREE.Vector3(x, -D * 0.38, z), k: 0, c: 0 });
   145→        n++;
   146→      }
   147→    }
   148→    const neutral = D * 0.38;
   149→    const kTotal = this.mass * G / neutral;
   150→    for (const p of this.probes) {
   151→      p.k = kTotal / n;
   152→      p.c = 2 * Math.sqrt(p.k * (this.mass / n)) * 0.5; // демпфер качки
   153→    }
   154→    // Резервные зонды по скулам борта чуть выше спокойной ватерлинии:
   155→    // моделируют запас плавучести корпуса до планширя — гребень, накрывший
   156→    // палубу, встречает сильное сопротивление и кренящий момент вправку,
   157→    // «полностью по борт» корабль больше не погружается.
   158→    let rn = 0;
   159→    for (let i = 0; i < NZ; i++) {
   160→      const t = i / (NZ - 1);
   161→      const z = THREE.MathUtils.lerp(-L * 0.44, L * 0.44, t);
   162→      const prof = this.halfWidthAt(z) * 0.85;
   163→      for (const x of [-prof, prof]) {
   164→        this.reserve.push({ off: new THREE.Vector3(x, F * 0.34, z), k: 0, c: 0 });
   165→        rn++;
   166→      }
   167→    }
   168→    // жёстче прежнего: полный вес запаса уже на 15 см погружения борта —
   169→    // волна не может накрыть палубу, корпус «всплывает» на гребень
   170→    const rkTotal = this.mass * G / 0.15;
   171→    for (const p of this.reserve) {
   172→      p.k = rkTotal / rn;
   173→      p.c = 2 * Math.sqrt(p.k * (this.mass / rn)) * 1.3;
   174→    }
   175→
   176→    this.foam = new WaterlineFoam(this.visual.outline);
   177→    this.wake.setWaterHeight(waterHeightAt);
   178→    if (isPlayer) enableLantern(this.visual);
   179→
   180→    // экипаж на палубе: процедурные типажи, сид фиксирован по модели
   181→    const seed = Array.from(presetId).reduce((a, ch) => a + ch.charCodeAt(0) * 31, 11)
   182→      + Math.round(this.preset.length * 7);
   183→    this.crew3d = buildCrew(this.preset, seed);
   184→    this.visual.group.add(this.crew3d.group);
   185→    for (let i = 0; i < this.visual.masts.length; i++) {
   186→      this.mastHP.push(this.mastMaxHP);
   187→      this.mastFall.push(-1);
   188→    }
   189→  }
   190→
   191→  halfWidthAt(z: number): number {
   192→    const t = clamp((z + this.preset.length / 2) / this.preset.length, 0, 1);
   193→    let w: number;
   194→    if (t < 0.45) w = THREE.MathUtils.lerp(0.58, 1.0, t / 0.45);
   195→    else w = THREE.MathUtils.lerp(1.0, 0.0, Math.pow(clamp((t - 0.45) / 0.55, 0, 1), 0.75));
   196→    return Math.max(0.3, w * this.preset.beam * 0.5);
   197→  }
   198→
   199→  // ---------- геометрия ориентации ----------
   200→  get forward(): THREE.Vector3 {
   201→    const e = new THREE.Euler().setFromQuaternion(this.quaternion, 'YXZ');
   202→    return new THREE.Vector3(Math.sin(e.y), 0, Math.cos(e.y)).normalize();
   203→  }
   204→  get heading(): number {
   205→    const e = new THREE.Euler().setFromQuaternion(this.quaternion, 'YXZ');
   206→    return e.y;
   207→  }
   208→  get euler(): THREE.Euler { return new THREE.Euler().setFromQuaternion(this.quaternion, 'YXZ'); }
   209→  speed(): number { return Math.hypot(this.velocity.x, this.velocity.z); }
   210→  speedKnots(): number { return this.speed() * 1.9438; }
   211→
   212→  /** Угол между направлением ОТКУДА дует (кажущийся) ветер и курсом, 0..PI */
   213→  apparentWindAngle(): number {
   214→    const wf = this.tmpV.set(Math.cos(this.wind.from), 0, Math.sin(this.wind.from));
   215→    // кажущийся ветер = истинный - скорость корабля
   216→    const av = this.tmpV2.set(-wf.x * this.wind.speed - this.velocity.x, 0, -wf.z * this.wind.speed - this.velocity.z);
   217→    // вектор ОТКУДА дует кажущийся ветер:
   218→    const aFromX = -av.x, aFromZ = -av.z;
   219→    const len = Math.hypot(aFromX, aFromZ) || 1;
   220→    const f = this.forward;
   221→    const dot = (aFromX / len) * f.x + (aFromZ / len) * f.z;
   222→    return Math.acos(clamp(dot, -1, 1));
   223→  }
   224→
   225→  /** Полярная эффективность парусов по углу (рад): лавировка/бейдевинд/галфвинд/фордевинд */
   226→  static polar(awa: number): number {
   227→    const d = awa * 180 / Math.PI;
   228→    if (d < 25) return 0.02;                       // левентик — почти нет хода
   229→    if (d < 60) return THREE.MathUtils.lerp(0.08, 0.7, (d - 25) / 35);
   230→    if (d < 110) return THREE.MathUtils.lerp(0.7, 1.0, (d - 60) / 50);  // бейдевинд-галфвинд
   231→    if (d < 150) return THREE.MathUtils.lerp(1.0, 0.82, (d - 110) / 40);
   232→    return THREE.MathUtils.lerp(0.82, 0.68, (d - 150) / 30);            // фордевинд
   233→  }
   234→
   235→  // ---------- фиксированный шаг физики ----------
   236→  fixedStep(dt: number, time: number) {
   237→    if (this.sunk) return;
   238→    this.force.set(0, 0, 0);
   239→    this.torque.set(0, 0, 0);
   240→
   241→    const f = this._f.copy(this.forward);
   242→    // штирборт (правый борт): forward × up = (−f.z, 0, f.x)
   243→    const right = this._r.set(-f.z, 0, f.x).normalize();
   244→    const e = this.euler;
   245→    const spd = this.speed();
   246→
   247→    const m = this.mass + this.water * 1000; // вода в трюме утяжеляет
   248→
   249→    // --- гравитация ---
   250→    this.force.y -= m * G;
   251→
   252→    // --- плавучесть по зондам (τ = r × F, F = (0, fl, 0)) ---
   253→    for (const p of this.probes) {
   254→      const wp = this._t1.copy(p.off).applyQuaternion(this.quaternion).add(this.position);
   255→      const wy = waterHeightAt(wp.x, wp.z, time);
   256→      const depth = wy - wp.y;
   257→      if (depth > 0) {
   258→        const rx = wp.x - this.position.x, rz = wp.z - this.position.z;
   259→        // вертикальная скорость точки: v = ω × r → vy = ωz*rx − ωx*rz
   260→        const vy = this.velocity.y + (this.angular.z * rx - this.angular.x * rz);
   261→        let fl = p.k * Math.min(depth, 6) - p.c * vy;
   262→        if (fl < 0) fl = 0;
   263→        this.force.y += fl;
   264→        this.torque.x += -rz * fl;
   265→        this.torque.z += rx * fl;
   266→      }
   267→    }
   268→    // --- запас плавучести бортов (пока не тонем) ---
   269→    if (!this.sinking) {
   270→      for (const p of this.reserve) {
   271→        const wp = this._t1.copy(p.off).applyQuaternion(this.quaternion).add(this.position);
   272→        const wy = waterHeightAt(wp.x, wp.z, time);
   273→        const depth = wy - wp.y;
   274→        if (depth > 0) {
   275→          const rx = wp.x - this.position.x, rz = wp.z - this.position.z;
   276→          const vy = this.velocity.y + (this.angular.z * rx - this.angular.x * rz);
   277→          let fl = p.k * Math.min(depth, 2.5) - p.c * vy;
   278→          if (fl < 0) fl = 0;
   279→          this.force.y += fl;
   280→          this.torque.x += -rz * fl;
   281→          this.torque.z += rx * fl;
   282→        }
   283→      }
   284→    }
   285→
   286→    // --- ветер на парусах ---
   287→    this.sailFrac += (this.sailTarget / 3 - this.sailFrac) * Math.min(1, dt * 0.7);
   288→    const awa = this.apparentWindAngle();
   289→    const polar = Ship.polar(awa);
   290→    const sailEffMul = this.sailHP / this.preset.sailHP;
   291→    const damaged = sailEffMul < 0.5 ? 0.45 : sailEffMul < 0.85 ? 0.8 : 1.0;
   292→    const windPower = Math.pow(this.wind.speed / 8.0, 1.6);
   293→    let thrust = 0;
   294→    if (!this.anchored && !this.sinking && this.mastsBroken < 2) {
   295→      thrust = this.sailFrac * polar * windPower * this.preset.maxSpeed * 7.0 * damaged * (this.sailHP > 0 ? 1 : 0);
   296→    }
   297→    if (this.reverseThrust && !this.anchored) thrust = -this.preset.maxSpeed * 0.45; // задний ход на вёслах
   298→    // сила тяги вдоль forward
   299→    this.force.x += f.x * thrust * m * 0.06;
   300→    this.force.z += f.z * thrust * m * 0.06;
   301→    // крен от ветра: боковая составляющая ветра на парусах
   302→    const wf = new THREE.Vector3(Math.cos(this.wind.from), 0, Math.sin(this.wind.from));
   303→    const windTravel = wf.clone().multiplyScalar(-this.wind.speed);
   304→    const lateral = windTravel.dot(right);
   305→    const heelF = lateral * this.sailFrac * windPower * m * 0.05;
   306→    this.torque.z += heelF;
   307→    // ... и небольшое боковое сносовое усилие
   308→    this.force.x += right.x * lateral * this.sailFrac * m * 0.004;
   309→    this.force.z += right.z * lateral * this.sailFrac * m * 0.004;
   310→
   311→    // --- руль ---
   312→    this.rudder += (this.rudderTarget - this.rudder) * Math.min(1, dt * 1.8);
   313→    const rudderAngle = this.rudder * 0.6; // рад, макс ~34°
   314→    const yawAuth = clamp(spd / 4.0, 0, 1.15); // авторитеты руля от скорости
   315→    // положительный руль (D) = поворот ВПРАВО (штирборт):
   316→    // вращение вокруг −Y, т.к. +Y = поворот влево (к порту)
   317→    this.torque.y -= rudderAngle * yawAuth * m * 12.0; // ~6°/с на полном ходу
   318→    // крен на повороте (отцентробежный)
   319→    this.torque.z += this.angular.y * spd * m * 0.35;
   320→    // лёгкое зарывание носа на скорости (слабое: сильное топило борт
   321→    // в волну на 3+ метра — см. жалобы «корабль погружается по борт»)
   322→    this.torque.x += spd * spd * m * 0.0012;
   323→
   324→    // --- сопротивление (по связным осям) ---
   325→    const vLocal = this.velocity.clone();
   326→    const fwdSpd = vLocal.dot(f);
   327→    const latSpd = vLocal.dot(right);
   328→    const fwdDrag = -Math.sign(fwdSpd) * (18 * fwdSpd + 55 * fwdSpd * Math.abs(fwdSpd)) * m * 0.001;
   329→    const latDrag = -latSpd * (3.2) * m * 0.1;
   330→    this.force.x += f.x * fwdDrag + right.x * latDrag;
   331→    this.force.z += f.z * fwdDrag + right.z * latDrag;
   332→    this.force.y -= this.velocity.y * m * 0.12; // вертикальное демпфирование
   333→    if (this.anchored) {
   334→      this.force.x -= this.velocity.x * m * 1.2;
   335→      this.force.z -= this.velocity.z * m * 1.2;
   336→    }
   337→    if (this.grounded) {
   338→      this.force.x -= this.velocity.x * m * 2.5;
   339→      this.force.z -= this.velocity.z * m * 2.5;
   340→    }
   341→
   342→    // --- угловое демпфирование по связным осям ---
   343→    const invQ = this.tmpQ.copy(this.quaternion).invert();
   344→    const ab = this._t3.copy(this.angular).applyQuaternion(invQ); // связные
   345→    ab.x *= Math.exp(-1.6 * dt);  // дифферент
   346→    ab.y *= Math.exp(-0.9 * dt);  // рыскание
   347→    ab.z *= Math.exp(-1.1 * dt);  // крен
   348→    this.angular.copy(ab.applyQuaternion(this.quaternion));
   349→
   350→    // --- ограничение крена/дифферента (мягкие стопы) ---
   351→    // дифферент жёстко: 0.14 рад (8°) — нос не может нырнуть под гребень;
   352→    // крен 0.36 рад (20°) — подветренный борт остаётся над водой
   353→    const roll = e.z, pitch = e.x;
   354→    const rollLim = 0.36, pitchLim = 0.14;
   355→    if (Math.abs(roll) > rollLim) this.torque.z -= (roll - Math.sign(roll) * rollLim) * m * 9.0;
   356→    if (Math.abs(pitch) > pitchLim) this.torque.x -= (pitch - Math.sign(pitch) * pitchLim) * m * 12.0;
   357→    // затопление кренит корабль в сторону пробоин
   358→    this.floodRoll(m);
   359→    if (this.sinking) {
   360→      this.sinkT += dt;
   361→      this.force.y -= m * G * Math.min(0.9, this.sinkT / 9); // неотвратимое погружение
   362→      this.torque.x += m * 0.25 * Math.sign(roll || 1);
   363→      this.torque.z += m * 0.4 * Math.sign(roll || 1);
   364→    }
   365→
   366→    // --- интеграция ---
   367→    this.velocity.addScaledVector(this.force, dt / m);
   368→    this.position.addScaledVector(this.velocity, dt);
   369→    // α = I⁻¹ τ в связных осях
   370→    const tb = this._t3.copy(this.torque).applyQuaternion(invQ);
   371→    tb.x /= this.inertia.x; tb.y /= this.inertia.y; tb.z /= this.inertia.z;
   372→    this.angular.addScaledVector(tb.applyQuaternion(this.quaternion), dt);
   373→    // ограничим угловые скорости
   374→    const wLen = this.angular.length();
   375→    if (wLen > 1.1) this.angular.multiplyScalar(1.1 / wLen);
   376→    // q̇ = 0.5 · ω⊗q → q_new = q + 0.5·dt·(ωx,ωy,ωz,0)⊗q, потом нормализация
   377→    this.wq.set(this.angular.x, this.angular.y, this.angular.z, 0);
   378→    this.dq.copy(this.wq).multiply(this.quaternion); // dq = ω ⊗ q
   379→    const qs = 0.5 * dt;
   380→    this.quaternion.set(
   381→      this.quaternion.x + this.dq.x * qs,
   382→      this.quaternion.y + this.dq.y * qs,
   383→      this.quaternion.z + this.dq.z * qs,
   384→      this.quaternion.w + this.dq.w * qs
   385→    ).normalize();
   386→
   387→    // --- посадка на мель / столкновение с землёй ---
   388→    this.checkGrounding(dt);
   389→
   390→    // --- течение воды в трюме ---
   391→    if (this.leakRate > 0) {
   392→      this.water = Math.min(this.holdCap, this.water + this.leakRate * dt);
   393→    }
   394→    if (this.onFire) {
   395→      this.fireT += dt;
   396→      this.hullHP -= dt * 1.6;
   397→      if (this.fireT > 26) { this.onFire = false; this.fireT = 0; } // выгорит
   398→      if (Math.random() < dt * 1.2) {
   399→        const p = this.particles;
   400→        const wp = this.position.clone();
   401→        p.fire(wp.x + (Math.random() - 0.5) * 4, this.waterlineY() + 1 + Math.random() * 2, wp.z + (Math.random() - 0.5) * 10);
   402→        p.smoke(wp.x + (Math.random() - 0.5) * 4, this.waterlineY() + 3, wp.z + (Math.random() - 0.5) * 10, this.wind ? Math.cos(this.wind.from + Math.PI) * 2 : 0, 2.2, this.wind ? Math.sin(this.wind.from + Math.PI) * 2 : 0, 2.8, 3.0);
   403→      }
   404→    }
   405→    // порог затопления
   406→    if (!this.sinking && this.water > this.holdCap * 0.62) this.startSinking();
   407→    if (this.hullHP <= 0 && !this.sinking) this.startSinking();
   408→
   409→    // --- сдался? ---
   410→    if (!this.isPlayer && !this.surrendered && !this.sinking) {
   411→      if (this.hullHP < this.hullMax * 0.34 || this.crew < this.preset.maxCrew * 0.25) {
   412→        this.surrendered = true;
   413→        this.sailTarget = 0;
   414→        this.onSurrender?.(this);
   415→      }
   416→    }
   417→
   418→    if (this.sinking && this.sinkT > 14 && !this.sunk) {
   419→      this.sunk = true;
   420→      this.visual.group.visible = false;
   421→      this.onSunk?.(this);
   422→    }
   423→  }
   424→
   425→  private checkGrounding(dt: number) {
   426→    if (this.sinking || this.sunk) return;
   427→    const f = this.forward;
   428→    const bow = this.tmpV.copy(this.position).addScaledVector(f, this.preset.length * 0.42);
   429→    const stern = this.tmpV2.copy(this.position).addScaledVector(f, -this.preset.length * 0.42);
   430→    let hit = false;
   431→    for (const p of [bow, stern]) {
   432→      const th = this.terrainHeight(p.x, p.z);
   433→      if (th > this.position.y - this.preset.draft * 0.6) {
   434→        hit = true;
   435→        // нормаль склона
   436→        const e = 2;
   437→        const nx = this.terrainHeight(p.x - e, p.z) - this.terrainHeight(p.x + e, p.z);
   438→        const nz = this.terrainHeight(p.x, p.z - e) - this.terrainHeight(p.x, p.z + e);
   439→        const n = new THREE.Vector3(nx, 2 * e, nz).normalize();
   440→        this.force.addScaledVector(n, this.mass * 22);
   441→        this.velocity.multiplyScalar(0.94);
   442→      }
   443→    }
   444→    if (hit) {
   445→      if (!this.grounded && this.speed() > 3.5) {
   446→        this.applyDamage({ kind: 'ground', amount: 5 + this.speed() });
   447→      }
   448→      this.grounded = true;
   449→    } else this.grounded = false;
   450→  }
   451→
   452→  waterlineY(): number {
   453→    return this.position.y;
   454→  }

   456→  // ---------- повреждения ----------
   457→  applyDamage(d: DamageInput) {
   458→    if (this.sunk || this.sinking) return;
   459→    const rng = Math.random;
   460→    switch (d.kind) {
   461→      case 'ball': {
   462→        this.hullHP -= d.amount;
   463→        if (d.hitPoint && d.hitPoint.y < 0.45 + this.position.y) this.addLeak(0.06 + rng() * 0.05);
   464→        if (rng() < 0.3) this.crew = Math.max(0, this.crew - 1);
   465→        // случайный «рикошет в рангоут» — редкий, основную работу делают
   466→        // прицельные попадания из CombatSystem (mast hits)
   467→        if (rng() < 0.10) this.breakMast();
   468→        break;
   469→      }
   470→      case 'chain': {
   471→        this.sailHP = Math.max(0, this.sailHP - d.amount);
   472→        if (rng() < 0.25) this.crew = Math.max(0, this.crew - 1);
   473→        break;
   474→      }
   475→      case 'grape': {
   476→        this.crew = Math.max(0, this.crew - Math.round(d.amount));
   477→        this.hullHP -= d.amount * 0.4;
   478→        break;
   479→      }
   480→      case 'fire': {
   481→        this.hullHP -= d.amount;
   482→        if (rng() < 0.35 && !this.onFire) { this.onFire = true; this.fireT = 0; }
   483→        if (rng() < 0.08) { this.addLeak(0.03); }
   484→        break;
   485→      }
   486→      case 'ram': {
   487→        this.hullHP -= d.amount;
   488→        if (rng() < 0.5) this.addLeak(0.04);
   489→        break;
   490→      }
   491→      case 'ground': this.hullHP -= d.amount * 0.5; break;
   492→    }
   493→    if (this.hullHP <= 0 && !this.sinking) this.startSinking();
   494→    if (!this.isPlayer && !this.noSurrender && !this.surrendered && this.hullHP < this.hullMax * 0.34) {
   495→      this.surrendered = true; this.sailTarget = 0; this.onSurrender?.(this);
   496→    }
   497→  }

   499→  /** Урон в КОНКРЕТНУЮ мачту (прицельное попадание ядром/книппелем) */
   500→  damageMast(idx: number, amount: number): boolean {
   501→    if (idx < 0 || idx >= this.mastHP.length) return false;
   502→    if (this.mastFall[idx] >= 0) return false; // уже сломана
   503→    this.mastHP[idx] -= amount;
   504→    if (this.mastHP[idx] <= 0) { this.breakMast(idx); return true; }
   505→    return false;
   506→  }
   507→
   508→  addLeak(rate: number) {
   509→    this.leakRate = Math.min(1.8, this.leakRate + rate);
   510→    if (this.leakRate < 0) this.leakRate = 0;
   511→  }

   513→  /** Пробоина в борту: декаль + течь (ниже ватерлинии) + кренящий момент */
   514→  addHole(localPos: THREE.Vector3, belowWL: boolean) {
   515→    const side = localPos.x >= 0 ? 1 : -1;
   516→    // тёмная «рваная» дыра: круг + обломки досок вокруг
   517→    const g = new THREE.CircleGeometry(0.34 + Math.random() * 0.14, 9);
   518→    const mat = new THREE.MeshStandardMaterial({ color: 0x120d08, roughness: 1 });
   519→    const mesh = new THREE.Mesh(g, mat);
   520→    const w = Math.max(0.3, this.halfWidthAt(localPos.z) * 0.985);
   521→    mesh.position.set(side * w, localPos.y, localPos.z);
   522→    mesh.rotation.y = side * Math.PI / 2;
   523→    mesh.rotation.x = (Math.random() - 0.5) * 0.5;
   524→    this.visual.group.add(mesh);
   525→    this.hullHoles.push({ side, belowWL, mesh });
   526→    if (belowWL) {
   527→      this.addLeak(0.16 + Math.random() * 0.10);
   528→      this.leakBias = clamp(this.leakBias + side * 0.35, -1, 1);
   529→    }
   530→  }

   532→  /** Затопление: крен в сторону пробоин + вода в трюме утяжеляет */
   533→  private floodRoll(m: number) {
   534→    if (this.water <= 0 || this.sinking) return;
   535→    const frac = this.water / this.holdCap;
   536→    this.torque.z += this.leakBias * frac * m * 0.06;
   537→  }

   539→  breakMast(idx?: number) {
   540→    let i = idx;
   541→    if (i === undefined || i < 0 || i >= this.visual.masts.length || this.mastFall[i] >= 0) {
   542→      // без указания — случайная целая мачта
   543→      const intact = this.mastFall.map((f, k) => f < 0 ? k : -1).filter(k => k >= 0);
   544→      if (!intact.length) return;
   545→      i = intact[Math.floor(Math.random() * intact.length)];
   546→    }
   547→    this.mastFall[i] = 0;
   548→    this.mastsBroken = this.mastFall.filter(f => f >= 0).length;
   549→    this.sailHP = Math.max(0, this.sailHP - 30);
   550→    if (this.sailTarget > 0) this.sailTarget = Math.max(0, this.sailTarget - 1);
   551→  }

   553→  /** Сброс ВСЕХ повреждений (полигон/возрождение): корпус, мачты,
   554→   *  паруса, пробоины, течь, пожар, экипаж */
   555→  clearDamage() {
   556→    this.hullHP = this.hullMax;
   557→    this.sailHP = this.preset.sailHP;
   558→    this.water = 0;
   559→    this.leakRate = 0;
   560→    this.leakBias = 0;
   561→    this.onFire = false;
   562→    this.fireT = 0;
   563→    this.sinking = false;
   564→    this.sinkT = 0;
   565→    this.sunk = false;
   566→    this.surrendered = false;
   567→    this.mastsBroken = 0;
   568→    this.mastAnim = -1;
   569→    for (let i = 0; i < this.visual.masts.length; i++) {
   570→      this.mastHP[i] = this.mastMaxHP;
   571→      this.mastFall[i] = -1;
   572→      const m = this.visual.masts[i];
   573→      m.rotation.set(0, 0, 0);
   574→      m.position.y = 0;
   575→    }
   576→    for (const h of this.hullHoles) {
   577→      this.visual.group.remove(h.mesh);
   578→      h.mesh.geometry.dispose();
   579→      (h.mesh.material as THREE.Material).dispose();
   580→    }
   581→    this.hullHoles = [];
   582→    this.crew = Math.round(this.preset.maxCrew * 0.8);
   583→    this.crew3d.reset();
   584→    this.visual.group.visible = true;
   585→    this.velocity.set(0, 0, 0);
   586→    this.angular.set(0, 0, 0);
   587→    this.sailTarget = 0;
   588→    this.rudder = 0;
   589→    this.rudderTarget = 0;
   590→  }

   592→  startSinking() {
   593→    if (this.sinking) return;
   594→    this.sinking = true;
   595→    this.sinkT = 0;
   596→    this.onFire = false;
   597→  }

   599→  /** Действие команды (R): пожар -> течь -> ремонт */
   600→  crewAction(dt: number): string | null {
   601→    if (this.onFire) { this.fireT += dt * 3; if (this.fireT > 4) { this.onFire = false; this.fireT = 0; return 'Пожар потушен'; } return 'Тушение пожара...'; }
   602→    if (this.water > 0.01) {
   603→      const pump = 0.05 + this.crew * 0.004;
   604→      this.water = Math.max(0, this.water - pump * dt);
   605→      this.leakRate = Math.max(0, this.leakRate - 0.004 * dt);
   606→      return this.water > 0.05 ? 'Откачка воды...' : 'Трюм осушён';
   607→    }
   608→    if (this.hullHP < this.hullMax && this.planks > 0) {
   609→      this.hullHP = Math.min(this.hullMax, this.hullHP + 0.9 * dt);
   610→      this.planks -= dt * 0.15;
   611→      return 'Ремонт корпуса...';
   612→    }
   613→    return null;
   614→  }

   616→  // ---------- визуал ----------
   617→  updateVisual(time: number, dt: number, particles: Particles) {
   618→    const v = this.visual;
   619→    v.group.position.copy(this.position);
   620→    v.group.quaternion.copy(this.quaternion);
   621→
   622→    // паруса: количество поставленных парусов соответствует ФАКТИЧЕСКОЙ
   623→    // СКОРОСТИ — экипаж ставит паруса по мере разгона и убирает по мере
   624→    // остановки; уровень W/S задаёт максимум. Рея брасопится по ветру,
   625→    // «пузо» — по кажущемуся ветру. Паруса СЛОМАННОЙ мачты обвисают.
   626→    const awa = this.apparentWindAngle();
   627→    const polar = Ship.polar(awa);
   628→    const windClamp = clamp(this.wind.speed / 9, 0.3, 1.4);
   629→    // скорость, достижимая под полным парусным вооружением в этот ветер
   630→    // (корень из тяги — сопротивление квадратично): ~0.85·maxSpeed
   631→    const windPow = Math.pow(clamp(this.wind.speed / 8.0, 0.05, 1.6), 1.6);
   632→    const attainable = this.preset.maxSpeed * 0.85 * Math.sqrt(clamp(polar * windPow, 0, 1.5));
   633→    // в левентик/штиль паруса остаются на мачтах и трепещут (не убираются
   634→    // механически) — убираются только когда есть чем идти, но нет хода
   635→    const spdSetFrac = attainable > 1.5 ? clamp(this.speed() / attainable, 0, 1) : 1;
   636→    const setTotal = Math.min(this.sailFrac, spdSetFrac);
   637→    // локальный вектор «откуда дует» (x — борт, z — нос): ψ = φ + курс
   638→    const psi = this.wind.from + this.heading;
   639→    const wlX = Math.cos(psi); // >0 — ветер с порта (левый борт)
   640→    // цель брасопки: рея поперёк ветра (нормаль паруса по ветру), ±55°
   641→    let bt = Math.atan2(wlX, Math.sin(psi)) - Math.PI;
   642→    bt = Math.atan2(Math.sin(bt), Math.cos(bt)); // кратчайшая дуга
   643→    const braceTarget = clamp(bt, -0.96, 0.96);
   644→    const bellyBase = (0.35 + 0.95 * windClamp) * (0.15 + 0.85 * polar);
   645→    const flutterBase = 0.012 * windClamp * (1 + (1 - polar) * 2.2);
   646→    for (const s of v.sails.sails) {
   647→      // мачта-носитель сломана → парус безжизненно обвисает
   648→      const mastDead = s.mastIndex !== undefined && this.mastFall[s.mastIndex] >= 0;
   649→      // доля постановки этого паруса при текущей скорости хода
   650→      const setFrac = mastDead ? 0 : smoothstep(s.setAt - 0.38, s.setAt, setTotal);
   651→      s.furl += ((1 - setFrac) - s.furl) * Math.min(1, dt * (mastDead ? 2.2 : 1.1));
   652→      const sf = 1 - s.furl;
   653→      s.mat.uniforms.uTime.value = time;
   654→      s.mat.uniforms.uFurl.value = s.furl;
   655→      s.mat.uniforms.uDamage.value = 1 - this.sailHP / this.preset.sailHP;
   656→      s.mat.uniforms.uFlutter.value = flutterBase * (0.25 + 0.75 * sf);
   657→      if (s.square && s.brace) {
   658→        // реи разворачиваются медленно — как будто работают брашпили
   659→        s.braceAngle += ((mastDead ? 0 : braceTarget) - s.braceAngle) * Math.min(1, dt * 0.5);
   660→        s.brace.rotation.y = s.braceAngle;
   661→        s.mat.uniforms.uBelly.value = mastDead ? 0 : bellyBase * sf;
   662→      } else {
   663→        // кливер: пузо в подветренную сторону (локальный Z → борт)
   664→        s.mat.uniforms.uBelly.value = mastDead ? 0 : bellyBase * sf * (-wlX);
   665→      }
   666→    }
   667→
   668→    // флаг — по ветру
   669→    const wf = Math.atan2(-Math.cos(this.wind.from), -Math.sin(this.wind.from));
   670→    v.flag.rotation.y = wf - this.heading + Math.PI;
   671→    v.flagMat.uniforms.uTime.value = time;
   672→
   673→    // штурвал крутится от руля: D (руль вправо) — по часовой от рулевого
   674→    v.wheel.rotation.z = this.rudder * 2.2;
   675→
   676→    // падение мачт: каждая валится вперёд-вбок со своим наклоном,
   677→    // реи и паруса едут вместе с мачтой (они её потомки)
   678→    for (let i = 0; i < v.masts.length; i++) {
   679→      const f = this.mastFall[i];
   680→      if (f >= 0 && f < 1) {
   681→        this.mastFall[i] = Math.min(1, f + dt * 0.42);
   682→        const k = this.mastFall[i];
   683→        const lean = (i % 2 === 0 ? 1 : -1) * 0.3;
   684→        v.masts[i].rotation.x = k * 1.25;
   685→        v.masts[i].rotation.z = k * lean;
   686→        v.masts[i].position.y = -k * 1.6;
   687→      }
   688→    }
   689→
   690→    // пенный воротник по ватерлинии
   691→    if (!this.sunk) {
   692→      const wy = waterHeightAt(this.position.x, this.position.z, time);
   693→      this.foam.mesh.position.set(this.position.x, wy + 0.28, this.position.z);
   694→      this.foam.mesh.rotation.y = this.heading;
   695→      const spdFrac = clamp(this.speed() / this.preset.maxSpeed, 0, 1);
   696→      this.foam.update(time, spdFrac, this.euler.z);
   697→    } else {
   698→      this.foam.mesh.visible = false;
   699→    }
   700→
   701→    // кильватерный след
   702→    const sternW = this.tmpV.set(0, 0, -this.preset.length * 0.48)
   703→      .applyQuaternion(this.quaternion).add(this.position);
   704→    const spdFrac = clamp(this.speed() / this.preset.maxSpeed, 0, 1);
   705→    if (!this.sunk && !this.grounded) {
   706→      this.wake.push(sternW, spdFrac, dt, time);
   707→    }
   708→    this.wake.update(time);
   709→
   710→    // носовой бурун: брызги при ударе носа о волну
   711→    if (!this.sunk) {
   712→      const bowW = this.tmpV2.set(0, -this.preset.draft * 0.3, this.preset.length * 0.46)
   713→        .applyQuaternion(this.quaternion).add(this.position);
   714→      const waveY = waterHeightAt(bowW.x, bowW.z, time);
   715→      const relV = this.velocity.y - (waveY - this.lastStern.y) * 0; // грубая оценка
   716→      const impact = (waveY - bowW.y) * (0.3 + spdFrac);
   717→      if (impact > 0.55 && Math.random() < 0.6) {
   718→        particles.splash(bowW.x, waveY, bowW.z, 0.5 + spdFrac * 0.9);
   719→      }
   720→      if (spdFrac > 0.35 && Math.random() < spdFrac * 0.5) {
   721→        particles.spawn(bowW.x + (Math.random() - 0.5) * 2, waveY + 0.1, bowW.z + (Math.random() - 0.5) * 2,
   722→          (Math.random() - 0.5) * 2, 1.5 + Math.random() * 2.5 * spdFrac, (Math.random() - 0.5) * 2,
   723→          0.5, 0.3 + Math.random() * 0.35, 0, 0.4, 10);
   724→      }
   725→      // крен: всплески у борта
   726→      const roll = this.euler.z;
   727→      this.rollSmooth += (roll - this.rollSmooth) * Math.min(1, dt * 3);
   728→      if (Math.abs(this.rollSmooth) > 0.28 && Math.random() < 0.3) {
   729→        const side = Math.sign(this.rollSmooth) * this.preset.beam * 0.5;
   730→        const px = this.position.x + Math.cos(this.heading) * side;
   731→        const pz = this.position.z - Math.sin(this.heading) * side;
   732→        particles.splash(px, waterHeightAt(px, pz, time), pz, 0.5);
   733→      }
   734→    }
   735→
   736→    // экипаж: анимация, потери (трупы/разрывы), эвакуация при потоплении
   737→    this.crew3d.update(dt, time, {
   738→      crew: this.crew, maxCrew: this.preset.maxCrew,
   739→      sinking: this.sinking, sunk: this.sunk,
   740→      position: this.position, quaternion: this.quaternion,
   741→    });
   742→  }

   744→  /** Сдвиг мира (floating origin) */
   745→  shiftWorld(delta: THREE.Vector3) {
   746→    this.position.add(delta);
   747→    this.wake.shiftWorld(delta);
   748→  }

   750→  dispose() {
   751→    this.visual.group.traverse((o) => {
   752→      const mesh = o as THREE.Mesh;
   753→      if (mesh.geometry) mesh.geometry.dispose();
   754→    });
   755→  }
