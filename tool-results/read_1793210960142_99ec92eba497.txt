    22→/** Расстояние от точки (px,pz) до отрезка (ax,az)-(bx,bz) в плоскости XZ */
    23→function distPointSeg2D(px: number, pz: number, ax: number, az: number, bx: number, bz: number): number {
    24→  const dx = bx - ax, dz = bz - az;
    25→  const len2 = dx * dx + dz * dz;
    26→  let t = len2 > 1e-9 ? ((px - ax) * dx + (pz - az) * dz) / len2 : 0;
    27→  t = Math.max(0, Math.min(1, t));
    28→  const cx = ax + dx * t, cz = az + dz * t;
    29→  return Math.hypot(px - cx, pz - cz);
    30→}

   158→  /** Книппель: два полушара ядра на цепи — кувыркается в полёте */
   159→  private makeChainMesh(): THREE.Group {
   160→    if (!this.chainGeoCache) {
   161→      // прототип: ось X — ось полёта, полушария по краям, цепь между ними
   162→      const proto = new THREE.Group();
   163→      const ballMat = new THREE.MeshStandardMaterial({ color: 0x1b1b20, roughness: 0.5, metalness: 0.72 });
   164→      const linkMat = new THREE.MeshStandardMaterial({ color: 0x454b54, roughness: 0.55, metalness: 0.85 });
   165→      const R = 0.34;
   166→      // полусфера (z≥0) + плоская крышка
   167→      const hemi = new THREE.SphereGeometry(R, 12, 8, 0, Math.PI);
   168→      const cap = new THREE.CircleGeometry(R, 12);
   169→      for (const dir of [1, -1] as const) {
   170→        const half = new THREE.Group();
   171→        const dome = new THREE.Mesh(hemi, ballMat);
   172→        dome.rotation.y = dir > 0 ? Math.PI / 2 : -Math.PI / 2; // купол наружу по X
   173→        half.add(dome);
   174→        const c = new THREE.Mesh(cap, ballMat);
   175→        c.rotation.y = dir > 0 ? -Math.PI / 2 : Math.PI / 2; // крышка к центру
   176→        half.add(c);
   177→        half.position.x = dir * 0.62;
   178→        proto.add(half);
   179→      }
   180→      // цепь: 4 звена-тора перпендикулярно друг другу
   181→      for (let i = 0; i < 4; i++) {
   182→        const link = new THREE.Mesh(new THREE.TorusGeometry(0.1, 0.03, 5, 12), linkMat);
   183→        link.position.x = -0.23 + i * 0.155;
   184→        link.rotation.y = i % 2 === 0 ? 0 : Math.PI / 2;
   185→        proto.add(link);
   186→      }
   187→      this.chainGeoCache = proto;
   188→    }
   189→    // копируем прототип (меши шарят геометрию/материал)
   190→    const g = new THREE.Group();
   191→    for (const child of this.chainGeoCache.children) g.add(child.clone());
   192→    return g;
   193→  }

   206→  /** Мировое направление прицела */
   207→  aimDirection(out: THREE.Vector3): THREE.Vector3 {
   208→    const heading = this.player.heading;
   209→    const side = this.aim.side; // 1 = правый борт (штирборт = мир −X при курсе 0)
   210→    // штирборт: (−cos h, 0, sin h); порт: зеркально
   211→    const baseAz = Math.atan2(-side * Math.cos(heading), side * Math.sin(heading));
   212→    // baseAz: направление правого/левого борта
   213→    const az = baseAz + this.aim.azimuth;
   214→    const cosE = Math.cos(this.aim.elevation);
   215→    out.set(Math.sin(az) * cosE, Math.sin(this.aim.elevation), Math.cos(az) * cosE);
   216→    return out.normalize();
   217→  }

   228→  fireOne() {
   229→    if (!this.aim.active) return;
   230→    const guns = this.aim.side === 1 ? this.player.visual.gunsStarboard : this.player.visual.gunsPort;
   231→    for (let i = 0; i < guns.length; i++) {
   232→      const idx = this.gunIndex(this.aim.side, i);
   233→      if (this.reloads[idx] <= 0) {
   234→        this.fireGun(idx, guns[i]);
   235→        return;
   236→      }
   237→    }
   238→    this.emit(GameEvents.MESSAGE, { text: 'Орудия перезаряжаются!' });
   239→  }

   241→  fireBroadside() {
   242→    if (!this.aim.active) return;
   243→    const guns = this.aim.side === 1 ? this.player.visual.gunsStarboard : this.player.visual.gunsPort;
   244→    let fired = 0;
   245→    for (let i = 0; i < guns.length; i++) {
   246→      const idx = this.gunIndex(this.aim.side, i);
   247→      if (this.reloads[idx] <= 0) {
   248→        // лёгкий разброс залпа
   249→        const az0 = this.aim.azimuth, el0 = this.aim.elevation;
   250→        this.aim.azimuth += (Math.random() - 0.5) * 0.06;
   251→        this.fireGun(idx, guns[i], i * 0.05);
   252→        this.aim.azimuth = az0; this.aim.elevation = el0;
   253→        fired++;
   254→      }
   255→    }
   256→    if (!fired) this.emit(GameEvents.MESSAGE, { text: 'Борт перезаряжается!' });
   257→  }

   259→  private fireGun(idx: number, gun: THREE.Object3D, delay = 0) {
   260→    const ship = this.player;
   261→    const info = AMMO_INFO[this.ammo];
   262→    if (ship.ammo[this.ammo] <= 0) {
   263→      this.emit(GameEvents.MESSAGE, { text: `${info.name}: нет боеприпасов!` });
   264→      return;
   265→    }
   266→    ship.ammo[this.ammo]--;
   267→    this.reloads[idx] = this.reloadDuration();
   268→    if (delay > 0) {
   269→      // отложенный выстрел для залпа
   270→      setTimeout(() => this.doFire(gun, info), delay * 1000);
   271→    } else {
   272→      this.doFire(gun, info);
   273→    }
   274→  }

   276→  private doFire(gun: THREE.Object3D, info: typeof AMMO_INFO[AmmoType]) {
   277→    const ship = this.player;
   278→    // позиция дула в МИРОВЫХ координатах (без matrixWorld — floating origin!)
   279→    const sideDir = gun.userData.side > 0 ? 1 : -1;
   280→    this.muzzleWorld.copy(gun.position);
   281→    this.muzzleWorld.x += sideDir * 1.35;
   282→    this.muzzleWorld.y += 0.12;
   283→    this.muzzleWorld.applyQuaternion(ship.quaternion).add(ship.position);
   284→    const h = ship.heading;
   285→
   286→    this.aimDirection(this.aimDir);
   287→    // отдача: крен от залпа (борт, из которого стреляем, поднимается)
   288→    ship.angular.z += sideDir * 0.006 * info.muzzle / 175;
   289→
   290→    for (let p = 0; p < info.pellets; p++) {
   291→      const pr = this.projectiles.find(q => !q.active);
   292→      if (!pr) break;
   293→      pr.active = true;
   294→      pr.pos.copy(this.muzzleWorld);
   295→      pr.prev.copy(this.muzzleWorld);
   296→      pr.type = this.ammo;
   297→      pr.shooter = ship;
   298→      pr.age = 0;
   299→      pr.trailT = 0;
   300→      pr.spin = Math.random() * Math.PI * 2;
   301→      pr.pierced = new Map();
   302→      // картечь — ШИРОКИЙ веер пуль (горизонтальный), остальное — точно
   303→      const spreadH = this.ammo === 'grape' ? 0.075 : 0.008;
   304→      const spreadV = this.ammo === 'grape' ? 0.05 : 0.006;
   305→      const dir = this.aimDir.clone();
   306→      dir.x += (Math.random() - 0.5) * spreadH * 2;
   307→      dir.z += (Math.random() - 0.5) * spreadH * 2;
   308→      dir.y += (Math.random() - 0.5) * spreadV * 2;
   309→      dir.normalize();
   310→      pr.vel.copy(dir).multiplyScalar(info.muzzle * (0.94 + Math.random() * 0.12));
   311→      pr.vel.add(ship.velocity);
   312→    }
   313→    // вспышка выстрела: короткая и компактная (аддитивное свечение)
   314→    this.particles.spawn(this.muzzleWorld.x, this.muzzleWorld.y, this.muzzleWorld.z,
   315→      this.aimDir.x * 7, 1.0, this.aimDir.z * 7, 0.11, 1.15, 2, 2.4, -1.5);
   316→    // пороховой дым: два небольших полупрозрачных клубка у дула,
   317→    // разнесённых по высоте и направлению — не сливаются в одно облако
   318→    for (let s = 0; s < 2; s++) {
   319→      const jx = (Math.random() - 0.5) * 1.1;
   320→      this.particles.smoke(
   321→        this.muzzleWorld.x + this.aimDir.x * (0.3 + s * 0.85) + jx,
   322→        this.muzzleWorld.y + 0.1 + s * 0.3,
   323→        this.muzzleWorld.z + this.aimDir.z * (0.3 + s * 0.85) + jx,
   324→        this.aimDir.x * (1.5 - s * 0.5) + (Math.random() - 0.5) * 1.1 + ship.velocity.x * 0.25,
   325→        0.3 + Math.random() * 0.35,
   326→        this.aimDir.z * (1.5 - s * 0.5) + (Math.random() - 0.5) * 1.1 + ship.velocity.z * 0.25,
   327→        0.95 + Math.random() * 0.3, 0.9 + Math.random() * 0.4);
   328→    }
   329→    this.emit(GameEvents.CANNON_FIRED, { pos: this.muzzleWorld.clone(), power: info.muzzle / 175 });
   330→  }

   342→      pr.type = 'grape';
   343→      pr.shooter = shooter;

   349→      dir.z += (Math.random() - 0.5) * 0.12;
   350→      dir.y += (Math.random() - 0.5) * 0.08;

   357→  /** Залп ИИ ядрами по игроку (с упреждением и разбросом) */
   358→  aiBallVolley(shooter: Ship, target: Ship) {
   359→    const to = new THREE.Vector3().subVectors(target.position, shooter.position);
   360→    const dist = to.length();
   361→    if (dist > 400) return;
   362→    to.normalize();
   363→    const flight = dist / 165;
   364→    const aim = target.position.clone().addScaledVector(target.velocity, flight).sub(shooter.position).normalize();
   365→    const n = 3; // бортовой залп из 3 ядер
   366→    for (let i = 0; i < n; i++) {
   367→      const pr = this.projectiles.find(q => !q.active);
   368→      if (!pr) break;
   369→      pr.active = true;
   370→      pr.pos.copy(shooter.position).addScaledVector(aim, shooter.preset.beam * 0.55);
   371→      pr.prev.copy(pr.pos);
   372→      pr.pos.y += 1.0;
   373→      pr.type = 'ball';
   374→      pr.shooter = shooter;
   375→      pr.age = 0;
   376→      pr.trailT = 0;
   377→      pr.pierced = new Map();
   378→      const dir = aim.clone();
   379→      dir.x += (Math.random() - 0.5) * 0.05;
   380→      dir.y += 0.04 + (Math.random() - 0.5) * 0.03;
   381→      dir.z += (Math.random() - 0.5) * 0.05;
   382→      dir.normalize();
   383→      pr.vel.copy(dir).multiplyScalar(170).add(shooter.velocity);
   384→    }
   385→    // дым и звук
   386→    const muzzle = shooter.position.clone().addScaledVector(aim, shooter.preset.beam * 0.7);
   387→    muzzle.y += 1.2;
   388→    for (let s = 0; s < 2; s++) {
   389→      this.particles.smoke(muzzle.x, muzzle.y, muzzle.z,
   390→        aim.x * 1.6 + (Math.random() - 0.5) * 0.6, 0.3 + Math.random() * 0.25,
   391→        aim.z * 1.6 + (Math.random() - 0.5) * 0.6, 1.05 + Math.random() * 0.25, 1.0 + Math.random() * 0.2);
   392→    }
   393→    this.emit(GameEvents.CANNON_FIRED, { pos: muzzle, power: 0.9 });
   394→  }

   399→    onGroundHit: ((x: number, y: number, z: number, radius: number) => number) | null = null) {
   400→    // перезарядка
   401→    for (let i = 0; i < this.reloads.length; i++) {
   402→      if (this.reloads[i] > 0) this.reloads[i] -= dt;
   403→    }
   404→
   405→    // снаряды
   406→    let count = 0;
   407→    let grapeCount = 0;
   408→    let chainIdx = 0;
   409→    for (const pr of this.projectiles) {
   410→      if (!pr.active) continue;
   411→      pr.age += dt;
   412→      const info = AMMO_INFO[pr.type];
   413→      pr.prev.copy(pr.pos);
   414→      // интеграция с сопротивлением
   415→      const dragF = Math.exp(-info.drag * dt * 3);
   416→      pr.vel.multiplyScalar(dragF);
   417→      pr.vel.y -= 9.81 * dt;
   418→      pr.pos.addScaledVector(pr.vel, dt);
   419→      // книппель кувыркается вокруг оси полёта
   420→      if (pr.type === 'chain') pr.spin += dt * 22;
   421→
   422→      // дымный след — ядро видно в полёте (кроме картечи — её дробин много)
   423→      pr.trailT += dt;
   424→      if (pr.trailT > 0.18 && pr.type !== 'grape') {
   425→        pr.trailT = 0;
   426→        this.particles.smoke(pr.pos.x, pr.pos.y, pr.pos.z, 0, 0.08, 0, 0.42, 0.5);
   427→      }
   428→
   429→      // вода
   430→      const wy = waterHeight(pr.pos.x, pr.pos.z, time);
   431→      if (pr.pos.y < wy) {
   432→        pr.active = false;
   433→        this.particles.splash(pr.pos.x, wy, pr.pos.z, pr.type === 'ball' ? 1.0 : 0.5);
   434→        this.emit(GameEvents.PROJECTILE_SPLASH, { pos: pr.pos.clone() });
   435→        continue;
   436→      }
   437→
   438→      // земля (острова): фугасный удар + повал деревьев
   439→      if (groundHeight && pr.pos.y < groundHeight(pr.pos.x, pr.pos.z)) {
   440→        pr.active = false;
   441→        const r = pr.type === 'ball' || pr.type === 'chain' ? 2.6 : pr.type === 'fire' ? 3.2 : 1.6;
   442→        const trees = onGroundHit ? onGroundHit(pr.pos.x, pr.pos.y, pr.pos.z, r) : 0;
   443→        // пыль/земля + щепа, если деревья сломаны
   444→        this.particles.smoke(pr.pos.x, pr.pos.y + 0.4, pr.pos.z, 0, 0.6, 0, 1.6, 1.2);
   445→        if (trees > 0) this.particles.debris(pr.pos.x, pr.pos.y + 1, pr.pos.z, 10, 1.1);
   446→        this.emit(GameEvents.TERRAIN_HIT, { pos: pr.pos.clone(), trees });
   447→        continue;
   448→      }
   449→
   450→      // цели: ПАРУСА прошиваются насквозь (дыры + урон такелажу,
   451→      // снаряд летит дальше), затем мачты (рангоут перекрывает
   452→      // корпус), корпус — сегментом от prev до pos (быстрое ядро
   453→      // не «проскакивает"); КНИППЕЛЬ пробивает рангоут насквозь:
   454→      // круг его разлёта ШИРЕ корпуса (3.2 м > полуширины) — летя
   455→      // вдоль борта, сносит ВСЕ мачты коридором
   456→      let hit: HitInfo | null = null;
   457→      let piercedMast = false;
   458→      const targets: Ship[] = pr.shooter === this.player ? this.others : [this.player];
   459→      for (const t of targets) {
   460→        if (t.sunk) continue;
   461→        // паруса: картечь дырявит полотно россвелью пробоин, книппель
   462→        // рвёт широкими прорехами, ядро пробивает круглую дыру
   463→        this.sailPass(pr, t);
   464→        const done = pr.type === 'chain' ? (pr.pierced.get(t) ?? []) : [];
   465→        hit = this.hitTest(pr, t, done);
   466→        if (hit) {
   467→          this.applyHit(pr, t, info, hit);
   468→          if (pr.type === 'chain' && hit.mast >= 0) {
   469→            // сквозной пролёт: пометить разорванные мачты, летим дальше
   470→            const masts = pr.pierced.get(t) ?? [];
   471→            for (const mi of this.chainMasts) masts.push(mi);
   472→            pr.pierced.set(t, masts);
   473→            piercedMast = true;
   474→          }
   475→          break;
   476→        }
   477→      }
   478→      if (hit && !piercedMast) { pr.active = false; continue; }
   479→      if (!hit && pr.age > 12) { pr.active = false; continue; }
   480→
   481→      // визуал
   482→      if (pr.type === 'chain') {
   483→        const mesh = this.chainMeshes[chainIdx];
   484→        if (mesh) {
   485→          mesh.visible = true;
   486→          mesh.position.copy(pr.pos);
   487→          // ось полёта → локальная X, затем кувырок вокруг неё
   488→          const vlen = pr.vel.length();
   489→          if (vlen > 0.01) {
   490→            this.chainDummy.position.copy(pr.pos);
   491→            this._vN.copy(pr.vel).divideScalar(vlen);
   492→            this.chainDummy.quaternion.setFromUnitVectors(this.xAxis, this._vN);
   493→            this.chainDummy.rotateX(pr.spin);
   494→            mesh.quaternion.copy(this.chainDummy.quaternion);
   495→          }
   496→          chainIdx++;
   497→        }
   498→      } else if (pr.type === 'grape') {
   499→        // свинцовые пульки — рой МЕЛКИХ шаров (не ядра!)
   500→        if (grapeCount < 192) {
   501→          this.dummy.position.copy(pr.pos);
   502→          this.dummy.updateMatrix();
   503→          this.grapeMesh.setMatrixAt(grapeCount, this.dummy.matrix);
   504→          grapeCount++;
   505→        }
   506→      } else {
   507→        this.dummy.position.copy(pr.pos);
   508→        this.dummy.updateMatrix();
   509→        this.projMesh.setMatrixAt(count, this.dummy.matrix);
   510→        count++;
   511→      }
   512→    }
   513→    this.projMesh.count = count;
   514→    this.projMesh.instanceMatrix.needsUpdate = true;
   515→    this.grapeMesh.count = grapeCount;
   516→    this.grapeMesh.instanceMatrix.needsUpdate = true;
   517→    for (let i = chainIdx; i < this.chainMeshes.length; i++) this.chainMeshes[i].visible = false;
   518→
   519→    // прицеливание: траектория и упреждение
   520→    if (this.aim.active) {
   521→      this.updateAimAssist(time, waterHeight);
   522→    } else {
   523→      this.trajLine.visible = false;
   524→      this.impactRing.visible = false;
   525→      this.leadMarker.visible = false;
   526→    }
   527→  }

   529→  private lprev = new THREE.Vector3();
   530→  private lcur = new THREE.Vector3();
   531→  private invQ = new THREE.Quaternion();
   532→  private hitDir = new THREE.Vector3();
   533→  private pathB = new THREE.Vector3();
   534→  private _vN = new THREE.Vector3();
   535→  /** мачты в круге разлёта книппеля (заполняет hitTest) */
   536→  private chainMasts: number[] = [];
   537→
   538→  /** Проверка попадания: мачты (стоячие, swept) → корпус.
   539→   *  Книппель ловит мачты ВЕСЬ СВОЙ КРУГ (радиус 3.2 м — шире корпуса);
   540→   *  skipMasts — уже разорванные этим снарядом (сквозной полёт). */
   541→  private hitTest(pr: Projectile, t: Ship, skipMasts: number[] = []): HitInfo | null {
   542→    this.invQ.copy(t.quaternion).invert();
   543→    this.lprev.subVectors(pr.prev, t.position).applyQuaternion(this.invQ);
   544→    this.lcur.subVectors(pr.pos, t.position).applyQuaternion(this.invQ);
   545→    // направление полёта в связных координатах цели
   546→    this.hitDir.subVectors(this.lcur, this.lprev);
   547→    if (this.hitDir.lengthSq() < 1e-9) this.hitDir.set(0, 0, 1);
   548→    else this.hitDir.normalize();
   549→
   550→    // --- мачты: вертикальный цилиндр в XZ, путь ядра — отрезок prev→pos ---
   551→    const masts = t.visual.masts;
   552→    this.chainMasts.length = 0;
   553→    const mastR = pr.type === 'chain' ? 3.2 : 0.8; // книппель бьёт кругом разлёта (шире корпуса!)
   554→    let first = -1;
   555→    for (let i = 0; i < masts.length; i++) {
   556→      if (t.mastFall[i] >= 0 || skipMasts.includes(i)) continue;
   557→      const mz = masts[i].position.z;
   558→      const h = ((masts[i].userData.height as number) || 20) - t.preset.draft * 0.3;
   559→      const d = distPointSeg2D(0, mz, this.lprev.x, this.lprev.z, this.lcur.x, this.lcur.z);
   560→      if (d < mastR && Math.max(this.lprev.y, this.lcur.y) > 0.7
   561→        && Math.min(this.lprev.y, this.lcur.y) < h + 0.4) {
   562→        if (first < 0) first = i;
   563→        if (pr.type === 'chain') this.chainMasts.push(i);
   564→      }
   565→    }
   566→    if (first >= 0) {
   567→      return { mast: first, lp: this.lcur.clone(), dir: this.hitDir.clone() };
   568→    }
   569→
   570→    // --- корпус (бокс в связных координатах) ---
   571→    const L = t.preset.length, B = t.preset.beam;
   572→    const c = this.lcur;
   573→    if (Math.abs(c.z) < L * 0.52 && Math.abs(c.x) < B * 0.62
   574→      && c.y > -t.preset.draft * 0.8 && c.y < t.preset.freeboard + L * 0.35) {
   575→      return { mast: -1, lp: c.clone(), dir: this.hitDir.clone() };
   576→    }
   577→    return null;
   578→  }

   580→  private applyHit(pr: Projectile, t: Ship, info: typeof AMMO_INFO[AmmoType], hit: HitInfo) {
   581→    const amount = 0.75 + Math.random() * 0.5;
   582→    const P = this.particles;
   583→    if (hit.mast >= 0) {
   584→      // --- попадание в РАНГОУТ: мачта трещит и ломается ---
   585→      // книппель — специализированный «обрыватель» такелажа: бьёт
   586→      // ВСЕМ КРУГОМ разлёта (ломает несколько мачт за раз),
   587→      // ядро — одну, за 2-3 попадания
   588→      if (pr.type === 'chain') {
   589→        const hitSet = this.chainMasts.length ? this.chainMasts : [hit.mast];
   590→        for (const mi of hitSet) {
   591→          const dmg = mi === hit.mast ? 30 + Math.random() * 12 : 22 + Math.random() * 10;
   592→          const broke = t.damageMast(mi, dmg);
   593→          t.applyDamage({ kind: 'chain', amount: info.dmgSail * amount * (mi === hit.mast ? 1 : 0.7) });
   594→          if (broke) this.emit(GameEvents.MAST_BROKEN, { target: t, mast: mi, pos: pr.pos.clone() });
   595→        }
   596→      } else {
   597→        const dmg = pr.type === 'ball' ? 11 + Math.random() * 8
   598→          : pr.type === 'fire' ? 6 : 1.6;
   599→        const broke = t.damageMast(hit.mast, dmg);
   600→        if (pr.type === 'grape') t.applyDamage({ kind: 'grape', amount: info.dmgCrew * amount });
   601→        if (broke) this.emit(GameEvents.MAST_BROKEN, { target: t, mast: hit.mast, pos: pr.pos.clone() });
   602→      }
   603→      // щепа летит вдоль мачты
   604→      for (let i = 0; i < 9; i++) {
   605→        P.spawn(pr.pos.x, pr.pos.y + i * 0.35, pr.pos.z,
   606→          (Math.random() - 0.5) * 7, 2 + Math.random() * 5, (Math.random() - 0.5) * 7,
   607→          0.8, 0.3, 0, 0.6, 10);
   608→      }
   609→      P.smoke(pr.pos.x, pr.pos.y, pr.pos.z, 0, 0.8, 0, 1.0, 1.3);
   610→    } else {
   611→      // --- попадание в КОРПУС: пробоина, у ватерлинии/ниже — течь ---
   612→      // (порог 0.6 м: при качке «у самой воды» дыры также дают течь)
   613→      const belowWL = hit.lp.y < 0.6;
   614→      // --- поражение ЭКИПАЖА вдоль пути снаряда через палубу ---
   615→      // картечь косят палубу (радиус широкий), ядро/зажигательное
   616→      // разрывает тех, кто на линии полёта
   617→      this.pathB.copy(hit.lp).addScaledVector(hit.dir, t.preset.length * 1.25);
   618→      let crewKilled = 0;
   619→      if (pr.type === 'grape') {
   620→        crewKilled = t.crew3d.killNearPath(hit.lp, this.pathB, 'grape', 2, 2.4,
   621→          t.position, t.quaternion);
   622→      } else if (pr.type === 'ball' || pr.type === 'fire') {
   623→        crewKilled = t.crew3d.killNearPath(hit.lp, this.pathB, 'ball', 2, 1.35,
   624→          t.position, t.quaternion);
   625→      }
   626→      if (pr.type === 'grape') {
   627→        // потери команды согласованы с убитыми фигурами
   628→        const per = Math.max(1, t.preset.maxCrew / Math.max(1, t.crew3d.total));
   629→        t.applyDamage({ kind: 'grape', amount: Math.max(0.6, crewKilled * per * 0.9) });
   630→      } else {
   631→        t.applyDamage({
   632→          kind: pr.type === 'fire' && Math.random() < info.fireChance ? 'fire' : pr.type,
   633→          amount: info.dmgHull * amount,
   634→          hitPoint: pr.pos.clone(),
   635→        });
   636→      }
   637→      if (pr.type === 'chain') t.applyDamage({ kind: 'chain', amount: info.dmgSail * amount });
   638→      // декаль пробоины (ядро и зажигательное)
   639→      if (pr.type === 'ball' || pr.type === 'fire') t.addHole(hit.lp, belowWL);
   640→      // щепки и дым
   641→      for (let i = 0; i < 6; i++) {
   642→        P.spawn(pr.pos.x, pr.pos.y, pr.pos.z,
   643→          (Math.random() - 0.5) * 6, 2 + Math.random() * 4, (Math.random() - 0.5) * 6,
   644→          0.7, 0.28, 0, 0.6, 10);
   645→      }
   646→      P.smoke(pr.pos.x, pr.pos.y, pr.pos.z, 0, 0.7, 0, 0.95, 1.2);
   647→    }
   648→    this.emit(GameEvents.PROJECTILE_HIT, { pos: pr.pos.clone(), target: t });
   649→  }

   651→  private updateAimAssist(time: number, waterHeight: (x: number, z: number, t: number) => number) {
   652→    const ship = this.player;
   653→    const info = AMMO_INFO[this.ammo];
   654→    const start = new THREE.Vector3().copy(ship.position);
   655→    start.y += ship.preset.freeboard * 0.55;
   656→    const vel = this.aimDirection(new THREE.Vector3()).multiplyScalar(info.muzzle).add(ship.velocity);
   657→
   658→    // симуляция траектории
   659→    const attr = this.trajLine.geometry.getAttribute('position') as THREE.BufferAttribute;
   660→    const arr = attr.array as Float32Array;
   661→    let px = start.x, py = start.y, pz = start.z;
   662→    let vx = vel.x, vy = vel.y, vz = vel.z;
   663→    const stp = 0.06;
   664→    let range = 0;
   665→    let i = 0;
   666→    for (; i < 48; i++) {
   667→      arr[i * 3] = px; arr[i * 3 + 1] = py; arr[i * 3 + 2] = pz;
   668→      for (let s = 0; s < 4; s++) {
   669→        const d = Math.exp(-info.drag * stp * 3);
   670→        vx *= d; vy *= d; vz *= d;
   671→        vy -= 9.81 * stp;
   672→        px += vx * stp; py += vy * stp; pz += vz * stp;
   673→        range += Math.hypot(vx, vy, vz) * stp;
   674→        if (py < waterHeight(px, pz, time)) break;
   675→      }
   676→      if (py < waterHeight(px, pz, time)) { i++; if (i < 48) { arr[i * 3] = px; arr[i * 3 + 1] = Math.max(py, waterHeight(px, pz, time)); arr[i * 3 + 2] = pz; } break; }
   677→    }
   678→    attr.needsUpdate = true;
   679→    this.trajLine.geometry.setDrawRange(0, Math.max(2, i));
   680→    this.aim.range = range;
   681→
   682→    // кольцо в точке падения
   683→    if (i > 0 && py < waterHeight(px, pz, time)) {
   684→      const wy = waterHeight(px, pz, time);
   685→      this.impactRing.position.set(px, wy + 0.3, pz);
   686→      this.impactRing.scale.setScalar(1 + Math.sin(time * 6) * 0.12);
   687→      this.impactRing.visible = true;
   688→    } else {
   689→      this.impactRing.visible = false;
   690→    }
   691→
   692→    // упреждение: ближайшая цель в секторе
   693→    const dir = this.aimDirection(new THREE.Vector3());
   694→    this.aim.target = null;
   695→    this.aim.targetLead = null;
   696→    let best: Ship | null = null;
   697→    let bestScore = Infinity;
   698→    const toT = new THREE.Vector3();
   699→    for (const t of this.others) {
   700→      if (t.sunk || t.sinking) continue;
   701→      toT.subVectors(t.position, ship.position);
   702→      const dist = toT.length();
   703→      if (dist > 900) continue;
   704→      toT.normalize();
   705→      const dot = toT.dot(dir);
   706→      if (dot < 0.92) continue; // в пределах ~23° от прицела
   707→      const score = dist * (2 - dot);
   708→      if (score < bestScore) { bestScore = score; best = t; }
   709→    }
   710→    if (best) {
   711→      const dist = best.position.distanceTo(ship.position);
   712→      const flight = dist / info.muzzle;
   713→      const lead = new THREE.Vector3().copy(best.position).addScaledVector(best.velocity, flight);
   714→      lead.y += 1.5;
   715→      this.aim.target = best;
   716→      this.aim.targetLead = lead;
   717→      this.leadMarker.position.copy(lead);
   718→      this.leadMarker.rotation.y = time * 2;
   719→      this.leadMarker.scale.setScalar(1 + Math.sin(time * 5) * 0.15);
   720→      this.leadMarker.visible = true;
   721→    } else {
   722→      this.leadMarker.visible = false;
   723→    }
   724→  }

   726→  /** Готовность орудий (0..1) для HUD */
   727→  readiness(): number {
   728→    let ready = 0;
   729→    for (const r of this.reloads) if (r <= 0) ready++;
   730→    return ready / this.reloads.length;
   731→  }
