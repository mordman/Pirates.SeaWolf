    22→/** Расстояние от точки (px,pz) до отрезка (ax,az)-(bx,bz) в плоскости XZ */
    23→function distPointSeg2D(px: number, pz: number, ax: number, az: number, bx: number, bz: number): number {
    24→  const dx = bx - ax, dz = bz - az;
    25→  const len2 = dx * dx + dz * dz;
    26→  let t = len2 > 1e-9 ? ((px - ax) * dx + (pz - az) * dz) / len2 : 0;
    27→  t = Math.max(0, Math.min(1, t));
    28→  const cx = ax + dx * t, cz = az + dz * t;
    29→  return Math.hypot(px - cx, pz - cz);
    30→}

   158→  /** Книппель: два полушара ядра на цепи — кувыркается в полёте */
   159→  private makeChainMesh(): THREE.Group {
   160→    if (!this.chainGeoCache) {
   161→      // прототип: ось X — ось полёта, полушария по краям, цепь между ними
   162→      const proto = new THREE.Group();
   163→      const ballMat = new THREE.MeshStandardMaterial({ color: 0x1b1b20, roughness: 0.5, metalness: 0.72 });
   164→      const linkMat = new THREE.MeshStandardMaterial({ color: 0x454b54, roughness: 0.55, metalness: 0.85 });
   165→      const R = 0.34;
   166→      // полусфера (z≥0) + плоская крышка
   167→      const hemi = new THREE.SphereGeometry(R, 12, 8, 0, Math.PI);
   168→      const cap = new THREE.CircleGeometry(R, 12);
   169→      for (const dir of [1, -1] as const) {
   170→        const half = new THREE.Group();
   171→        const dome = new THREE.Mesh(hemi, ballMat);
   172→        dome.rotation.y = dir > 0 ? Math.PI / 2 : -Math.PI / 2; // купол наружу по X
   173→        half.add(dome);
   174→        const c = new THREE.Mesh(cap, ballMat);
   175→        c.rotation.y = dir > 0 ? -Math.PI / 2 : Math.PI / 2; // крышка к центру
   176→        half.add(c);
   177→        half.position.x = dir * 0.62;
   178→        proto.add(half);
   179→      }
   180→      // цепь: 4 звена-тора перпендикулярно друг другу
   181→      for (let i = 0; i < 4; i++) {
   182→        const link = new THREE.Mesh(new THREE.TorusGeometry(0.1, 0.03, 5, 12), linkMat);
   183→        link.position.x = -0.23 + i * 0.155;
   184→        link.rotation.y = i % 2 === 0 ? 0 : Math.PI / 2;
   185→        proto.add(link);
   186→      }
   187→      this.chainGeoCache = proto;
   188→    }
   189→    // копируем прототип (меши шарят геометрию/материал)
   190→    const g = new THREE.Group();
   191→    for (const child of this.chainGeoCache.children) g.add(child.clone());
   192→    return g;
   193→  }

   206→  /** Мировое направление прицела */
   207→  aimDirection(out: THREE.Vector3): THREE.Vector3 {
   208→    const heading = this.player.heading;
   209→    const side = this.aim.side; // 1 = правый борт (штирборт = мир −X при курсе 0)
   210→    // штирборт: (−cos h, 0, sin h); порт: зеркально
   211→    const baseAz = Math.atan2(-side * Math.cos(heading), side * Math.sin(heading));
   212→    // baseAz: направление правого/левого борта
   213→    const az = baseAz + this.aim.azimuth;
   214→    const cosE = Math.cos(this.aim.elevation);
   215→    out.set(Math.sin(az) * cosE, Math.sin(this.aim.elevation), Math.cos(az) * cosE);
   216→    return out.normalize();
   217→  }

   228→  fireOne() {
   229→    if (!this.aim.active) return;
   230→    const guns = this.aim.side === 1 ? this.player.visual.gunsStarboard : this.player.visual.gunsPort;
   231→    for (let i = 0; i < guns.length; i++) {
   232→      const idx = this.gunIndex(this.aim.side, i);
   233→      if (this.reloads[idx] <= 0) {
   234→        this.fireGun(idx, guns[i]);
   235→        return;
   236→      }
   237→    }
   238→    this.emit(GameEvents.MESSAGE, { text: 'Орудия перезаряжаются!' });
   239→  }
   240→
   241→  fireBroadside() {
   242→    if (!this.aim.active) return;
   243→    const guns = this.aim.side === 1 ? this.player.visual.gunsStarboard : this.player.visual.gunsPort;
   244→    let fired = 0;
   245→    for (let i = 0; i < guns.length; i++) {
   246→      const idx = this.gunIndex(this.aim.side, i);
   247→      if (this.reloads[idx] <= 0) {
   248→        // лёгкий разброс залпа
   249→        const az0 = this.aim.azimuth, el0 = this.aim.elevation;
   250→        this.aim.azimuth += (Math.random() - 0.5) * 0.06;
   251→        this.fireGun(idx, guns[i], i * 0.05);
   252→        this.aim.azimuth = az0; this.aim.elevation = el0;
   253→        fired++;
   254→      }
   255→    }
   256→    if (!fired) this.emit(GameEvents.MESSAGE, { text: 'Борт перезаряжается!' });
   257→  }

   259→  private fireGun(idx: number, gun: THREE.Object3D, delay = 0) {
   260→    const ship = this.player;
   261→    const info = AMMO_INFO[this.ammo];
   262→    if (ship.ammo[this.ammo] <= 0) {
   263→      this.emit(GameEvents.MESSAGE, { text: `${info.name}: нет боеприпасов!` });
   264→      return;
   265→    }
   266→    ship.ammo[this.ammo]--;
   267→    this.reloads[idx] = this.reloadDuration();
   268→    if (delay > 0) {
   269→      // отложенный выстрел для залпа
   270→      setTimeout(() => this.doFire(gun, info), delay * 1000);
   271→    } else {
   272→      this.doFire(gun, info);
   273→    }
   274→  }

   276→  private doFire(gun: THREE.Object3D, info: typeof AMMO_INFO[AmmoType]) {
   277→    const ship = this.player;
   278→    // позиция дула в МИРОВЫХ координатах (без matrixWorld — floating origin!)
   279→    const sideDir = gun.userData.side > 0 ? 1 : -1;
   280→    this.muzzleWorld.copy(gun.position);
   281→    this.muzzleWorld.x += sideDir * 1.35;
   282→    this.muzzleWorld.y += 0.12;
   283→    this.muzzleWorld.applyQuaternion(ship.quaternion).add(ship.position);
   284→    const h = ship.heading;
   285→
   286→    this.aimDirection(this.aimDir);
   287→    // отдача: крен от залпа (борт, из которого стреляем, поднимается)
   288→    ship.angular.z += sideDir * 0.006 * info.muzzle / 175;
   289→
   290→    for (let p = 0; p < info.pellets; p++) {
   291→      const pr = this.projectiles.find(q => !q.active);
   292→      if (!pr) break;
   293→      pr.active = true;
   294→      pr.pos.copy(this.muzzleWorld);
   295→      pr.prev.copy(this.muzzleWorld);
   296→      pr.type = this.ammo;
   297→      pr.shooter = ship;
   298→      pr.age = 0;
   299→      pr.trailT = 0;
   300→      pr.spin = Math.random() * Math.PI * 2;
   301→      pr.pierced = new Map();
   302→      // картечь — ШИРОКИЙ веер пуль (горизонтальный), остальное — точно
   303→      const spreadH = this.ammo === 'grape' ? 0.075 : 0.008;
   304→      const spreadV = this.ammo === 'grape' ? 0.05 : 0.006;
   305→      const dir = this.aimDir.clone();
   306→      dir.x += (Math.random() - 0.5) * spreadH * 2;
   307→      dir.z += (Math.random() - 0.5) * spreadH * 2;
   308→      dir.y += (Math.random() - 0.5) * spreadV * 2;
   309→      dir.normalize();
   310→      pr.vel.copy(dir).multiplyScalar(info.muzzle * (0.94 + Math.random() * 0.12));
   311→      pr.vel.add(ship.velocity);
   312→    }
   313→    // вспышка выстрела: короткая и компактная (аддитивное свечение)
   314→    this.particles.spawn(this.muzzleWorld.x, this.muzzleWorld.y, this.muzzleWorld.z,
   315→      this.aimDir.x * 7, 1.0, this.aimDir.z * 7, 0.11, 1.15, 2, 2.4, -1.5);
   316→    // пороховой дым: два небольших полупрозрачных клубка у дула,
   317→    // разнесённых по высоте и направлению — не сливаются в одно облако
   318→    for (let s = 0; s < 2; s++) {
   319→      const jx = (Math.random() - 0.5) * 1.1;
   320→      this.particles.smoke(
   321→        this.muzzleWorld.x + this.aimDir.x * (0.3 + s * 0.85) + jx,
   322→        this.muzzleWorld.y + 0.1 + s * 0.3,
   323→        this.muzzleWorld.z + this.aimDir.z * (0.3 + s * 0.85) + jx,
   324→        this.aimDir.x * (1.5 - s * 0.5) + (Math.random() - 0.5) * 1.1 + ship.velocity.x * 0.25,
   325→        0.3 + Math.random() * 0.35,
   326→        this.aimDir.z * (1.5 - s * 0.5) + (Math.random() - 0.5) * 1.1 + ship.velocity.z * 0.25,
   327→        0.95 + Math.random() * 0.3, 0.9 + Math.random() * 0.4);
   328→    }
   329→    this.emit(GameEvents.CANNON_FIRED, { pos: this.muzzleWorld.clone(), power: info.muzzle / 175 });
   330→  }

   333→  aiGrapeBurst(shooter: Ship) {

   335→    const muzzle = shooter.position.clone().addScaledVector(dirTo, shooter.preset.beam * 0.6);
   336→    muzzle.y += 1.2;
   337→    for (let i = 0; i < 8; i++) {
   338→      const pr = this.projectiles.find(q => !q.active);
   339→      if (!pr) break;
   340→      pr.active = true;
   341→      pr.pos.copy(shooter.position).addScaledVector(dirTo, shooter.preset.beam * 0.6);
   342→      pr.prev.copy(pr.pos);
   343→      pr.pos.y += 1.2;
   344→      pr.type = 'grape';
   345→      pr.shooter = shooter;
   346→      pr.age = 0;
   347→      pr.trailT = 0;
   348→      pr.pierced = new Map();
   349→      const dir = dirTo.clone();
   350→      dir.x += (Math.random() - 0.5) * 0.12;
   351→      dir.z += (Math.random() - 0.5) * 0.12;
   352→      dir.y += (Math.random() - 0.5) * 0.08;
   353→      dir.normalize();
   354→      pr.vel.copy(dir).multiplyScalar(100 + Math.random() * 18).add(shooter.velocity);
   355→    }
   356→    this.emit(GameEvents.CANNON_FIRED, { pos: muzzle, power: 0.6 });

   359→  /** Залп ИИ ядрами по игроку (с упреждением и разбросом) */
   360→  aiBallVolley(shooter: Ship, target: Ship) {
   361→    const to = new THREE.Vector3().subVectors(target.position, shooter.position);
   362→    const dist = to.length();
   363→    if (dist > 400) return;
   364→    to.normalize();
   365→    const flight = dist / 165;
   366→    const aim = target.position.clone().addScaledVector(target.velocity, flight).sub(shooter.position).normalize();
   367→    const n = 3; // бортовой залп из 3 ядер
   368→    for (let i = 0; i < n; i++) {
   369→      const pr = this.projectiles.find(q => !q.active);
   370→      if (!pr) break;
   371→      pr.active = true;
   372→      pr.pos.copy(shooter.position).addScaledVector(aim, shooter.preset.beam * 0.55);
   373→      pr.prev.copy(pr.pos);
   374→      pr.pos.y += 1.0;
   375→      pr.type = 'ball';
   376→      pr.shooter = shooter;
   377→      pr.age = 0;
   378→      pr.trailT = 0;
   379→      pr.pierced = new Map();
   380→      const dir = aim.clone();
   381→      dir.x += (Math.random() - 0.5) * 0.05;
   382→      dir.y += 0.04 + (Math.random() - 0.5) * 0.03;
   383→      dir.z += (Math.random() - 0.5) * 0.05;
   384→      dir.normalize();
   385→      pr.vel.copy(dir).multiplyScalar(170).add(shooter.velocity);
   386→    }
   387→    // дым и звук
   388→    const muzzle = shooter.position.clone().addScaledVector(aim, shooter.preset.beam * 0.7);
   389→    muzzle.y += 1.2;
   390→    for (let s = 0; s < 2; s++) {
   391→      this.particles.smoke(muzzle.x, muzzle.y, muzzle.z,
   392→        aim.x * 1.6 + (Math.random() - 0.5) * 0.6, 0.3 + Math.random() * 0.25,
   393→        aim.z * 1.6 + (Math.random() - 0.5) * 0.6, 1.05 + Math.random() * 0.25, 1.0 + Math.random() * 0.2);
   394→    }
   395→    this.emit(GameEvents.CANNON_FIRED, { pos: muzzle, power: 0.9 });
   396→  }

   399→  update(dt: number, time: number, waterHeight: (x: number, z: number, t: number) => number,
   400→    groundHeight: ((x: number, z: number) => number) | null = null,
   401→    onGroundHit: ((x: number, y: number, z: number, radius: number) => number) | null = null) {
   402→    // перезарядка
   403→    for (let i = 0; i < this.reloads.length; i++) {
   404→      if (this.reloads[i] > 0) this.reloads[i] -= dt;
   405→    }
   406→
   407→    // снаряды
   408→    let count = 0;
   409→    let grapeCount = 0;
   410→    let chainIdx = 0;
   411→    for (const pr of this.projectiles) {
   412→      if (!pr.active) continue;
   413→      pr.age += dt;
   414→      const info = AMMO_INFO[pr.type];
   415→      pr.prev.copy(pr.pos);
   416→      // интеграция с сопротивлением
   417→      const dragF = Math.exp(-info.drag * dt * 3);
   418→      pr.vel.multiplyScalar(dragF);
   419→      pr.vel.y -= 9.81 * dt;
   420→      pr.pos.addScaledVector(pr.vel, dt);
   421→      // книппель кувыркается вокруг оси полёта
   422→      if (pr.type === 'chain') pr.spin += dt * 22;
   423→
   424→      // дымный след — ядро видно в полёте (кроме картечи — её дробин много)
   425→      pr.trailT += dt;
   426→      if (pr.trailT > 0.18 && pr.type !== 'grape') {
   427→        pr.trailT = 0;
   428→        this.particles.smoke(pr.pos.x, pr.pos.y, pr.pos.z, 0, 0.08, 0, 0.42, 0.5);
   429→      }
   430→
   431→      // вода
   432→      const wy = waterHeight(pr.pos.x, pr.pos.z, time);
   433→      if (pr.pos.y < wy) {
   434→        pr.active = false;
   435→        this.particles.splash(pr.pos.x, wy, pr.pos.z, pr.type === 'ball' ? 1.0 : 0.5);
   436→        this.emit(GameEvents.PROJECTILE_SPLASH, { pos: pr.pos.clone() });
   437→        continue;
   438→      }
   439→
   440→      // земля (острова): фугасный удар + повал деревьев
   441→      if (groundHeight && pr.pos.y < groundHeight(pr.pos.x, pr.pos.z)) {
   442→        pr.active = false;
   443→        const r = pr.type === 'ball' || pr.type === 'chain' ? 2.6 : pr.type === 'fire' ? 3.2 : 1.6;
   444→        const trees = onGroundHit ? onGroundHit(pr.pos.x, pr.pos.y, pr.pos.z, r) : 0;
   445→        // пыль/земля + щепа, если деревья сломаны
   446→        this.particles.smoke(pr.pos.x, pr.pos.y + 0.4, pr.pos.z, 0, 0.6, 0, 1.6, 1.2);
   447→        if (trees > 0) this.particles.debris(pr.pos.x, pr.pos.y + 1, pr.pos.z, 10, 1.1);
   448→        this.emit(GameEvents.TERRAIN_HIT, { pos: pr.pos.clone(), trees });
   449→        continue;
   450→      }
   451→
   452→      // цели: ПАРУСА прошиваются насквозь (дыры + урон такелажу,
   453→      // снаряд летит дальше), затем мачты (рангоут перекрывает
   454→      // корпус), корпус — сегментом от prev до pos (быстрое ядро
   455→      // не «проскакивает"); КНИППЕЛЬ пробивает рангоут насквозь:
   456→      // круг его разлёта ШИРЕ корпуса (3.2 м > полуширины) — летя
   457→      // вдоль борта, сносит ВСЕ мачты коридором
   458→      let hit: HitInfo | null = null;
   459→      let piercedMast = false;
   460→      const targets: Ship[] = pr.shooter === this.player ? this.others : [this.player];
   461→      for (const t of targets) {
   462→        if (t.sunk) continue;
   463→        // паруса: картечь дырявит полотно россвелью пробоин, книппель
   464→        // рвёт широкими прорехами, ядро пробивает круглую дыру
   465→        this.sailPass(pr, t);
   466→        const done = pr.type === 'chain' ? (pr.pierced.get(t) ?? []) : [];
   467→        hit = this.hitTest(pr, t, done);
   468→        if (hit) {
   469→          this.applyHit(pr, t, info, hit);
   470→          if (pr.type === 'chain' && hit.mast >= 0) {
   471→            // сквозной пролёт: пометить разорванные мачты, летим дальше
   472→            const masts = pr.pierced.get(t) ?? [];
   473→            for (const mi of this.chainMasts) masts.push(mi);
   474→            pr.pierced.set(t, masts);
   475→            piercedMast = true;
   476→          }
   477→          break;
   478→        }
   479→      }
   480→      if (hit && !piercedMast) { pr.active = false; continue; }
   481→      if (!hit && pr.age > 12) { pr.active = false; continue; }
   482→
   483→      // визуал
   484→      if (pr.type === 'chain') {
   485→        const mesh = this.chainMeshes[chainIdx];
   486→        if (mesh) {
   487→          mesh.visible = true;
   488→          mesh.position.copy(pr.pos);
   489→          // ось полёта → локальная X, затем кувырок вокруг неё
   490→          const vlen = pr.vel.length();
   491→          if (vlen > 0.01) {
   492→            this.chainDummy.position.copy(pr.pos);
   493→            this._vN.copy(pr.vel).divideScalar(vlen);
   494→            this.chainDummy.quaternion.setFromUnitVectors(this.xAxis, this._vN);
   495→            this.chainDummy.rotateX(pr.spin);
   496→            mesh.quaternion.copy(this.chainDummy.quaternion);
   497→          }
   498→          chainIdx++;
   499→        }
   500→      } else if (pr.type === 'grape') {
   501→        // свинцовые пульки — рой МЕЛКИХ шаров (не ядра!)
   502→        if (grapeCount < 192) {
   503→          this.dummy.position.copy(pr.pos);
   504→          this.dummy.updateMatrix();
   505→          this.grapeMesh.setMatrixAt(grapeCount, this.dummy.matrix);
   506→          grapeCount++;
   507→        }
   508→      } else {
   509→        this.dummy.position.copy(pr.pos);
   510→        this.dummy.updateMatrix();
   511→        this.projMesh.setMatrixAt(count, this.dummy.matrix);
   512→        count++;
   513→      }
   514→    }
   515→    this.projMesh.count = count;
   516→    this.projMesh.instanceMatrix.needsUpdate = true;
   517→    this.grapeMesh.count = grapeCount;
   518→    this.grapeMesh.instanceMatrix.needsUpdate = true;
   519→    for (let i = chainIdx; i < this.chainMeshes.length; i++) this.chainMeshes[i].visible = false;
   520→
   521→    // прицеливание: траектория и упреждение
   522→    if (this.aim.active) {
   523→      this.updateAimAssist(time, waterHeight);
   524→    } else {
   525→      this.trajLine.visible = false;
   526→      this.impactRing.visible = false;
   527→      this.leadMarker.visible = false;
   528→    }
   529→  }

   531→  private lprev = new THREE.Vector3();
   532→  private lcur = new THREE.Vector3();
   533→  private invQ = new THREE.Quaternion();
   534→  private hitDir = new THREE.Vector3();
   535→  private pathB = new THREE.Vector3();
   536→  private _vN = new THREE.Vector3();
   537→  /** мачты в круге разлёта книппеля (заполняет hitTest) */
   538→  private chainMasts: number[] = [];
   539→  // temporaries for sailPass
   540→  private sqA = new THREE.Vector3();
   541→  private sqB = new THREE.Vector3();
   542→  private sqC0 = new THREE.Vector3();
   543→  private sqE1 = new THREE.Vector3();
   544→  private sqE2 = new THREE.Vector3();
   545→  private sqN = new THREE.Vector3();
   546→  private sqW = new THREE.Vector3();
   547→  private sqP = new THREE.Vector3();
   548→
   549→  /** Прострел ПАРУСОВ насквозь: путь снаряда (prev→pos) против
   550→   *  полотен-квадов (с поворотом брасопки). Попадание — пробоина
   551→   *  (sailAddHole) + урон такелажу; снаряд ЛЕТИТ ДАЛЬШЕ — паруса
   552→   *  не останавливают ни картучь, ни ядро. Возвращает true, если
   553→   *  хотя бы одно полотно прошито (для звука/телеметрии). */
   554→  private sailPass(pr: Projectile, t: Ship): boolean {
   555→    if (t.sinking || t.sunk) return false;
   556→    const sails = t.visual.sails.sails;
   557→    if (!sails.length) return false;
   558→    // путь снаряда в СВЯЗНЫХ координатах цели
   559→    this.invQ.copy(t.quaternion).invert();
   560→    this.sqA.subVectors(pr.prev, t.position).applyQuaternion(this.invQ);
   561→    this.sqB.subVectors(pr.pos, t.position).applyQuaternion(this.invQ);
   562→    let torn = false;
   563→    for (const s of sails) {
   564→      // мачта сломана — парус обвис: не простреливаем
   565→      if (s.mastIndex !== undefined && t.mastFall[s.mastIndex] >= 0) continue;
   566→      // сильно убран (собран у реи) — полотно мало, пропускаем
   567→      if (s.furl > 0.72) continue;
   568→      const q = s.quad;
   569→      // углы полотна в координатах КОРАБЛЯ (+текущая брасопка реи)
   570→      const ba = q.rotates ? s.braceAngle : 0;
   571→      const cb = Math.cos(ba), sb = Math.sin(ba);
   572→      // C0 = uv(0,0), E1 → uv(1,0), E2 → uv(0,1) (параллелограмм)
   573→      const loc = (i: number, out: THREE.Vector3): THREE.Vector3 => {
   574→        const c = q.corners[i];
   575→        // поворот вокруг оси мачты (Y) на угол брасопки
   576→        const rx = c.x * cb + c.z * sb;
   577→        const rz = -c.x * sb + c.z * cb;
   578→        out.set(q.pivot.x + rx, q.pivot.y + c.y, q.pivot.z + rz);
   579→        return out;
   580→      };
   581→      loc(0, this.sqC0);
   582→      loc(1, this.sqE1).sub(this.sqC0);
   583→      loc(3, this.sqE2).sub(this.sqC0);
   584→      // плоскость полотна
   585→      this.sqN.crossVectors(this.sqE1, this.sqE2);
   586→      const nLen2 = this.sqN.lengthSq();
   587→      if (nLen2 < 1e-9) continue;
   588→      const dA = this.sqW.subVectors(this.sqA, this.sqC0).dot(this.sqN);
   589→      const dB = this.sqW.subVectors(this.sqB, this.sqC0).dot(this.sqN);
   590→      if (dA * dB > 0) continue;              // обе точки по одну сторону
   591→      const k = dA / (dA - dB);               // 0..1 — точка пересечения
   592→      if (k < 0 || k > 1) continue;
   593→      this.sqP.lerpVectors(this.sqA, this.sqB, k).sub(this.sqC0);
   594→      // билинейные координаты в параллелограмме: P = u·E1 + v·E2
   595→      const e11 = this.sqE1.dot(this.sqE1), e22 = this.sqE2.dot(this.sqE2);
   596→      const e12 = this.sqE1.dot(this.sqE2);
   597→      const pu = this.sqP.dot(this.sqE1), pv = this.sqP.dot(this.sqE2);
   598→      const det = e11 * e22 - e12 * e12;
   599→      if (Math.abs(det) < 1e-9) continue;
   600→      const u = (pu * e22 - pv * e12) / det;
   601→      const v = (pv * e11 - pu * e12) / det;
   602→      const tri = q.uvs[2].equals(q.uvs[3]);  // латинский/кливер — треугольник
   603→      if (u < -0.02 || u > 1.02 || v < -0.02 || v > 1.02) continue;
   604→      if (tri && u + v > 1.04) continue;
   605→      // === ПОПАДАНИЕ В ПОЛОТНО ===
   606→      torn = true;
   607→      const uu = clamp(u, 0, 1), vv = clamp(v, 0, 1);
   608→      // радиус пробоины (в долях ВЫСОТЫ полотна — см. шейдер uHoles)
   609→      let r: number;
   610→      if (pr.type === 'grape') r = 0.035 + Math.random() * 0.025;   // россыпь малых дыр
   611→      else if (pr.type === 'chain') r = 0.11 + Math.random() * 0.07; // широкая прореха
   612→      else if (pr.type === 'fire') r = 0.045;
   613→      else r = 0.055 + Math.random() * 0.03;                         // ядро
   614→      sailAddHole(s, uu, vv, r);
   615→      // урон такелажу: учесть долю поставленного полотна
   616→      const setFrac = (1 - s.furl) * 0.6 + 0.4;
   617→      const dmg = AMMO_INFO[pr.type].dmgSail * setFrac * (0.8 + Math.random() * 0.4);
   618→      t.damageSails(dmg);
   619→    }
   620→    return torn;
   621→  }
   622→
   623→  /** Проверка попадания: мачты (стоячие, swept) → корпус.
   624→   *  Книппель ловит мачты ВЕСЬ СВОЙ КРУГ (радиус 3.2 м — шире корпуса);
   625→   *  skipMasts — уже разорванные этим снарядом (сквозной полёт). */
   626→  private hitTest(pr: Projectile, t: Ship, skipMasts: number[] = []): HitInfo | null {
   627→    this.invQ.copy(t.quaternion).invert();
   628→    this.lprev.subVectors(pr.prev, t.position).applyQuaternion(this.invQ);
   629→    this.lcur.subVectors(pr.pos, t.position).applyQuaternion(this.invQ);
   630→    // направление полёта в связных координатах цели
   631→    this.hitDir.subVectors(this.lcur, this.lprev);
   632→    if (this.hitDir.lengthSq() < 1e-9) this.hitDir.set(0, 0, 1);
   633→    else this.hitDir.normalize();
   634→
   635→    // --- мачты: вертикальный цилиндр в XZ, путь ядра — отрезок prev→pos ---
   636→    const masts = t.visual.masts;
   637→    this.chainMasts.length = 0;
   638→    const mastR = pr.type === 'chain' ? 3.2 : 0.8; // книппель бьёт кругом разлёта (шире корпуса!)
   639→    let first = -1;
   640→    for (let i = 0; i < masts.length; i++) {
   641→      if (t.mastFall[i] >= 0 || skipMasts.includes(i)) continue;
   642→      const mz = masts[i].position.z;
   643→      const h = ((masts[i].userData.height as number) || 20) - t.preset.draft * 0.3;
   644→      const d = distPointSeg2D(0, mz, this.lprev.x, this.lprev.z, this.lcur.x, this.lcur.z);
   645→      if (d < mastR && Math.max(this.lprev.y, this.lcur.y) > 0.7
   646→        && Math.min(this.lprev.y, this.lcur.y) < h + 0.4) {
   647→        if (first < 0) first = i;
   648→        if (pr.type === 'chain') this.chainMasts.push(i);
   649→      }
   650→    }
   651→    if (first >= 0) {
   652→      return { mast: first, lp: this.lcur.clone(), dir: this.hitDir.clone() };
   653→    }
   654→
   655→    // --- корпус (бокс в связных координатах) ---
   656→    const L = t.preset.length, B = t.preset.beam;
   657→    const c = this.lcur;
   658→    if (Math.abs(c.z) < L * 0.52 && Math.abs(c.x) < B * 0.62
   659→      && c.y > -t.preset.draft * 0.8 && c.y < t.preset.freeboard + L * 0.35) {
   660→      return { mast: -1, lp: c.clone(), dir: this.hitDir.clone() };
   661→    }
   662→    return null;
   663→  }

   665→  private applyHit(pr: Projectile, t: Ship, info: typeof AMMO_INFO[AmmoType], hit: HitInfo) {
   666→    const amount = 0.75 + Math.random() * 0.5;
   667→    const P = this.particles;
   668→    if (hit.mast >= 0) {
   669→      // --- попадание в РАНГОУТ: мачта трещит и ломается ---
   670→      // книппель — специализированный «обрыватель» такелажа: бьёт
   671→      // ВСЕМ КРУГОМ разлёта (ломает несколько мачт за раз),
   672→      // ядро — одну, за 2-3 попадания
   673→      if (pr.type === 'chain') {
   674→        const hitSet = this.chainMasts.length ? this.chainMasts : [hit.mast];
   675→        for (const mi of hitSet) {
   676→          const dmg = mi === hit.mast ? 30 + Math.random() * 12 : 22 + Math.random() * 10;
   677→          const broke = t.damageMast(mi, dmg);
   678→          t.applyDamage({ kind: 'chain', amount: info.dmgSail * amount * (mi === hit.mast ? 1 : 0.7) });
   679→          if (broke) this.emit(GameEvents.MAST_BROKEN, { target: t, mast: mi, pos: pr.pos.clone() });
   680→        }
   681→      } else {
   682→        const dmg = pr.type === 'ball' ? 11 + Math.random() * 8
   683→          : pr.type === 'fire' ? 6 : 1.6;
   684→        const broke = t.damageMast(hit.mast, dmg);
   685→        if (pr.type === 'grape') t.applyDamage({ kind: 'grape', amount: info.dmgCrew * amount });
   686→        if (broke) this.emit(GameEvents.MAST_BROKEN, { target: t, mast: hit.mast, pos: pr.pos.clone() });
   687→      }
   688→      // щепа летит вдоль мачты
   689→      for (let i = 0; i < 9; i++) {
   690→        P.spawn(pr.pos.x, pr.pos.y + i * 0.35, pr.pos.z,
   691→          (Math.random() - 0.5) * 7, 2 + Math.random() * 5, (Math.random() - 0.5) * 7,
   692→          0.8, 0.3, 0, 0.6, 10);
   693→      }
   694→      P.smoke(pr.pos.x, pr.pos.y, pr.pos.z, 0, 0.8, 0, 1.0, 1.3);
   695→    } else {
   696→      // --- попадание в КОРПУС: пробоина, у ватерлинии/ниже — течь ---
   697→      // (порог 0.6 м: при качке «у самой воды» дыры также дают течь)
   698→      const belowWL = hit.lp.y < 0.6;
   699→      // --- поражение ЭКИПАЖА вдоль пути снаряда через палубу ---
   700→      // картечь косят палубу (радиус широкий), ядро/зажигательное
   701→      // разрывает тех, кто на линии полёта
   702→      this.pathB.copy(hit.lp).addScaledVector(hit.dir, t.preset.length * 1.25);
   703→      let crewKilled = 0;
   704→      if (pr.type === 'grape') {
   705→        crewKilled = t.crew3d.killNearPath(hit.lp, this.pathB, 'grape', 2, 2.4,
   706→          t.position, t.quaternion);
   707→      } else if (pr.type === 'ball' || pr.type === 'fire') {
   708→        crewKilled = t.crew3d.killNearPath(hit.lp, this.pathB, 'ball', 2, 1.35,
   709→          t.position, t.quaternion);
   710→      }
   711→      if (pr.type === 'grape') {
   712→        // потери команды согласованы с убитыми фигурами
   713→        const per = Math.max(1, t.preset.maxCrew / Math.max(1, t.crew3d.total));
   714→        t.applyDamage({ kind: 'grape', amount: Math.max(0.6, crewKilled * per * 0.9) });
   715→      } else {
   716→        t.applyDamage({
   717→          kind: pr.type === 'fire' && Math.random() < info.fireChance ? 'fire' : pr.type,
   718→          amount: info.dmgHull * amount,
   719→          hitPoint: pr.pos.clone(),
   720→        });
   721→      }
   722→      if (pr.type === 'chain') t.applyDamage({ kind: 'chain', amount: info.dmgSail * amount });
   723→      // декаль пробоины (ядро и зажигательное)
   724→      if (pr.type === 'ball' || pr.type === 'fire') t.addHole(hit.lp, belowWL);
   725→      // щепки и дым
   726→      for (let i = 0; i < 6; i++) {
   727→        P.spawn(pr.pos.x, pr.pos.y, pr.pos.z,
   728→          (Math.random() - 0.5) * 6, 2 + Math.random() * 4, (Math.random() - 0.5) * 6,
   729→          0.7, 0.28, 0, 0.6, 10);
   730→      }
   731→      P.smoke(pr.pos.x, pr.pos.y, pr.pos.z, 0, 0.7, 0, 0.95, 1.2);
   732→    }
   733→    this.emit(GameEvents.PROJECTILE_HIT, { pos: pr.pos.clone(), target: t });
   734→  }

   736→  private updateAimAssist(time: number, waterHeight: (x: number, z: number, t: number) => number) {
   737→    const ship = this.player;
   738→    const info = AMMO_INFO[this.ammo];
   739→    const start = new THREE.Vector3().copy(ship.position);
   740→    start.y += ship.preset.freeboard * 0.55;
   741→    const vel = this.aimDirection(new THREE.Vector3()).multiplyScalar(info.muzzle).add(ship.velocity);
   742→
   743→    // симуляция траектории
   744→    const attr = this.trajLine.geometry.getAttribute('position') as THREE.BufferAttribute;
   745→    const arr = attr.array as Float32Array;
   746→    let px = start.x, py = start.y, pz = start.z;
   747→    let vx = vel.x, vy = vel.y, vz = vel.z;
   748→    const stp = 0.06;
   749→    let range = 0;
   750→    let i = 0;
   751→    for (; i < 48; i++) {
   752→      arr[i * 3] = px; arr[i * 3 + 1] = py; arr[i * 3 + 2] = pz;
   753→      for (let s = 0; s < 4; s++) {
   754→        const d = Math.exp(-info.drag * stp * 3);
   755→        vx *= d; vy *= d; vz *= d;
   756→        vy -= 9.81 * stp;
   757→        px += vx * stp; py += vy * stp; pz += vz * stp;
   758→        range += Math.hypot(vx, vy, vz) * stp;
   759→        if (py < waterHeight(px, pz, time)) break;
   760→      }
   761→      if (py < waterHeight(px, pz, time)) { i++; if (i < 48) { arr[i * 3] = px; arr[i * 3 + 1] = Math.max(py, waterHeight(px, pz, time)); arr[i * 3 + 2] = pz; } break; }
   762→    }
   763→    attr.needsUpdate = true;
   764→    this.trajLine.geometry.setDrawRange(0, Math.max(2, i));
   765→    this.aim.range = range;
   766→
   767→    // кольцо в точке падения
   768→    if (i > 0 && py < waterHeight(px, pz, time)) {
   769→      const wy = waterHeight(px, pz, time);
   770→      this.impactRing.position.set(px, wy + 0.3, pz);
   771→      this.impactRing.scale.setScalar(1 + Math.sin(time * 6) * 0.12);
   772→      this.impactRing.visible = true;
   773→    } else {
   774→      this.impactRing.visible = false;
   775→    }
   776→
   777→    // упреждение: ближайшая цель в секторе
   778→    const dir = this.aimDirection(new THREE.Vector3());
   779→    this.aim.target = null;
   780→    this.aim.targetLead = null;
   781→    let best: Ship | null = null;
   782→    let bestScore = Infinity;
   783→    const toT = new THREE.Vector3();
   784→    for (const t of this.others) {
   785→      if (t.sunk || t.sinking) continue;
   786→      toT.subVectors(t.position, ship.position);
   787→      const dist = toT.length();
   788→      if (dist > 900) continue;
   789→      toT.normalize();
   790→      const dot = toT.dot(dir);
   791→      if (dot < 0.92) continue; // в пределах ~23° от прицела
   792→      const score = dist * (2 - dot);
   793→      if (score < bestScore) { bestScore = score; best = t; }
   794→    }
   795→    if (best) {
   796→      const dist = best.position.distanceTo(ship.position);
   797→      const flight = dist / info.muzzle;
   798→      const lead = new THREE.Vector3().copy(best.position).addScaledVector(best.velocity, flight);
   799→      lead.y += 1.5;
   800→      this.aim.target = best;
   801→      this.aim.targetLead = lead;
   802→      this.leadMarker.position.copy(lead);
   803→      this.leadMarker.rotation.y = time * 2;
   804→      this.leadMarker.scale.setScalar(1 + Math.sin(time * 5) * 0.15);
   805→      this.leadMarker.visible = true;
   806→    } else {
   807→      this.leadMarker.visible = false;
   808→    }
   809→  }
   810→
   811→  /** Готовность орудий (0..1) для HUD */
   812→  readiness(): number {
   813→    let ready = 0;
   814→    for (const r of this.reloads) if (r <= 0) ready++;
   815→    return ready / this.reloads.length;
   816→  }
