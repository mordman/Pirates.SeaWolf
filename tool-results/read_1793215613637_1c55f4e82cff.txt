    88→
    89→  useEffect(() => {

   104→      game?.dispose();

   112→      const c = settings.controls;
   113→      if (e.code === 'Escape') {
   114→        setSettingsOpen(false); setMapOpen(false); setHelpOpen(false);
   115→        const gg = gameRef.current; if (gg) { gg.uiOpen = false; gg.input.keys.clear(); }
   116→        return;
   117→      }
   118→      if (settingsOpen) return;
   119→      if (e.code === c.map) { e.preventDefault(); setMapOpen(v => !v); }
   120→      if (e.code === c.help) setHelpOpen(v => !v);

   128→    if (gg) gg.uiOpen = true;
   129→    setSettingsOpen(true);

   136→  }, []);
   137→
   138→  const applySettings = useCallback((s: GameSettings) => {
   139→    setSettings(s);
   140→    saveSettings(s);
   141→    gameRef.current?.applySettings(s);
   142→  }, []);
   143→
   144→  // отрисовка карты
   145→  useEffect(() => {
