    35→export class Ship {
    36→  preset: ShipPreset;
    37→  visual: ShipVisual;
    38→  isPlayer: boolean;
    39→
    40→  // --- физика ---
    41→  position = new THREE.Vector3();
    42→  quaternion = new THREE.Quaternion();
    43→  velocity = new THREE.Vector3();
    44→  angular = new THREE.Vector3(); // мировая угловая скорость
    45→  mass: number;
    46→  inertia = new THREE.Vector3(); // диагональ в связных осях (x=right, y=up, z=forward)
    47→  private probes: Probe[] = [];
    48→  private reserve: Probe[] = [];   // запас плавучести надводного борта
    49→  private force = new THREE.Vector3();
    50→  private torque = new THREE.Vector3();
    51→  private _f = new THREE.Vector3();
    52→  private _r = new THREE.Vector3();
    53→  private _t1 = new THREE.Vector3();
    54→  private _t2 = new THREE.Vector3();
    55→  private _t3 = new THREE.Vector3();
    56→  private tmpV = new THREE.Vector3();
    57→  private tmpV2 = new THREE.Vector3();
    58→  private tmpQ = new THREE.Quaternion();
    59→  private wq = new THREE.Quaternion();
    60→  private dq = new THREE.Quaternion();
    61→  private _e2 = new THREE.Euler();   // разбор кватерниона для посадки на волну
    62→
    63→  // --- управление ---
    64→  sailTarget = 0;          // 0..3
    65→  /** Принудительная доля постановки парусов (0..1) — мишень полигона
    66→   *  стоит на якоре с поставленным рангоутом; null = обычная логика */
    67→  sailOverride: number | null = null;
    68→  sailFrac = 0;            // плавное 0..1
    69→  rudder = 0;              // -1..1 (реальный угол)
    70→  rudderTarget = 0;
    71→  anchored = false;
    72→  reverseThrust = false;
    73→  helmActive = false;
    74→
    75→  // --- состояние ---
    76→  hullHP: number; hullMax: number;
    77→  sailHP: number;
    78→  /** максимум парусов с учётом множителя прочности (clearDamage) */
    79→  sailMax: number;
    80→  crew: number;
    81→  water = 0;               // м³ в трюме
    82→  holdCap: number;
    83→  leakRate = 0;            // м³/с суммарная течь
    84→  onFire = false;
    85→  fireT = 0;
    86→  mastsBroken = 0;         // число сломанных (для тяги)
    87→  mastAnim = -1;           // (легаси) для совместимости
    88→  mastHP: number[] = [];   // HP каждой мачты
    89→  mastFall: number[] = []; // прогресс падения: -1 стоит, 0..1 валится/лежит
    90→  mastMaxHP = 32;
    91→  hullHoles: { side: number; belowWL: boolean; mesh: THREE.Mesh }[] = [];
    92→  leakBias = 0;            // суммарный кренящий момент пробоин (−1..1 порт/штирборт)

    95→  sinking = false;
    96→  sinkT = 0;
    97→  sunk = false;
    98→  surrendered = false;
    99→  grounded = false;
   100→  gold = 0;
   101→  cargo: Record<string, number> = {};
   102→  ammo: Record<string, number> = { ball: 40, chain: 12, grape: 20, fire: 8 };
   103→  planks = 12;
   104→  pumpAction = false;
   105→

   109→  particles: Particles;
   110→

   112→  foam: WaterlineFoam;
   113→  wake = new WakeTrail();
   114→  private lastStern = new THREE.Vector3();
   115→  private rollSmooth = 0;

   117→  // --- события колбэки ---
   118→  onSurrender?: (s: Ship) => void;
   119→  onSunk?: (s: Ship) => void;

   121→  constructor(presetId: string, isPlayer: boolean, wind: WindState,
   122→    terrainHeight: (x: number, z: number) => number, particles: Particles,
   123→    opts?: ShipOpts) {
   124→    this.preset = PRESETS[presetId];
   125→    this.isPlayer = isPlayer;
   126→    this.wind = wind;
   127→    this.terrainHeight = terrainHeight;
   128→    this.particles = particles;
   129→    this.visual = buildShip(this.preset);
   130→    // прочность по настройке игры (по умолчанию 1× — исторические ТТХ)
   131→    const mul = opts?.durabilityMul ?? 1;
   132→    this.hullHP = this.preset.hullHP * mul;
   133→    this.hullMax = this.hullHP;
   134→    this.sailHP = this.preset.sailHP * mul;
   135→    this.sailMax = this.sailHP;
   136→    this.mastMaxHP = 32 * mul;
   137→    if (opts?.ammo) this.ammo = { ...opts.ammo };
   138→    this.crew = Math.round(this.preset.maxCrew * 0.8);
   139→    this.holdCap = this.preset.cargo * 0.8;
   140→    this.mass = this.preset.mass;
   141→
   142→    const { length: L, beam: B, draft: D, freeboard: F } = this.preset;
   143→    // моменты инерции (связные оси: x — крен? нет: x=право, y=верх, z=нос)
   144→    // крен — вокруг z, дифферент — вокруг x, рыскание — вокруг y
   145→    this.inertia.set(
   146→      this.mass * (B * B + F * F) / 12 * 1.5,   // pitch (x)
   147→      this.mass * (L * L + B * B) / 12 * 1.2,   // yaw (y)
   148→      this.mass * (B * B + F * F) / 12 * 1.5,   // roll (z)
   149→    );
   150→
   151→    // зонды плавучести: 9 шпангоутов × 3 точки (ниже ватерлинии),
   152→    // вплоть до носа/кормы (±0.47 полудлины). Пружины жёсткие
   153→    // (нейтраль 0.38 осадки, запас ×1.35) и демпфер ζ≈0.62 — корпус
   154→    // БЫСТРО всходит на гребень и не проваливается внутрь волны.
   155→    const NZ = 9;
   156→    let n = 0;
   157→    for (let i = 0; i < NZ; i++) {
   158→      const t = i / (NZ - 1);
   159→      const z = THREE.MathUtils.lerp(-L * 0.47, L * 0.47, t);
   160→      const prof = this.halfWidthAt(z);
   161→      const offs = [0, -prof * 0.75, prof * 0.75];
   162→      for (const x of offs) {
   163→        this.probes.push({ off: new THREE.Vector3(x, -D * 0.38, z), k: 0, c: 0 });
   164→        n++;
   165→      }
   166→    }
   167→    const neutral = D * 0.38;
   168→    const kTotal = this.mass * G / neutral * 1.35;
   169→    for (const p of this.probes) {
   170→      p.k = kTotal / n;
   171→      p.c = 2 * Math.sqrt(p.k * (this.mass / n)) * 0.62; // демпфер качки
   172→    }
   173→    // Резервные зонды по скулам борта чуть выше спокойной ватерлинии:
   174→    // моделируют запас плавучести корпуса до планширя — гребень, накрывший
   175→    // палубу, встречает сильное сопротивление и кренящий момент вправку,
   176→    // «полностью по борт» корабль больше не погружается.
   177→    let rn = 0;
   178→    for (let i = 0; i < NZ; i++) {
   179→      const t = i / (NZ - 1);
   180→      const z = THREE.MathUtils.lerp(-L * 0.44, L * 0.44, t);
   181→      const prof = this.halfWidthAt(z) * 0.85;
   182→      for (const x of [-prof, prof]) {
   183→        this.reserve.push({ off: new THREE.Vector3(x, F * 0.34, z), k: 0, c: 0 });
   184→        rn++;
   185→      }
   186→    }
   187→    // жёстче прежнего: полный вес запаса уже на 15 см погружения борта —
   188→    // волна не может накрыть палубу, корпус «всплывает» на гребень
   189→    const rkTotal = this.mass * G / 0.15;
   190→    for (const p of this.reserve) {
   191→      p.k = rkTotal / rn;
   192→      p.c = 2 * Math.sqrt(p.k * (this.mass / rn)) * 1.3;
   193→    }
   194→
   195→    this.foam = new WaterlineFoam(this.visual.outline);
   196→    this.wake.setWaterHeight(waterHeightAt);
   197→    if (isPlayer) enableLantern(this.visual);
   198→
   199→    // экипаж на палубе: процедурные типажи, сид фиксирован по модели
   200→    const seed = Array.from(presetId).reduce((a, ch) => a + ch.charCodeAt(0) * 31, 11)
   201→      + Math.round(this.preset.length * 7);
   202→    this.crew3d = buildCrew(this.preset, seed);
   203→    this.visual.group.add(this.crew3d.group);
   204→    for (let i = 0; i < this.visual.masts.length; i++) {
   205→      this.mastHP.push(this.mastMaxHP);
   206→      this.mastFall.push(-1);
   207→    }
   208→  }
   209→
   210→  halfWidthAt(z: number): number {
   211→    const t = clamp((z + this.preset.length / 2) / this.preset.length, 0, 1);
   212→    let w: number;
   213→    if (t < 0.45) w = THREE.MathUtils.lerp(0.58, 1.0, t / 0.45);
   214→    else w = THREE.MathUtils.lerp(1.0, 0.0, Math.pow(clamp((t - 0.45) / 0.55, 0, 1), 0.75));
   215→    return Math.max(0.3, w * this.preset.beam * 0.5);
   216→  }
   217→
   218→  // ---------- геометрия ориентации ----------
   219→  get forward(): THREE.Vector3 {
   220→    const e = new THREE.Euler().setFromQuaternion(this.quaternion, 'YXZ');
   221→    return new THREE.Vector3(Math.sin(e.y), 0, Math.cos(e.y)).normalize();
   222→  }
   223→  get heading(): number {
   224→    const e = new THREE.Euler().setFromQuaternion(this.quaternion, 'YXZ');
   225→    return e.y;
   226→  }
   227→  get euler(): THREE.Euler { return new THREE.Euler().setFromQuaternion(this.quaternion, 'YXZ'); }
   228→  speed(): number { return Math.hypot(this.velocity.x, this.velocity.z); }
   229→  speedKnots(): number { return this.speed() * 1.9438; }
   230→
   231→  /** Угол между направлением ОТКУДА дует (кажущийся) ветер и курсом, 0..PI */
   232→  apparentWindAngle(): number {
   233→    const wf = this.tmpV.set(Math.cos(this.wind.from), 0, Math.sin(this.wind.from));
   234→    // кажущийся ветер = истинный - скорость корабля
   235→    const av = this.tmpV2.set(-wf.x * this.wind.speed - this.velocity.x, 0, -wf.z * this.wind.speed - this.velocity.z);
   236→    // вектор ОТКУДА дует кажущийся ветер:
   237→    const aFromX = -av.x, aFromZ = -av.z;
   238→    const len = Math.hypot(aFromX, aFromZ) || 1;
   239→    const f = this.forward;
   240→    const dot = (aFromX / len) * f.x + (aFromZ / len) * f.z;
   241→    return Math.acos(clamp(dot, -1, 1));
   242→  }
   243→
   244→  /** Полярная эффективность парусов по углу (рад): лавировка/бейдевинд/галфвинд/фордевинд */
   245→  static polar(awa: number): number {
   246→    const d = awa * 180 / Math.PI;
   247→    if (d < 25) return 0.02;                       // левентик — почти нет хода
   248→    if (d < 60) return THREE.MathUtils.lerp(0.08, 0.7, (d - 25) / 35);
   249→    if (d < 110) return THREE.MathUtils.lerp(0.7, 1.0, (d - 60) / 50);  // бейдевинд-галфвинд
   250→    if (d < 150) return THREE.MathUtils.lerp(1.0, 0.82, (d - 110) / 40);
   251→    return THREE.MathUtils.lerp(0.82, 0.68, (d - 150) / 30);            // фордевинд
   252→  }
   253→
   254→  // ---------- фиксированный шаг физики ----------
   255→  fixedStep(dt: number, time: number) {
   256→    if (this.sunk) return;
   257→    this.force.set(0, 0, 0);
   258→    this.torque.set(0, 0, 0);
   259→
   260→    const f = this._f.copy(this.forward);
   261→    // штирборт (правый борт): forward × up = (−f.z, 0, f.x)
   262→    const right = this._r.set(-f.z, 0, f.x).normalize();
   263→    const e = this.euler;
   264→    const spd = this.speed();
   265→
   266→    const m = this.mass + this.water * 1000; // вода в трюме утяжеляет
   267→
   268→    // --- гравитация ---
   269→    this.force.y -= m * G;
   270→
   271→    // --- плавучесть по зондам (τ = r × F, F = (0, fl, 0)) ---
   272→    for (const p of this.probes) {
   273→      const wp = this._t1.copy(p.off).applyQuaternion(this.quaternion).add(this.position);
   274→      const wy = waterHeightAt(wp.x, wp.z, time);
   275→      const depth = wy - wp.y;
   276→      if (depth > 0) {
   277→        const rx = wp.x - this.position.x, rz = wp.z - this.position.z;
   278→        // вертикальная скорость точки: v = ω × r → vy = ωz*rx − ωx*rz
   279→        const vy = this.velocity.y + (this.angular.z * rx - this.angular.x * rz);
   280→        let fl = p.k * Math.min(depth, 6) - p.c * vy;
   281→        if (fl < 0) fl = 0;
   282→        this.force.y += fl;
   283→        this.torque.x += -rz * fl;
   284→        this.torque.z += rx * fl;
   285→      }
   286→    }
   287→    // --- запас плавучести бортов (пока не тонем) ---
   288→    if (!this.sinking) {
   289→      for (const p of this.reserve) {
   290→        const wp = this._t1.copy(p.off).applyQuaternion(this.quaternion).add(this.position);
   291→        const wy = waterHeightAt(wp.x, wp.z, time);
   292→        const depth = wy - wp.y;
   293→        if (depth > 0) {
   294→          const rx = wp.x - this.position.x, rz = wp.z - this.position.z;
   295→          const vy = this.velocity.y + (this.angular.z * rx - this.angular.x * rz);
   296→          let fl = p.k * Math.min(depth, 2.5) - p.c * vy;
   297→          if (fl < 0) fl = 0;
   298→          this.force.y += fl;
   299→          this.torque.x += -rz * fl;
   300→          this.torque.z += rx * fl;
   301→        }
   302→      }
   303→    }
   304→
   305→    // --- ветер на парусах ---
   306→    this.sailFrac += (this.sailTarget / 3 - this.sailFrac) * Math.min(1, dt * 0.7);
   307→    const awa = this.apparentWindAngle();
   308→    const polar = Ship.polar(awa);
   309→    const sailEffMul = this.sailHP / this.preset.sailHP;
   310→    const damaged = sailEffMul < 0.5 ? 0.45 : sailEffMul < 0.85 ? 0.8 : 1.0;
   311→    const windPower = Math.pow(this.wind.speed / 8.0, 1.6);
   312→    let thrust = 0;
   313→    if (!this.anchored && !this.sinking && this.mastsBroken < 2) {
   314→      thrust = this.sailFrac * polar * windPower * this.preset.maxSpeed * 7.0 * damaged * (this.sailHP > 0 ? 1 : 0);
   315→    }
   316→    if (this.reverseThrust && !this.anchored) thrust = -this.preset.maxSpeed * 0.45; // задний ход на вёслах
   317→    // ГРЕБНЫЕ суда (шебека, галера): вёсла дают ход даже в полный штиль;
   318→    // под парусами вёсла помогают меньше — экипаж занят парусами
   319→    if (this.preset.oars && !this.anchored && !this.sinking) {
   320→      const crewFrac = clamp(this.crew / this.preset.maxCrew, 0.2, 1);
   321→      const oarThrust = this.preset.maxSpeed * 0.34 * crewFrac * (1 - 0.55 * this.sailFrac);
   322→      thrust += oarThrust;
   323→    }
   324→    // сила тяги вдоль forward
   325→    this.force.x += f.x * thrust * m * 0.06;
   326→    this.force.z += f.z * thrust * m * 0.06;
   327→    // крен от ветра: боковая составляющая ветра на парусах
   328→    // (умеренный: в шторм подветренный борт не зарывается в волну)
   329→    const wf = new THREE.Vector3(Math.cos(this.wind.from), 0, Math.sin(this.wind.from));
   330→    const windTravel = wf.clone().multiplyScalar(-this.wind.speed);
   331→    const lateral = windTravel.dot(right);
   332→    const heelF = lateral * this.sailFrac * windPower * m * 0.034;
   333→    this.torque.z += heelF;
   334→    // ... и небольшое боковое сносовое усилие
   335→    this.force.x += right.x * lateral * this.sailFrac * m * 0.004;
   336→    this.force.z += right.z * lateral * this.sailFrac * m * 0.004;
   337→
   338→    // --- руль ---
   339→    this.rudder += (this.rudderTarget - this.rudder) * Math.min(1, dt * 1.8);
   340→    const rudderAngle = this.rudder * 0.6; // рад, макс ~34°
   341→    const yawAuth = clamp(spd / 4.0, 0, 1.15); // авторитеты руля от скорости
   342→    // положительный руль (D) = поворот ВПРАВО (штирборт):
   343→    // вращение вокруг −Y, т.к. +Y = поворот влево (к порту)
   344→    this.torque.y -= rudderAngle * yawAuth * m * 12.0; // ~6°/с на полном ходу
   345→    // крен на повороте (отцентробежный)
   346→    this.torque.z += this.angular.y * spd * m * 0.35;
   347→    // лёгкое зарывание носа на скорости (слабое: сильное топило борт
   348→    // в волну на 3+ метра — см. жалобы «корабль погружается по борт»)
   349→    this.torque.x += spd * spd * m * 0.0012;
   350→
   351→    // --- сопротивление (по связным осям) ---
   352→    const vLocal = this.velocity.clone();
   353→    const fwdSpd = vLocal.dot(f);
   354→    const latSpd = vLocal.dot(right);
   355→    const fwdDrag = -Math.sign(fwdSpd) * (18 * fwdSpd + 55 * fwdSpd * Math.abs(fwdSpd)) * m * 0.001;
   356→    const latDrag = -latSpd * (3.2) * m * 0.1;
   357→    this.force.x += f.x * fwdDrag + right.x * latDrag;
   358→    this.force.z += f.z * fwdDrag + right.z * latDrag;
   359→    this.force.y -= this.velocity.y * m * 0.12; // вертикальное демпфирование
   360→    if (this.anchored) {
   361→      this.force.x -= this.velocity.x * m * 1.2;
   362→      this.force.z -= this.velocity.z * m * 1.2;
   363→    }
   364→    if (this.grounded) {
   365→      this.force.x -= this.velocity.x * m * 2.5;
   366→      this.force.z -= this.velocity.z * m * 2.5;
   367→    }
   368→
   369→    // --- угловое демпфирование по связным осям ---
   370→    const invQ = this.tmpQ.copy(this.quaternion).invert();
   371→    const ab = this._t3.copy(this.angular).applyQuaternion(invQ); // связные
   372→    ab.x *= Math.exp(-1.6 * dt);  // дифферент
   373→    ab.y *= Math.exp(-0.9 * dt);  // рыскание
   374→    ab.z *= Math.exp(-1.1 * dt);  // крен
   375→    this.angular.copy(ab.applyQuaternion(this.quaternion));
   376→
   377→    // --- ограничение крена/дифферента (мягкие стопы) ---
   378→    // дифферент свободнее прежнего: 0.21 рад (12°) — нос всходит
   379→    // на крутой фронт волны, а не прорезает его;
   380→    // крен 0.36 рад (20°) — подветренный борт остаётся над водой
   381→    const roll = e.z, pitch = e.x;
   382→    const rollLim = 0.36, pitchLim = 0.21;
   383→    if (Math.abs(roll) > rollLim) this.torque.z -= (roll - Math.sign(roll) * rollLim) * m * 9.0;
   384→    if (Math.abs(pitch) > pitchLim) this.torque.x -= (pitch - Math.sign(pitch) * pitchLim) * m * 12.0;
   385→    // затопление кренит корабль в сторону пробоин
   386→    this.floodRoll(m);
   387→    if (this.sinking) {
   388→      this.sinkT += dt;
   389→      this.force.y -= m * G * Math.min(0.9, this.sinkT / 9); // неотвратимое погружение
   390→      this.torque.x += m * 0.25 * Math.sign(roll || 1);
   391→      this.torque.z += m * 0.4 * Math.sign(roll || 1);
   392→    }
   393→
   394→    // --- интеграция ---
   395→    this.velocity.addScaledVector(this.force, dt / m);
   396→    this.position.addScaledVector(this.velocity, dt);
   397→    // α = I⁻¹ τ в связных осях
   398→    const tb = this._t3.copy(this.torque).applyQuaternion(invQ);
   399→    tb.x /= this.inertia.x; tb.y /= this.inertia.y; tb.z /= this.inertia.z;
   400→    this.angular.addScaledVector(tb.applyQuaternion(this.quaternion), dt);
   401→    // ограничим угловые скорости
   402→    const wLen = this.angular.length();
   403→    if (wLen > 1.1) this.angular.multiplyScalar(1.1 / wLen);
   404→    // q̇ = 0.5 · ω⊗q → q_new = q + 0.5·dt·(ωx,ωy,ωz,0)⊗q, потом нормализация
   405→    this.wq.set(this.angular.x, this.angular.y, this.angular.z, 0);
   406→    this.dq.copy(this.wq).multiply(this.quaternion); // dq = ω ⊗ q
   407→    const qs = 0.5 * dt;
   408→    this.quaternion.set(
   409→      this.quaternion.x + this.dq.x * qs,
   410→      this.quaternion.y + this.dq.y * qs,
   411→      this.quaternion.z + this.dq.z * qs,
   412→      this.quaternion.w + this.dq.w * qs
   413→    ).normalize();
   414→
   415→    // === КИНЕМАТИЧЕСКАЯ ПОСАДКА НА ВОЛНУ ===
   416→    // Пружины зондов дают равновесие на СРЕДНЕМ уровне воды — корпус
   417→    // «сидел» ВНУТРИ гребня. Здесь корпус прямо следует ПОВЕРХНОСТИ:
   418→    // вертикаль — к уровню 4 точек (нос/корма/оба борта), дифферент —
   419→    // к плоскости волны вдоль корпуса. Крен ОСТАЁТСЯ динамическим
   420→    // (ветер, течь, поворот) — зондов-пружин на крен достаточно.
   421→    // Скорости λy 3.2/с и λp 2.4/с: успевают за зыбью (период 4–8 с),
   422→    // короткую рябь усредняют; поправка ограничена 8 м/с и 0.5 рад/с.
   423→    if (!this.sinking && !this.sunk && !this.grounded) {
   424→      const Lh = this.preset.length * 0.38, Bh = this.preset.beam * 0.36;
   425→      const hB = waterHeightAt(this.position.x + f.x * Lh, this.position.z + f.z * Lh, time);
   426→      const hS = waterHeightAt(this.position.x - f.x * Lh, this.position.z - f.z * Lh, time);
   427→      const hR = waterHeightAt(this.position.x + right.x * Bh, this.position.z + right.z * Bh, time);
   428→      const hP = waterHeightAt(this.position.x - right.x * Bh, this.position.z - right.z * Bh, time);
   429→      const surfY = (hB + hS + hR + hP) * 0.25;
   430→      // равновесие пружин: осадка зондов 0.28·D → центр на 0.10·D над водой;
   431→      // затопленный трюм садит корпус ниже (до 30 см)
   432→      const sink = clamp(this.water / Math.max(1, this.holdCap), 0, 1) * 0.30;
   433→      const targetY = surfY + this.preset.draft * 0.10 - sink;
   434→      const kY = 1 - Math.exp(-dt * 4.2);
   435→      const corrY = clamp((targetY - this.position.y) * kY, -11 * dt, 11 * dt);
   436→      this.position.y += corrY;
   437→      // согласованное затухание вертикальной скорости — пружины
   438→      // не раскачивают корпус вокруг кинематической посадки
   439→      this.velocity.y *= Math.exp(-dt * 3.0);
   440→      // дифферент: нос/корма по склону волны (вода выше у носа → нос вверх)
   441→      const wPitch = -Math.atan2(hB - hS, 2 * Lh);
   442→      const e2 = this._e2.setFromQuaternion(this.quaternion, 'YXZ');
   443→      const kP = 1 - Math.exp(-dt * 3.2);
   444→      e2.x += clamp((wPitch - e2.x) * kP, -0.65 * dt, 0.65 * dt);
   445→      this.quaternion.setFromEuler(e2);
   446→    }
   447→
   448→    // --- посадка на мель / столкновение с землёй ---
   449→    this.checkGrounding(dt);
   450→
   451→    // --- течение воды в трюме ---
   452→    if (this.leakRate > 0) {
   453→      this.water = Math.min(this.holdCap, this.water + this.leakRate * dt);
   454→    }
   455→    if (this.onFire) {
   456→      this.fireT += dt;
   457→      this.hullHP -= dt * 1.6;
   458→      if (this.fireT > 26) { this.onFire = false; this.fireT = 0; } // выгорит
   459→      if (Math.random() < dt * 1.2) {
   460→        const p = this.particles;
   461→        const wp = this.position.clone();
   462→        p.fire(wp.x + (Math.random() - 0.5) * 4, this.waterlineY() + 1 + Math.random() * 2, wp.z + (Math.random() - 0.5) * 10);
   463→        p.smoke(wp.x + (Math.random() - 0.5) * 4, this.waterlineY() + 3, wp.z + (Math.random() - 0.5) * 10, this.wind ? Math.cos(this.wind.from + Math.PI) * 2 : 0, 2.2, this.wind ? Math.sin(this.wind.from + Math.PI) * 2 : 0, 2.8, 3.0);
   464→      }
   465→    }
   466→    // порог затопления
   467→    if (!this.sinking && this.water > this.holdCap * 0.62) this.startSinking();
   468→    if (this.hullHP <= 0 && !this.sinking) this.startSinking();
   469→
   470→    // --- сдался? ---
   471→    if (!this.isPlayer && !this.surrendered && !this.sinking) {
   472→      if (this.hullHP < this.hullMax * 0.34 || this.crew < this.preset.maxCrew * 0.25) {
   473→        this.surrendered = true;
   474→        this.sailTarget = 0;
   475→        this.onSurrender?.(this);
   476→      }
   477→    }
   478→
   479→    if (this.sinking && this.sinkT > 14 && !this.sunk) {
   480→      this.sunk = true;
   481→      this.visual.group.visible = false;
   482→      this.onSunk?.(this);
   483→    }
   484→  }

   486→  private checkGrounding(dt: number) {
   487→    if (this.sinking || this.sunk) return;
   488→    const f = this.forward;
   489→    const bow = this.tmpV.copy(this.position).addScaledVector(f, this.preset.length * 0.42);
   490→    const stern = this.tmpV2.copy(this.position).addScaledVector(f, -this.preset.length * 0.42);
   491→    let hit = false;
   492→    for (const p of [bow, stern]) {
   493→      const th = this.terrainHeight(p.x, p.z);
   494→      if (th > this.position.y - this.preset.draft * 0.6) {
   495→        hit = true;
   496→        // нормаль склона
   497→        const e = 2;
   498→        const nx = this.terrainHeight(p.x - e, p.z) - this.terrainHeight(p.x + e, p.z);
   499→        const nz = this.terrainHeight(p.x, p.z - e) - this.terrainHeight(p.x, p.z + e);
   500→        const n = new THREE.Vector3(nx, 2 * e, nz).normalize();
   501→        this.force.addScaledVector(n, this.mass * 22);
   502→        this.velocity.multiplyScalar(0.94);
   503→      }
   504→    }
   505→    if (hit) {
   506→      if (!this.grounded && this.speed() > 3.5) {
   507→        this.applyDamage({ kind: 'ground', amount: 5 + this.speed() });
   508→      }
   509→      this.grounded = true;
   510→    } else this.grounded = false;
   511→  }
   512→
   513→  waterlineY(): number {
   514→    return this.position.y;
   515→  }

   517→  // ---------- повреждения ----------
   518→  applyDamage(d: DamageInput) {
   519→    if (this.sunk || this.sinking) return;
   520→    const rng = Math.random;
   521→    switch (d.kind) {
   522→      case 'ball': {
   523→        this.hullHP -= d.amount;
   524→        if (d.hitPoint && d.hitPoint.y < 0.45 + this.position.y) this.addLeak(0.06 + rng() * 0.05);
   525→        if (rng() < 0.3) this.crew = Math.max(0, this.crew - 1);
   526→        // случайный «рикошет в рангоут» — редкий, основную работу делают
   527→        // прицельные попадания из CombatSystem (mast hits)
   528→        if (rng() < 0.10) this.breakMast();
   529→        break;
   530→      }
   531→      case 'chain': {
   532→        this.sailHP = Math.max(0, this.sailHP - d.amount);
   533→        if (rng() < 0.25) this.crew = Math.max(0, this.crew - 1);
   534→        break;
   535→      }
   536→      case 'grape': {
   537→        this.crew = Math.max(0, this.crew - Math.round(d.amount));
   538→        this.hullHP -= d.amount * 0.4;
   539→        break;
   540→      }
   541→      case 'fire': {
   542→        this.hullHP -= d.amount;
   543→        if (rng() < 0.35 && !this.onFire) { this.onFire = true; this.fireT = 0; }
   544→        if (rng() < 0.08) { this.addLeak(0.03); }
   545→        break;
   546→      }
   547→      case 'ram': {
   548→        this.hullHP -= d.amount;
   549→        if (rng() < 0.5) this.addLeak(0.04);
   550→        break;
   551→      }
   552→      case 'ground': this.hullHP -= d.amount * 0.5; break;
   553→    }
   554→    if (this.hullHP <= 0 && !this.sinking) this.startSinking();
   555→    if (!this.isPlayer && !this.noSurrender && !this.surrendered && this.hullHP < this.hullMax * 0.34) {
   556→      this.surrendered = true; this.sailTarget = 0; this.onSurrender?.(this);
   557→    }
   558→  }

   560→  /** Урон в КОНКРЕТНУЮ мачту (прицельное попадание ядром/книппелем) */
   561→  damageMast(idx: number, amount: number): boolean {
   562→    if (idx < 0 || idx >= this.mastHP.length) return false;
   563→    if (this.mastFall[idx] >= 0) return false; // уже сломана
   564→    this.mastHP[idx] -= amount;
   565→    if (this.mastHP[idx] <= 0) { this.breakMast(idx); return true; }
   566→    return false;
   567→  }
   568→
   569→  addLeak(rate: number) {
   570→    this.leakRate = Math.min(1.8, this.leakRate + rate);
   571→    if (this.leakRate < 0) this.leakRate = 0;
   572→  }

   574→  /** Урон ПАРУСАМ: пробоины картечи/книппеля снижают тягу (sailHP).
   575→   *  Дыры рисуются отдельно — см. combat.applySailHit + sailAddHole. */
   576→  damageSails(n: number) {
   577→    this.sailHP = Math.max(0, this.sailHP - n);
   578→  }

   580→  /** Пробоина в борту: декаль + течь (ниже ватерлинии) + кренящий момент */
   581→  addHole(localPos: THREE.Vector3, belowWL: boolean) {
   582→    const side = localPos.x >= 0 ? 1 : -1;
   583→    // тёмная «рваная» дыра: круг + обломки досок вокруг
   584→    const g = new THREE.CircleGeometry(0.34 + Math.random() * 0.14, 9);
   585→    const mat = new THREE.MeshStandardMaterial({ color: 0x120d08, roughness: 1 });
   586→    const mesh = new THREE.Mesh(g, mat);
   587→    const w = Math.max(0.3, this.halfWidthAt(localPos.z) * 0.985);
   588→    mesh.position.set(side * w, localPos.y, localPos.z);
   589→    mesh.rotation.y = side * Math.PI / 2;
   590→    mesh.rotation.x = (Math.random() - 0.5) * 0.5;
   591→    this.visual.group.add(mesh);
   592→    this.hullHoles.push({ side, belowWL, mesh });
   593→    if (belowWL) {
   594→      this.addLeak(0.16 + Math.random() * 0.10);
   595→      this.leakBias = clamp(this.leakBias + side * 0.35, -1, 1);
   596→    }
   597→  }

   599→  /** Затопление: крен в сторону пробоин + вода в трюме утяжеляет */
   600→  private floodRoll(m: number) {
   601→    if (this.water <= 0 || this.sinking) return;
   602→    const frac = this.water / this.holdCap;
   603→    this.torque.z += this.leakBias * frac * m * 0.06;
   604→  }

   606→  breakMast(idx?: number) {
   607→    let i = idx;
   608→    if (i === undefined || i < 0 || i >= this.visual.masts.length || this.mastFall[i] >= 0) {
   609→      // без указания — случайная целая мачта
   610→      const intact = this.mastFall.map((f, k) => f < 0 ? k : -1).filter(k => k >= 0);
   611→      if (!intact.length) return;
   612→      i = intact[Math.floor(Math.random() * intact.length)];
   613→    }
   614→    this.mastFall[i] = 0;
   615→    this.mastsBroken = this.mastFall.filter(f => f >= 0).length;
   616→    this.sailHP = Math.max(0, this.sailHP - 30);
   617→    if (this.sailTarget > 0) this.sailTarget = Math.max(0, this.sailTarget - 1);
   618→  }

   620→  /** Сброс ВСЕХ повреждений (полигон/возрождение): корпус, мачты,
   621→   *  паруса, пробоины, течь, пожар, экипаж */
   622→  clearDamage() {
   623→    this.hullHP = this.hullMax;
   624→    this.sailHP = this.sailMax;
   625→    this.water = 0;
   626→    this.leakRate = 0;
   627→    this.leakBias = 0;
   628→    this.onFire = false;
   629→    this.fireT = 0;
   630→    this.sinking = false;
   631→    this.sinkT = 0;
   632→    this.sunk = false;
   633→    this.surrendered = false;
   634→    this.mastsBroken = 0;
   635→    this.mastAnim = -1;
   636→    for (let i = 0; i < this.visual.masts.length; i++) {
   637→      this.mastHP[i] = this.mastMaxHP;
   638→      this.mastFall[i] = -1;
   639→      const m = this.visual.masts[i];
   640→      m.rotation.set(0, 0, 0);
   641→      m.position.y = 0;
   642→    }
   643→    for (const h of this.hullHoles) {
   644→      this.visual.group.remove(h.mesh);
   645→      h.mesh.geometry.dispose();
   646→      (h.mesh.material as THREE.Material).dispose();
   647→    }
   648→    this.hullHoles = [];
   649→    this.crew = Math.round(this.preset.maxCrew * 0.8);
   650→    this.crew3d.reset();
   651→    this.visual.group.visible = true;
   652→    this.velocity.set(0, 0, 0);
   653→    this.angular.set(0, 0, 0);
   654→    this.sailTarget = 0;
   655→    this.rudder = 0;
   656→    this.rudderTarget = 0;
   657→  }

   659→  startSinking() {
   660→    if (this.sinking) return;
   661→    this.sinking = true;
   662→    this.sinkT = 0;
   663→    this.onFire = false;
   664→  }

   666→  /** Действие команды (R): пожар -> течь -> ремонт */
   667→  crewAction(dt: number): string | null {
   668→    if (this.onFire) { this.fireT += dt * 3; if (this.fireT > 4) { this.onFire = false; this.fireT = 0; return 'Пожар потушен'; } return 'Тушение пожара...'; }
   669→    if (this.water > 0.01) {
   670→      const pump = 0.05 + this.crew * 0.004;
   671→      this.water = Math.max(0, this.water - pump * dt);
   672→      this.leakRate = Math.max(0, this.leakRate - 0.004 * dt);
   673→      return this.water > 0.05 ? 'Откачка воды...' : 'Трюм осушён';
   674→    }
   675→    if (this.hullHP < this.hullMax && this.planks > 0) {
   676→      this.hullHP = Math.min(this.hullMax, this.hullHP + 0.9 * dt);
   677→      this.planks -= dt * 0.15;
   678→      return 'Ремонт корпуса...';
   679→    }
   680→    return null;
   681→  }

   683→  // ---------- визуал ----------
   684→  updateVisual(time: number, dt: number, particles: Particles) {
   685→    const v = this.visual;
   686→    v.group.position.copy(this.position);
   687→    v.group.quaternion.copy(this.quaternion);
   688→
   689→    // паруса: количество поставленных парусов соответствует ФАКТИЧЕСКОЙ
   690→    // СКОРОСТИ — экипаж ставит паруса по мере разгона и убирает по мере
   691→    // остановки; уровень W/S задаёт максимум. Рея брасопится по ветру,
   692→    // «пузо» — по кажущемуся ветру. Паруса СЛОМАННОЙ мачты обвисают.
   693→    const awa = this.apparentWindAngle();
   694→    const polar = Ship.polar(awa);
   695→    const windClamp = clamp(this.wind.speed / 9, 0.3, 1.4);
   696→    // скорость, достижимая под полным парусным вооружением в этот ветер
   697→    // (корень из тяги — сопротивление квадратично): ~0.85·maxSpeed
   698→    const windPow = Math.pow(clamp(this.wind.speed / 8.0, 0.05, 1.6), 1.6);
   699→    const attainable = this.preset.maxSpeed * 0.85 * Math.sqrt(clamp(polar * windPow, 0, 1.5));
   700→    // в левентик/штиль паруса остаются на мачтах и трепещут (не убираются
   701→    // механически) — убираются только когда есть чем идти, но нет хода
   702→    const spdSetFrac = attainable > 1.5 ? clamp(this.speed() / attainable, 0, 1) : 1;
   703→    const setTotal = this.sailOverride ?? Math.min(this.sailFrac, spdSetFrac);
   704→    // локальный вектор «откуда дует» (x — борт, z — нос): ψ = φ + курс
   705→    const psi = this.wind.from + this.heading;
   706→    const wlX = Math.cos(psi); // >0 — ветер с порта (левый борт)
   707→    // цель брасопки: рея поперёк ветра (нормаль паруса по ветру), ±55°
   708→    let bt = Math.atan2(wlX, Math.sin(psi)) - Math.PI;
   709→    bt = Math.atan2(Math.sin(bt), Math.cos(bt)); // кратчайшая дуга
   710→    const braceTarget = clamp(bt, -0.96, 0.96);
   711→    const bellyBase = (0.35 + 0.95 * windClamp) * (0.15 + 0.85 * polar);
   712→    const flutterBase = 0.012 * windClamp * (1 + (1 - polar) * 2.2);
   713→    for (const s of v.sails.sails) {
   714→      // мачта-носитель сломана → парус безжизненно обвисает
   715→      const mastDead = s.mastIndex !== undefined && this.mastFall[s.mastIndex] >= 0;
   716→      // доля постановки этого паруса при текущей скорости хода
   717→      const setFrac = mastDead ? 0 : smoothstep(s.setAt - 0.38, s.setAt, setTotal);
   718→      s.furl += ((1 - setFrac) - s.furl) * Math.min(1, dt * (mastDead ? 2.2 : 1.1));
   719→      const sf = 1 - s.furl;
   720→      s.mat.uniforms.uTime.value = time;
   721→      s.mat.uniforms.uFurl.value = s.furl;
   722→      s.mat.uniforms.uDamage.value = 1 - this.sailHP / this.preset.sailHP;
   723→      s.mat.uniforms.uFlutter.value = flutterBase * (0.25 + 0.75 * sf);
   724→      if (s.square && s.brace) {
   725→        // реи разворачиваются медленно — как будто работают брашпили
   726→        s.braceAngle += ((mastDead ? 0 : braceTarget) - s.braceAngle) * Math.min(1, dt * 0.5);
   727→        s.brace.rotation.y = s.braceAngle;
   728→        s.mat.uniforms.uBelly.value = mastDead ? 0 : bellyBase * sf;
   729→      } else {
   730→        // кливер: пузо в подветренную сторону (локальный Z → борт)
   731→        s.mat.uniforms.uBelly.value = mastDead ? 0 : bellyBase * sf * (-wlX);
   732→      }
   733→    }
   734→
   735→    // флаг — по ветру
   736→    const wf = Math.atan2(-Math.cos(this.wind.from), -Math.sin(this.wind.from));
   737→    v.flag.rotation.y = wf - this.heading + Math.PI;
   738→    v.flagMat.uniforms.uTime.value = time;
   739→
   740→    // штурвал крутится от руля: D (руль вправо) — по часовой от рулевого
   741→    v.wheel.rotation.z = this.rudder * 2.2;
   742→
   743→    // руль за кормой следует за рулём
   744→    if (v.rudderMesh) v.rudderMesh.rotation.y = this.rudder * 0.45;
   745→
   746→    // вёсла: гребки в противофазе бортов (сила гребков — от хода)
   747→    if (v.oars && v.oars.length) {
   748→      const rowing = clamp(this.speed() / Math.max(1, this.preset.maxSpeed * 0.4), 0.12, 1);
   749→      for (const o of v.oars) {
   750→        const base = o.userData.base as number;
   751→        const phase = o.userData.phase as number;
   752→        o.rotation.z = base + Math.sin(time * 2.3 + phase) * 0.16 * rowing;
   753→      }
   754→    }
   755→
   756→    // падение мачт: каждая валится вперёд-вбок со своим наклоном,
   757→    // реи и паруса едут вместе с мачтой (они её потомки)
   758→    for (let i = 0; i < v.masts.length; i++) {
   759→      const f = this.mastFall[i];
   760→      if (f >= 0 && f < 1) {
   761→        this.mastFall[i] = Math.min(1, f + dt * 0.42);
   762→        const k = this.mastFall[i];
   763→        const lean = (i % 2 === 0 ? 1 : -1) * 0.3;
   764→        v.masts[i].rotation.x = k * 1.25;
   765→        v.masts[i].rotation.z = k * lean;
   766→        v.masts[i].position.y = -k * 1.6;
   767→      }
   768→    }
   769→
   770→    // пенный воротник по ватерлинии
   771→    if (!this.sunk) {
   772→      const wy = waterHeightAt(this.position.x, this.position.z, time);
   773→      this.foam.mesh.position.set(this.position.x, wy + 0.28, this.position.z);
   774→      this.foam.mesh.rotation.y = this.heading;
   775→      const spdFrac = clamp(this.speed() / this.preset.maxSpeed, 0, 1);
   776→      this.foam.update(time, spdFrac, this.euler.z);
   777→    } else {
   778→      this.foam.mesh.visible = false;
   779→    }
   780→
   781→    // кильватерный след
   782→    const sternW = this.tmpV.set(0, 0, -this.preset.length * 0.48)
   783→      .applyQuaternion(this.quaternion).add(this.position);
   784→    const spdFrac = clamp(this.speed() / this.preset.maxSpeed, 0, 1);
   785→    if (!this.sunk && !this.grounded) {
   786→      this.wake.push(sternW, spdFrac, dt, time);
   787→    }
   788→    this.wake.update(time);
   789→
   790→    // носовой бурун: брызги при ударе носа о волну
   791→    if (!this.sunk) {
   792→      const bowW = this.tmpV2.set(0, -this.preset.draft * 0.3, this.preset.length * 0.46)
   793→        .applyQuaternion(this.quaternion).add(this.position);
   794→      const waveY = waterHeightAt(bowW.x, bowW.z, time);
   795→      const relV = this.velocity.y - (waveY - this.lastStern.y) * 0; // грубая оценка
   796→      const impact = (waveY - bowW.y) * (0.3 + spdFrac);
   797→      if (impact > 0.55 && Math.random() < 0.6) {
   798→        particles.splash(bowW.x, waveY, bowW.z, 0.5 + spdFrac * 0.9);
   799→      }
   800→      if (spdFrac > 0.35 && Math.random() < spdFrac * 0.5) {
   801→        particles.spawn(bowW.x + (Math.random() - 0.5) * 2, waveY + 0.1, bowW.z + (Math.random() - 0.5) * 2,
   802→          (Math.random() - 0.5) * 2, 1.5 + Math.random() * 2.5 * spdFrac, (Math.random() - 0.5) * 2,
   803→          0.5, 0.3 + Math.random() * 0.35, 0, 0.4, 10);
   804→      }
   805→      // крен: всплески у борта
   806→      const roll = this.euler.z;
   807→      this.rollSmooth += (roll - this.rollSmooth) * Math.min(1, dt * 3);
   808→      if (Math.abs(this.rollSmooth) > 0.28 && Math.random() < 0.3) {
   809→        const side = Math.sign(this.rollSmooth) * this.preset.beam * 0.5;
   810→        const px = this.position.x + Math.cos(this.heading) * side;
   811→        const pz = this.position.z - Math.sin(this.heading) * side;
   812→        particles.splash(px, waterHeightAt(px, pz, time), pz, 0.5);
   813→      }
   814→    }
   815→
   816→    // экипаж: анимация, потери (трупы/разрывы), эвакуация при потоплении
   817→    this.crew3d.update(dt, time, {
   818→      crew: this.crew, maxCrew: this.preset.maxCrew,
   819→      sinking: this.sinking, sunk: this.sunk,
   820→      position: this.position, quaternion: this.quaternion,
   821→    });
   822→  }
   823→
   824→  /** Сдвиг мира (floating origin) */
   825→  shiftWorld(delta: THREE.Vector3) {
   826→    this.position.add(delta);
   827→    this.wake.shiftWorld(delta);
   828→  }
   829→
   830→  dispose() {
   831→    this.visual.group.traverse((o) => {
   832→      const mesh = o as THREE.Mesh;
   833→      if (mesh.geometry) mesh.geometry.dispose();
   834→    });
   835→  }
   836→}
