    35→export class Ship {
    36→  preset: ShipPreset;
    37→  visual: ShipVisual;
    38→  isPlayer: boolean;
    39→
    40→  // --- физика ---
    41→  position = new THREE.Vector3();
    42→  quaternion = new THREE.Quaternion();
    43→  velocity = new THREE.Vector3();
    44→  angular = new THREE.Vector3(); // мировая угловая скорость
    45→  mass: number;
    46→  inertia = new THREE.Vector3(); // диагональ в связных осях (x=right, y=up, z=forward)
    47→  private probes: Probe[] = [];
    48→  private reserve: Probe[] = [];   // запас плавучести надводного борта
    49→  private force = new THREE.Vector3();
    50→  private torque = new THREE.Vector3();
    51→  private _f = new THREE.Vector3();
    52→  private _r = new THREE.Vector3();
    53→  private _t1 = new THREE.Vector3();
    54→  private _t2 = new THREE.Vector3();
    55→  private _t3 = new THREE.Vector3();
    56→  private tmpV = new THREE.Vector3();
    57→  private tmpV2 = new THREE.Vector3();
    58→  private tmpQ = new THREE.Quaternion();
    59→  private wq = new THREE.Quaternion();
    60→  private dq = new THREE.Quaternion();
    61→  private _e2 = new THREE.Euler();   // разбор кватерниона для посадки на волну
    62→  private _st = [-0.46, -0.23, 0, 0.23, 0.46]; // доли полудлины станций корпуса
    63→  private _hs = new Float64Array(5);  // высоты воды на станциях
    64→
    65→  // --- управление ---
    66→  sailTarget = 0;          // 0..3
    67→  /** Принудительная доля постановки парусов (0..1) — мишень полигона
    68→   *  стоит на якоре с поставленным рангоутом; null = обычная логика */
    69→  sailOverride: number | null = null;
    70→  sailFrac = 0;            // плавное 0..1
    71→  rudder = 0;              // -1..1 (реальный угол)
    72→  rudderTarget = 0;
    73→  anchored = false;
    74→  reverseThrust = false;
    75→  helmActive = false;
    76→
    77→  // --- состояние ---
    78→  hullHP: number; hullMax: number;
    79→  sailHP: number;
    80→  /** максимум парусов с учётом множителя прочности (clearDamage) */
    81→  sailMax: number;
    82→  crew: number;
    83→  water = 0;               // м³ в трюме
    84→  holdCap: number;
    85→  leakRate = 0;            // м³/с суммарная течь
    86→  onFire = false;
    87→  fireT = 0;
    88→  mastsBroken = 0;         // число сломанных (для тяги)
    89→  mastAnim = -1;           // (легаси) для совместимости
    90→  mastHP: number[] = [];   // HP каждой мачты
    91→  mastFall: number[] = []; // прогресс падения: -1 стоит, 0..1 валится/лежит
    92→  mastMaxHP = 32;
    93→  hullHoles: { side: number; belowWL: boolean; mesh: THREE.Mesh }[] = [];
    94→  leakBias = 0;            // суммарный кренящий момент пробоин (−1..1 порт/штирборт)
    95→  noSurrender = false;     // мишень полигона не сдаётся
    96→  crew3d: CrewVisual;      // фигуры на палубе
    97→  sinking = false;
    98→  sinkT = 0;
    99→  sunk = false;
   100→  surrendered = false;
   101→  grounded = false;
   102→  gold = 0;
   103→  cargo: Record<string, number> = {};
   104→  ammo: Record<string, number> = { ball: 90, chain: 30, grape: 60, fire: 20 };
   105→  planks = 12;
   106→  pumpAction = false;
   107→
   108→  // --- мир ---
   109→  wind: WindState;
   110→  terrainHeight: (x: number, z: number) => number;
   111→  particles: Particles;
   112→
   113→  // --- вода/след ---
   114→  foam: WaterlineFoam;
   115→  wake = new WakeTrail();
   116→  private lastStern = new THREE.Vector3();
   117→  private rollSmooth = 0;
   118→
   119→  // --- события колбэки ---
   120→  onSurrender?: (s: Ship) => void;
   121→  onSunk?: (s: Ship) => void;
   122→
   123→  constructor(presetId: string, isPlayer: boolean, wind: WindState,
   124→    terrainHeight: (x: number, z: number) => number, particles: Particles,
   125→    opts?: ShipOpts) {
   126→    this.preset = PRESETS[presetId];
   127→    this.isPlayer = isPlayer;
   128→    this.wind = wind;
   129→    this.terrainHeight = terrainHeight;
   130→    this.particles = particles;
   131→    this.visual = buildShip(this.preset);
   132→    // прочность по настройке игры (по умолчанию 1× — исторические ТТХ)
   133→    const mul = opts?.durabilityMul ?? 1;
   134→    this.hullHP = this.preset.hullHP * mul;
   135→    this.hullMax = this.hullHP;
   136→    this.sailHP = this.preset.sailHP * mul;
   137→    this.sailMax = this.sailHP;
   138→    this.mastMaxHP = 32 * mul;
   139→    if (opts?.ammo) this.ammo = { ...opts.ammo };
   140→    this.crew = Math.round(this.preset.maxCrew * 0.8);
   141→    this.holdCap = this.preset.cargo * 0.8;
   142→    this.mass = this.preset.mass;
   143→
   144→    const { length: L, beam: B, draft: D, freeboard: F } = this.preset;
   145→    // моменты инерции (связные оси: x — крен? нет: x=право, y=верх, z=нос)
   146→    // крен — вокруг z, дифферент — вокруг x, рыскание — вокруг y
   147→    this.inertia.set(
   148→      this.mass * (B * B + F * F) / 12 * 1.5,   // pitch (x)
   149→      this.mass * (L * L + B * B) / 12 * 1.2,   // yaw (y)
   150→      this.mass * (B * B + F * F) / 12 * 1.5,   // roll (z)
   151→    );
   152→
   153→    // зонды плавучести: 9 шпангоутов × 3 точки (ниже ватерлинии),
   154→    // вплоть до носа/кормы (±0.47 полудлины). Пружины жёсткие
   155→    // (нейтраль 0.38 осадки, запас ×1.35) и демпфер ζ≈0.62 — корпус
   156→    // БЫСТРО всходит на гребень и не проваливается внутрь волны.
   157→    const NZ = 9;
   158→    let n = 0;
   159→    for (let i = 0; i < NZ; i++) {
   160→      const t = i / (NZ - 1);
   161→      const z = THREE.MathUtils.lerp(-L * 0.47, L * 0.47, t);
   162→      const prof = this.halfWidthAt(z);
   163→      const offs = [0, -prof * 0.75, prof * 0.75];
   164→      for (const x of offs) {
   165→        this.probes.push({ off: new THREE.Vector3(x, -D * 0.38, z), k: 0, c: 0 });
   166→        n++;
   167→      }
   168→    }
   169→    const neutral = D * 0.38;
   170→    const kTotal = this.mass * G / neutral * 1.35;
   171→    for (const p of this.probes) {
   172→      p.k = kTotal / n;
   173→      p.c = 2 * Math.sqrt(p.k * (this.mass / n)) * 0.62; // демпфер качки
   174→    }
   175→    // Резервные зонды по скулам борта чуть выше спокойной ватерлинии:
   176→    // моделируют запас плавучести корпуса до планширя — гребень, накрывший
   177→    // палубу, встречает сильное сопротивление и кренящий момент вправку,
   178→    // «полностью по борт» корабль больше не погружается.
   179→    let rn = 0;
   180→    for (let i = 0; i < NZ; i++) {
   181→      const t = i / (NZ - 1);
   182→      const z = THREE.MathUtils.lerp(-L * 0.44, L * 0.44, t);
   183→      const prof = this.halfWidthAt(z) * 0.85;
   184→      for (const x of [-prof, prof]) {
   185→        this.reserve.push({ off: new THREE.Vector3(x, F * 0.34, z), k: 0, c: 0 });
   186→        rn++;
   187→      }
   188→    }
   189→    // жёстче прежнего: полный вес запаса уже на 15 см погружения борта —
   190→    // волна не может накрыть палубу, корпус «всплывает» на гребень
   191→    const rkTotal = this.mass * G / 0.15;
   192→    for (const p of this.reserve) {
   193→      p.k = rkTotal / rn;
   194→      p.c = 2 * Math.sqrt(p.k * (this.mass / rn)) * 1.3;
   195→    }
   196→
   197→    this.foam = new WaterlineFoam(this.visual.outline);
   198→    this.wake.setWaterHeight(waterHeightAt);
   199→    if (isPlayer) enableLantern(this.visual);
   200→
   201→    // экипаж на палубе: процедурные типажи, сид фиксирован по модели
   202→    const seed = Array.from(presetId).reduce((a, ch) => a + ch.charCodeAt(0) * 31, 11)
   203→      + Math.round(this.preset.length * 7);
   204→    this.crew3d = buildCrew(this.preset, seed);
   205→    this.visual.group.add(this.crew3d.group);
   206→    for (let i = 0; i < this.visual.masts.length; i++) {
   207→      this.mastHP.push(this.mastMaxHP);
   208→      this.mastFall.push(-1);
   209→    }
   210→  }
   211→
   212→  halfWidthAt(z: number): number {
   213→    const t = clamp((z + this.preset.length / 2) / this.preset.length, 0, 1);
   214→    let w: number;
   215→    if (t < 0.45) w = THREE.MathUtils.lerp(0.58, 1.0, t / 0.45);
   216→    else w = THREE.MathUtils.lerp(1.0, 0.0, Math.pow(clamp((t - 0.45) / 0.55, 0, 1), 0.75));
   217→    return Math.max(0.3, w * this.preset.beam * 0.5);
   218→  }
   219→
   220→  // ---------- геометрия ориентации ----------
   221→  get forward(): THREE.Vector3 {
   222→    const e = new THREE.Euler().setFromQuaternion(this.quaternion, 'YXZ');
   223→    return new THREE.Vector3(Math.sin(e.y), 0, Math.cos(e.y)).normalize();
   224→  }
   225→  get heading(): number {
   226→    const e = new THREE.Euler().setFromQuaternion(this.quaternion, 'YXZ');
   227→    return e.y;
   228→  }
   229→  get euler(): THREE.Euler { return new THREE.Euler().setFromQuaternion(this.quaternion, 'YXZ'); }
   230→  speed(): number { return Math.hypot(this.velocity.x, this.velocity.z); }
   231→  speedKnots(): number { return this.speed() * 1.9438; }
   232→
   233→  /** Угол между направлением ОТКУДА дует (кажущийся) ветер и курсом, 0..PI */
   234→  apparentWindAngle(): number {
   235→    const wf = this.tmpV.set(Math.cos(this.wind.from), 0, Math.sin(this.wind.from));
   236→    // кажущийся ветер = истинный - скорость корабля
   237→    const av = this.tmpV2.set(-wf.x * this.wind.speed - this.velocity.x, 0, -wf.z * this.wind.speed - this.velocity.z);
   238→    // вектор ОТКУДА дует кажущийся ветер:
   239→    const aFromX = -av.x, aFromZ = -av.z;
   240→    const len = Math.hypot(aFromX, aFromZ) || 1;
   241→    const f = this.forward;
   242→    const dot = (aFromX / len) * f.x + (aFromZ / len) * f.z;
   243→    return Math.acos(clamp(dot, -1, 1));
   244→  }
   245→
   246→  /** Полярная эффективность парусов по углу (рад): лавировка/бейдевинд/галфвинд/фордевинд */
   247→  static polar(awa: number): number {
   248→    const d = awa * 180 / Math.PI;
   249→    if (d < 25) return 0.02;                       // левентик — почти нет хода
   250→    if (d < 60) return THREE.MathUtils.lerp(0.08, 0.7, (d - 25) / 35);
   251→    if (d < 110) return THREE.MathUtils.lerp(0.7, 1.0, (d - 60) / 50);  // бейдевинд-галфвинд
   252→    if (d < 150) return THREE.MathUtils.lerp(1.0, 0.82, (d - 110) / 40);
   253→    return THREE.MathUtils.lerp(0.82, 0.68, (d - 150) / 30);            // фордевинд
   254→  }
   255→
   256→  // ---------- фиксированный шаг физики ----------
   257→  fixedStep(dt: number, time: number) {
   258→    if (this.sunk) return;
   259→    this.force.set(0, 0, 0);
   260→    this.torque.set(0, 0, 0);
   261→
   262→    const f = this._f.copy(this.forward);
   263→    // штирборт (правый борт): forward × up = (−f.z, 0, f.x)
   264→    const right = this._r.set(-f.z, 0, f.x).normalize();
   265→    const e = this.euler;
   266→    const spd = this.speed();
   267→
   268→    const m = this.mass + this.water * 1000; // вода в трюме утяжеляет
   269→
   270→    // --- гравитация ---
   271→    this.force.y -= m * G;
   272→
   273→    // --- плавучесть по зондам (τ = r × F, F = (0, fl, 0)) ---
   274→    for (const p of this.probes) {
   275→      const wp = this._t1.copy(p.off).applyQuaternion(this.quaternion).add(this.position);
   276→      const wy = waterHeightAt(wp.x, wp.z, time);
   277→      const depth = wy - wp.y;
   278→      if (depth > 0) {
   279→        const rx = wp.x - this.position.x, rz = wp.z - this.position.z;
   280→        // вертикальная скорость точки: v = ω × r → vy = ωz*rx − ωx*rz
   281→        const vy = this.velocity.y + (this.angular.z * rx - this.angular.x * rz);
   282→        let fl = p.k * Math.min(depth, 6) - p.c * vy;
   283→        if (fl < 0) fl = 0;
   284→        this.force.y += fl;
   285→        this.torque.x += -rz * fl;
   286→        this.torque.z += rx * fl;
   287→      }
   288→    }
   289→    // --- запас плавучести бортов (пока не тонем) ---
   290→    if (!this.sinking) {
   291→      for (const p of this.reserve) {
   292→        const wp = this._t1.copy(p.off).applyQuaternion(this.quaternion).add(this.position);
   293→        const wy = waterHeightAt(wp.x, wp.z, time);
   294→        const depth = wy - wp.y;
   295→        if (depth > 0) {
   296→          const rx = wp.x - this.position.x, rz = wp.z - this.position.z;
   297→          const vy = this.velocity.y + (this.angular.z * rx - this.angular.x * rz);
   298→          let fl = p.k * Math.min(depth, 2.5) - p.c * vy;
   299→          if (fl < 0) fl = 0;
   300→          this.force.y += fl;
   301→          this.torque.x += -rz * fl;
   302→          this.torque.z += rx * fl;
   303→        }
   304→      }
   305→    }
   306→
   307→    // --- ветер на парусах ---
   308→    this.sailFrac += (this.sailTarget / 3 - this.sailFrac) * Math.min(1, dt * 0.7);
   309→    const awa = this.apparentWindAngle();
   310→    const polar = Ship.polar(awa);
   311→    const sailEffMul = this.sailHP / this.preset.sailHP;
   312→    const damaged = sailEffMul < 0.5 ? 0.45 : sailEffMul < 0.85 ? 0.8 : 1.0;
   313→    const windPower = Math.pow(this.wind.speed / 8.0, 1.6);
   314→    let thrust = 0;
   315→    if (!this.anchored && !this.sinking && this.mastsBroken < 2) {
   316→      thrust = this.sailFrac * polar * windPower * this.preset.maxSpeed * 7.0 * damaged * (this.sailHP > 0 ? 1 : 0);
   317→    }
   318→    if (this.reverseThrust && !this.anchored) thrust = -this.preset.maxSpeed * 0.45; // задний ход на вёслах
   319→    // ГРЕБНЫЕ суда (шебека, галера): вёсла дают ход даже в полный штиль;
   320→    // под парусами вёсла помогают меньше — экипаж занят парусами
   321→    if (this.preset.oars && !this.anchored && !this.sinking) {
   322→      const crewFrac = clamp(this.crew / this.preset.maxCrew, 0.2, 1);
   323→      const oarThrust = this.preset.maxSpeed * 0.34 * crewFrac * (1 - 0.55 * this.sailFrac);
   324→      thrust += oarThrust;
   325→    }
   326→    // сила тяги вдоль forward
   327→    this.force.x += f.x * thrust * m * 0.06;
   328→    this.force.z += f.z * thrust * m * 0.06;
   329→    // крен от ветра: боковая составляющая ветра на парусах
   330→    // (умеренный: в шторм подветренный борт не зарывается в волну)
   331→    const wf = new THREE.Vector3(Math.cos(this.wind.from), 0, Math.sin(this.wind.from));
   332→    const windTravel = wf.clone().multiplyScalar(-this.wind.speed);
   333→    const lateral = windTravel.dot(right);
   334→    const heelF = lateral * this.sailFrac * windPower * m * 0.034;
   335→    this.torque.z += heelF;
   336→    // ... и небольшое боковое сносовое усилие
   337→    this.force.x += right.x * lateral * this.sailFrac * m * 0.004;
   338→    this.force.z += right.z * lateral * this.sailFrac * m * 0.004;
   339→
   340→    // --- руль ---
   341→    this.rudder += (this.rudderTarget - this.rudder) * Math.min(1, dt * 1.8);
   342→    const rudderAngle = this.rudder * 0.6; // рад, макс ~34°
   343→    const yawAuth = clamp(spd / 4.0, 0, 1.15); // авторитеты руля от скорости
   344→    // положительный руль (D) = поворот ВПРАВО (штирборт):
   345→    // вращение вокруг −Y, т.к. +Y = поворот влево (к порту)
   346→    this.torque.y -= rudderAngle * yawAuth * m * 12.0; // ~6°/с на полном ходу
   347→    // крен на повороте (отцентробежный)
   348→    this.torque.z += this.angular.y * spd * m * 0.35;
   349→    // лёгкое зарывание носа на скорости (слабое: сильное топило борт
   350→    // в волну на 3+ метра — см. жалобы «корабль погружается по борт»)
   351→    this.torque.x += spd * spd * m * 0.0012;
   352→
   353→    // --- сопротивление (по связным осям) ---
   354→    const vLocal = this.velocity.clone();
   355→    const fwdSpd = vLocal.dot(f);
   356→    const latSpd = vLocal.dot(right);
   357→    const fwdDrag = -Math.sign(fwdSpd) * (18 * fwdSpd + 55 * fwdSpd * Math.abs(fwdSpd)) * m * 0.001;
   358→    const latDrag = -latSpd * (3.2) * m * 0.1;
   359→    this.force.x += f.x * fwdDrag + right.x * latDrag;
   360→    this.force.z += f.z * fwdDrag + right.z * latDrag;
   361→    this.force.y -= this.velocity.y * m * 0.12; // вертикальное демпфирование
   362→    if (this.anchored) {
   363→      this.force.x -= this.velocity.x * m * 1.2;
   364→      this.force.z -= this.velocity.z * m * 1.2;
   365→    }
   366→    if (this.grounded) {
   367→      this.force.x -= this.velocity.x * m * 2.5;
   368→      this.force.z -= this.velocity.z * m * 2.5;
   369→    }
   370→
   371→    // --- угловое демпфирование по связным осям ---
   372→    const invQ = this.tmpQ.copy(this.quaternion).invert();
   373→    const ab = this._t3.copy(this.angular).applyQuaternion(invQ); // связные
   374→    ab.x *= Math.exp(-1.6 * dt);  // дифферент
   375→    ab.y *= Math.exp(-0.9 * dt);  // рыскание
   376→    ab.z *= Math.exp(-1.1 * dt);  // крен
   377→    this.angular.copy(ab.applyQuaternion(this.quaternion));
   378→
   379→    // --- ограничение крена/дифферента (мягкие стопы) ---
   380→    // дифферент 0.30 рад (17°) — нос всходит на передний склон
   381→    // штормового гребня, а не прорезает его;
   382→    // крен 0.36 рад (20°) — подветренный борт остаётся над водой
   383→    const roll = e.z, pitch = e.x;
   384→    const rollLim = 0.36, pitchLim = 0.30;
   385→    if (Math.abs(roll) > rollLim) this.torque.z -= (roll - Math.sign(roll) * rollLim) * m * 9.0;
   386→    if (Math.abs(pitch) > pitchLim) this.torque.x -= (pitch - Math.sign(pitch) * pitchLim) * m * 12.0;
   387→    // затопление кренит корабль в сторону пробоин
   388→    this.floodRoll(m);
   389→    if (this.sinking) {
   390→      this.sinkT += dt;
   391→      this.force.y -= m * G * Math.min(0.9, this.sinkT / 9); // неотвратимое погружение
   392→      this.torque.x += m * 0.25 * Math.sign(roll || 1);
   393→      this.torque.z += m * 0.4 * Math.sign(roll || 1);
   394→    }
   395→
   396→    // --- интеграция ---
   397→    this.velocity.addScaledVector(this.force, dt / m);
   398→    this.position.addScaledVector(this.velocity, dt);
   399→    // α = I⁻¹ τ в связных осях
   400→    const tb = this._t3.copy(this.torque).applyQuaternion(invQ);
   401→    tb.x /= this.inertia.x; tb.y /= this.inertia.y; tb.z /= this.inertia.z;
   402→    this.angular.addScaledVector(tb.applyQuaternion(this.quaternion), dt);
   403→    // ограничим угловые скорости
   404→    const wLen = this.angular.length();
   405→    if (wLen > 1.1) this.angular.multiplyScalar(1.1 / wLen);
   406→    // q̇ = 0.5 · ω⊗q → q_new = q + 0.5·dt·(ωx,ωy,ωz,0)⊗q, потом нормализация
   407→    this.wq.set(this.angular.x, this.angular.y, this.angular.z, 0);
   408→    this.dq.copy(this.wq).multiply(this.quaternion); // dq = ω ⊗ q
   409→    const qs = 0.5 * dt;
   410→    this.quaternion.set(
   411→      this.quaternion.x + this.dq.x * qs,
   412→      this.quaternion.y + this.dq.y * qs,
   413→      this.quaternion.z + this.dq.z * qs,
   414→      this.quaternion.w + this.dq.w * qs
   415→    ).normalize();
   416→
   417→    // === КИНЕМАТИЧЕСКАЯ ПОСАДКА НА ВОЛНУ (плывёт ПОВЕРХНОСТЬЮ, без парения) ===
   418→    // Итог трёх итераций:
   419→    //  1) среднее 4 точек — гребень проходил СКВОЗЬ корпус (жалоба);
   420→    //  2) жёсткая огибающая (max по станциям) — корабль «повис над водой»,
   421→    //     киль ни разу не доходил до нормальной осадки (VLM: «парит»);
   422→    //  3) текущий баланс:
   423→    //   - ВЕРТИКАЛЬ — средний уровень, ВЗВЕШЕННЫЙ по ширине корпуса
   424→    //     (широкий мидель важнее узких оконечностей — как настоящая плавучесть);
   425→    //   - МЯГКАЯ огибающая: если гребень под какой-то станцией зарывает киль
   426→    //     глубже (0.80+0.35)·осадки — корпус мягко поднимается, но суммарный
   427→    //     подъём ограничен 0.45·осадки → парение исключено, а зарывание
   428→    //     носа в передний склон гребня остаётся в пределах «реза воды»;
   429→    //   - ДИФФЕРЕНТ — МНК-наклон поверхности (вес по ширине): нос задирается
   430→    //     заранее, на переднем склоне приближающегося гребня.
   431→    // Крен остаётся живым (ветер/течь/поворот) — пружины бортов.
   432→    if (!this.sinking && !this.sunk && !this.grounded) {
   433→      const L = this.preset.length, D = this.preset.draft;
   434→      const FR = this._st;
   435→      const sinP = Math.sin(e.x);
   436→      const halfB = Math.max(0.5, this.preset.beam * 0.5);
   437→      // проход 1: высоты воды и веса станций (ширина корпуса в точке)
   438→      let wSum = 0, whSum = 0, whzSum = 0, wz2Sum = 0;
   439→      for (let i = 0; i < FR.length; i++) {
   440→        const zz = FR[i] * L;
   441→        const h = waterHeightAt(this.position.x + f.x * zz, this.position.z + f.z * zz, time);
   442→        // вес 0.25..1: оконечности узкие — их гребни корпус не поднимают всем весом
   443→        const w = Math.max(0.25, this.halfWidthAt(zz) / halfB);
   444→        this._hs[i] = h;
   445→        wSum += w;
   446→        whSum += w * h;
   447→        whzSum += w * h * zz;
   448→        wz2Sum += w * zz * zz;
   449→      }
   450→      // спокойная посадка: центр на 0.10·D над средневзвешенным уровнем;
   451→      // затопленный трюм садит корпус ниже (до 30 см)
   452→      const sink = clamp(this.water / Math.max(1, this.holdCap), 0, 1) * 0.30;
   453→      const meanY = whSum / wSum + D * 0.10 - sink;
   454→      // проход 2: мягкая огибающая против зарывания оконечностей.
   455→      // Заглубление киля станции при среднем уровне: dig = h + z·sinP + 0.9D − meanY;
   456→      // нормальная осадка даёт dig ≈ 0.8D — лишнее свыше 0.35D выталкиваем,
   457→      // но подъём capped 0.45D (выше — парение, см. VLM-проверку v31)
   458→      let lift = 0;
   459→      for (let i = 0; i < FR.length; i++) {
   460→        const zz = FR[i] * L;
   461→        const dig = this._hs[i] + zz * sinP + 0.9 * D - meanY;
   462→        const l = dig - 1.15 * D;
   463→        if (l > lift) lift = l;
   464→      }
   465→      const targetY = meanY + clamp(lift, 0, 0.45 * D);
   466→      const kY = 1 - Math.exp(-dt * 5.5);
   467→      const corrY = clamp((targetY - this.position.y) * kY, -14 * dt, 14 * dt);
   468→      this.position.y += corrY;
   469→      // согласованное затухание вертикальной скорости — пружины
   470→      // не раскачивают корпус вокруг кинематической посадки
   471→      this.velocity.y *= Math.exp(-dt * 3.0);
   472→      // дифферент: МНК-наклон поверхности вдоль ДП (вода выше у носа → нос вверх)
   473→      const wPitch = -Math.atan2(whzSum, wz2Sum);
   474→      const e2 = this._e2.setFromQuaternion(this.quaternion, 'YXZ');
   475→      const kP = 1 - Math.exp(-dt * 5.0);
   476→      e2.x += clamp((wPitch - e2.x) * kP, -1.2 * dt, 1.2 * dt);
   477→      this.quaternion.setFromEuler(e2);
   478→    }
   479→
   480→    // --- посадка на мель / столкновение с землёй ---
   481→    this.checkGrounding(dt);
   482→
   483→    // --- течение воды в трюме ---
   484→    if (this.leakRate > 0) {
   485→      this.water = Math.min(this.holdCap, this.water + this.leakRate * dt);
   486→    }
   487→    if (this.onFire) {
   488→      this.fireT += dt;
   489→      this.hullHP -= dt * 1.6;
   490→      if (this.fireT > 26) { this.onFire = false; this.fireT = 0; } // выгорит
   491→      if (Math.random() < dt * 1.2) {
   492→        const p = this.particles;
   493→        const wp = this.position.clone();
   494→        p.fire(wp.x + (Math.random() - 0.5) * 4, this.waterlineY() + 1 + Math.random() * 2, wp.z + (Math.random() - 0.5) * 10);
   495→        p.smoke(wp.x + (Math.random() - 0.5) * 4, this.waterlineY() + 3, wp.z + (Math.random() - 0.5) * 10, this.wind ? Math.cos(this.wind.from + Math.PI) * 2 : 0, 2.2, this.wind ? Math.sin(this.wind.from + Math.PI) * 2 : 0, 2.8, 3.0);
   496→      }
   497→    }
   498→    // порог затопления
   499→    if (!this.sinking && this.water > this.holdCap * 0.62) this.startSinking();
   500→    if (this.hullHP <= 0 && !this.sinking) this.startSinking();
   501→
   502→    // --- сдался? ---
   503→    if (!this.isPlayer && !this.surrendered && !this.sinking) {
   504→      if (this.hullHP < this.hullMax * 0.34 || this.crew < this.preset.maxCrew * 0.25) {
   505→        this.surrendered = true;
   506→        this.sailTarget = 0;
   507→        this.onSurrender?.(this);
   508→      }
   509→    }
   510→
   511→    if (this.sinking && this.sinkT > 14 && !this.sunk) {
   512→      this.sunk = true;
   513→      this.visual.group.visible = false;
   514→      this.onSunk?.(this);
   515→    }
   516→  }
   517→
   518→  private checkGrounding(dt: number) {
   519→    if (this.sinking || this.sunk) return;
   520→    const f = this.forward;
   521→    const bow = this.tmpV.copy(this.position).addScaledVector(f, this.preset.length * 0.42);
   522→    const stern = this.tmpV2.copy(this.position).addScaledVector(f, -this.preset.length * 0.42);
   523→    let hit = false;
   524→    for (const p of [bow, stern]) {
   525→      const th = this.terrainHeight(p.x, p.z);
   526→      if (th > this.position.y - this.preset.draft * 0.6) {
   527→        hit = true;
   528→        // нормаль склона
   529→        const e = 2;
   530→        const nx = this.terrainHeight(p.x - e, p.z) - this.terrainHeight(p.x + e, p.z);
   531→        const nz = this.terrainHeight(p.x, p.z - e) - this.terrainHeight(p.x, p.z + e);
   532→        const n = new THREE.Vector3(nx, 2 * e, nz).normalize();
   533→        this.force.addScaledVector(n, this.mass * 22);
   534→        this.velocity.multiplyScalar(0.94);
   535→      }
   536→    }
   537→    if (hit) {
   538→      if (!this.grounded && this.speed() > 3.5) {
   539→        this.applyDamage({ kind: 'ground', amount: 5 + this.speed() });
   540→      }
   541→      this.grounded = true;
   542→    } else this.grounded = false;
   543→  }
   544→
   545→  waterlineY(): number {
   546→    return this.position.y;
   547→  }

   549→  // ---------- повреждения ----------
   550→  applyDamage(d: DamageInput) {
   551→    if (this.sunk || this.sinking) return;
   552→    const rng = Math.random;
   553→    switch (d.kind) {
   554→      case 'ball': {
   555→        this.hullHP -= d.amount;
   556→        if (d.hitPoint && d.hitPoint.y < 0.45 + this.position.y) this.addLeak(0.06 + rng() * 0.05);
   557→        if (rng() < 0.3) this.crew = Math.max(0, this.crew - 1);
   558→        // случайный «рикошет в рангоут» — редкий, основную работу делают
   559→        // прицельные попадания из CombatSystem (mast hits)
   560→        if (rng() < 0.10) this.breakMast();
   561→        break;
   562→      }
   563→      case 'chain': {
   564→        this.sailHP = Math.max(0, this.sailHP - d.amount);
   565→        if (rng() < 0.25) this.crew = Math.max(0, this.crew - 1);
   566→        break;
   567→      }
   568→      case 'grape': {
   569→        this.crew = Math.max(0, this.crew - Math.round(d.amount));
   570→        this.hullHP -= d.amount * 0.4;
   571→        break;
   572→      }
   573→      case 'fire': {
   574→        this.hullHP -= d.amount;
   575→        if (rng() < 0.35 && !this.onFire) { this.onFire = true; this.fireT = 0; }
   576→        if (rng() < 0.08) { this.addLeak(0.03); }
   577→        break;
   578→      }
   579→      case 'ram': {
   580→        this.hullHP -= d.amount;
   581→        if (rng() < 0.5) this.addLeak(0.04);
   582→        break;
   583→      }
   584→      case 'ground': this.hullHP -= d.amount * 0.5; break;
   585→    }
   586→    if (this.hullHP <= 0 && !this.sinking) this.startSinking();
   587→    if (!this.isPlayer && !this.noSurrender && !this.surrendered && this.hullHP < this.hullMax * 0.34) {
   588→      this.surrendered = true; this.sailTarget = 0; this.onSurrender?.(this);
   589→    }
   590→  }

   592→  /** Урон в КОНКРЕТНУЮ мачту (прицельное попадание ядром/книппелем) */
   593→  damageMast(idx: number, amount: number): boolean {
   594→    if (idx < 0 || idx >= this.mastHP.length) return false;
   595→    if (this.mastFall[idx] >= 0) return false; // уже сломана
   596→    this.mastHP[idx] -= amount;
   597→    if (this.mastHP[idx] <= 0) { this.breakMast(idx); return true; }
   598→    return false;
   599→  }
   600→
   601→  addLeak(rate: number) {
   602→    this.leakRate = Math.min(1.8, this.leakRate + rate);
   603→    if (this.leakRate < 0) this.leakRate = 0;
   604→  }

   606→  /** Урон ПАРУСАМ: пробоины картечи/книппеля снижают тягу (sailHP).
   607→   *  Дыры рисуются отдельно — см. combat.applySailHit + sailAddHole. */
   608→  damageSails(n: number) {
   609→    this.sailHP = Math.max(0, this.sailHP - n);
   610→  }

   612→  /** Пробоина в борту: декаль + течь (ниже ватерлинии) + кренящий момент */
   613→  addHole(localPos: THREE.Vector3, belowWL: boolean) {
   614→    const side = localPos.x >= 0 ? 1 : -1;
   615→    // тёмная «рваная» дыра: круг + обломки досок вокруг
   616→    const g = new THREE.CircleGeometry(0.34 + Math.random() * 0.14, 9);
   617→    const mat = new THREE.MeshStandardMaterial({ color: 0x120d08, roughness: 1 });
   618→    const mesh = new THREE.Mesh(g, mat);
   619→    const w = Math.max(0.3, this.halfWidthAt(localPos.z) * 0.985);
   620→    mesh.position.set(side * w, localPos.y, localPos.z);
   621→    mesh.rotation.y = side * Math.PI / 2;
   622→    mesh.rotation.x = (Math.random() - 0.5) * 0.5;
   623→    this.visual.group.add(mesh);
   624→    this.hullHoles.push({ side, belowWL, mesh });
   625→    if (belowWL) {
   626→      this.addLeak(0.16 + Math.random() * 0.10);
   627→      this.leakBias = clamp(this.leakBias + side * 0.35, -1, 1);
   628→    }
   629→  }

   631→  /** Затопление: крен в сторону пробоин + вода в трюме утяжеляет */
   632→  private floodRoll(m: number) {
   633→    if (this.water <= 0 || this.sinking) return;
   634→    const frac = this.water / this.holdCap;
   635→    this.torque.z += this.leakBias * frac * m * 0.06;
   636→  }

   638→  breakMast(idx?: number) {
   639→    let i = idx;
   640→    if (i === undefined || i < 0 || i >= this.visual.masts.length || this.mastFall[i] >= 0) {
   641→      // без указания — случайная целая мачта
   642→      const intact = this.mastFall.map((f, k) => f < 0 ? k : -1).filter(k => k >= 0);
   643→      if (!intact.length) return;
   644→      i = intact[Math.floor(Math.random() * intact.length)];
   645→    }
   646→    this.mastFall[i] = 0;
   647→    this.mastsBroken = this.mastFall.filter(f => f >= 0).length;
   648→    this.sailHP = Math.max(0, this.sailHP - 30);
   649→    if (this.sailTarget > 0) this.sailTarget = Math.max(0, this.sailTarget - 1);
   650→  }

   652→  /** Сброс ВСЕХ повреждений (полигон/возрождение): корпус, мачты,
   653→   *  паруса, пробоины, течь, пожар, экипаж */
   654→  clearDamage() {
   655→    this.hullHP = this.hullMax;
   656→    this.sailHP = this.sailMax;
   657→    this.water = 0;
   658→    this.leakRate = 0;
   659→    this.leakBias = 0;
   660→    this.onFire = false;
   661→    this.fireT = 0;
   662→    this.sinking = false;
   663→    this.sinkT = 0;
   664→    this.sunk = false;
   665→    this.surrendered = false;
   666→    this.mastsBroken = 0;
   667→    this.mastAnim = -1;
   668→    for (let i = 0; i < this.visual.masts.length; i++) {
   669→      this.mastHP[i] = this.mastMaxHP;
   670→      this.mastFall[i] = -1;
   671→      const m = this.visual.masts[i];
   672→      m.rotation.set(0, 0, 0);
   673→      m.position.y = 0;
   674→    }
   675→    for (const h of this.hullHoles) {
   676→      this.visual.group.remove(h.mesh);
   677→      h.mesh.geometry.dispose();
   678→      (h.mesh.material as THREE.Material).dispose();
   679→    }
   680→    this.hullHoles = [];
   681→    this.crew = Math.round(this.preset.maxCrew * 0.8);
   682→    this.crew3d.reset();
   683→    this.visual.group.visible = true;
   684→    this.velocity.set(0, 0, 0);
   685→    this.angular.set(0, 0, 0);
   686→    this.sailTarget = 0;
   687→    this.rudder = 0;
   688→    this.rudderTarget = 0;
   689→  }

   691→  startSinking() {
   692→    if (this.sinking) return;
   693→    this.sinking = true;
   694→    this.sinkT = 0;
   695→    this.onFire = false;
   696→  }

   698→  /** Действие команды (R): пожар -> течь -> ремонт */
   699→  crewAction(dt: number): string | null {
   700→    if (this.onFire) { this.fireT += dt * 3; if (this.fireT > 4) { this.onFire = false; this.fireT = 0; return 'Пожар потушен'; } return 'Тушение пожара...'; }
   701→    if (this.water > 0.01) {
   702→      const pump = 0.05 + this.crew * 0.004;
   703→      this.water = Math.max(0, this.water - pump * dt);
   704→      this.leakRate = Math.max(0, this.leakRate - 0.004 * dt);
   705→      return this.water > 0.05 ? 'Откачка воды...' : 'Трюм осушён';
   706→    }
   707→    if (this.hullHP < this.hullMax && this.planks > 0) {
   708→      this.hullHP = Math.min(this.hullMax, this.hullHP + 0.9 * dt);
   709→      this.planks -= dt * 0.15;
   710→      return 'Ремонт корпуса...';
   711→    }
   712→    return null;
   713→  }

   715→  // ---------- визуал ----------
   716→  updateVisual(time: number, dt: number, particles: Particles, negShift?: THREE.Vector3) {
   717→    const v = this.visual;
   718→    v.group.position.copy(this.position);
   719→    v.group.quaternion.copy(this.quaternion);
   720→
   721→    // паруса: количество поставленных парусов соответствует ФАКТИЧЕСКОЙ
   722→    // СКОРОСТИ — экипаж ставит паруса по мере разгона и убирает по мере
   723→    // остановки; уровень W/S задаёт максимум. Рея брасопится по ветру,
   724→    // «пузо» — по кажущемуся ветру. Паруса СЛОМАННОЙ мачты обвисают.
   725→    const awa = this.apparentWindAngle();
   726→    const polar = Ship.polar(awa);
   727→    const windClamp = clamp(this.wind.speed / 9, 0.3, 1.4);
   728→    // скорость, достижимая под полным парусным вооружением в этот ветер
   729→    // (корень из тяги — сопротивление квадратично): ~0.85·maxSpeed
   730→    const windPow = Math.pow(clamp(this.wind.speed / 8.0, 0.05, 1.6), 1.6);
   731→    const attainable = this.preset.maxSpeed * 0.85 * Math.sqrt(clamp(polar * windPow, 0, 1.5));
   732→    // в левентик/штиль паруса остаются на мачтах и трепещут (не убираются
   733→    // механически) — убираются только когда есть чем идти, но нет хода
   734→    const spdSetFrac = attainable > 1.5 ? clamp(this.speed() / attainable, 0, 1) : 1;
   735→    const setTotal = this.sailOverride ?? Math.min(this.sailFrac, spdSetFrac);
   736→    // локальный вектор «откуда дует» (x — борт, z — нос): ψ = φ + курс
   737→    const psi = this.wind.from + this.heading;
   738→    const wlX = Math.cos(psi); // >0 — ветер с порта (левый борт)
   739→    // цель брасопки: рея поперёк ветра (нормаль паруса по ветру), ±55°
   740→    let bt = Math.atan2(wlX, Math.sin(psi)) - Math.PI;
   741→    bt = Math.atan2(Math.sin(bt), Math.cos(bt)); // кратчайшая дуга
   742→    const braceTarget = clamp(bt, -0.96, 0.96);
   743→    const bellyBase = (0.35 + 0.95 * windClamp) * (0.15 + 0.85 * polar);
   744→    const flutterBase = 0.012 * windClamp * (1 + (1 - polar) * 2.2);
   745→    for (const s of v.sails.sails) {
   746→      // мачта-носитель сломана → парус безжизненно обвисает
   747→      const mastDead = s.mastIndex !== undefined && this.mastFall[s.mastIndex] >= 0;
   748→      // доля постановки этого паруса при текущей скорости хода
   749→      const setFrac = mastDead ? 0 : smoothstep(s.setAt - 0.38, s.setAt, setTotal);
   750→      s.furl += ((1 - setFrac) - s.furl) * Math.min(1, dt * (mastDead ? 2.2 : 1.1));
   751→      const sf = 1 - s.furl;
   752→      s.mat.uniforms.uTime.value = time;
   753→      s.mat.uniforms.uFurl.value = s.furl;
   754→      s.mat.uniforms.uDamage.value = 1 - this.sailHP / this.preset.sailHP;
   755→      s.mat.uniforms.uFlutter.value = flutterBase * (0.25 + 0.75 * sf);
   756→      if (s.square && s.brace) {
   757→        // реи разворачиваются медленно — как будто работают брашпили
   758→        s.braceAngle += ((mastDead ? 0 : braceTarget) - s.braceAngle) * Math.min(1, dt * 0.5);
   759→        s.brace.rotation.y = s.braceAngle;
   760→        s.mat.uniforms.uBelly.value = mastDead ? 0 : bellyBase * sf;
   761→      } else {
   762→        // кливер: пузо в подветренную сторону (локальный Z → борт)
   763→        s.mat.uniforms.uBelly.value = mastDead ? 0 : bellyBase * sf * (-wlX);
   764→      }
   765→    }
   766→
   767→    // флаг — по ветру
   768→    const wf = Math.atan2(-Math.cos(this.wind.from), -Math.sin(this.wind.from));
   769→    v.flag.rotation.y = wf - this.heading + Math.PI;
   770→    v.flagMat.uniforms.uTime.value = time;
   771→
   772→    // штурвал крутится от руля: D (руль вправо) — по часовой от рулевого
   773→    v.wheel.rotation.z = this.rudder * 2.2;
   774→
   775→    // руль за кормой следует за рулём
   776→    if (v.rudderMesh) v.rudderMesh.rotation.y = this.rudder * 0.45;
   777→
   778→    // вёсла: гребки в противофазе бортов (сила гребков — от хода)
   779→    if (v.oars && v.oars.length) {
   780→      const rowing = clamp(this.speed() / Math.max(1, this.preset.maxSpeed * 0.4), 0.12, 1);
   781→      for (const o of v.oars) {
   782→        const base = o.userData.base as number;
   783→        const phase = o.userData.phase as number;
   784→        o.rotation.z = base + Math.sin(time * 2.3 + phase) * 0.16 * rowing;
   785→      }
   786→    }
   787→
   788→    // падение мачт: каждая валится вперёд-вбок со своим наклоном,
   789→    // реи и паруса едут вместе с мачтой (они её потомки)
   790→    for (let i = 0; i < v.masts.length; i++) {
   791→      const f = this.mastFall[i];
   792→      if (f >= 0 && f < 1) {
   793→        this.mastFall[i] = Math.min(1, f + dt * 0.42);
   794→        const k = this.mastFall[i];
   795→        const lean = (i % 2 === 0 ? 1 : -1) * 0.3;
   796→        v.masts[i].rotation.x = k * 1.25;
   797→        v.masts[i].rotation.z = k * lean;
   798→        v.masts[i].position.y = -k * 1.6;
   799→      }
   800→    }
   801→
   802→    // пенный воротник по ватерлинии: высоту волн добавляет ВЕРШИННЫЙ
   803→    // шейдер (тот же Gerstner, что у океана) — кольцо лежит на гребнях;
   804→    // на CPU остаётся только горизонтальная привязка к кораблю
   805→    if (!this.sunk) {
   806→      this.foam.mesh.position.set(this.position.x, 0.30, this.position.z);
   807→      this.foam.mesh.rotation.y = this.heading;
   808→      const spdFrac = clamp(this.speed() / this.preset.maxSpeed, 0, 1);
   809→      this.foam.update(time, spdFrac, this.euler.z, negShift);
   810→    } else {
   811→      this.foam.mesh.visible = false;
   812→    }
   813→
   814→    // кильватерный след
   815→    const sternW = this.tmpV.set(0, 0, -this.preset.length * 0.48)
   816→      .applyQuaternion(this.quaternion).add(this.position);
   817→    const spdFrac = clamp(this.speed() / this.preset.maxSpeed, 0, 1);
   818→    if (!this.sunk && !this.grounded) {
   819→      this.wake.push(sternW, spdFrac, dt, time);
   820→    }
   821→    this.wake.update(time);
   822→
   823→    // носовой бурун: брызги при ударе носа о волну
   824→    if (!this.sunk) {
   825→      const bowW = this.tmpV2.set(0, -this.preset.draft * 0.3, this.preset.length * 0.46)
   826→        .applyQuaternion(this.quaternion).add(this.position);
   827→      const waveY = waterHeightAt(bowW.x, bowW.z, time);
   828→      const relV = this.velocity.y - (waveY - this.lastStern.y) * 0; // грубая оценка
   829→      const impact = (waveY - bowW.y) * (0.3 + spdFrac);
   830→      if (impact > 0.55 && Math.random() < 0.6) {
   831→        particles.splash(bowW.x, waveY, bowW.z, 0.5 + spdFrac * 0.9);
   832→      }
   833→      if (spdFrac > 0.35 && Math.random() < spdFrac * 0.5) {
   834→        particles.spawn(bowW.x + (Math.random() - 0.5) * 2, waveY + 0.1, bowW.z + (Math.random() - 0.5) * 2,
   835→          (Math.random() - 0.5) * 2, 1.5 + Math.random() * 2.5 * spdFrac, (Math.random() - 0.5) * 2,
   836→          0.5, 0.3 + Math.random() * 0.35, 0, 0.4, 10);
   837→      }
   838→      // крен: всплески у борта
   839→      const roll = this.euler.z;
   840→      this.rollSmooth += (roll - this.rollSmooth) * Math.min(1, dt * 3);
   841→      if (Math.abs(this.rollSmooth) > 0.28 && Math.random() < 0.3) {
   842→        const side = Math.sign(this.rollSmooth) * this.preset.beam * 0.5;
   843→        const px = this.position.x + Math.cos(this.heading) * side;
   844→        const pz = this.position.z - Math.sin(this.heading) * side;
   845→        particles.splash(px, waterHeightAt(px, pz, time), pz, 0.5);
   846→      }
   847→    }
   848→
   849→    // экипаж: анимация, потери (трупы/разрывы), эвакуация при потоплении
   850→    this.crew3d.update(dt, time, {
   851→      crew: this.crew, maxCrew: this.preset.maxCrew,
   852→      sinking: this.sinking, sunk: this.sunk,
   853→      position: this.position, quaternion: this.quaternion,
   854→    });
   855→  }
   856→
   857→  /** Сдвиг мира (floating origin) */
   858→  shiftWorld(delta: THREE.Vector3) {
   859→    this.position.add(delta);
   860→    this.wake.shiftWorld(delta);
   861→  }
   862→
   863→  dispose() {
   864→    this.visual.group.traverse((o) => {
   865→      const mesh = o as THREE.Mesh;
   866→      if (mesh.geometry) mesh.geometry.dispose();
   867→    });
   868→  }
   869→}
