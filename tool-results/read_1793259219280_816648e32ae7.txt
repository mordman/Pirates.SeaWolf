    22→/** Расстояние от точки (px,pz) до отрезка (ax,az)-(bx,bz) в плоскости XZ */
    23→function distPointSeg2D(px: number, pz: number, ax: number, az: number, bx: number, bz: number): number {
    24→  const dx = bx - ax, dz = bz - az;
    25→  const len2 = dx * dx + dz * dz;
    26→  let t = len2 > 1e-9 ? ((px - ax) * dx + (pz - az) * dz) / len2 : 0;
    27→  t = Math.max(0, Math.min(1, t));
    28→  const cx = ax + dx * t, cz = az + dz * t;
    29→  return Math.hypot(px - cx, pz - cz);
    30→}

   160→  /** Книппель: два полушара ядра на цепи — кувыркается в полёте */
   161→  private makeChainMesh(): THREE.Group {
   162→    if (!this.chainGeoCache) {
   163→      // прототип: ось X — ось полёта, полушария по краям, цепь между ними
   164→      const proto = new THREE.Group();
   165→      const ballMat = new THREE.MeshStandardMaterial({ color: 0x1b1b20, roughness: 0.5, metalness: 0.72 });
   166→      const linkMat = new THREE.MeshStandardMaterial({ color: 0x454b54, roughness: 0.55, metalness: 0.85 });
   167→      const R = 0.34;
   168→      // полусфера (z≥0) + плоская крышка
   169→      const hemi = new THREE.SphereGeometry(R, 12, 8, 0, Math.PI);
   170→      const cap = new THREE.CircleGeometry(R, 12);
   171→      for (const dir of [1, -1] as const) {
   172→        const half = new THREE.Group();
   173→        const dome = new THREE.Mesh(hemi, ballMat);
   174→        dome.rotation.y = dir > 0 ? Math.PI / 2 : -Math.PI / 2; // купол наружу по X
   175→        half.add(dome);
   176→        const c = new THREE.Mesh(cap, ballMat);
   177→        c.rotation.y = dir > 0 ? -Math.PI / 2 : Math.PI / 2; // крышка к центру
   178→        half.add(c);
   179→        half.position.x = dir * 0.62;
   180→        proto.add(half);
   181→      }
   182→      // цепь: 4 звена-тора перпендикулярно друг другу
   183→      for (let i = 0; i < 4; i++) {
   184→        const link = new THREE.Mesh(new THREE.TorusGeometry(0.1, 0.03, 5, 12), linkMat);
   185→        link.position.x = -0.23 + i * 0.155;
   186→        link.rotation.y = i % 2 === 0 ? 0 : Math.PI / 2;
   187→        proto.add(link);
   188→      }
   189→      this.chainGeoCache = proto;
   190→    }
   191→    // копируем прототип (меши шарят геометрию/материал)
   192→    const g = new THREE.Group();
   193→    for (const child of this.chainGeoCache.children) g.add(child.clone());
   194→    return g;
   195→  }

   208→  /** Мировое направление прицела */
   209→  aimDirection(out: THREE.Vector3): THREE.Vector3 {
   210→    const heading = this.player.heading;
   211→    const side = this.aim.side; // 1 = правый борт (штирборт = мир −X при курсе 0)
   212→    // штирборт: (−cos h, 0, sin h); порт: зеркально
   213→    const baseAz = Math.atan2(-side * Math.cos(heading), side * Math.sin(heading));
   214→    // baseAz: направление правого/левого борта
   215→    const az = baseAz + this.aim.azimuth;
   216→    const cosE = Math.cos(this.aim.elevation);
   217→    out.set(Math.sin(az) * cosE, Math.sin(this.aim.elevation), Math.cos(az) * cosE);
   218→    return out.normalize();
   219→  }

   230→  fireOne() {
   231→    if (!this.aim.active) return;
   232→    const guns = this.aim.side === 1 ? this.player.visual.gunsStarboard : this.player.visual.gunsPort;
   233→    for (let i = 0; i < guns.length; i++) {
   234→      const idx = this.gunIndex(this.aim.side, i);

   236→        this.fireGun(idx, guns[i]);

   238→      }
   239→    }
   240→    this.emit(GameEvents.MESSAGE, { text: 'Орудия перезаряжаются!' });

   243→  fireBroadside() {
   244→    if (!this.aim.active) return;
   245→    const guns = this.aim.side === 1 ? this.player.visual.gunsStarboard : this.player.visual.gunsPort;
   246→    let fired = 0;
   247→    for (let i = 0; i < guns.length; i++) {
   248→      const idx = this.gunIndex(this.aim.side, i);
   249→      if (this.reloads[idx] <= 0) {
   250→        // лёгкий разброс залпа
   251→        const az0 = this.aim.azimuth, el0 = this.aim.elevation;
   252→        this.aim.azimuth += (Math.random() - 0.5) * 0.06;
   253→        this.fireGun(idx, guns[i], i * 0.05);
   254→        this.aim.azimuth = az0; this.aim.elevation = el0;
   255→        fired++;
   256→      }
   257→    }
   258→    if (!fired) this.emit(GameEvents.MESSAGE, { text: 'Борт перезаряжается!' });
   259→  }

   261→  private fireGun(idx: number, gun: THREE.Object3D, delay = 0) {
   262→    const ship = this.player;
   263→    const info = AMMO_INFO[this.ammo];
   264→    if (ship.ammo[this.ammo] <= 0) {
   265→      this.emit(GameEvents.MESSAGE, { text: `${info.name}: нет боеприпасов!` });
   266→      return;
   267→    }
   268→    if (!this.infiniteAmmo) ship.ammo[this.ammo]--; // расход боезапаса
   269→    this.reloads[idx] = this.reloadDuration();
   270→    if (delay > 0) {
   271→      // отложенный выстрел для залпа
   272→      setTimeout(() => this.doFire(gun, info), delay * 1000);
   273→    } else {
   274→      this.doFire(gun, info);
   275→    }
   276→  }

   278→  private doFire(gun: THREE.Object3D, info: typeof AMMO_INFO[AmmoType]) {
   279→    const ship = this.player;
   280→    // позиция дула в МИРОВЫХ координатах (без matrixWorld — floating origin!)
   281→    const sideDir = gun.userData.side > 0 ? 1 : -1;
   282→    this.muzzleWorld.copy(gun.position);
   283→    this.muzzleWorld.x += sideDir * 1.35;
   284→    this.muzzleWorld.y += 0.12;
   285→    this.muzzleWorld.applyQuaternion(ship.quaternion).add(ship.position);
   286→    const h = ship.heading;
   287→
   288→    this.aimDirection(this.aimDir);
   289→    // отдача: крен от залпа (борт, из которого стреляем, поднимается)
   290→    ship.angular.z += sideDir * 0.006 * info.muzzle / 175;
   291→
   292→    for (let p = 0; p < info.pellets; p++) {
   293→      const pr = this.projectiles.find(q => !q.active);
   294→      if (!pr) break;
   295→      pr.active = true;
   296→      pr.pos.copy(this.muzzleWorld);
   297→      pr.prev.copy(this.muzzleWorld);
   298→      pr.type = this.ammo;
   299→      pr.shooter = ship;
   300→      pr.age = 0;
   301→      pr.trailT = 0;
   302→      pr.spin = Math.random() * Math.PI * 2;
   303→      pr.pierced = new Map();
   304→      // картечь — ШИРОКИЙ веер пуль (горизонтальный), остальное — точно
   305→      const spreadH = this.ammo === 'grape' ? 0.075 : 0.008;
   306→      const spreadV = this.ammo === 'grape' ? 0.05 : 0.006;
   307→      const dir = this.aimDir.clone();
   308→      dir.x += (Math.random() - 0.5) * spreadH * 2;
   309→      dir.z += (Math.random() - 0.5) * spreadH * 2;
   310→      dir.y += (Math.random() - 0.5) * spreadV * 2;
   311→      dir.normalize();
   312→      pr.vel.copy(dir).multiplyScalar(info.muzzle * (0.94 + Math.random() * 0.12));
   313→      pr.vel.add(ship.velocity);
   314→    }
   315→    // вспышка выстрела: короткая и компактная (аддитивное свечение)
   316→    this.particles.spawn(this.muzzleWorld.x, this.muzzleWorld.y, this.muzzleWorld.z,
   317→      this.aimDir.x * 7, 1.0, this.aimDir.z * 7, 0.11, 1.15, 2, 2.4, -1.5);
   318→    // пороховой дым: два небольших полупрозрачных клубка у дула,
   319→    // разнесённых по высоте и направлению — не сливаются в одно облако
   320→    for (let s = 0; s < 2; s++) {
   321→      const jx = (Math.random() - 0.5) * 1.1;
   322→      this.particles.smoke(
   323→        this.muzzleWorld.x + this.aimDir.x * (0.3 + s * 0.85) + jx,
   324→        this.muzzleWorld.y + 0.1 + s * 0.3,
   325→        this.muzzleWorld.z + this.aimDir.z * (0.3 + s * 0.85) + jx,
   326→        this.aimDir.x * (1.5 - s * 0.5) + (Math.random() - 0.5) * 1.1 + ship.velocity.x * 0.25,
   327→        0.3 + Math.random() * 0.35,
   328→        this.aimDir.z * (1.5 - s * 0.5) + (Math.random() - 0.5) * 1.1 + ship.velocity.z * 0.25,
   329→        0.95 + Math.random() * 0.3, 0.9 + Math.random() * 0.4);
   330→    }
   331→    this.emit(GameEvents.CANNON_FIRED, { pos: this.muzzleWorld.clone(), power: info.muzzle / 175 });
   332→  }

   361→  /** Залп ИИ ядрами по игроку (с упреждением и разбросом) */
   362→  aiBallVolley(shooter: Ship, target: Ship) {
   363→    const to = new THREE.Vector3().subVectors(target.position, shooter.position);
   364→    const dist = to.length();
   365→    if (dist > 400) return;
   366→    to.normalize();
   367→    const flight = dist / 165;
   368→    const aim = target.position.clone().addScaledVector(target.velocity, flight).sub(shooter.position).normalize();
   369→    // размер залпа — от числа орудий борта (шлюп 5 → 3 ядра, бриг 14 → 5)
   370→    const n = Math.max(2, Math.min(5, Math.round(shooter.preset.gunsPerSide * 0.55)));
   371→    for (let i = 0; i < n; i++) {
   372→      const pr = this.projectiles.find(q => !q.active);
   373→      if (!pr) break;
   374→      pr.active = true;
   375→      pr.pos.copy(shooter.position).addScaledVector(aim, shooter.preset.beam * 0.55);
   376→      pr.prev.copy(pr.pos);
   377→      pr.pos.y += 1.0;
   378→      pr.type = 'ball';
   379→      pr.shooter = shooter;
   380→      pr.age = 0;
   381→      pr.trailT = 0;
   382→      pr.pierced = new Map();
   383→      const dir = aim.clone();
   384→      dir.x += (Math.random() - 0.5) * 0.05;
   385→      dir.y += 0.04 + (Math.random() - 0.5) * 0.03;
   386→      dir.z += (Math.random() - 0.5) * 0.05;
   387→      dir.normalize();
   388→      pr.vel.copy(dir).multiplyScalar(170).add(shooter.velocity);
   389→    }
   390→    // дым и звук
   391→    const muzzle = shooter.position.clone().addScaledVector(aim, shooter.preset.beam * 0.7);
   392→    muzzle.y += 1.2;
   393→    for (let s = 0; s < 2; s++) {
   394→      this.particles.smoke(muzzle.x, muzzle.y, muzzle.z,
   395→        aim.x * 1.6 + (Math.random() - 0.5) * 0.6, 0.3 + Math.random() * 0.25,
   396→        aim.z * 1.6 + (Math.random() - 0.5) * 0.6, 1.05 + Math.random() * 0.25, 1.0 + Math.random() * 0.2);
   397→    }
   398→    this.emit(GameEvents.CANNON_FIRED, { pos: muzzle, power: 0.9 });
   399→  }
   400→
   401→  // ---------- обновление ----------
   402→  update(dt: number, time: number, waterHeight: (x: number, z: number, t: number) => number,
   403→    groundHeight: ((x: number, z: number) => number) | null = null,
   404→    onGroundHit: ((x: number, y: number, z: number, radius: number) => number) | null = null) {
   405→    // перезарядка
   406→    for (let i = 0; i < this.reloads.length; i++) {
   407→      if (this.reloads[i] > 0) this.reloads[i] -= dt;
   408→    }
   409→
   410→    // снаряды
   411→    let count = 0;
   412→    let grapeCount = 0;
   413→    let chainIdx = 0;
   414→    for (const pr of this.projectiles) {
   415→      if (!pr.active) continue;
   416→      pr.age += dt;
   417→      const info = AMMO_INFO[pr.type];
   418→      pr.prev.copy(pr.pos);
   419→      // интеграция с сопротивлением
   420→      const dragF = Math.exp(-info.drag * dt * 3);
   421→      pr.vel.multiplyScalar(dragF);
   422→      pr.vel.y -= 9.81 * dt;
   423→      pr.pos.addScaledVector(pr.vel, dt);
   424→      // книппель кувыркается вокруг оси полёта
   425→      if (pr.type === 'chain') pr.spin += dt * 22;
   426→
   427→      // дымный след — ядро видно в полёте (кроме картечи — её дробин много)
   428→      pr.trailT += dt;
   429→      if (pr.trailT > 0.18 && pr.type !== 'grape') {
   430→        pr.trailT = 0;
   431→        this.particles.smoke(pr.pos.x, pr.pos.y, pr.pos.z, 0, 0.08, 0, 0.42, 0.5);
   432→      }
   433→
   434→      // вода
   435→      const wy = waterHeight(pr.pos.x, pr.pos.z, time);
   436→      if (pr.pos.y < wy) {
   437→        pr.active = false;
   438→        this.particles.splash(pr.pos.x, wy, pr.pos.z, pr.type === 'ball' ? 1.0 : 0.5);
   439→        this.emit(GameEvents.PROJECTILE_SPLASH, { pos: pr.pos.clone() });
   440→        continue;
   441→      }
   442→
   443→      // земля (острова): фугасный удар + повал деревьев
   444→      if (groundHeight && pr.pos.y < groundHeight(pr.pos.x, pr.pos.z)) {
   445→        pr.active = false;
   446→        const r = pr.type === 'ball' || pr.type === 'chain' ? 2.6 : pr.type === 'fire' ? 3.2 : 1.6;
   447→        const trees = onGroundHit ? onGroundHit(pr.pos.x, pr.pos.y, pr.pos.z, r) : 0;
   448→        // пыль/земля + щепа, если деревья сломаны
   449→        this.particles.smoke(pr.pos.x, pr.pos.y + 0.4, pr.pos.z, 0, 0.6, 0, 1.6, 1.2);
   450→        if (trees > 0) this.particles.debris(pr.pos.x, pr.pos.y + 1, pr.pos.z, 10, 1.1);
   451→        this.emit(GameEvents.TERRAIN_HIT, { pos: pr.pos.clone(), trees });
   452→        continue;
   453→      }
   454→
   455→      // цели: ПАРУСА прошиваются насквозь (дыры + урон такелажу,
   456→      // снаряд летит дальше), затем мачты (рангоут перекрывает
   457→      // корпус), корпус — сегментом от prev до pos (быстрое ядро
   458→      // не «проскакивает"); КНИППЕЛЬ пробивает рангоут насквозь:
   459→      // круг его разлёта ШИРЕ корпуса (3.2 м > полуширины) — летя
   460→      // вдоль борта, сносит ВСЕ мачты коридором
   461→      let hit: HitInfo | null = null;
   462→      let piercedMast = false;
   463→      const targets: Ship[] = pr.shooter === this.player ? this.others : [this.player];
   464→      for (const t of targets) {
   465→        if (t.sunk) continue;
   466→        // паруса: картечь дырявит полотно россвелью пробоин, книппель
   467→        // рвёт широкими прорехами, ядро пробивает круглую дыру
   468→        this.sailPass(pr, t);
   469→        const done = pr.type === 'chain' ? (pr.pierced.get(t) ?? []) : [];
   470→        hit = this.hitTest(pr, t, done);
   471→        if (hit) {
   472→          this.applyHit(pr, t, info, hit);
   473→          if (pr.type === 'chain' && hit.mast >= 0) {
   474→            // сквозной пролёт: пометить разорванные мачты, летим дальше
   475→            const masts = pr.pierced.get(t) ?? [];
   476→            for (const mi of this.chainMasts) masts.push(mi);
   477→            pr.pierced.set(t, masts);
   478→            piercedMast = true;
   479→          }
   480→          break;
   481→        }
   482→      }
   483→      if (hit && !piercedMast) { pr.active = false; continue; }
   484→      if (!hit && pr.age > 12) { pr.active = false; continue; }
   485→
   486→      // визуал
   487→      if (pr.type === 'chain') {
   488→        const mesh = this.chainMeshes[chainIdx];
   489→        if (mesh) {
   490→          mesh.visible = true;
   491→          mesh.position.copy(pr.pos);
   492→          // ось полёта → локальная X, затем кувырок вокруг неё
   493→          const vlen = pr.vel.length();
   494→          if (vlen > 0.01) {
   495→            this.chainDummy.position.copy(pr.pos);
   496→            this._vN.copy(pr.vel).divideScalar(vlen);
   497→            this.chainDummy.quaternion.setFromUnitVectors(this.xAxis, this._vN);
   498→            this.chainDummy.rotateX(pr.spin);
   499→            mesh.quaternion.copy(this.chainDummy.quaternion);
   500→          }
   501→          chainIdx++;
   502→        }
   503→      } else if (pr.type === 'grape') {
   504→        // свинцовые пульки — рой МЕЛКИХ шаров (не ядра!)
   505→        if (grapeCount < 192) {
   506→          this.dummy.position.copy(pr.pos);
   507→          this.dummy.updateMatrix();
   508→          this.grapeMesh.setMatrixAt(grapeCount, this.dummy.matrix);
   509→          grapeCount++;
   510→        }
   511→      } else {
   512→        this.dummy.position.copy(pr.pos);
   513→        this.dummy.updateMatrix();
   514→        this.projMesh.setMatrixAt(count, this.dummy.matrix);
   515→        count++;
   516→      }
   517→    }
   518→    this.projMesh.count = count;
   519→    this.projMesh.instanceMatrix.needsUpdate = true;
   520→    this.grapeMesh.count = grapeCount;
   521→    this.grapeMesh.instanceMatrix.needsUpdate = true;
   522→    for (let i = chainIdx; i < this.chainMeshes.length; i++) this.chainMeshes[i].visible = false;
   523→
   524→    // прицеливание: траектория и упреждение
   525→    if (this.aim.active) {
   526→      this.updateAimAssist(time, waterHeight);
   527→    } else {
   528→      this.trajLine.visible = false;
   529→      this.impactRing.visible = false;
   530→      this.leadMarker.visible = false;
   531→    }
   532→  }

   534→  private lprev = new THREE.Vector3();
   535→  private lcur = new THREE.Vector3();
   536→  private invQ = new THREE.Quaternion();
   537→  private hitDir = new THREE.Vector3();
   538→  private pathB = new THREE.Vector3();
   539→  private _vN = new THREE.Vector3();
   540→  /** мачты в круге разлёта книппеля (заполняет hitTest) */
   541→  private chainMasts: number[] = [];
   542→  // temporaries for sailPass
   543→  private sqA = new THREE.Vector3();
   544→  private sqB = new THREE.Vector3();
   545→  private sqC0 = new THREE.Vector3();
   546→  private sqE1 = new THREE.Vector3();
   547→  private sqE2 = new THREE.Vector3();
   548→  private sqN = new THREE.Vector3();
   549→  private sqW = new THREE.Vector3();
   550→  private sqP = new THREE.Vector3();
   551→
   552→  /** Прострел ПАРУСОВ насквозь: путь снаряда (prev→pos) против
   553→   *  полотен-квадов (с поворотом брасопки). Попадание — пробоина
   554→   *  (sailAddHole) + урон такелажу; снаряд ЛЕТИТ ДАЛЬШЕ — паруса
   555→   *  не останавливают ни картучь, ни ядро. Возвращает true, если
   556→   *  хотя бы одно полотно прошито (для звука/телеметрии). */
   557→  private sailPass(pr: Projectile, t: Ship): boolean {
   558→    if (t.sinking || t.sunk) return false;
   559→    const sails = t.visual.sails.sails;
   560→    if (!sails.length) return false;
   561→    // путь снаряда в СВЯЗНЫХ координатах цели
   562→    this.invQ.copy(t.quaternion).invert();
   563→    this.sqA.subVectors(pr.prev, t.position).applyQuaternion(this.invQ);
   564→    this.sqB.subVectors(pr.pos, t.position).applyQuaternion(this.invQ);
   565→    let torn = false;
   566→    for (const s of sails) {
   567→      // мачта сломана — парус обвис: не простреливаем
   568→      if (s.mastIndex !== undefined && t.mastFall[s.mastIndex] >= 0) continue;
   569→      // сильно убран (собран у реи) — полотно мало, пропускаем
   570→      if (s.furl > 0.72) continue;
   571→      const q = s.quad;
   572→      // углы полотна в координатах КОРАБЛЯ (+текущая брасопка реи)
   573→      const ba = q.rotates ? s.braceAngle : 0;
   574→      const cb = Math.cos(ba), sb = Math.sin(ba);
   575→      // C0 = uv(0,0), E1 → uv(1,0), E2 → uv(0,1) (параллелограмм)
   576→      const loc = (i: number, out: THREE.Vector3): THREE.Vector3 => {
   577→        const c = q.corners[i];
   578→        // поворот вокруг оси мачты (Y) на угол брасопки
   579→        const rx = c.x * cb + c.z * sb;
   580→        const rz = -c.x * sb + c.z * cb;
   581→        out.set(q.pivot.x + rx, q.pivot.y + c.y, q.pivot.z + rz);
   582→        return out;
   583→      };
   584→      loc(0, this.sqC0);
   585→      loc(1, this.sqE1).sub(this.sqC0);
   586→      loc(3, this.sqE2).sub(this.sqC0);
   587→      // плоскость полотна
   588→      this.sqN.crossVectors(this.sqE1, this.sqE2);
   589→      const nLen2 = this.sqN.lengthSq();
   590→      if (nLen2 < 1e-9) continue;
   591→      const dA = this.sqW.subVectors(this.sqA, this.sqC0).dot(this.sqN);
   592→      const dB = this.sqW.subVectors(this.sqB, this.sqC0).dot(this.sqN);
   593→      if (dA * dB > 0) continue;              // обе точки по одну сторону
   594→      const k = dA / (dA - dB);               // 0..1 — точка пересечения
   595→      if (k < 0 || k > 1) continue;
   596→      this.sqP.lerpVectors(this.sqA, this.sqB, k).sub(this.sqC0);
   597→      // билинейные координаты в параллелограмме: P = u·E1 + v·E2
   598→      const e11 = this.sqE1.dot(this.sqE1), e22 = this.sqE2.dot(this.sqE2);
   599→      const e12 = this.sqE1.dot(this.sqE2);
   600→      const pu = this.sqP.dot(this.sqE1), pv = this.sqP.dot(this.sqE2);
   601→      const det = e11 * e22 - e12 * e12;
   602→      if (Math.abs(det) < 1e-9) continue;
   603→      const u = (pu * e22 - pv * e12) / det;
   604→      const v = (pv * e11 - pu * e12) / det;
   605→      const tri = q.uvs[2].equals(q.uvs[3]);  // латинский/кливер — треугольник
   606→      if (u < -0.02 || u > 1.02 || v < -0.02 || v > 1.02) continue;
   607→      if (tri && u + v > 1.04) continue;
   608→      // === ПОПАДАНИЕ В ПОЛОТНО ===
   609→      torn = true;
   610→      const uu = clamp(u, 0, 1), vv = clamp(v, 0, 1);
   611→      // радиус пробоины (в долях ВЫСОТЫ полотна — см. шейдер uHoles)
   612→      let r: number;
   613→      if (pr.type === 'grape') r = 0.035 + Math.random() * 0.025;   // россыпь малых дыр
   614→      else if (pr.type === 'chain') r = 0.11 + Math.random() * 0.07; // широкая прореха
   615→      else if (pr.type === 'fire') r = 0.045;
   616→      else r = 0.055 + Math.random() * 0.03;                         // ядро
   617→      sailAddHole(s, uu, vv, r);
   618→      // урон такелажу: учесть долю поставленного полотна
   619→      const setFrac = (1 - s.furl) * 0.6 + 0.4;
   620→      const dmg = AMMO_INFO[pr.type].dmgSail * setFrac * (0.8 + Math.random() * 0.4);
   621→      t.damageSails(dmg);
   622→    }
   623→    return torn;
   624→  }
   625→
   626→  /** Проверка попадания: мачты (стоячие, swept) → корпус.
   627→   *  Книппель ловит мачты ВЕСЬ СВОЙ КРУГ (радиус 3.2 м — шире корпуса);
   628→   *  skipMasts — уже разорванные этим снарядом (сквозной полёт). */
   629→  private hitTest(pr: Projectile, t: Ship, skipMasts: number[] = []): HitInfo | null {
   630→    this.invQ.copy(t.quaternion).invert();
   631→    this.lprev.subVectors(pr.prev, t.position).applyQuaternion(this.invQ);
   632→    this.lcur.subVectors(pr.pos, t.position).applyQuaternion(this.invQ);
   633→    // направление полёта в связных координатах цели
   634→    this.hitDir.subVectors(this.lcur, this.lprev);
   635→    if (this.hitDir.lengthSq() < 1e-9) this.hitDir.set(0, 0, 1);
   636→    else this.hitDir.normalize();
   637→
   638→    // --- мачты: вертикальный цилиндр в XZ, путь ядра — отрезок prev→pos ---
   639→    const masts = t.visual.masts;
   640→    this.chainMasts.length = 0;
   641→    const mastR = pr.type === 'chain' ? 3.2 : 0.8; // книппель бьёт кругом разлёта (шире корпуса!)
   642→    let first = -1;
   643→    for (let i = 0; i < masts.length; i++) {
   644→      if (t.mastFall[i] >= 0 || skipMasts.includes(i)) continue;
   645→      const mz = masts[i].position.z;
   646→      const h = ((masts[i].userData.height as number) || 20) - t.preset.draft * 0.3;
   647→      const d = distPointSeg2D(0, mz, this.lprev.x, this.lprev.z, this.lcur.x, this.lcur.z);
   648→      if (d < mastR && Math.max(this.lprev.y, this.lcur.y) > 0.7
   649→        && Math.min(this.lprev.y, this.lcur.y) < h + 0.4) {
   650→        if (first < 0) first = i;
   651→        if (pr.type === 'chain') this.chainMasts.push(i);
   652→      }
   653→    }
   654→    if (first >= 0) {
   655→      return { mast: first, lp: this.lcur.clone(), dir: this.hitDir.clone() };
   656→    }
   657→
   658→    // --- корпус (бокс в связных координатах) ---
   659→    const L = t.preset.length, B = t.preset.beam;
   660→    const c = this.lcur;
   661→    if (Math.abs(c.z) < L * 0.52 && Math.abs(c.x) < B * 0.62
   662→      && c.y > -t.preset.draft * 0.8 && c.y < t.preset.freeboard + L * 0.35) {
   663→      return { mast: -1, lp: c.clone(), dir: this.hitDir.clone() };
   664→    }
   665→    return null;
   666→  }

   668→  private applyHit(pr: Projectile, t: Ship, info: typeof AMMO_INFO[AmmoType], hit: HitInfo) {
   669→    const amount = 0.75 + Math.random() * 0.5;
   670→    const P = this.particles;
   671→    if (hit.mast >= 0) {
   672→      // --- попадание в РАНГОУТ: мачта трещит и ломается ---
   673→      // книппель — специализированный «обрыватель» такелажа: бьёт
   674→      // ВСЕМ КРУГОМ разлёта (ломает несколько мачт за раз),
   675→      // ядро — одну, за 2-3 попадания
   676→      if (pr.type === 'chain') {
   677→        const hitSet = this.chainMasts.length ? this.chainMasts : [hit.mast];
   678→        for (const mi of hitSet) {
   679→          const dmg = mi === hit.mast ? 30 + Math.random() * 12 : 22 + Math.random() * 10;
   680→          const broke = t.damageMast(mi, dmg);
   681→          t.applyDamage({ kind: 'chain', amount: info.dmgSail * amount * (mi === hit.mast ? 1 : 0.7) });
   682→          if (broke) this.emit(GameEvents.MAST_BROKEN, { target: t, mast: mi, pos: pr.pos.clone() });
   683→        }
   684→      } else {
   685→        const dmg = pr.type === 'ball' ? 11 + Math.random() * 8
   686→          : pr.type === 'fire' ? 6 : 1.6;
   687→        const broke = t.damageMast(hit.mast, dmg);
   688→        if (pr.type === 'grape') t.applyDamage({ kind: 'grape', amount: info.dmgCrew * amount });
   689→        if (broke) this.emit(GameEvents.MAST_BROKEN, { target: t, mast: hit.mast, pos: pr.pos.clone() });
   690→      }
   691→      // щепа летит вдоль мачты
   692→      for (let i = 0; i < 9; i++) {
   693→        P.spawn(pr.pos.x, pr.pos.y + i * 0.35, pr.pos.z,
   694→          (Math.random() - 0.5) * 7, 2 + Math.random() * 5, (Math.random() - 0.5) * 7,
   695→          0.8, 0.3, 0, 0.6, 10);
   696→      }
   697→      P.smoke(pr.pos.x, pr.pos.y, pr.pos.z, 0, 0.8, 0, 1.0, 1.3);
   698→    } else {
   699→      // --- попадание в КОРПУС: пробоина, у ватерлинии/ниже — течь ---
   700→      // (порог 0.6 м: при качке «у самой воды» дыры также дают течь)
   701→      const belowWL = hit.lp.y < 0.6;
   702→      // --- поражение ЭКИПАЖА вдоль пути снаряда через палубу ---
   703→      // картечь косят палубу (радиус широкий), ядро/зажигательное
   704→      // разрывает тех, кто на линии полёта
   705→      this.pathB.copy(hit.lp).addScaledVector(hit.dir, t.preset.length * 1.25);
   706→      let crewKilled = 0;
   707→      if (pr.type === 'grape') {
   708→        crewKilled = t.crew3d.killNearPath(hit.lp, this.pathB, 'grape', 2, 2.4,
   709→          t.position, t.quaternion);
   710→      } else if (pr.type === 'ball' || pr.type === 'fire') {
   711→        crewKilled = t.crew3d.killNearPath(hit.lp, this.pathB, 'ball', 2, 1.35,
   712→          t.position, t.quaternion);
   713→      }
   714→      if (pr.type === 'grape') {
   715→        // потери команды согласованы с убитыми фигурами
   716→        const per = Math.max(1, t.preset.maxCrew / Math.max(1, t.crew3d.total));
   717→        t.applyDamage({ kind: 'grape', amount: Math.max(0.6, crewKilled * per * 0.9) });
   718→      } else {
   719→        t.applyDamage({
   720→          kind: pr.type === 'fire' && Math.random() < info.fireChance ? 'fire' : pr.type,
   721→          amount: info.dmgHull * amount,
   722→          hitPoint: pr.pos.clone(),
   723→        });
   724→      }
   725→      if (pr.type === 'chain') t.applyDamage({ kind: 'chain', amount: info.dmgSail * amount });
   726→      // декаль пробоины (ядро и зажигательное)
   727→      if (pr.type === 'ball' || pr.type === 'fire') t.addHole(hit.lp, belowWL);
   728→      // щепки и дым
   729→      for (let i = 0; i < 6; i++) {
   730→        P.spawn(pr.pos.x, pr.pos.y, pr.pos.z,
   731→          (Math.random() - 0.5) * 6, 2 + Math.random() * 4, (Math.random() - 0.5) * 6,
   732→          0.7, 0.28, 0, 0.6, 10);
   733→      }
   734→      P.smoke(pr.pos.x, pr.pos.y, pr.pos.z, 0, 0.7, 0, 0.95, 1.2);
   735→    }
   736→    this.emit(GameEvents.PROJECTILE_HIT, { pos: pr.pos.clone(), target: t });
   737→  }

   739→  private updateAimAssist(time: number, waterHeight: (x: number, z: number, t: number) => number) {
   740→    const ship = this.player;
   741→    const info = AMMO_INFO[this.ammo];
   742→    const start = new THREE.Vector3().copy(ship.position);
   743→    start.y += ship.preset.freeboard * 0.55;
   744→    const vel = this.aimDirection(new THREE.Vector3()).multiplyScalar(info.muzzle).add(ship.velocity);
   745→
   746→    // симуляция траектории
   747→    const attr = this.trajLine.geometry.getAttribute('position') as THREE.BufferAttribute;
   748→    const arr = attr.array as Float32Array;
   749→    let px = start.x, py = start.y, pz = start.z;
   750→    let vx = vel.x, vy = vel.y, vz = vel.z;
   751→    const stp = 0.06;
   752→    let range = 0;
   753→    let i = 0;
   754→    for (; i < 48; i++) {
   755→      arr[i * 3] = px; arr[i * 3 + 1] = py; arr[i * 3 + 2] = pz;
   756→      for (let s = 0; s < 4; s++) {
   757→        const d = Math.exp(-info.drag * stp * 3);
   758→        vx *= d; vy *= d; vz *= d;
   759→        vy -= 9.81 * stp;
   760→        px += vx * stp; py += vy * stp; pz += vz * stp;
   761→        range += Math.hypot(vx, vy, vz) * stp;
   762→        if (py < waterHeight(px, pz, time)) break;
   763→      }
   764→      if (py < waterHeight(px, pz, time)) { i++; if (i < 48) { arr[i * 3] = px; arr[i * 3 + 1] = Math.max(py, waterHeight(px, pz, time)); arr[i * 3 + 2] = pz; } break; }
   765→    }
   766→    attr.needsUpdate = true;
   767→    this.trajLine.geometry.setDrawRange(0, Math.max(2, i));
   768→    this.aim.range = range;
   769→
   770→    // кольцо в точке падения
   771→    if (i > 0 && py < waterHeight(px, pz, time)) {
   772→      const wy = waterHeight(px, pz, time);
   773→      this.impactRing.position.set(px, wy + 0.3, pz);
   774→      this.impactRing.scale.setScalar(1 + Math.sin(time * 6) * 0.12);
   775→      this.impactRing.visible = true;
   776→    } else {
   777→      this.impactRing.visible = false;
   778→    }
   779→
   780→    // упреждение: ближайшая цель в секторе
   781→    const dir = this.aimDirection(new THREE.Vector3());
   782→    this.aim.target = null;
   783→    this.aim.targetLead = null;
   784→    let best: Ship | null = null;
   785→    let bestScore = Infinity;
   786→    const toT = new THREE.Vector3();
   787→    for (const t of this.others) {
   788→      if (t.sunk || t.sinking) continue;
   789→      toT.subVectors(t.position, ship.position);
   790→      const dist = toT.length();
   791→      if (dist > 900) continue;
   792→      toT.normalize();
   793→      const dot = toT.dot(dir);
   794→      if (dot < 0.92) continue; // в пределах ~23° от прицела
   795→      const score = dist * (2 - dot);
   796→      if (score < bestScore) { bestScore = score; best = t; }
   797→    }
   798→    if (best) {
   799→      const dist = best.position.distanceTo(ship.position);
   800→      const flight = dist / info.muzzle;
   801→      const lead = new THREE.Vector3().copy(best.position).addScaledVector(best.velocity, flight);
   802→      lead.y += 1.5;
   803→      this.aim.target = best;
   804→      this.aim.targetLead = lead;
   805→      this.leadMarker.position.copy(lead);
   806→      this.leadMarker.rotation.y = time * 2;
   807→      this.leadMarker.scale.setScalar(1 + Math.sin(time * 5) * 0.15);
   808→      this.leadMarker.visible = true;
   809→    } else {
   810→      this.leadMarker.visible = false;
   811→    }
   812→  }
   813→
   814→  /** Готовность орудий (0..1) для HUD */
   815→  readiness(): number {
   816→    let ready = 0;
   817→    for (const r of this.reloads) if (r <= 0) ready++;
   818→    return ready / this.reloads.length;
   819→  }
