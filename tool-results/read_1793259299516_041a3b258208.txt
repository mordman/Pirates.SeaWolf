    77→function MiniMap({ hud }: { hud: HudState }) {
    78→  const ref = useRef<HTMLCanvasElement>(null);
    79→  useEffect(() => {
    80→    const cv = ref.current;
    81→    if (!cv) return;
    82→    const ctx = cv.getContext('2d')!;
    83→    const S = cv.width;
    84→    const c = S / 2;
    85→    const RANGE = 900;                       // метров от центра до края
    86→    const scale = c / RANGE;
    87→    ctx.clearRect(0, 0, S, S);
    88→    // морской фон + картушка
    89→    ctx.save();
    90→    ctx.beginPath(); ctx.arc(c, c, c - 2, 0, Math.PI * 2); ctx.clip();
    91→    ctx.fillStyle = '#0c242e';
    92→    ctx.fillRect(0, 0, S, S);
    93→    ctx.strokeStyle = 'rgba(255,255,255,0.05)';
    94→    for (let r = RANGE / 3; r <= RANGE; r += RANGE / 3) {
    95→      ctx.beginPath(); ctx.arc(c, c, r * scale, 0, Math.PI * 2); ctx.stroke();
    96→    }
    97→    const px = (x: number) => c + (x - hud.mapPlayer.x) * scale;
    98→    const pz = (z: number) => c + (z - hud.mapPlayer.z) * scale;
    99→    // острова: открытые — в цвете, неизвестные — пунктиром
   100→    for (const isl of hud.mapIslands) {
   101→      const ix = px(isl.x), iz = pz(isl.z);
   102→      const r = Math.max(3, isl.r * scale);
   103→      if (ix < -r || ix > S + r || iz < -r || iz > S + r) continue;
   104→      if (!isl.discovered) {
   105→        ctx.strokeStyle = 'rgba(220,200,150,0.28)';
   106→        ctx.setLineDash([3, 4]);
   107→        ctx.beginPath(); ctx.arc(ix, iz, r, 0, Math.PI * 2); ctx.stroke();
   108→        ctx.setLineDash([]);
   109→        continue;
   110→      }
   111→      const grad = ctx.createRadialGradient(ix, iz, r * 0.2, ix, iz, r);
   112→      grad.addColorStop(0, '#4a7a3a');
   113→      grad.addColorStop(0.75, '#c9b26a');
   114→      grad.addColorStop(1, '#1d5d6e');
   115→      ctx.fillStyle = grad;
   116→      ctx.beginPath(); ctx.arc(ix, iz, r, 0, Math.PI * 2); ctx.fill();
   117→    }
   118→    // корабли ИИ: военные — красные, торговцы — янтарные, сдавшиеся — белые
   119→    for (const s of hud.mapShips) {
   120→      const sx = px(s.x), sz = pz(s.z);
   121→      if (Math.hypot(sx - c, sz - c) > c - 5) continue;
   122→      ctx.fillStyle = s.surrendered ? '#e8e4d8' : s.cat === 'war' ? '#c23b3b' : '#d8a03c';
   123→      ctx.beginPath(); ctx.arc(sx, sz, 3, 0, Math.PI * 2); ctx.fill();
   124→    }
   125→    ctx.restore();
   126→    // игрок — носом по курсу (курс 0 = +Z = вниз карты)
   127→    ctx.save();
   128→    ctx.translate(c, c);
   129→    ctx.rotate(Math.PI - hud.mapPlayer.h);
   130→    ctx.fillStyle = '#7dd3fc';
   131→    ctx.beginPath();
   132→    ctx.moveTo(0, -7); ctx.lineTo(5, 6); ctx.lineTo(0, 3); ctx.lineTo(-5, 6);
   133→    ctx.closePath(); ctx.fill();
   134→    ctx.restore();
   135→    // ветер: стрелка по ободу (куда ДУЕТ)
   136→    const wa = ((hud.windFromDeg + 180) % 360) * Math.PI / 180;
   137→    const wx = c + Math.sin(wa) * (c - 13), wy = c - Math.cos(wa) * (c - 13);
   138→    ctx.save();
   139→    ctx.translate(wx, wy);
   140→    ctx.rotate(wa);
   141→    ctx.fillStyle = 'rgba(125,211,252,0.9)';
   142→    ctx.beginPath();
   143→    ctx.moveTo(0, -6); ctx.lineTo(4, 3); ctx.lineTo(-4, 3);
   144→    ctx.closePath(); ctx.fill();
   145→    ctx.restore();
   146→    // север
   147→    ctx.fillStyle = '#e8c87a';
   148→    ctx.font = 'bold 10px serif';
   149→    ctx.textAlign = 'center';
   150→    ctx.fillText('С', c, 12);
   151→  }, [hud]);
   152→  return (
   153→    <canvas ref={ref} width={168} height={168}
   154→      className="rounded-full border-2 border-amber-900/60 bg-[#0c242e] shadow-lg shadow-black/50" />
   155→  );
   156→}

   161→function ShipViewer3D({ presetId }: { presetId: string }) {
   162→  const ref = useRef<HTMLCanvasElement>(null);
   163→  useEffect(() => {
   164→    let disposed = false;
   165→    let raf = 0;
   166→    let renderer: import('three').WebGLRenderer | null = null;
   167→    let visual: import('@/game/ship/buildShip').ShipVisual | null = null;
   168→    let three: typeof import('three') | null = null;
   169→    (async () => {
   170→      three = await import('three');
   171→      const bs = await import('@/game/ship/buildShip');
   172→      if (disposed || !ref.current || !three) return;
   173→      const T = three;
   174→      const P = bs.PRESETS[presetId];
   175→      if (!P) return;
   176→      renderer = new T.WebGLRenderer({ canvas: ref.current, antialias: true, alpha: true });
   177→      renderer.setPixelRatio(Math.min(window.devicePixelRatio, 1.75));
   178→      renderer.toneMapping = T.ACESFilmicToneMapping;
   179→      const scene = new T.Scene();
   180→      const cam = new T.PerspectiveCamera(36, 1.5, 0.5, 800);
   181→      // свет: небо + солнце + подсветка
   182→      scene.add(new T.HemisphereLight(0xcfe4f2, 0x40331f, 1.05));
   183→      const sun = new T.DirectionalLight(0xfff0d8, 1.9);
   184→      sun.position.set(0.6, 1, 0.5);
   185→      scene.add(sun);
   186→      const rim = new T.DirectionalLight(0x9db8d8, 0.6);
   187→      rim.position.set(-0.8, 0.4, -0.6);
   188→      scene.add(rim);
   189→      // модель
   190→      visual = bs.buildShip(P);
   191→      scene.add(visual.group);
   192→      const L = P.length;
   193→      const H = P.freeboard + L * 0.78; // высота рангоута
   194→      let az = 0.85, elev = 0.30, dist = Math.max(L * 1.5, H * 1.32);
   195→      let targetDist = dist;
   196→      // паруса частично поставлены, реи чуть брасоплены — «живой» силуэт
   197→      for (const s of visual.sails.sails) {
   198→        s.mat.uniforms.uFurl.value = 0.16;
   199→        s.mat.uniforms.uBelly.value = 0.55;
   200→        if (s.brace) s.brace.rotation.y = 0.22;
   201→      }
   202→      const resize = () => {
   203→        if (!renderer || !ref.current) return;
   204→        const w = ref.current.clientWidth || 300;
   205→        const h = ref.current.clientHeight || 210;
   206→        renderer.setSize(w, h, false);
   207→        cam.aspect = w / h;
   208→        cam.updateProjectionMatrix();
   209→      };
   210→      resize();
   211→      // drag-вращение и зум колесом
   212→      let dragging = false, lastX = 0, lastY = 0;
   213→      const el = ref.current;
   214→      const onDown = (e: PointerEvent) => { dragging = true; lastX = e.clientX; lastY = e.clientY; el.setPointerCapture(e.pointerId); };
   215→      const onMove = (e: PointerEvent) => {
   216→        if (!dragging) return;
   217→        az -= (e.clientX - lastX) * 0.008;
   218→        elev = Math.min(1.35, Math.max(0.06, elev + (e.clientY - lastY) * 0.005));
   219→        lastX = e.clientX; lastY = e.clientY;
   220→      };
   221→      const onUp = () => { dragging = false; };
   222→      const onWheel = (e: WheelEvent) => {
   223→        e.preventDefault();
   224→        targetDist = Math.min(Math.max(L * 0.6, targetDist * (1 + Math.sign(e.deltaY) * 0.12)), L * 3.2);
   225→      };
   226→      el.addEventListener('pointerdown', onDown);
   227→      el.addEventListener('pointermove', onMove);
   228→      el.addEventListener('pointerup', onUp);
   229→      el.addEventListener('wheel', onWheel, { passive: false });
   230→      const t0 = performance.now();
   231→      const loop = () => {
   232→        if (disposed) return;
   233→        raf = requestAnimationFrame(loop);
   234→        const t = (performance.now() - t0) / 1000;
   235→        if (!dragging) az += 0.0028;
   236→        dist += (targetDist - dist) * 0.15;
   237→        const ty = H * 0.42;
   238→        cam.position.set(
   239→          Math.sin(az) * Math.cos(elev) * dist,
   240→          ty + Math.sin(elev) * dist,
   241→          Math.cos(az) * Math.cos(elev) * dist);
   242→        cam.lookAt(0, ty * 0.8, 0);
   243→        // анимация парусов/флага
   244→        if (visual) {
   245→          for (const s of visual.sails.sails) s.mat.uniforms.uTime.value = t;
   246→          visual.flagMat.uniforms.uTime.value = t;
   247→        }
   248→        renderer?.render(scene, cam);
   249→      };
   250→      loop();
   251→      // cleanup-обработчики вынесены в return замыкания
   252→      (el as unknown as { __cleanup3d?: () => void }).__cleanup3d = () => {
   253→        el.removeEventListener('pointerdown', onDown);
   254→        el.removeEventListener('pointermove', onMove);
   255→        el.removeEventListener('pointerup', onUp);
   256→        el.removeEventListener('wheel', onWheel);
   257→      };
   258→    })();
   259→    return () => {
   260→      disposed = true;
   261→      cancelAnimationFrame(raf);
   262→      const el = ref.current;
   263→      (el as unknown as { __cleanup3d?: () => void })?.__cleanup3d?.();
   264→      // освободить GPU-ресурсы модели
   265→      if (visual && three) {
   266→        visual.group.traverse(o => {
   267→          const mesh = o as import('three').Mesh;
   268→          if (mesh.geometry) mesh.geometry.dispose();
   269→          const mats = Array.isArray(mesh.material) ? mesh.material : mesh.material ? [mesh.material] : [];
   270→          for (const m of mats) {
   271→            const mat = m as import('three').MeshStandardMaterial & { map?: { dispose(): void } };
   272→            mat.map?.dispose();
   273→            mat.dispose();
   274→          }
   275→        });
   276→      }
   277→      renderer?.dispose();
   278→    };
   279→  }, [presetId]);
   280→  return (
   281→    <canvas ref={ref}
   282→      className="h-full w-full cursor-grab rounded-lg bg-gradient-to-b from-[#1c2f3d] to-[#0d1a22] active:cursor-grabbing" />
   283→  );
   284→}
