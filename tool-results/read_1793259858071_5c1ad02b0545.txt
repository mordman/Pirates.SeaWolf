   417→   417→function sailMaterial(baseColor: number): THREE.ShaderMaterial {
   418→   418→  return new THREE.ShaderMaterial({
   419→   419→    uniforms: {
   420→   420→      uBelly: { value: 0.5 },
   421→   421→      uFlutter: { value: 0.3 },
   422→   422→      uTime: { value: 0 },
   423→   423→      uDamage: { value: 0 },
   424→   424→      uColor: { value: new THREE.Color(baseColor) },
   425→   425→      uSunDir: { value: new THREE.Vector3(0.4, 0.8, 0.3) },
   426→   426→      uFurl: { value: 0 },
   427→   427→      // пробоины от картечи/ядер/книппеля: (u, v, радиус в долях высоты)
   428→   428→      uHoles: { value: Array.from({ length: 8 }, () => new THREE.Vector3(0, 0, 0)) },
   429→   429→      uHoleN: { value: 0 },
   430→   430→      uAspect: { value: 1 },      // ширина/высота — дыры круглые в метрах
   431→   431→    },
   432→   432→    vertexShader: /* glsl */`
   433→   433→      uniform float uBelly, uTime, uFurl, uFlutter;
   434→   434→      varying vec2 vUv;
   435→   435→      varying vec3 vN;
   436→   436→      varying float vDmgU;
   437→   437→      uniform float uDamage;
   438→   438→      void main() {
   439→   439→        vUv = uv;
   440→   440→        vDmgU = uDamage;
   441→   441→        vec3 p = position;
   442→   442→        // «пузо» паруса: синус по ширине и высоте, кромки закреплены.
   443→   443→        // асимметрия по высоте: максимум ниже середины — наполненное
   444→   444→        // полотно не выглядит плоским мешком. uBelly задан на CPU с
   445→   445→        // учётом ветра и степени постановки
   446→   446→        float belly = uBelly * sin(uv.x * 3.14159) * sin(uv.y * 3.14159)
   447→   447→                    * (0.72 + 0.28 * (1.0 - uv.y));
   448→   448→        // трепет свободной кромки: две волны, сильнее при слабом пузе (флюктер)
   449→   449→        float flut = uFlutter * (
   450→   450→          sin(uv.x * 9.0 + uTime * 7.3) * 0.6 +
   451→   451→          sin(uv.x * 17.0 - uTime * 11.7) * 0.4
   452→   452→        ) * sin(uv.y * 3.14159) * (1.0 - uv.y * 0.75);
   453→   453→        float dz = belly + flut;
   454→   454→        p.z += dz;
   455→   455→        // нормаль аналитически (по пуза) — с полным наклоном, чтобы
   456→   456→        // выпуклость читалась и в освещении, а не только в силуэте
   457→   457→        float dzu = uBelly * cos(uv.x * 3.14159) * 3.14159 * sin(uv.y * 3.14159);
   458→   458→        float dzv = uBelly * sin(uv.x * 3.14159) * 3.14159 * cos(uv.y * 3.14159);
   459→   459→        vN = normalize(vec3(-dzu * 0.8, -dzv * 0.8, 1.0));
   460→   460→        // рифление: при uFurl парус собирается к рее/штагу
   461→   461→        p.y *= mix(1.0, 0.10, uFurl);
   462→   462→        p.x *= mix(1.0, 0.55, uFurl);
   463→   463→        p.z *= mix(1.0, 0.08, uFurl);
   464→   464→        gl_Position = projectionMatrix * modelViewMatrix * vec4(p, 1.0);
   465→   465→      }
   466→   466→    `,
   467→   467→    fragmentShader: /* glsl */`
   468→   468→      varying vec2 vUv;
   469→   469→      varying vec3 vN;
   470→   470→      varying float vDmgU;
   471→   471→      uniform vec3 uColor;
   472→   472→      uniform vec3 uSunDir;
   473→   473→      uniform vec3 uHoles[8];
   474→   474→      uniform float uHoleN;
   475→   475→      uniform float uAspect;
   476→   476→      float hash12(vec2 p){ vec3 p3 = fract(vec3(p.xyx)*0.1031); p3+=dot(p3,p3.yzx+33.33); return fract((p3.x+p3.y)*p3.z); }
   477→   477→      float vnoise(vec2 p){ vec2 i=floor(p), f=fract(p); vec2 u=f*f*(3.0-2.0*f);
   478→   478→        return mix(mix(hash12(i),hash12(i+vec2(1,0)),u.x), mix(hash12(i+vec2(0,1)),hash12(i+vec2(1,1)),u.x), u.y); }
   479→   479→      void main() {
   480→   480→        vec3 col = uColor;
   481→   481→        // швы (горизонтальные полотнища)
   482→   482→        float seams = smoothstep(0.035, 0.0, abs(fract(vUv.y * 5.0) - 0.5) - 0.46);
   483→   483→        col *= 1.0 - seams * 0.16;
   484→   484→        // заплатки и рваные края от повреждений
   485→   485→        float dmg = vDmgU;
   486→   486→        float patches = step(0.55, vnoise(vUv * 4.7 + 7.3)) * dmg;
   487→   487→        col = mix(col, col * 0.55, patches);
   488→   488→        float holes = step(0.82 - dmg * 0.15, vnoise(vUv * 7.1 + 2.9)) * dmg * 1.4;
   489→   489→        // ===== ПРОБОИНЫ от снарядов (картечь/ядро/книппель) =====
   490→   490→        // рваный тёмный край + сквозная дыра (discard);
   491→   491→        // uAspect приводит UV к метрам — дыры круглые
   492→   492→        for (int i = 0; i < 8; i++) {
   493→   493→          if (float(i) >= uHoleN) break;
   494→   494→          vec2 hd = (vUv - uHoles[i].xy) * vec2(uAspect, 1.0);
   495→   495→          float hdist = length(hd);
   496→   496→          float n = vnoise(vUv * 17.0 + float(i) * 7.31);
   497→   497→          float rr = uHoles[i].z * (0.78 + 0.5 * n);
   498→   498→          if (hdist < rr) {
   499→   499→            col *= 0.42;
   500→   500→            if (hdist < rr * 0.58) discard;   // рваный сквозной проём
   501→   501→          }
   502→   502→        }
   503→   503→        // простое освещение
   504→   504→        float ndl = max(dot(normalize(vN), normalize(uSunDir)), 0.0) * 0.5 + 0.55;
   505→   505→        col *= ndl;
   506→   506→        float a = 1.0 - clamp(holes, 0.0, 0.9);
   507→   507→        if (a < 0.55) discard; // дыры — без полупрозрачности (нет проблем сортировки)
   508→   508→        gl_FragColor = vec4(col, 1.0);
   509→   509→      }
   510→   510→    `,
   511→   511→    side: THREE.DoubleSide,
   512→   512→  });
   513→   513→}

   571→   571→export function hullProfile(t: number, P: ShipPreset) {
   572→   572→  // t: 0 корма -> 1 нос. Полуширина, осадка, высота фальшборта
   573→   573→  const B = P.beam / 2;
   574→   574→  let w: number;
   575→   575→  if (t < 0.45) w = THREE.MathUtils.lerp(0.58, 1.0, smooth01(t / 0.45));
   576→   576→  else w = THREE.MathUtils.lerp(1.0, 0.0, Math.pow(smooth01((t - 0.45) / 0.55), 0.75));
   577→   577→  // пинасс: сужение только в самом конце — почти вертикальный плоский нос
   578→   578→  if (P.flatBow && t > 0.82) w = Math.max(w, THREE.MathUtils.lerp(0.74, 0.42, smooth01((t - 0.82) / 0.18)));
   579→   579→  const draft = P.draft * (0.72 + 0.28 * Math.sin(Math.PI * Math.min(1, t * 1.15)));
   580→   580→  let bulwark: number;
   581→   581→  if (P.beakhead) {
   582→   582→    // галеон: высокая корма, низкий бак — характерный силуэт с гальюном
   583→   583→    bulwark = P.freeboard * (0.85 + 0.36 * Math.pow(1 - t, 2.4) + 0.08 * Math.pow(Math.max(0, t - 0.7) / 0.3, 2));
   584→   584→  } else {
   585→   585→    bulwark = P.freeboard * (0.82 + 0.35 * Math.pow(Math.max(0, t - 0.75) / 0.25, 1.4) + 0.5 * Math.pow(1 - t, 2.2) * 0.6);
   586→   586→  }
   587→   587→  return { w: w * B, draft, bulwark };
   588→   588→}
   589→   589→function smooth01(x: number) { const t = Math.min(1, Math.max(0, x)); return t * t * (3 - 2 * t); }

   592→   592→/** Слот орудия: z — вдоль корпуса (м, локально), tier 0 = открытая
   593→   593→ *  (верхнепалубная) батарея с лафетами, 1.. — закрытые орудийные деки,

   597→   597→/** Раскладка ВСЕХ орудий борта (число всегда = gunsPerSide — отрисовка
   598→   598→ *  соответствует количеству). Без gunportRows всё на палубе (шлюп/флейт);
   599→   599→ *  1 ряд портов — палуба + одна закрытая батарея; 2–4 ряда — палуба +
   600→   600→ *  (rows−1) деков, порт-ряды идут сверху вниз (0.62F … 0.28F). */
   601→   601→export function gunLayout(P: ShipPreset): GunSlot[] {
