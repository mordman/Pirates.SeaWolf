  1252→/** Вёсла гребных судов (шебека, галера): вальные порты по борту +
  1253→ *  вёсла с лопастями, наклонённые к воде; гребки анимирует Ship.updateVisual */
  1254→function buildOars(P: ShipPreset): { group: THREE.Group; oars: THREE.Object3D[] } {
  1255→  const group = new THREE.Group();
  1256→  const oars: THREE.Object3D[] = [];
  1257→  const n = P.oars || 0;
  1258→  if (!n) return { group, oars };
  1259→  const L = P.length;
  1260→  const shaftMat = new THREE.MeshStandardMaterial({ color: 0x8a6a42, roughness: 0.85 });
  1261→  const portMat = new THREE.MeshStandardMaterial({ color: 0x14100d, roughness: 0.9 });
  1262→  const oarLen = clamp2(L * 0.17, 4.5, 11);
  1263→  const z0 = -L * 0.26, z1 = L * 0.30;
  1264→  for (let i = 0; i < n; i++) {
  1265→    const t = n === 1 ? 0.5 : i / (n - 1);
  1266→    const z = THREE.MathUtils.lerp(z0, z1, t);
  1267→    const tt = clamp2((z + L / 2) / L, 0, 1);
  1268→    const y = P.freeboard * 0.30; // вальная палуба — ниже орудийной
  1269→    const hw = hullHalfWidthAt(P, tt, y);
  1270→    for (const side of [-1, 1]) {
  1271→      // тёмный вальный порт
  1272→      const port = new THREE.Mesh(new THREE.BoxGeometry(0.06, 0.26, 0.3), portMat);
  1273→      port.position.set(side * (hw + 0.01), y, z);
  1274→      group.add(port);
  1275→      // весло: вал + лопасть, наклонены вниз-наружу к воде
  1276→      const oar = new THREE.Group();
  1277→      const shaft = new THREE.Mesh(new THREE.CylinderGeometry(0.05, 0.065, oarLen, 5), shaftMat);
  1278→      shaft.rotation.z = Math.PI / 2;
  1279→      shaft.position.x = oarLen * 0.42;
  1280→      oar.add(shaft);
  1281→      const blade = new THREE.Mesh(new THREE.BoxGeometry(oarLen * 0.3, 0.03, 0.2), shaftMat);
  1282→      blade.position.x = oarLen * 0.92;
  1283→      oar.add(blade);
  1284→      oar.position.set(side * (hw + 0.04), y, z);
  1285→      oar.rotation.z = side * -0.34;
  1286→      oar.userData.phase = t * Math.PI * 2 + (side > 0 ? Math.PI : 0);
  1287→      oar.userData.side = side;
  1288→      oar.userData.base = side * -0.34;
  1289→      group.add(oar);
  1290→      oars.push(oar);
  1291→    }
  1292→  }
  1293→  return { group, oars };
  1294→}

  1328→export function buildShip(P: ShipPreset): ShipVisual {
  1329→  const group = new THREE.Group();
  1330→  const L = P.length;
  1331→
  1332→  // корпус
  1333→  const { geometry: hullGeo, outline } = buildHull(P);
  1334→  // DoubleSide: даже если камера (или волна) заглянет под корпус/за корму —
  1335→  // стенки видны, «прозрачной кормы» нет (изнутри односторонние грани куллятся)
  1336→  const hullMat = new THREE.MeshStandardMaterial({ vertexColors: true, roughness: 0.82, metalness: 0.05, side: THREE.DoubleSide });
  1337→  const hull = new THREE.Mesh(hullGeo, hullMat);
  1338→  hull.castShadow = true;
  1339→  group.add(hull);
  1340→
  1341→  // палуба: текстура досок с конопачением (DoubleSide — страховка от куллинга)
  1342→  const deckMat = new THREE.MeshStandardMaterial({
  1343→    map: makeDeckTexture(P.colors.deck, P.colors.deck + P.length),
  1344→    roughness: 0.93, side: THREE.DoubleSide,
  1345→  });

  1347→  group.add(deck);
  1348→
  1349→  // планширь по верхней кромке фальшборта
  1350→  const capRail = new THREE.Mesh(buildCapRail(P),
  1351→    new THREE.MeshStandardMaterial({ vertexColors: true, roughness: 0.7, metalness: 0.05 }));
  1352→  group.add(capRail);
  1353→
  1354→  const woodMat = new THREE.MeshStandardMaterial({ color: 0x6e5433, roughness: 0.8 });
  1355→  const darkMat = new THREE.MeshStandardMaterial({ color: P.colors.trim, roughness: 0.75 });
  1356→
  1357→  // бушприт
  1358→  const bowsprit = new THREE.Mesh(new THREE.CylinderGeometry(0.09, 0.16, L * 0.32, 6), woodMat);
  1359→  bowsprit.rotation.x = Math.PI / 2 - 0.18;
  1360→  bowsprit.position.set(0, P.freeboard * 0.75, L / 2 + L * 0.12);
  1361→  group.add(bowsprit);
  1362→  const bowspritTip = new THREE.Vector3(0, P.freeboard * 0.75 + L * 0.32 * 0.18, L / 2 + L * 0.12 + L * 0.32 * Math.cos(0.18) * 0.98);
  1363→
  1364→  // мачты
  1365→  const mastDefs: { z: number; h: number }[] = [];
  1366→  const H = P.freeboard + L * 0.78;
  1367→  if (P.masts === 2) {
  1368→    mastDefs.push({ z: L * 0.16, h: H * 0.88 });
  1369→    mastDefs.push({ z: -L * 0.10, h: H });
  1370→  } else {
  1371→    mastDefs.push({ z: L * 0.18, h: H * 0.8 });
  1372→    mastDefs.push({ z: -L * 0.05, h: H });
  1373→    mastDefs.push({ z: -L * 0.28, h: H * 0.62 });
  1374→  }
  1375→  const masts: THREE.Object3D[] = [];
  1376→  for (const md of mastDefs) {
  1377→    const mast = new THREE.Group();
  1378→    const pole = new THREE.Mesh(new THREE.CylinderGeometry(0.11, 0.24, md.h, 7), woodMat);
  1379→    pole.position.y = md.h / 2 - P.draft * 0.3;
  1380→    pole.castShadow = true;
  1381→    mast.add(pole);
  1382→    // марс (платформа)
  1383→    const top = new THREE.Mesh(new THREE.CylinderGeometry(0.75, 0.6, 0.28, 8), darkMat);
  1384→    top.position.y = md.h * 0.72 - P.draft * 0.3;
  1385→    mast.add(top);
  1386→    mast.position.set(0, -P.draft * 0.3 > 0 ? 0 : 0, md.z);
  1387→    mast.userData.height = md.h;
  1388→    masts.push(mast);
  1389→    group.add(mast);
  1390→  }
  1391→
  1392→  // паруса: каждый со своим материалом (своё рифление), квадратные —
  1393→  // на поворотной группе «рея+парус» для брасопки по ветру
  1394→  const sails: SailInfo[] = [];
  1395→  const addSquareSail = (mastIndex: number, level: number, width: number, height: number, setAt: number) => {
  1396→    const md = mastDefs[mastIndex];
  1397→    const g = new THREE.PlaneGeometry(width, height, 10, 7);
  1398→    g.translate(0, -height / 2, 0); // верхняя кромка у реи
  1399→    const mat = sailMaterial(P.colors.sail);
  1400→    mat.uniforms.uAspect.value = width / height;
  1401→    const mesh = new THREE.Mesh(g, mat);
  1402→    const y = md.h * level - P.draft * 0.3;
  1403→    // поворотная группа: ось вращения — мачта (брасопка рея по ветру)
  1404→    // рея+парус — потомок МАЧТЫ: сломанная мачта валится вместе с ними
  1405→    const pivot = new THREE.Group();
  1406→    pivot.position.set(0, y + 0.05, 0);
  1407→    // рея
  1408→    const yard = new THREE.Mesh(new THREE.CylinderGeometry(0.07, 0.07, width * 1.04, 5), woodMat);
  1409→    yard.rotation.z = Math.PI / 2;
  1410→    pivot.add(yard);
  1411→    mesh.position.set(0, 0, -0.25);
  1412→    pivot.add(mesh);
  1413→    masts[mastIndex].add(pivot);
  1414→    // зона прострела: прямоугольник полотна в координатах корабля (brace=0);
  1415→    // при проверке поворачивается на текущий угол брасопки вокруг мачты
  1416→    const quad: SailHitQuad = {
  1417→      pivot: new THREE.Vector3(0, y + 0.05, md.z),
  1418→      corners: [
  1419→        new THREE.Vector3(-width / 2, -height, -0.25), // uv(0,0) — нижний-лево
  1420→        new THREE.Vector3(width / 2, -height, -0.25),  // uv(1,0)
  1421→        new THREE.Vector3(width / 2, 0, -0.25),        // uv(1,1)
  1422→        new THREE.Vector3(-width / 2, 0, -0.25),       // uv(0,1)
  1423→      ],
  1424→      uvs: [
  1425→        new THREE.Vector2(0, 0), new THREE.Vector2(1, 0),
  1426→        new THREE.Vector2(1, 1), new THREE.Vector2(0, 1),
  1427→      ],
  1428→      rotates: true,
  1429→    };
  1430→    sails.push({ mesh, mat, brace: pivot, setAt, furl: 1, square: true, braceAngle: 0, mastIndex, quad, holeW: width, holeH: height });
  1431→  };
  1432→  // латинский парус (каравелла): длинная рейка от мачты к носу,
  1433→  // треугольное полотно в диаметральной плоскости, «пузо» — на борт
  1434→  const addLateenSail = (mastIndex: number, setAt: number, yLen: number, drop: number, aftFrac: number) => {
  1435→    const md = mastDefs[mastIndex];
  1436→    // локальная плоскость XY: +X -> нос, +Y вверх; поворот меша
  1437→    // развернёт её в диаметральную, «пузо» (локальный +Z) — на борт
  1438→    const H = { x: 0, y: 0 };
  1439→    const T = { x: yLen, y: -drop * 0.62 };        // нижний конец рея — вперёд и вниз
  1440→    const C = { x: -yLen * aftFrac, y: -drop };     // шкотовый угол — к корме
  1441→    const NU = 9, NV = 6;
  1442→    const pos: number[] = [], uvs: number[] = [], idx: number[] = [];
  1443→    for (let i = 0; i <= NU; i++) {
  1444→      const u = i / NU;
  1445→      const ex = THREE.MathUtils.lerp(H.x, T.x, u), ey = THREE.MathUtils.lerp(H.y, T.y, u);
  1446→      const fx = THREE.MathUtils.lerp(H.x, C.x, u), fy = THREE.MathUtils.lerp(H.y, C.y, u);
  1447→      for (let j = 0; j <= NV; j++) {
  1448→        const v = j / NV;
  1449→        pos.push(THREE.MathUtils.lerp(ex, fx, v), THREE.MathUtils.lerp(ey, fy, v), 0);
  1450→        uvs.push(u, v);
  1451→      }
  1452→    }
  1453→    const row = NV + 1;
  1454→    for (let i = 0; i < NU; i++) {
  1455→      for (let j = 0; j < NV; j++) {
  1456→        const p0 = i * row + j, p1 = p0 + 1, p2 = p0 + row, p3 = p2 + 1;
  1457→        idx.push(p0, p1, p2, p1, p3, p2);
  1458→      }
  1459→    }
  1460→    const g = new THREE.BufferGeometry();
  1461→    g.setAttribute('position', new THREE.Float32BufferAttribute(pos, 3));
  1462→    g.setAttribute('uv', new THREE.Float32BufferAttribute(uvs, 2));
  1463→    g.setIndex(idx);
  1464→    g.computeVertexNormals();
  1465→    const mat = sailMaterial(P.colors.sail);
  1466→    // зона прострела латинского паруса — треугольник H(голова)-T(конец реи)-C(шкот):
  1467→    // в локалях меша +X -> нос после поворота меша на -90° вокруг Y
  1468→    const yardLenM = Math.hypot(T.x - H.x, T.y - H.y);
  1469→    mat.uniforms.uAspect.value = Math.max(0.3, (yardLenM + Math.abs(C.x)) / Math.max(1, drop));
  1470→    const mesh = new THREE.Mesh(g, mat);
  1471→    mesh.rotation.y = -Math.PI / 2; // локальный +X -> нос, «пузо» -> борт
  1472→    mesh.frustumCulled = false;
  1473→    const pivot = new THREE.Group();
  1474→    pivot.position.set(0, md.h * 0.72 - P.draft * 0.3, 0);
  1475→    // рея: от головы у мачты к носу-вниз
  1476→    const yardLen = Math.hypot(T.x - H.x, T.y - H.y) * 1.06;
  1477→    const yard = new THREE.Mesh(new THREE.CylinderGeometry(0.055, 0.085, yardLen, 5), woodMat);
  1478→    yard.rotation.z = Math.atan2(T.y - H.y, T.x - H.x) - Math.PI / 2;
  1479→    yard.position.set(T.x / 2, T.y / 2, 0);
  1480→    pivot.add(yard);
  1481→    pivot.add(mesh);
  1482→    masts[mastIndex].add(pivot);
  1483→    // поворот меша -90° вокруг Y: (x, y, 0) -> (0, y, x)
  1484→    const rotP = (x: number, y: number) => new THREE.Vector3(0, y, x);
  1485→    const pv = new THREE.Vector3(0, md.h * 0.72 - P.draft * 0.3, md.z);
  1486→    const quad: SailHitQuad = {
  1487→      pivot: pv,
  1488→      corners: [rotP(H.x, H.y), rotP(T.x, T.y), rotP(C.x, C.y), rotP(C.x, C.y)],
  1489→      uvs: [
  1490→        new THREE.Vector2(0, 0), new THREE.Vector2(1, 0),
  1491→        new THREE.Vector2(1, 1), new THREE.Vector2(1, 1),
  1492→      ],
  1493→      rotates: false,
  1494→    };
  1495→    sails.push({
  1496→      mesh, mat, setAt, furl: 1, square: false, braceAngle: 0, mastIndex,
  1497→      quad, holeW: yardLenM + Math.abs(C.x), holeH: drop,
  1498→    });
  1499→  };
  1500→  if (P.rig === 'single') {
  1501→    // когг: один большой прямой парус на мачту
  1502→    addSquareSail(0, 0.7, L * 0.42, L * 0.24, 0.5);
  1503→    addSquareSail(1, 0.76, L * 0.4, L * 0.22, 0.72);
  1504→  } else if (P.rig === 'lateenMix') {
  1505→    // каравелла редонда: прямой фок + латинские грот и бизань
  1506→    addSquareSail(0, 0.6, L * 0.28, L * 0.13, 0.55);
  1507→    addLateenSail(1, 0.45, L * 0.5, mastDefs[1].h * 0.5, 0.85);
  1508→    addLateenSail(2, 0.8, L * 0.32, mastDefs[2].h * 0.5, 0.55);
  1509→  } else if (P.rig === 'lateen') {
  1510→    // шебека / галера: огромные латинские паруса на всех мачтах
  1511→    addLateenSail(0, 0.68, L * 0.34, mastDefs[0].h * 0.46, 0.5);
  1512→    addLateenSail(1, 0.45, L * 0.52, mastDefs[1].h * 0.52, 0.85);
  1513→    addLateenSail(2, 0.78, L * 0.3, mastDefs[2].h * 0.44, 0.5);
  1514→  } else if (P.masts === 2) {
  1515→    addSquareSail(0, 0.66, L * 0.3, L * 0.16, 0.62);   // фок
  1516→    addSquareSail(1, 0.72, L * 0.36, L * 0.2, 0.45);   // грот нижний — ставится первым
  1517→    addSquareSail(1, 0.45, L * 0.26, L * 0.14, 0.92);  // грот верхний (брамсель)
  1518→  } else {
  1519→    addSquareSail(0, 0.62, L * 0.24, L * 0.13, 0.60);
  1520→    addSquareSail(1, 0.7, L * 0.34, L * 0.18, 0.45);
  1521→    addSquareSail(1, 0.44, L * 0.24, L * 0.13, 0.92);
  1522→    addSquareSail(2, 0.5, L * 0.18, L * 0.11, 0.78);
  1523→  }
  1524→
  1525→  // кливера (треугольные) от бушприта к фок-мачте: сетка с uv —
  1526→  // есть «пузо» и трепет; при рифлении собираются к голове
  1527→  const addJib = (tip: THREE.Vector3, foot: THREE.Vector3, headY: number, mastZ: number, setAt: number) => {
  1528→    const head = new THREE.Vector3(0, headY, mastZ - 0.3);
  1529→    // локальная система: X = −мирZ (продольная), Y = мирY, Z = мирX (пузо вбок)
  1530→    const a = { x: -(tip.z - head.z), y: tip.y - head.y };
  1531→    const b = { x: -(foot.z - head.z), y: foot.y - head.y };
  1532→    const NU = 8, NV = 5;
  1533→    const pos: number[] = [], uvs: number[] = [], idx: number[] = [];
  1534→    for (let i = 0; i <= NU; i++) {
  1535→      const u = i / NU;
  1536→      // точка на штаге (к галсовому углу) и на шкаторине
  1537→      const lx = a.x * u, ly = a.y * u;
  1538→      const rx = b.x * u, ry = b.y * u;
  1539→      for (let j = 0; j <= NV; j++) {
  1540→        const v = j / NV;
  1541→        pos.push(THREE.MathUtils.lerp(lx, rx, v), THREE.MathUtils.lerp(ly, ry, v), 0);
  1542→        uvs.push(u, v);
  1543→      }
  1544→    }
  1545→    const row = NV + 1;
  1546→    for (let i = 0; i < NU; i++) {
  1547→      for (let j = 0; j < NV; j++) {
  1548→        const p0 = i * row + j, p1 = p0 + 1, p2 = p0 + row, p3 = p2 + 1;
  1549→        idx.push(p0, p1, p2, p1, p3, p2);
  1550→      }
  1551→    }
  1552→    const g = new THREE.BufferGeometry();
  1553→    g.setAttribute('position', new THREE.Float32BufferAttribute(pos, 3));
  1554→    g.setAttribute('uv', new THREE.Float32BufferAttribute(uvs, 2));
  1555→    g.setIndex(idx);
  1556→    g.computeVertexNormals();
  1557→    const mat = sailMaterial(P.colors.sail);
  1558→    mat.uniforms.uAspect.value = Math.max(0.3, Math.hypot(a.x, a.y) / Math.max(1, Math.hypot(b.x, b.y)));
  1559→    const mesh = new THREE.Mesh(g, mat);
  1560→    mesh.position.set(head.x, head.y, head.z - mastZ);
  1561→    mesh.rotation.y = Math.PI / 2; // локальный Z (пузо) -> мир X (борт)
  1562→    mesh.frustumCulled = false;
  1563→    masts[0].add(mesh); // потомок фок-мачты — падает вместе с ней
  1564→    // зона прострела: треугольник голова-(конец штага)-(галсовый угол);
  1565→    // поворот меша +90° вокруг Y: (x, y, 0) -> (0, y, -x)
  1566→    const rotP = (x: number, y: number) => new THREE.Vector3(0, y, -x);
  1567→    const pv = new THREE.Vector3(head.x, head.y, head.z);
  1568→    const quad: SailHitQuad = {
  1569→      pivot: pv,
  1570→      corners: [rotP(0, 0), rotP(a.x, a.y), rotP(b.x, b.y), rotP(b.x, b.y)],
  1571→      uvs: [
  1572→        new THREE.Vector2(0, 0), new THREE.Vector2(1, 0),
  1573→        new THREE.Vector2(1, 1), new THREE.Vector2(1, 1),
  1574→      ],
  1575→      rotates: false,
  1576→    };
  1577→    sails.push({
  1578→      mesh, mat, setAt, furl: 1, square: false, braceAngle: 0, mastIndex: 0,
  1579→      quad, holeW: Math.hypot(a.x, a.y), holeH: Math.hypot(b.x, b.y),
  1580→    });
  1581→  };
  1582→  // кливера — только на «полных» вооружениях (у когга, каравеллы и шебеки их нет)
  1583→  const foreTop = mastDefs[0];
  1584→  if (P.rig !== 'lateenMix' && P.rig !== 'single' && P.rig !== 'lateen') {
  1585→    addJib(bowspritTip, new THREE.Vector3(0, P.freeboard * 0.72, L / 2 - 1.0), foreTop.h * 0.62 - P.draft * 0.3, foreTop.z, 0.72);
  1586→    addJib(new THREE.Vector3(bowspritTip.x * 0.55, bowspritTip.y - (bowspritTip.y - P.freeboard * 0.72) * 0.45, bowspritTip.z - L * 0.1),
  1587→      new THREE.Vector3(0, P.freeboard * 0.72, L / 2 - 2.0), foreTop.h * 0.78 - P.draft * 0.3, foreTop.z, 1.0);
  1588→  }
  1589→
  1590→  // флаг на топе грот-мачты (потомок мачты — валится вместе с ней)
  1591→  const mainMastIdx = 1; // грот-мачта и у 2-, и у 3-мачтовых
  1592→  const mainMast = mastDefs[mainMastIdx];
  1593→  const { mesh: flag, mat: flagMat } = flagMesh(P.flagKind);
  1594→  flag.position.set(0.05, mainMast.h - P.draft * 0.3 - 0.4, 0);
  1595→  masts[mainMastIdx].add(flag);
  1596→
  1597→  // пушки по бортам — по раскладке gunLayout: ИТОГО ровно gunsPerSide
  1598→  // на борт. Верхняя палуба — полные лафеты, закрытые деки — стволы,
  1599→  // высунутые из портов (лафет внутри корпуса не виден)
  1600→  const gunsPort: THREE.Object3D[] = [];
  1601→  const gunsStarboard: THREE.Object3D[] = [];
  1602→  for (const slot of gunLayout(P)) {
  1603→    const tt = clamp2((slot.z + L / 2) / L, 0, 1);
  1604→    const isDeck = slot.tier === 0;
  1605→    const prof = hullProfile(tt, P);
  1606→    for (const side of [-1, 1]) {
  1607→      const gun = new THREE.Group();
  1608→      const barrel = new THREE.Mesh(new THREE.CylinderGeometry(0.11, 0.15, 1.15, 8),
  1609→        new THREE.MeshStandardMaterial({ color: 0x1f1f22, roughness: 0.5, metalness: 0.75 }));
  1610→      barrel.rotation.z = Math.PI / 2;
  1611→      barrel.position.set(side * 0.62, 0.08, 0);
  1612→      gun.add(barrel);
  1613→      const carriage = new THREE.Mesh(new THREE.BoxGeometry(0.5, 0.34, 0.55), darkMat);
  1614→      gun.add(carriage);
  1615→      if (isDeck) {
  1616→        gun.position.set(side * (prof.w * 0.9 - 0.15), prof.bulwark - 0.72, slot.z);
  1617→      } else {
  1618→        // закрытая батарея: дуло высунуто из порта за борт
  1619→        const hw = hullHalfWidthAt(P, tt, slot.y);
  1620→        gun.position.set(side * (hw - 0.85), slot.y, slot.z);
  1621→      }
  1622→      gun.userData.side = side;
  1623→      group.add(gun);
  1624→      // локальный −X = штирборт (правый борт), локальный +X = порт (левый)
  1625→      (side < 0 ? gunsStarboard : gunsPort).push(gun);
  1626→    }
  1627→  }
  1628→
  1629→  // штурвал: пьедестал (не вращается) + колесо, начало координат
  1630→  // вращающейся группы — в ступице, чтобы крутился вокруг СВОЕЙ оси
  1631→  const wheelAsm = new THREE.Group();
  1632→  const wheel = new THREE.Group();
  1633→  const wheelR = 0.55;
  1634→  const hubY = 0.98; // высота оси над палубой
  1635→  const wheelTorus = new THREE.Mesh(new THREE.TorusGeometry(wheelR, 0.045, 6, 18), darkMat);
  1636→  wheel.add(wheelTorus);
  1637→  for (let i = 0; i < 4; i++) {
  1638→    const ang = (i / 4) * Math.PI;
  1639→    const spoke = new THREE.Mesh(new THREE.CylinderGeometry(0.028, 0.028, wheelR * 2.3, 4), darkMat);
  1640→    spoke.rotation.z = ang;
  1641→    wheel.add(spoke);
  1642→    // рукоятки наружу, к рулевому (в сторону кормы, -Z)
  1643→    const handle = new THREE.Mesh(new THREE.CylinderGeometry(0.035, 0.035, 0.26, 5), woodMat);
  1644→    handle.rotation.x = Math.PI / 2;
  1645→    handle.position.set(-Math.sin(ang) * (wheelR + 0.02), Math.cos(ang) * (wheelR + 0.02), -0.14);
  1646→    wheel.add(handle);
  1647→  }
  1648→  const wheelHub = new THREE.Mesh(new THREE.CylinderGeometry(0.1, 0.1, 0.34, 8), woodMat);
  1649→  wheelHub.rotation.x = Math.PI / 2;
  1650→  wheel.add(wheelHub);
  1651→  wheel.position.set(0, hubY, 0);
  1652→  wheelAsm.add(wheel);
  1653→  // стойка, подкосы и ось — вне вращающейся группы
  1654→  const post = new THREE.Mesh(new THREE.BoxGeometry(0.14, hubY, 0.12), darkMat);
  1655→  post.position.set(0, hubY / 2, 0);
  1656→  wheelAsm.add(post);
  1657→  for (const sgn of [-1, 1]) {
  1658→    const brace = new THREE.Mesh(new THREE.BoxGeometry(0.08, hubY * 1.2, 0.08), darkMat);
  1659→    brace.position.set(sgn * 0.24, hubY * 0.5, 0);
  1660→    brace.rotation.z = sgn * 0.55;
  1661→    wheelAsm.add(brace);
  1662→  }
  1663→  const wheelAxle = new THREE.Mesh(new THREE.CylinderGeometry(0.04, 0.04, 0.55, 6), darkMat);
  1664→  wheelAxle.rotation.x = Math.PI / 2;
  1665→  wheelAxle.position.set(0, hubY, 0.12);
  1666→  wheelAsm.add(wheelAxle);
  1667→  const wheelT = P.castleH ? 0.29 : 0.2;
  1668→  wheelAsm.position.set(0, hullProfile(wheelT, P).bulwark - 1.0, THREE.MathUtils.lerp(-L / 2, L / 2, wheelT));
  1669→  group.add(wheelAsm);
  1670→
  1671→  // палубный реквизит: гратинг-люк, бочки, канаты, кнехты, колокол, ядра,
  1672→  // сундук, ведро — одним мешом
  1673→  const props = new THREE.Mesh(buildDeckProps(P),
  1674→    new THREE.MeshStandardMaterial({ vertexColors: true, roughness: 0.85, side: THREE.DoubleSide }));
  1675→  props.castShadow = true;
  1676→  group.add(props);
  1677→
  1678→  // шпиль, фонарь, якоря — строго на палубе
  1679→  const deckYAt = (t: number) => hullProfile(t, P).bulwark - 1.0;
  1680→  const capstan = new THREE.Mesh(new THREE.CylinderGeometry(0.4, 0.5, 0.7, 8), woodMat);
  1681→  capstan.position.set(0, deckYAt(0.36) + 0.35, -L * 0.16);
  1682→  group.add(capstan);
  1683→  const sternLantern = new THREE.Mesh(new THREE.SphereGeometry(0.22, 8, 8),
  1684→    new THREE.MeshStandardMaterial({ color: 0xffd88a, emissive: 0xffaa33, emissiveIntensity: 1.6 }));
  1685→  sternLantern.position.set(0, P.freeboard * 1.15, -L / 2 - 0.15);
  1686→  group.add(sternLantern);
  1687→  let lantern: THREE.PointLight | undefined;
  1688→  // фигура на носу
  1689→  const figure = new THREE.Mesh(new THREE.ConeGeometry(0.22, 0.8, 6),
  1690→    new THREE.MeshStandardMaterial({ color: 0xc9a24a, roughness: 0.4, metalness: 0.6 }));
  1691→  figure.position.set(0, P.freeboard * 1.12, L / 2 - 0.2);
  1692→  figure.rotation.x = -0.5;
  1693→  group.add(figure);
  1694→  // (простые якоря-коробки заменены настоящими якорями в группе мелких деталей)
  1695→
  1696→  // такелаж
  1697→  group.add(buildRigging(P, mastDefs, bowspritTip));
  1698→
  1699→  // надстройки-«замки» (когг) — силуэт виден издалека
  1700→  if (P.castleH) {
  1701→    const castles = new THREE.Mesh(buildCastles(P),
  1702→      new THREE.MeshStandardMaterial({ vertexColors: true, roughness: 0.85 }));
  1703→    castles.castShadow = true;
  1704→    group.add(castles);
  1705→  }
  1706→  // наружный декор: пушечные порты, вельсы, кормовая галерея, нос/гальюн
  1707→  const ornGeo = buildHullOrnaments(P);
  1708→  if (ornGeo) {
  1709→    group.add(new THREE.Mesh(ornGeo,
  1710→      new THREE.MeshStandardMaterial({ vertexColors: true, roughness: 0.8, metalness: 0.15 })));
  1711→  }
  1712→  // мелкие детали — только вблизи (LOD, переключает game.ts)
  1713→  const detail = buildShipDetail(P, mastDefs);
  1714→  detail.visible = false;
  1715→  group.add(detail);
  1716→
  1717→  // вёсла (шебека, галера)
  1718→  let oarsVis: THREE.Object3D[] | undefined;
  1719→  if (P.oars) {
  1720→    const o = buildOars(P);
  1721→    group.add(o.group);
  1722→    oarsVis = o.oars;
  1723→  }
  1724→
  1725→  // руль за транцем: перо на петлях, поворачивается вместе с рулём
  1726→  const rudLen = Math.min(2.6, Math.max(1.4, L * 0.045));
  1727→  const rudGeo = new THREE.BoxGeometry(0.15, Math.max(1.2, P.draft * 0.72), rudLen);
  1728→  rudGeo.translate(0, 0, -rudLen / 2); // ось вращения — передняя кромка
  1729→  const rudderMesh = new THREE.Mesh(rudGeo, darkMat);
  1730→  rudderMesh.position.set(0, -P.draft * 0.40, -L / 2 + 0.03);
  1731→  rudderMesh.castShadow = true;
  1732→  group.add(rudderMesh);
  1733→
  1734→  const rig: SailRig = { sails };
  1735→
  1736→  return {
  1737→    group, sails: rig, gunsPort, gunsStarboard, flag, flagMat,
  1738→    dims: { length: L, beam: P.beam, draft: P.draft }, outline,
  1739→    masts, lantern, wheel, detail,
  1740→    oars: oarsVis, rudderMesh,
  1741→  };
  1742→}
