     4→// Процедурная постройка кораблей (исторические силуэты
     5→// XVI–XVIII веков): шлюп, флейт, бриг (игрок/торговцы/патруль)

     8→// батареями, фрегат. Корпус — по шпангоутам, паруса — шейдер
     9→// с «пузом», такелаж — линиями. Все размеры в метрах, 1u = 1м.
    10→// ============================================================
    11→
    12→export interface ShipPreset {
    13→  id: string;
    14→  name: string;
    15→  length: number; beam: number; draft: number; freeboard: number;
    16→  masts: 2 | 3;
    17→  gunsPerSide: number;
    18→  mass: number;          // кг
    19→  maxSpeed: number;      // м/с
    20→  hullHP: number; sailHP: number; maxCrew: number;
    21→  cargo: number;         // грузоподъёмность, т
    22→  colors: { hull: number; deck: number; trim: number; sail: number };
    23→  flagKind: 'pirate' | 'merchant' | 'navy';
    24→  // --- классификация (энциклопедия, мини-карта) ---
    25→  category: 'war' | 'merchant';          // военный / торговый
    26→  shipClass: string;                     // класс/ранг судна
    27→  year?: number;                         // год постройки
    28→  country?: string;                      // страна
    29→  // --- архитектура XVI–XVIII веков ---
    30→  rig?: 'full' | 'single' | 'lateenMix' | 'lateen'; // полное / один парус на мачту (когг) / фок прямой + латинские (каравелла) / все латинские (шебека, галера)
    31→  gunportRows?: number;  // рядов пушечных портов по борту (1–4)
    32→  castleH?: number;      // высота надстроек-«замков» (когг)
    33→  flatBow?: boolean;     // пинасс: почти плоская носовая переборка
    34→  beakhead?: boolean;    // галеон: низкий бак с гальюном
    35→  gallery?: boolean;     // кормовая галерея с окнами
    36→  oars?: number;         // вёсел на борт (гребные суда: шебека, галера)
    37→  stripes?: { y0: number; y1: number; color: number }[]; // цветные полосы борта (доли свободного борта): «Victory stripe»
    38→}

   406→export function sailAddHole(s: SailInfo, u: number, v: number, r: number) {
   407→  const arr = s.mat.uniforms.uHoles.value as THREE.Vector3[];
   408→  const n = s.mat.uniforms.uHoleN.value as number;
   409→  if (n < arr.length) {
   410→    arr[n].set(u, v, r);
   411→    s.mat.uniforms.uHoleN.value = n + 1;
   412→  } else {
   413→    // парус уже изрешечён — рвём существующюю дыру шире
   414→    const h = arr[(Math.random() * arr.length) | 0];
   415→    h.z = Math.min(0.34, h.z + r * 0.6);
   416→  }

   419→// ---------- шейдер паруса ----------
   420→function sailMaterial(baseColor: number): THREE.ShaderMaterial {
   421→  return new THREE.ShaderMaterial({
   422→    uniforms: {
   423→      uBelly: { value: 0.5 },
   424→      uFlutter: { value: 0.3 },
   425→      uTime: { value: 0 },
   426→      uDamage: { value: 0 },
   427→      uColor: { value: new THREE.Color(baseColor) },
   428→      uSunDir: { value: new THREE.Vector3(0.4, 0.8, 0.3) },
   429→      uFurl: { value: 0 },
   430→      // пробоины от картечи/ядер/книппеля: (u, v, радиус в долях высоты)
   431→      uHoles: { value: Array.from({ length: 8 }, () => new THREE.Vector3(0, 0, 0)) },
   432→      uHoleN: { value: 0 },
   433→      uAspect: { value: 1 },      // ширина/высота — дыры круглые в метрах
   434→    },
   435→    vertexShader: /* glsl */`
   436→      uniform float uBelly, uTime, uFurl, uFlutter;
   437→      varying vec2 vUv;
   438→      varying vec3 vN;
   439→      varying float vDmgU;
   440→      uniform float uDamage;
   441→      void main() {
   442→        vUv = uv;
   443→        vDmgU = uDamage;
   444→        vec3 p = position;
   445→        // «пузо» паруса: синус по ширине и высоте, кромки закреплены.
   446→        // асимметрия по высоте: максимум ниже середины — наполненное
   447→        // полотно не выглядит плоским мешком. uBelly задан на CPU с
   448→        // учётом ветра и степени постановки
   449→        float belly = uBelly * sin(uv.x * 3.14159) * sin(uv.y * 3.14159)
   450→                    * (0.72 + 0.28 * (1.0 - uv.y));
   451→        // трепет свободной кромки: две волны, сильнее при слабом пузе (флюктер)
   452→        float flut = uFlutter * (
   453→          sin(uv.x * 9.0 + uTime * 7.3) * 0.6 +
   454→          sin(uv.x * 17.0 - uTime * 11.7) * 0.4
   455→        ) * sin(uv.y * 3.14159) * (1.0 - uv.y * 0.75);
   456→        float dz = belly + flut;
   457→        p.z += dz;
   458→        // нормаль аналитически (по пуза) — с полным наклоном, чтобы
   459→        // выпуклость читалась и в освещении, а не только в силуэте
   460→        float dzu = uBelly * cos(uv.x * 3.14159) * 3.14159 * sin(uv.y * 3.14159);
   461→        float dzv = uBelly * sin(uv.x * 3.14159) * 3.14159 * cos(uv.y * 3.14159);
   462→        vN = normalize(vec3(-dzu * 0.8, -dzv * 0.8, 1.0));
   463→        // рифление: при uFurl парус собирается к рее/штагу
   464→        p.y *= mix(1.0, 0.10, uFurl);
   465→        p.x *= mix(1.0, 0.55, uFurl);
   466→        p.z *= mix(1.0, 0.08, uFurl);
   467→        gl_Position = projectionMatrix * modelViewMatrix * vec4(p, 1.0);
   468→      }
   469→    `,
   470→    fragmentShader: /* glsl */`
   471→      varying vec2 vUv;
   472→      varying vec3 vN;
   473→      varying float vDmgU;
   474→      uniform vec3 uColor;
   475→      uniform vec3 uSunDir;
   476→      uniform vec3 uHoles[8];
   477→      uniform float uHoleN;
   478→      uniform float uAspect;
   479→      float hash12(vec2 p){ vec3 p3 = fract(vec3(p.xyx)*0.1031); p3+=dot(p3,p3.yzx+33.33); return fract((p3.x+p3.y)*p3.z); }
   480→      float vnoise(vec2 p){ vec2 i=floor(p), f=fract(p); vec2 u=f*f*(3.0-2.0*f);
   481→        return mix(mix(hash12(i),hash12(i+vec2(1,0)),u.x), mix(hash12(i+vec2(0,1)),hash12(i+vec2(1,1)),u.x), u.y); }
   482→      void main() {
   483→        vec3 col = uColor;
   484→        // швы (горизонтальные полотнища)
   485→        float seams = smoothstep(0.035, 0.0, abs(fract(vUv.y * 5.0) - 0.5) - 0.46);
   486→        col *= 1.0 - seams * 0.16;
   487→        // заплатки и рваные края от повреждений
   488→        float dmg = vDmgU;
   489→        float patches = step(0.55, vnoise(vUv * 4.7 + 7.3)) * dmg;
   490→        col = mix(col, col * 0.55, patches);
   491→        float holes = step(0.82 - dmg * 0.15, vnoise(vUv * 7.1 + 2.9)) * dmg * 1.4;
   492→        // ===== ПРОБОИНЫ от снарядов (картечь/ядро/книппель) =====
   493→        // рваный тёмный край + сквозная дыра (discard);
   494→        // uAspect приводит UV к метрам — дыры круглые
   495→        for (int i = 0; i < 8; i++) {
   496→          if (float(i) >= uHoleN) break;
   497→          vec2 hd = (vUv - uHoles[i].xy) * vec2(uAspect, 1.0);
   498→          float hdist = length(hd);
   499→          float n = vnoise(vUv * 17.0 + float(i) * 7.31);
   500→          float rr = uHoles[i].z * (0.78 + 0.5 * n);
   501→          if (hdist < rr) {
   502→            col *= 0.42;
   503→            if (hdist < rr * 0.58) discard;   // рваный сквозной проём
   504→          }
   505→        }
   506→        // простое освещение
   507→        float ndl = max(dot(normalize(vN), normalize(uSunDir)), 0.0) * 0.5 + 0.55;
   508→        col *= ndl;
   509→        float a = 1.0 - clamp(holes, 0.0, 0.9);
   510→        if (a < 0.55) discard; // дыры — без полупрозрачности (нет проблем сортировки)
   511→        gl_FragColor = vec4(col, 1.0);
   512→      }
   513→    `,
   514→    side: THREE.DoubleSide,
   515→  });
   516→}

   573→// ---------- корпус по шпангоутам ----------
   574→export function hullProfile(t: number, P: ShipPreset) {
   575→  // t: 0 корма -> 1 нос. Полуширина, осадка, высота фальшборта
   576→  const B = P.beam / 2;
   577→  let w: number;
   578→  if (t < 0.45) w = THREE.MathUtils.lerp(0.58, 1.0, smooth01(t / 0.45));
   579→  else w = THREE.MathUtils.lerp(1.0, 0.0, Math.pow(smooth01((t - 0.45) / 0.55), 0.75));
   580→  // пинасс: сужение только в самом конце — почти вертикальный плоский нос
   581→  if (P.flatBow && t > 0.82) w = Math.max(w, THREE.MathUtils.lerp(0.74, 0.42, smooth01((t - 0.82) / 0.18)));
   582→  const draft = P.draft * (0.72 + 0.28 * Math.sin(Math.PI * Math.min(1, t * 1.15)));
   583→  let bulwark: number;
   584→  if (P.beakhead) {
   585→    // галеон: высокая корма, низкий бак — характерный силуэт с гальюном
   586→    bulwark = P.freeboard * (0.85 + 0.36 * Math.pow(1 - t, 2.4) + 0.08 * Math.pow(Math.max(0, t - 0.7) / 0.3, 2));
   587→  } else {
   588→    bulwark = P.freeboard * (0.82 + 0.35 * Math.pow(Math.max(0, t - 0.75) / 0.25, 1.4) + 0.5 * Math.pow(1 - t, 2.2) * 0.6);
   589→  }
   590→  return { w: w * B, draft, bulwark };
   591→}
   592→function smooth01(x: number) { const t = Math.min(1, Math.max(0, x)); return t * t * (3 - 2 * t); }

   594→// ---------- раскладка орудий по палубам ----------
   595→/** Слот орудия: z — вдоль корпуса (м, локально), tier 0 = открытая
   596→ *  (верхнепалубная) батарея с лафетами, 1.. — закрытые орудийные деки,
   597→ *  чьи стволы смотрят наружу из пушечных портов. */
   598→export interface GunSlot { z: number; tier: number; y: number; }
   599→
   600→/** Раскладка ВСЕХ орудий борта (число всегда = gunsPerSide — отрисовка
   601→ *  соответствует количеству). Без gunportRows всё на палубе (шлюп/флейт);
   602→ *  1 ряд портов — палуба + одна закрытая батарея; 2–4 ряда — палуба +
   603→ *  (rows−1) деков, порт-ряды идут сверху вниз (0.62F … 0.28F). */
   604→export function gunLayout(P: ShipPreset): GunSlot[] {
   605→  const N = Math.max(0, Math.round(P.gunsPerSide));
   606→  if (N === 0) return [];
   607→  const L = P.length;
   608→  const rows = Math.max(0, Math.min(4, P.gunportRows ?? 0));
   609→  const tiers = rows === 0 ? 1 : Math.max(2, rows);
   610→  // распределение стволов по ярусам: поровну, остаток — верхним ярусам
   611→  const counts = new Array<number>(tiers).fill(0);
   612→  for (let i = 0; i < N; i++) counts[i % tiers]++;
   613→  const slots: GunSlot[] = [];
   614→  for (let tier = 0; tier < tiers; tier++) {
   615→    const n = counts[tier];
   616→    if (n <= 0) continue;
   617→    if (tier === 0) {
   618→      // открытая палуба: от кормы (не задевая надстройки) до полубака
   619→      const z0 = -L * 0.22, z1 = P.castleH ? L * 0.18 : L * 0.24;
   620→      for (let i = 0; i < n; i++) {
   621→        const z = THREE.MathUtils.lerp(z0, z1, n === 1 ? 0.5 : i / (n - 1));
   622→        slots.push({ z, tier: 0, y: 0 }); // y палубы считает потребитель
   623→      }
   624→    } else {
   625→      // закрытая батарея: высота порт-ряда (tier−1), чуть уже по длине —
   626→      // корпус в оконечностях сужается, порты там не ставили
   627→      const y = P.freeboard * (rows === 1 ? 0.5 : 0.62 - (tier - 1) * (0.34 / (rows - 1)));
   628→      const z0 = -L * 0.24, z1 = L * 0.18;
   629→      for (let i = 0; i < n; i++) {
   630→        const z = THREE.MathUtils.lerp(z0, z1, n === 1 ? 0.5 : i / (n - 1));
   631→        slots.push({ z, tier, y });
   632→      }
   633→    }
   634→  }
   635→  return slots;
   636→}

   638→function buildHull(P: ShipPreset): { geometry: THREE.BufferGeometry; outline: THREE.Vector2[] } {
   639→  const S = 26, R = 8;
   640→  const L = P.length;
   641→  const pos: number[] = [];
   642→  const col: number[] = [];
   643→  const cHull = new THREE.Color(P.colors.hull);
   644→  const cBelow = new THREE.Color(0x33402a); // подводная часть — обрастание
   645→  const cBoot = new THREE.Color(0xd8d2c2); // пояс ватерлинии
   646→  const cWale = new THREE.Color(P.colors.trim);
   647→  const tmp = new THREE.Color();
   648→  // цветные полосы борта (доли свободного борта) — «Victory stripe»
   649→  const F = P.freeboard;
   650→  const stripes = (P.stripes ?? []).map(st => ({
   651→    y0: st.y0 * F, y1: st.y1 * F, c: new THREE.Color(st.color),
   652→  }));
   653→  // высотные зоны МАСШТАБИРУЮТСЯ под высоту борта (у линейных кораблей
   654→  // ватерлинионный пояс и вельс не должны лепиться у самой палубы)
   655→  const bootTop = Math.max(0.42, F * 0.13);        // верх ватерлинионного пояса
   656→  const waleY0 = Math.max(0.9, F * 0.22);          // вельс (дополнительный пояс прочности)
   657→  const waleY1 = waleY0 + Math.max(0.45, F * 0.09);
   658→
   659→  const stations: { z: number; w: number; draft: number; bulwark: number }[] = [];
   660→  for (let s = 0; s <= S; s++) {
   661→    const t = s / S;
   662→    const p = hullProfile(t, P);
   663→    stations.push({ z: THREE.MathUtils.lerp(-L / 2, L / 2, t), w: p.w, draft: p.draft, bulwark: p.bulwark });
   664→  }
   665→
   666→  const pushPoint = (x: number, y: number, z: number) => {
   667→    pos.push(x, y, z);
   668→    // раскраска по высоте: подводная часть -> ватерлинионный пояс ->
   669→    // (полосы пресета) -> вельс -> основной цвет борта
   670→    let stripeC: THREE.Color | null = null;
   671→    if (y > 0.1) {
   672→      for (const st of stripes) {
   673→        if (y >= st.y0 && y <= st.y1) { stripeC = st.c; break; }
   674→      }
   675→    }
   676→    if (y < -0.15) tmp.copy(cHull).lerp(cBelow, Math.min(1, (-y + 0.15) / P.draft));
   677→    else if (y < bootTop) tmp.copy(cBoot);
   678→    else if (stripeC) tmp.copy(stripeC);
   679→    else if (y > waleY0 && y < waleY1) tmp.copy(cWale);
   680→    else tmp.copy(cHull);
   681→    // лёгкая полосатость обшивки
   682→    const stripe = 1.0 - 0.06 * (Math.sin(y * 9.0) * 0.5 + 0.5);
   683→    col.push(tmp.r * stripe, tmp.g * stripe, tmp.b * stripe);
   684→  };
   685→
   686→  for (const st of stations) {
   687→    for (let r = 0; r <= R; r++) {
   688→      const u = r / R;
   689→      const y = THREE.MathUtils.lerp(-st.draft, st.bulwark, u);
   690→      const flare = Math.pow(u, 0.6);
   691→      const halfW = st.w * (0.16 + 0.84 * flare);
   692→      pushPoint(halfW, y, st.z);
   693→      pushPoint(-halfW, y, st.z);
   694→    }
   695→  }
   696→  const idx: number[] = [];
   697→  const rowLen = (R + 1) * 2;
   698→  for (let s = 0; s < S; s++) {
   699→    for (let r = 0; r < R; r++) {
   700→      const a = s * rowLen + r * 2, b = a + 1, c = a + 2, d = a + 3;
   701→      const e = a + rowLen, f = e + 1, g = e + 2, h = e + 3;
   702→      idx.push(a, c, e, c, g, e);
   703→      // зеркальная сторона (обратный порядок): два РАЗНЫХ треугольника квада —
   704→      // раньше (b,f,d) пушлся дважды (e+1===f!), второй (f,h,d) отсутствовал →
   705→      // левый борт был решетом из чередующихся треугольников
   706→      idx.push(b, f, d, f, h, d);
   707→    }
   708→  }
   709→  // транцевая корма: веер от центра к ПОСЛЕДОВАТЕЛЬНЫМ граничным вершинам
   710→  // (правые снизу вверх, затем левые сверху вниз) — без дыр между хордами.
   711→  // ВАЖНО: контур ЗАМКНУТ последним треугольником (левая-верх → правая-верх):
   712→  // раньше верхний клин транца не закрывался — сквозь корму «просвечивало»
   713→  const st0 = stations[0];
   714→  const centerIdx = pos.length / 3;
   715→  pushPoint(0, -st0.draft * 0.55, st0.z - 0.01);
   716→  const boundary: number[] = [];
   717→  for (let r = R; r >= 0; r--) boundary.push(r * 2);      // правые: верх → низ
   718→  for (let r = 0; r <= R; r++) boundary.push(r * 2 + 1);  // левые: низ → верх
   719→  boundary.push(boundary[0]);                             // замыкание контура
   720→  for (let i = 0; i < boundary.length - 1; i++) {
   721→    idx.push(centerIdx, boundary[i], boundary[i + 1]); // нормаль наружу (−Z)
   722→  }
   723→  // днище: замыкаем корпус снизу (r=0 — нижняя кромка шпангоутов),
   724→  // нормали вниз — при крене/под водой не «просвечивает»
   725→  for (let s = 0; s < S; s++) {
   726→    const a = s * rowLen + 1;        // левый борт, низ
   727→    const b = s * rowLen;            // правый борт, низ
   728→    const c = (s + 1) * rowLen + 1;
   729→    const d = (s + 1) * rowLen;
   730→    idx.push(a, b, c, b, d, c);
   731→  }
   732→  const g = new THREE.BufferGeometry();
   733→  g.setAttribute('position', new THREE.Float32BufferAttribute(pos, 3));
   734→  g.setAttribute('color', new THREE.Float32BufferAttribute(col, 3));
   735→  g.setIndex(idx);
   736→  g.computeVertexNormals();
   737→
   738→  // контур ватерлинии
   739→  const outline: THREE.Vector2[] = [];
   740→  for (let s = 0; s <= S; s += 2) {
   741→    const st = stations[s];
   742→    const u = (st.draft) / (st.draft + st.bulwark); // u при y=0
   743→    const flare = Math.pow(u, 0.6);
   744→    const halfW = Math.max(0.25, st.w * (0.16 + 0.84 * flare));
   745→    outline.push(new THREE.Vector2(halfW, st.z));
   746→  }
   747→  for (let s = S; s >= 0; s -= 2) {
   748→    const st = stations[s];
   749→    const u = st.draft / (st.draft + st.bulwark);
   750→    const flare = Math.pow(u, 0.6);
   751→    const halfW = Math.max(0.25, st.w * (0.16 + 0.84 * flare));
   752→    outline.push(new THREE.Vector2(-halfW, st.z));
   753→  }
   754→  return { geometry: g, outline };
   755→}

   757→function buildDeck(P: ShipPreset): THREE.BufferGeometry {
   758→  // палуба: полоса по длине чуть ниже фальшборта, вплотную к обшивке.
   759→  // ВАЖНО: порядок вершин (a,b,c)(b,d,c) даёт нормали ВВЕРХ —
   760→  // иначе палуба отсекается backface-куллингом и «просвечивает».
   761→  // UV: u — поперёк (0..1 = полная ширина), v — вдоль (тайл = ширина),
   762→  // доски идут вдоль корпуса, у краёв — тёмный водосток (в текстуре).
   763→  const S = 20, L = P.length;
   764→  const pos: number[] = [], uv: number[] = [], idx: number[] = [];
   765→  for (let s = 0; s <= S; s++) {
   766→    const t = s / S;
   767→    const p = hullProfile(t, P);
   768→    const z = THREE.MathUtils.lerp(-L / 2, L / 2, t);
   769→    const deckY = p.bulwark - 1.0;
   770→    // полуширина корпуса на высоте палубы — та же формула, что в buildHull,
   771→    // чтобы палуба доходила до борта без щели
   772→    const u = Math.min(1, Math.max(0, (deckY + p.draft) / (p.draft + p.bulwark)));
   773→    const flare = Math.pow(u, 0.6);
   774→    const w = Math.max(0.15, p.w * (0.16 + 0.84 * flare) - 0.05);
   775→    pos.push(w, deckY, z, -w, deckY, z);
   776→    const vv = z / P.beam;
   777→    uv.push(0.5 + w / P.beam, vv, 0.5 - w / P.beam, vv);
   778→  }
   779→  for (let s = 0; s < S; s++) {
   780→    const a = s * 2, b = a + 1, c = a + 2, d = a + 3;
   781→    idx.push(a, b, c, b, d, c);
   782→  }
   783→  const g = new THREE.BufferGeometry();
   784→  g.setAttribute('position', new THREE.Float32BufferAttribute(pos, 3));
   785→  g.setAttribute('uv', new THREE.Float32BufferAttribute(uv, 2));
   786→  g.setIndex(idx);
   787→  g.computeVertexNormals();
   788→  return g;
   789→}

   791→/** Текстура палубных досок: конопаченые швы, разбежка стыков, гвозди,
   792→ *  волокна, потёртости; тёмные водосточные планки по краям (u≈0 и u≈1) */
   793→function makeDeckTexture(baseColor: number, seed: number): THREE.Texture {
   794→  const S = 512;
   795→  const c = document.createElement('canvas');
   796→  c.width = S; c.height = S;
   797→  const x = c.getContext('2d')!;
   798→  const base = new THREE.Color(baseColor);
   799→  let rs = seed * 2654435761 % 2147483647;
   800→  const rnd = () => { rs = (rs * 16807) % 2147483647; return rs / 2147483647; };
   801→  const N = 30, pw = S / N;
   802→  for (let i = 0; i < N; i++) {
   803→    const v = 0.84 + rnd() * 0.26;
   804→    x.fillStyle = `rgb(${Math.round(base.r * 255 * v)},${Math.round(base.g * 255 * v)},${Math.round(base.b * 255 * v)})`;
   805→    x.fillRect(i * pw, 0, pw + 1, S);
   806→    // волокна дерева
   807→    x.strokeStyle = 'rgba(30,18,8,0.10)';
   808→    x.lineWidth = 1;
   809→    for (let k = 0; k < 3; k++) {
   810→      const gx = i * pw + pw * (0.2 + 0.6 * rnd());
   811→      x.beginPath();
   812→      x.moveTo(gx, 0);
   813→      for (let y = 0; y <= S; y += 32) x.lineTo(gx + Math.sin(y * 0.045 + i * 7 + k * 3) * 2.2, y);
   814→      x.stroke();
   815→    }
   816→    // стыки досок с разбежкой + гвозди
   817→    const b1 = (i * 137 + 61) % S, b2 = (b1 + Math.round(S * 0.55)) % S;
   818→    for (const by of [b1, b2]) {
   819→      x.fillStyle = 'rgba(22,14,7,0.6)';
   820→      x.fillRect(i * pw, by, pw + 1, 2);
   821→      x.fillStyle = 'rgba(32,32,36,0.85)';
   822→      x.beginPath(); x.arc(i * pw + pw * 0.3, by + 5, 1.4, 0, 7); x.fill();
   823→      x.beginPath(); x.arc(i * pw + pw * 0.7, by - 3, 1.4, 0, 7); x.fill();
   824→    }
   825→    // конопаченый шов между досками
   826→    x.fillStyle = 'rgba(20,13,7,0.72)';
   827→    x.fillRect(i * pw, 0, 1.6, S);
   828→  }
   829→  // водосточный жёлоб по краям палубы (тёмные продольные планки)
   830→  for (const [x0, wW] of [[0, S * 0.075], [S * 0.925, S * 0.075]] as [number, number][]) {
   831→    const gr = x.createLinearGradient(x0, 0, x0 + wW, 0);
   832→    gr.addColorStop(0, 'rgba(24,15,7,0.5)');
   833→    gr.addColorStop(1, 'rgba(24,15,7,0.12)');
   834→    x.fillStyle = gr;
   835→    x.fillRect(x0, 0, wW, S);
   836→  }
   837→  // потёртости и разводы
   838→  for (let k = 0; k < 26; k++) {
   839→    const rx = rnd() * S, ry = rnd() * S, rr = 18 + rnd() * 55;
   840→    const rg = x.createRadialGradient(rx, ry, 0, rx, ry, rr);
   841→    rg.addColorStop(0, 'rgba(28,18,8,0.09)');
   842→    rg.addColorStop(1, 'rgba(0,0,0,0)');
   843→    x.fillStyle = rg;
   844→    x.fillRect(rx - rr, ry - rr, rr * 2, rr * 2);
   845→  }
   846→  const t = new THREE.CanvasTexture(c);
   847→  t.wrapS = t.wrapT = THREE.RepeatWrapping;
   848→  t.colorSpace = THREE.SRGBColorSpace;
   849→  t.anisotropy = 4;
   850→  return t;
   851→}

   853→/** Планширь: планка, закрывающая верхнюю кромку фальшборта */
   854→function buildCapRail(P: ShipPreset): THREE.BufferGeometry {
   855→  const S = 26, L = P.length;
   856→  const pos: number[] = [], col: number[] = [], idx: number[] = [];
   857→  const cRail = new THREE.Color(P.colors.trim).multiplyScalar(1.25);
   858→  const cRail2 = new THREE.Color(P.colors.trim);
   859→  const push = (px: number, py: number, z: number, c: THREE.Color) => {
   860→    pos.push(px, py, z); col.push(c.r, c.g, c.b);
   861→  };
   862→  for (let s = 0; s <= S; s++) {
   863→    const t = s / S;
   864→    const p = hullProfile(t, P);
   865→    const z = THREE.MathUtils.lerp(-L / 2, L / 2, t);
   866→    const y = p.bulwark;
   867→    // внутренняя и внешняя кромки планки (лёгкий свес наружу)
   868→    push(p.w - 0.26, y + 0.07, z, cRail);
   869→    push(p.w + 0.13, y + 0.015, z, cRail2);
   870→    push(-p.w + 0.26, y + 0.07, z, cRail);
   871→    push(-p.w - 0.13, y + 0.015, z, cRail2);
   872→  }
   873→  for (let s = 0; s < S; s++) {
   874→    const a = s * 4;
   875→    // правый борт
   876→    idx.push(a, a + 4, a + 1, a + 1, a + 4, a + 5);
   877→    // левый борт
   878→    idx.push(a + 2, a + 3, a + 6, a + 3, a + 7, a + 6);
   879→  }
   880→  const g = new THREE.BufferGeometry();
   881→  g.setAttribute('position', new THREE.Float32BufferAttribute(pos, 3));
   882→  g.setAttribute('color', new THREE.Float32BufferAttribute(col, 3));
   883→  g.setIndex(idx);
   884→  g.computeVertexNormals();
   885→  return g;
   886→}

   888→/** Запечь геометрию в общие массивы (для слияния реквизита в один меш) */
   889→function bakeInto(target: { pos: number[]; col: number[] }, geo: THREE.BufferGeometry,
   890→  color: THREE.Color, x: number, y: number, z: number, euler?: THREE.Euler) {
   891→  const g = geo.index ? geo.toNonIndexed() : geo;
   892→  const p = g.attributes.position as THREE.BufferAttribute;
   893→  const v = new THREE.Vector3();
   894→  for (let i = 0; i < p.count; i++) {
   895→    v.fromBufferAttribute(p, i);
   896→    if (euler) v.applyEuler(euler);
   897→    target.pos.push(v.x + x, v.y + y, v.z + z);
   898→    target.col.push(color.r, color.g, color.b);
   899→  }
   900→}

   902→/** Палубный реквизит: решётчатый люк, бочки, канаты, кнехты, колокол,
   903→ *  ядра, сундук, ведро — всё сливается в ОДИН меш с vertex colors */
   904→function buildDeckProps(P: ShipPreset): THREE.BufferGeometry {
   905→  const L = P.length;
   906→  const t: { pos: number[]; col: number[] } = { pos: [], col: [] };
   907→  const deckYAt = (tt: number) => hullProfile(tt, P).bulwark - 1.0;
   908→  const cWood = new THREE.Color(0x7a5c38);
   909→  const cWoodD = new THREE.Color(0x4a3420);
   910→  const cDark = new THREE.Color(P.colors.trim);
   911→  const cIron = new THREE.Color(0x24262b);
   912→  const cBrass = new THREE.Color(0xb8912f);
   913→  const cRope = new THREE.Color(0xa08656);
   914→  const cBarrel = new THREE.Color(0x8a6a42);
   915→
   916→  // --- решётчатый люк (гратинг) с коомингом ---
   917→  const hx = L * 0.08, hz = L * 0.05; // пол-размера (между мачтами)
   918→  const hy = deckYAt(0.48);
   919→  // кооминг (рамка)
   920→  bakeInto(t, new THREE.BoxGeometry(hx * 2 + 0.24, 0.22, 0.12), cWoodD, 0, hy + 0.11, hz);
   921→  bakeInto(t, new THREE.BoxGeometry(hx * 2 + 0.24, 0.22, 0.12), cWoodD, 0, hy + 0.11, -hz);
   922→  bakeInto(t, new THREE.BoxGeometry(0.12, 0.22, hz * 2), cWoodD, hx, hy + 0.11, 0);
   923→  bakeInto(t, new THREE.BoxGeometry(0.12, 0.22, hz * 2), cWoodD, -hx, hy + 0.11, 0);
   924→  // продольные планки
   925→  const nL = 9;
   926→  for (let i = 0; i < nL; i++) {
   927→    const px = -hx + 0.12 + (i / (nL - 1)) * (hx * 2 - 0.24);
   928→    bakeInto(t, new THREE.BoxGeometry(0.09, 0.07, hz * 2 - 0.1), cWood, px, hy + 0.24, 0);
   929→  }
   930→  // поперечные планки
   931→  const nT = 5;
   932→  for (let i = 0; i < nT; i++) {
   933→    const pz = -hz + 0.1 + (i / (nT - 1)) * (hz * 2 - 0.2);
   934→    bakeInto(t, new THREE.BoxGeometry(hx * 2 - 0.16, 0.07, 0.09), cWood, 0, hy + 0.30, pz);
   935→  }
   936→
   937→  // --- бочки (2 стоя, 1 лёжа) ---
   938→  const barrelG = new THREE.CylinderGeometry(0.34, 0.34, 0.8, 10);
   939→  const hoopG = new THREE.TorusGeometry(0.345, 0.022, 5, 12);
   940→  const hoopE = new THREE.Euler(Math.PI / 2, 0, 0);
   941→  const mkBarrel = (x: number, z: number, lying = false) => {
   942→    const y = deckYAt(0.5 + z / L) + (lying ? 0.34 : 0.4);
   943→    const e = lying ? new THREE.Euler(0, lying ? 0.7 : 0, Math.PI / 2) : undefined;
   944→    bakeInto(t, barrelG, cBarrel, x, y, z, e);
   945→    if (lying) {
   946→      bakeInto(t, hoopG, cIron, x, y - 0.25, z, e);
   947→      bakeInto(t, hoopG, cIron, x, y + 0.25, z, e);
   948→    } else {
   949→      bakeInto(t, hoopG, cIron, x, y - 0.28, z, hoopE);
   950→      bakeInto(t, hoopG, cIron, x, y + 0.28, z, hoopE);
   951→    }
   952→  };
   953→  mkBarrel(P.beam * 0.24, L * 0.10);
   954→  mkBarrel(-P.beam * 0.26, L * 0.04);
   955→  mkBarrel(P.beam * 0.22, -L * 0.12, true);
   956→  // --- бухты канатов ---
   957→  const coilG = new THREE.TorusGeometry(0.30, 0.10, 6, 14);
   958→  const coilE = new THREE.Euler(Math.PI / 2, 0, 0);
   959→  bakeInto(t, coilG, cRope, P.beam * 0.20, deckYAt(0.75) + 0.10, P.castleH ? L * 0.2 : L * 0.30, coilE);
   960→  bakeInto(t, coilG, cRope, -P.beam * 0.18, deckYAt(0.62) + 0.10, L * 0.18, coilE);
   961→  bakeInto(t, coilG, cRope, P.beam * 0.34, deckYAt(0.36) + 0.10, -L * 0.15, coilE);
   962→
   963→  // --- кнехты (битенги) с поперечиной у носа (на кораблях с «замками» нос занят надстройкой) ---
   964→  const bittG = new THREE.BoxGeometry(0.15, 0.6, 0.15);
   965→  if (!P.castleH) for (const side of [-1, 1]) {
   966→    bakeInto(t, bittG, cWoodD, side * 0.42, deckYAt(0.85) + 0.3, L * 0.36);
   967→    bakeInto(t, bittG, cWoodD, side * 0.42, deckYAt(0.85) + 0.3, L * 0.36 - 0.5);
   968→    bakeInto(t, new THREE.BoxGeometry(0.13, 0.13, 0.75), cWoodD, side * 0.42, deckYAt(0.85) + 0.55, L * 0.36 - 0.25);
   969→  }
   970→
   971→  // --- корабельный колокол у штурвала ---
   972→  const bellZ = THREE.MathUtils.lerp(-L / 2, L / 2, 0.2) + 1.6;
   973→  bakeInto(t, new THREE.BoxGeometry(0.12, 0.8, 0.12), cWoodD, 0.8, deckYAt(0.22) + 0.4, bellZ);
   974→  bakeInto(t, new THREE.CylinderGeometry(0.05, 0.13, 0.2, 8), cBrass, 0.8, deckYAt(0.22) + 0.85, bellZ);
   975→
   976→  // --- пирамиды ядер у мачты ---
   977→  const ballG = new THREE.SphereGeometry(0.105, 7, 6);
   978→  const pile = (x: number, z: number) => {
   979→    const y0 = deckYAt(0.5 + z / L);
   980→    for (const [dx, dz] of [[-0.115, 0], [0, 0], [0.115, 0], [-0.058, 0.1], [0.058, 0.1], [0, 0.05]]) {
   981→      bakeInto(t, ballG, cIron, x + dx, y0 + 0.105, z + dz);
   982→    }
   983→  };
   984→  pile(0.85, -L * 0.16);
   985→  pile(-P.beam * 0.20, L * 0.09);
   986→
   987→  // --- сундук с инструментом ---
   988→  bakeInto(t, new THREE.BoxGeometry(0.72, 0.42, 0.5), cWoodD, -P.beam * 0.26, deckYAt(0.42) + 0.21, -L * 0.2);
   989→  bakeInto(t, new THREE.BoxGeometry(0.78, 0.1, 0.56), cDark, -P.beam * 0.26, deckYAt(0.42) + 0.46, -L * 0.2);
   990→
   991→  // --- ведро ---
   992→  bakeInto(t, new THREE.CylinderGeometry(0.17, 0.13, 0.3, 8, 1, true), cWood, -P.beam * 0.2, deckYAt(0.72) + 0.15, L * 0.24);
   993→
   994→  const g = new THREE.BufferGeometry();
   995→  g.setAttribute('position', new THREE.Float32BufferAttribute(t.pos, 3));
   996→  g.setAttribute('color', new THREE.Float32BufferAttribute(t.col, 3));
   997→  g.computeVertexNormals();
   998→  return g;
   999→}

  1001→/** Полуширина корпуса на высоте y в шпации t (та же формула развала бортов, что в buildHull) */
  1002→function hullHalfWidthAt(P: ShipPreset, t: number, y: number): number {
  1003→  const p = hullProfile(t, P);
  1004→  const u = Math.min(1, Math.max(0, (y + p.draft) / (p.draft + p.bulwark)));
  1005→  const flare = Math.pow(u, 0.6);
  1006→  return p.w * (0.16 + 0.84 * flare);
  1007→}

  1018→/** Надстройки-«замки» когга: массивые кормовый и носовой — ганзейский силуэт */
  1019→function buildCastles(P: ShipPreset): THREE.BufferGeometry {
  1020→  const L = P.length;
  1021→  const t = { pos: [] as number[], col: [] as number[] };
  1022→  const cWall = new THREE.Color(P.colors.trim).multiplyScalar(1.15);
  1023→  const cWoodD = new THREE.Color(0x4a3420);
  1024→  const cRail = new THREE.Color(P.colors.trim).multiplyScalar(1.35);
  1025→  const ch = P.castleH || 0;
  1026→
  1027→  const mkCastle = (t0: number, t1: number, extra: number) => {
  1028→    const zA = THREE.MathUtils.lerp(-L / 2, L / 2, t0);
  1029→    const zB = THREE.MathUtils.lerp(-L / 2, L / 2, t1);
  1030→    const zc = (zA + zB) / 2, d = Math.abs(zB - zA);
  1031→    const prof = hullProfile((t0 + t1) / 2, P);
  1032→    const w = Math.max(1.2, prof.w * 0.92);
  1033→    const deckY = prof.bulwark - 1.0;
  1034→    const topY = deckY + extra;
  1035→    // массивная надстройка от палубы до площадки
  1036→    bakeInto(t, new THREE.BoxGeometry(w * 2, topY - deckY, d), cWall, 0, (deckY + topY) / 2, zc);
  1037→    // площадка чуть шире стен
  1038→    bakeInto(t, new THREE.BoxGeometry(w * 2 + 0.3, 0.22, d + 0.35), cWoodD, 0, topY + 0.11, zc);
  1039→    // стойки ограждения + продольный поручень
  1040→    for (const side of [-1, 1]) {
  1041→      for (let k = 0; k <= 4; k++) {
  1042→        const zz = zA + (k / 4) * (zB - zA);
  1043→        bakeInto(t, new THREE.BoxGeometry(0.12, 0.55, 0.12), cRail, side * (w + 0.05), topY + 0.5, zz);
  1044→      }
  1045→      bakeInto(t, new THREE.BoxGeometry(0.08, 0.08, d), cRail, side * (w + 0.05), topY + 0.76, zc);
  1046→    }
  1047→  };
  1048→  mkCastle(0.08, 0.24, ch);          // кормовый (ахтеркастель) — выше
  1049→  mkCastle(0.74, 0.88, ch * 0.72);   // носовой (бак) — ниже
  1050→  return bakedGeo(t);
  1051→}

  1053→/** Наружный декор корпуса: пушечные порты с вельсовыми полосами
  1054→ *  (галеон/фрегат), кормовая галерея с окнами, плоская носовая
  1055→ *  переборка пинасса, гальюн галеона */
  1056→function buildHullOrnaments(P: ShipPreset): THREE.BufferGeometry | null {
  1057→  const L = P.length;
  1058→  const t = { pos: [] as number[], col: [] as number[] };
  1059→  const cPort = new THREE.Color(0x17100d);
  1060→  const cGold = new THREE.Color(P.colors.trim).lerp(new THREE.Color(0xd8b45a), 0.55);
  1061→  const cGlass = new THREE.Color(0x1d2a33);
  1062→  const cWoodD = new THREE.Color(0x4a3420);
  1063→  let any = false;
  1064→
  1065→  // пушечные порты: РОВНО под стволами закрытых батарей (gunLayout) —
  1066→  // число портов соответствует числу орудий деку + золочёный вельс над батареей
  1067→  const rows = P.gunportRows || 0;
  1068→  if (rows > 0) {
  1069→    const layout = gunLayout(P).filter(s => s.tier > 0);
  1070→    const decks = [...new Set(layout.map(s => s.tier))];
  1071→    for (const tier of decks) {
  1072→      const y = layout.find(s => s.tier === tier)!.y;
  1073→      for (const side of [-1, 1]) {
  1074→        for (const slot of layout.filter(s => s.tier === tier)) {
  1075→          const tt = clamp2((slot.z + L / 2) / L, 0, 1);
  1076→          if (tt < 0.12 || tt > 0.86) continue;
  1077→          const hw = hullHalfWidthAt(P, tt, y);
  1078→          bakeInto(t, new THREE.BoxGeometry(0.06, Math.min(0.5, P.freeboard * 0.09), 0.62), cPort, side * (hw + 0.02), y, slot.z);
  1079→        }
  1080→        const S2 = 26;
  1081→        for (let s = 0; s < S2; s++) {
  1082→          const zA = THREE.MathUtils.lerp(-L * 0.32, L * 0.32, s / S2);
  1083→          const zB = THREE.MathUtils.lerp(-L * 0.32, L * 0.32, (s + 1) / S2);
  1084→          const tt = ((zA + zB) / 2 + L / 2) / L;
  1085→          if (tt < 0.1 || tt > 0.88) continue;
  1086→          const hw = hullHalfWidthAt(P, tt, y + 0.42);
  1087→          bakeInto(t, new THREE.BoxGeometry(0.05, 0.1, (zB - zA) * 1.05), cGold,
  1088→            side * (hw + 0.015), y + 0.42, (zA + zB) / 2);
  1089→        }
  1090→      }
  1091→    }
  1092→    any = true;
  1093→  }
  1094→
  1095→  // кормовая галерея: два яруса окон с золочёными рамами
  1096→  if (P.gallery) {
  1097→    for (const y of [P.freeboard * 0.3, P.freeboard * 0.64]) {
  1098→      for (const xoff of [-P.beam * 0.24, 0, P.beam * 0.24]) {
  1099→        bakeInto(t, new THREE.BoxGeometry(0.62, 0.54, 0.1), cGold, xoff, y, -L / 2 + 0.02);
  1100→        bakeInto(t, new THREE.BoxGeometry(0.46, 0.38, 0.12), cGlass, xoff, y, -L / 2 + 0.03);
  1101→      }
  1102→    }
  1103→    any = true;
  1104→  }
  1105→
  1106→  // пинасс: почти плоская носовая переборка с планкирём
  1107→  if (P.flatBow) {
  1108→    const tt = 0.93;
  1109→    const prof = hullProfile(tt, P);
  1110→    const w = prof.w * 0.96;
  1111→    const z = THREE.MathUtils.lerp(-L / 2, L / 2, tt);
  1112→    bakeInto(t, new THREE.BoxGeometry(w * 2, prof.bulwark + P.draft * 0.5, 0.16), cWoodD,
  1113→      0, (prof.bulwark - P.draft * 0.5) / 2, z);
  1114→    bakeInto(t, new THREE.BoxGeometry(w * 2 + 0.14, 0.14, 0.2),
  1115→      new THREE.Color(P.colors.trim).multiplyScalar(1.3), 0, prof.bulwark + 0.05, z);
  1116→    any = true;
  1117→  }
  1118→
  1119→  // галеон: гальюн — решётчатая площадка от низкого бака к бушприту
  1120→  if (P.beakhead) {
  1121→    const y = hullProfile(0.92, P).bulwark * 0.55;
  1122→    const z0 = L / 2 - 1.2, z1 = L / 2 + L * 0.1;
  1123→    for (let k = 0; k < 3; k++) {
  1124→      bakeInto(t, new THREE.BoxGeometry(1.7 - k * 0.5, 0.09, 0.09), cWoodD, 0, y,
  1125→        THREE.MathUtils.lerp(z0, z1, k / 2));
  1126→    }
  1127→    bakeInto(t, new THREE.BoxGeometry(0.09, 0.09, z1 - z0), cWoodD, -0.55, y, (z0 + z1) / 2);
  1128→    bakeInto(t, new THREE.BoxGeometry(0.09, 0.09, z1 - z0), cWoodD, 0.55, y, (z0 + z1) / 2);
  1129→    any = true;
  1130→  }
  1131→
  1132→  if (!any) return null;
  1133→  return bakedGeo(t);
  1134→}

  1136→/** Мелкие детали, видимые только вблизи (LOD): выбленки на вантах,
  1137→ *  шлюпка вверх дном, настоящие якоря у скул, крышки пушечных портов,
  1138→ *  камбуз с трубой, кранцы, стол экипажа, блоки у мачт */
  1139→function buildShipDetail(P: ShipPreset, mastDefs: { z: number; h: number }[]): THREE.Group {
  1140→  const L = P.length;
  1141→  const group = new THREE.Group();
  1142→  const deckYAt = (tt: number) => hullProfile(tt, P).bulwark - 1.0;
  1143→  const cWood = new THREE.Color(0x7a5c38);
  1144→  const cWoodD = new THREE.Color(0x4a3420);
  1145→  const cIron = new THREE.Color(0x1e2024);
  1146→  const cRope = new THREE.Color(0xa08656);
  1147→
  1148→  // --- ванты с выбленками (верёвочная лестница) — линиями ---
  1149→  const pts: number[] = [];
  1150→  const addLine = (a: THREE.Vector3, b: THREE.Vector3) => { pts.push(a.x, a.y, a.z, b.x, b.y, b.z); };
  1151→  for (const m of mastDefs) {
  1152→    for (const side of [-1, 1]) {
  1153→      const N = 5;
  1154→      const tops: THREE.Vector3[] = [], bases: THREE.Vector3[] = [];
  1155→      for (let k = 0; k < N; k++) {
  1156→        tops.push(new THREE.Vector3(0, m.h * (0.55 + k * 0.09), m.z));
  1157→        bases.push(new THREE.Vector3(side * 2.6, 1.6, m.z - 2.0 + k * 1.0));
  1158→      }
  1159→      for (let k = 0; k < N; k++) addLine(tops[k], bases[k]); // плотный веер вант
  1160→      for (let k = 0; k < N - 1; k++) {
  1161→        for (let f = 0.15; f <= 0.78; f += 0.09) {
  1162→          const a = new THREE.Vector3().lerpVectors(bases[k], tops[k], f);
  1163→          const b = new THREE.Vector3().lerpVectors(bases[k + 1], tops[k + 1], f);
  1164→          addLine(a, b); // выбленки
  1165→        }
  1166→      }
  1167→    }
  1168→  }
  1169→
  1170→  const t = { pos: [] as number[], col: [] as number[] };
  1171→
  1172→  // --- шлюпка вверх дном у борта ---
  1173→  const boatG = new THREE.CapsuleGeometry(0.45, L * 0.14, 3, 8);
  1174→  boatG.rotateX(Math.PI / 2);
  1175→  boatG.scale(1, 0.5, 1);
  1176→  bakeInto(t, boatG, new THREE.Color(0x7d6446), P.beam * 0.16, deckYAt(0.63) + 0.22, L * 0.13);
  1177→  bakeInto(t, new THREE.BoxGeometry(0.08, 0.1, L * 0.3), cWoodD, P.beam * 0.16, deckYAt(0.63) + 0.44, L * 0.13);
  1178→
  1179→  // --- настоящие якоря на фалах у скул носа ---
  1180→  for (const side of [-1, 1]) {
  1181→    const z = L / 2 - L * 0.1;
  1182→    const prof = hullProfile(0.85, P);
  1183→    const ax = side * (prof.w + 0.3), ay = P.freeboard * 0.55;
  1184→    bakeInto(t, new THREE.BoxGeometry(0.1, 1.35, 0.1), cIron, ax, ay, z);          // веретено
  1185→    bakeInto(t, new THREE.BoxGeometry(0.9, 0.11, 0.11), cWoodD, ax, ay + 0.6, z);  // шток
  1186→    const armG = new THREE.TorusGeometry(0.4, 0.05, 5, 10, Math.PI);
  1187→    armG.rotateZ(Math.PI); // нижняя полуокружность — рога
  1188→    bakeInto(t, armG, cIron, ax, ay - 0.6, z);
  1189→    for (const s2 of [-1, 1]) { // лапы
  1190→      const fl = new THREE.ConeGeometry(0.15, 0.36, 4);
  1191→      fl.rotateZ(s2 * Math.PI / 2);
  1192→      bakeInto(t, fl, cIron, ax + s2 * 0.42, ay - 0.55, z);
  1193→    }
  1194→    const ring = new THREE.TorusGeometry(0.14, 0.03, 5, 10);
  1195→    bakeInto(t, ring, cIron, ax, ay + 0.75, z);                                    // рым
  1196→    addLine(new THREE.Vector3(side * prof.w * 0.7, prof.bulwark, z - 1.2),
  1197→      new THREE.Vector3(ax, ay + 0.75, z));                                        // фал
  1198→  }
  1199→
  1200→  // --- крышки пушечных портов (открытые, на верхних петлях) — только
  1201→  //     под ПАЛУБНЫМИ орудиями (закрытые деки стреляют из своих портов) ---
  1202→  for (const slot of gunLayout(P)) {
  1203→    if (slot.tier !== 0) continue;
  1204→    const tt = clamp2((slot.z + L / 2) / L, 0, 1);
  1205→    const y = hullProfile(tt, P).bulwark - 0.72;
  1206→    for (const side of [-1, 1]) {
  1207→      const hw = hullHalfWidthAt(P, tt, y);
  1208→      bakeInto(t, new THREE.BoxGeometry(0.06, 0.55, 0.6), cWoodD,
  1209→        side * (hw + 0.14), y + 0.3, slot.z, new THREE.Euler(0, 0, side * 0.55));
  1210→    }
  1211→  }
  1212→
  1213→  // --- камбуз (печь) с трубой ---
  1214→  bakeInto(t, new THREE.BoxGeometry(0.85, 0.75, 0.85), new THREE.Color(0x5c4034),
  1215→    -P.beam * 0.26, deckYAt(0.56) + 0.38, L * 0.06);
  1216→  bakeInto(t, new THREE.CylinderGeometry(0.09, 0.12, 1.5, 7), new THREE.Color(0x2b2d31),
  1217→    -P.beam * 0.26, deckYAt(0.56) + 1.35, L * 0.06);
  1218→
  1219→  // --- стол экипажа со скамьёй ---
  1220→  bakeInto(t, new THREE.BoxGeometry(1.5, 0.08, 0.7), cWood, P.beam * 0.12, deckYAt(0.4) + 0.62, -L * 0.1);
  1221→  for (const [lx, lz] of [[-0.65, -0.28], [0.65, -0.28], [-0.65, 0.28], [0.65, 0.28]] as [number, number][]) {
  1222→    bakeInto(t, new THREE.CylinderGeometry(0.045, 0.045, 0.62, 5), cWoodD,
  1223→      P.beam * 0.12 + lx, deckYAt(0.4) + 0.31, -L * 0.1 + lz);
  1224→  }
  1225→  bakeInto(t, new THREE.BoxGeometry(1.7, 0.09, 0.34), cWood, P.beam * 0.12, deckYAt(0.4) + 0.28, -L * 0.1 + 0.75);
  1226→
  1227→  // --- кранцы: канатные шары на фалинях у планширя ---
  1228→  for (let k = 0; k < 3; k++) {
  1229→    const z = L * (0.08 + k * 0.2);
  1230→    const prof = hullProfile(0.5 + z / L, P);
  1231→    for (const side of [-1, 1]) {
  1232→      const hx = side * (prof.w + 0.06), hy = prof.bulwark + 0.28;
  1233→      bakeInto(t, new THREE.SphereGeometry(0.13, 6, 5), cRope, hx, hy, z);
  1234→      addLine(new THREE.Vector3(side * prof.w * 0.92, prof.bulwark + 0.05, z),
  1235→        new THREE.Vector3(hx, hy, z));
  1236→    }
  1237→  }
  1238→
  1239→  // --- блоки у мачт ---
  1240→  for (const md of mastDefs) {
  1241→    bakeInto(t, new THREE.CylinderGeometry(0.09, 0.09, 0.14, 6), cWoodD, 0.35, deckYAt(0.5 + md.z / L) + 0.35, md.z - 0.6);
  1242→    bakeInto(t, new THREE.CylinderGeometry(0.09, 0.09, 0.14, 6), cWoodD, -0.35, deckYAt(0.5 + md.z / L) + 0.35, md.z - 0.6);
  1243→  }
  1244→
  1245→  const lg = new THREE.BufferGeometry();
  1246→  lg.setAttribute('position', new THREE.Float32BufferAttribute(pts, 3));
  1247→  const lines = new THREE.LineSegments(lg, new THREE.LineBasicMaterial({ color: 0x241c12, transparent: true, opacity: 0.9 }));
  1248→  group.add(lines);
  1249→  const mesh = new THREE.Mesh(bakedGeo(t),
  1250→    new THREE.MeshStandardMaterial({ vertexColors: true, roughness: 0.85, metalness: 0.2, side: THREE.DoubleSide }));
  1251→  group.add(mesh);
  1252→  return group;
  1253→}

  1255→/** Вёсла гребных судов (шебека, галера): вальные порты по борту +
  1256→ *  вёсла с лопастями, наклонённые к воде; гребки анимирует Ship.updateVisual */
  1257→function buildOars(P: ShipPreset): { group: THREE.Group; oars: THREE.Object3D[] } {
  1258→  const group = new THREE.Group();
  1259→  const oars: THREE.Object3D[] = [];
  1260→  const n = P.oars || 0;
  1261→  if (!n) return { group, oars };
  1262→  const L = P.length;
  1263→  const shaftMat = new THREE.MeshStandardMaterial({ color: 0x8a6a42, roughness: 0.85 });
  1264→  const portMat = new THREE.MeshStandardMaterial({ color: 0x14100d, roughness: 0.9 });
  1265→  const oarLen = clamp2(L * 0.17, 4.5, 11);
  1266→  const z0 = -L * 0.26, z1 = L * 0.30;
  1267→  for (let i = 0; i < n; i++) {
  1268→    const t = n === 1 ? 0.5 : i / (n - 1);
  1269→    const z = THREE.MathUtils.lerp(z0, z1, t);
  1270→    const tt = clamp2((z + L / 2) / L, 0, 1);
  1271→    const y = P.freeboard * 0.30; // вальная палуба — ниже орудийной
  1272→    const hw = hullHalfWidthAt(P, tt, y);
  1273→    for (const side of [-1, 1]) {
  1274→      // тёмный вальный порт
  1275→      const port = new THREE.Mesh(new THREE.BoxGeometry(0.06, 0.26, 0.3), portMat);
  1276→      port.position.set(side * (hw + 0.01), y, z);
  1277→      group.add(port);
  1278→      // весло: вал + лопасть, наклонены вниз-наружу к воде
  1279→      const oar = new THREE.Group();
  1280→      const shaft = new THREE.Mesh(new THREE.CylinderGeometry(0.05, 0.065, oarLen, 5), shaftMat);
  1281→      shaft.rotation.z = Math.PI / 2;
  1282→      shaft.position.x = oarLen * 0.42;
  1283→      oar.add(shaft);
  1284→      const blade = new THREE.Mesh(new THREE.BoxGeometry(oarLen * 0.3, 0.03, 0.2), shaftMat);
  1285→      blade.position.x = oarLen * 0.92;
  1286→      oar.add(blade);
  1287→      oar.position.set(side * (hw + 0.04), y, z);
  1288→      oar.rotation.z = side * -0.34;
  1289→      oar.userData.phase = t * Math.PI * 2 + (side > 0 ? Math.PI : 0);
  1290→      oar.userData.side = side;
  1291→      oar.userData.base = side * -0.34;
  1292→      group.add(oar);
  1293→      oars.push(oar);
  1294→    }
  1295→  }
  1296→  return { group, oars };
  1297→}
  1298→function clamp2(v: number, a: number, b: number) { return v < a ? a : v > b ? b : v; }

  1301→function buildRigging(P: ShipPreset, masts: { z: number; h: number }[], bowspritTip: THREE.Vector3): THREE.LineSegments {
  1302→  const pts: number[] = [];
  1303→  const L = P.length;
  1304→  const addLine = (a: THREE.Vector3, b: THREE.Vector3) => { pts.push(a.x, a.y, a.z, b.x, b.y, b.z); };
  1305→  for (const m of masts) {
  1306→    const top = new THREE.Vector3(0, m.h, m.z);
  1307→    // штаги к носу и корме
  1308→    addLine(top, bowspritTip);
  1309→    addLine(top, new THREE.Vector3(0, 1.2, -L / 2 + 0.4));
  1310→    // ванты к бортам
  1311→    for (const side of [-1, 1]) {
  1312→      for (let k = 0; k < 3; k++) {
  1313→        const zz = m.z - 1.5 + k * 1.5;
  1314→        addLine(new THREE.Vector3(side * 2.6, 1.6, zz), new THREE.Vector3(0, m.h * (0.55 + k * 0.15), m.z));
  1315→      }
  1316→    }
  1317→  }
  1318→  // бакштаги к корме от грот-мачты
  1319→  if (masts.length >= 2) {
  1320→    const main = masts[masts.length - 1 > 1 ? masts.length - 2 : 0];
  1321→    addLine(new THREE.Vector3(0, main.h * 0.8, main.z), new THREE.Vector3(2.4, 1.4, -L / 2 + 0.3));
  1322→    addLine(new THREE.Vector3(0, main.h * 0.8, main.z), new THREE.Vector3(-2.4, 1.4, -L / 2 + 0.3));
  1323→  }
  1324→  const g = new THREE.BufferGeometry();
  1325→  g.setAttribute('position', new THREE.Float32BufferAttribute(pts, 3));
  1326→  const mat = new THREE.LineBasicMaterial({ color: 0x241c12, transparent: true, opacity: 0.85 });
  1327→  return new THREE.LineSegments(g, mat);
  1328→}

  1331→export function buildShip(P: ShipPreset): ShipVisual {
  1332→  const group = new THREE.Group();
  1333→  const L = P.length;
  1334→
  1335→  // корпус
  1336→  const { geometry: hullGeo, outline } = buildHull(P);
  1337→  // DoubleSide: даже если камера (или волна) заглянет под корпус/за корму —
  1338→  // стенки видны, «прозрачной кормы» нет (изнутри односторонние грани куллятся)
  1339→  const hullMat = new THREE.MeshStandardMaterial({ vertexColors: true, roughness: 0.82, metalness: 0.05, side: THREE.DoubleSide });
  1340→  const hull = new THREE.Mesh(hullGeo, hullMat);
  1341→  hull.castShadow = true;
  1342→  group.add(hull);
  1343→
  1344→  // палуба: текстура досок с конопачением (DoubleSide — страховка от куллинга)
  1345→  const deckMat = new THREE.MeshStandardMaterial({
  1346→    map: makeDeckTexture(P.colors.deck, P.colors.deck + P.length),

  1354→    new THREE.MeshStandardMaterial({ vertexColors: true, roughness: 0.7, metalness: 0.05 }));

  1356→
  1357→  const woodMat = new THREE.MeshStandardMaterial({ color: 0x6e5433, roughness: 0.8 });
  1358→  const darkMat = new THREE.MeshStandardMaterial({ color: P.colors.trim, roughness: 0.75 });
  1359→
  1360→  // бушприт

  1366→

  1387→    top.position.y = md.h * 0.72 - P.draft * 0.3;

  1394→
  1395→  // паруса: каждый со своим материалом (своё рифление), квадратные —
  1396→  // на поворотной группе «рея+парус» для брасопки по ветру
  1397→  const sails: SailInfo[] = [];
  1398→  const addSquareSail = (mastIndex: number, level: number, width: number, height: number, setAt: number) => {
  1399→    const md = mastDefs[mastIndex];
  1400→    const g = new THREE.PlaneGeometry(width, height, 10, 7);
  1401→    g.translate(0, -height / 2, 0); // верхняя кромка у реи

  1405→    const y = md.h * level - P.draft * 0.3;
  1406→    // поворотная группа: ось вращения — мачта (брасопка рея по ветру)
  1407→    // рея+парус — потомок МАЧТЫ: сломанная мачта валится вместе с ними
  1408→    const pivot = new THREE.Group();
  1409→    pivot.position.set(0, y + 0.05, 0);
  1410→    // рея
  1411→    const yard = new THREE.Mesh(new THREE.CylinderGeometry(0.07, 0.07, width * 1.04, 5), woodMat);
  1412→    yard.rotation.z = Math.PI / 2;
  1413→    pivot.add(yard);
  1414→    mesh.position.set(0, 0, -0.25);
  1415→    pivot.add(mesh);
  1416→    masts[mastIndex].add(pivot);

  1418→    // при проверке поворачивается на текущий угол брасопки вокруг мачты
  1419→    const quad: SailHitQuad = {
  1420→      pivot: new THREE.Vector3(0, y + 0.05, md.z),
  1421→      corners: [
  1422→        new THREE.Vector3(-width / 2, -height, -0.25), // uv(0,0) — нижний-лево
  1423→        new THREE.Vector3(width / 2, -height, -0.25),  // uv(1,0)
  1424→        new THREE.Vector3(width / 2, 0, -0.25),        // uv(1,1)
  1425→        new THREE.Vector3(-width / 2, 0, -0.25),       // uv(0,1)
  1426→      ],
  1427→      uvs: [
  1428→        new THREE.Vector2(0, 0), new THREE.Vector2(1, 0),
  1429→        new THREE.Vector2(1, 1), new THREE.Vector2(0, 1),
  1430→      ],
  1431→      rotates: true,
  1432→    };
  1433→    sails.push({ mesh, mat, brace: pivot, setAt, furl: 1, square: true, braceAngle: 0, mastIndex, quad, holeW: width, holeH: height });
  1434→  };
  1435→  // латинский парус (каравелла): длинная рейка от мачты к носу,
  1436→  // треугольное полотно в диаметральной плоскости, «пузо» — на борт
  1437→  const addLateenSail = (mastIndex: number, setAt: number, yLen: number, drop: number, aftFrac: number) => {
  1438→    const md = mastDefs[mastIndex];
  1439→    // локальная плоскость XY: +X -> нос, +Y вверх; поворот меша
  1440→    // развернёт её в диаметральную, «пузо» (локальный +Z) — на борт
  1441→    const H = { x: 0, y: 0 };
  1442→    const T = { x: yLen, y: -drop * 0.62 };        // нижний конец рея — вперёд и вниз
  1443→    const C = { x: -yLen * aftFrac, y: -drop };     // шкотовый угол — к корме
  1444→    const NU = 9, NV = 6;
  1445→    const pos: number[] = [], uvs: number[] = [], idx: number[] = [];
  1446→    for (let i = 0; i <= NU; i++) {
  1447→      const u = i / NU;
  1448→      const ex = THREE.MathUtils.lerp(H.x, T.x, u), ey = THREE.MathUtils.lerp(H.y, T.y, u);
  1449→      const fx = THREE.MathUtils.lerp(H.x, C.x, u), fy = THREE.MathUtils.lerp(H.y, C.y, u);
  1450→      for (let j = 0; j <= NV; j++) {
  1451→        const v = j / NV;
  1452→        pos.push(THREE.MathUtils.lerp(ex, fx, v), THREE.MathUtils.lerp(ey, fy, v), 0);
  1453→        uvs.push(u, v);
  1454→      }
  1455→    }
  1456→    const row = NV + 1;
  1457→    for (let i = 0; i < NU; i++) {
  1458→      for (let j = 0; j < NV; j++) {
  1459→        const p0 = i * row + j, p1 = p0 + 1, p2 = p0 + row, p3 = p2 + 1;
  1460→        idx.push(p0, p1, p2, p1, p3, p2);
  1461→      }
  1462→    }
  1463→    const g = new THREE.BufferGeometry();
  1464→    g.setAttribute('position', new THREE.Float32BufferAttribute(pos, 3));
  1465→    g.setAttribute('uv', new THREE.Float32BufferAttribute(uvs, 2));
  1466→    g.setIndex(idx);
  1467→    g.computeVertexNormals();
  1468→    const mat = sailMaterial(P.colors.sail);
  1469→    // зона прострела латинского паруса — треугольник H(голова)-T(конец реи)-C(шкот):
  1470→    // в локалях меша +X -> нос после поворота меша на -90° вокруг Y
  1471→    const yardLenM = Math.hypot(T.x - H.x, T.y - H.y);
  1472→    mat.uniforms.uAspect.value = Math.max(0.3, (yardLenM + Math.abs(C.x)) / Math.max(1, drop));
  1473→    const mesh = new THREE.Mesh(g, mat);
  1474→    mesh.rotation.y = -Math.PI / 2; // локальный +X -> нос, «пузо» -> борт
  1475→    mesh.frustumCulled = false;
  1476→    const pivot = new THREE.Group();
  1477→    pivot.position.set(0, md.h * 0.72 - P.draft * 0.3, 0);
  1478→    // рея: от головы у мачты к носу-вниз
  1479→    const yardLen = Math.hypot(T.x - H.x, T.y - H.y) * 1.06;
  1480→    const yard = new THREE.Mesh(new THREE.CylinderGeometry(0.055, 0.085, yardLen, 5), woodMat);
  1481→    yard.rotation.z = Math.atan2(T.y - H.y, T.x - H.x) - Math.PI / 2;
  1482→    yard.position.set(T.x / 2, T.y / 2, 0);
  1483→    pivot.add(yard);
  1484→    pivot.add(mesh);
  1485→    masts[mastIndex].add(pivot);
  1486→    // поворот меша -90° вокруг Y: (x, y, 0) -> (0, y, x)
  1487→    const rotP = (x: number, y: number) => new THREE.Vector3(0, y, x);
  1488→    const pv = new THREE.Vector3(0, md.h * 0.72 - P.draft * 0.3, md.z);
  1489→    const quad: SailHitQuad = {
  1490→      pivot: pv,
  1491→      corners: [rotP(H.x, H.y), rotP(T.x, T.y), rotP(C.x, C.y), rotP(C.x, C.y)],
  1492→      uvs: [
  1493→        new THREE.Vector2(0, 0), new THREE.Vector2(1, 0),
  1494→        new THREE.Vector2(1, 1), new THREE.Vector2(1, 1),
  1495→      ],
  1496→      rotates: false,
  1497→    };
  1498→    sails.push({
  1499→      mesh, mat, setAt, furl: 1, square: false, braceAngle: 0, mastIndex,
  1500→      quad, holeW: yardLenM + Math.abs(C.x), holeH: drop,
  1501→    });
  1502→  };
  1503→  if (P.rig === 'single') {
  1504→    // когг: один большой прямой парус на мачту
  1505→    addSquareSail(0, 0.7, L * 0.42, L * 0.24, 0.5);
  1506→    addSquareSail(1, 0.76, L * 0.4, L * 0.22, 0.72);
  1507→  } else if (P.rig === 'lateenMix') {
  1508→    // каравелла редонда: прямой фок + латинские грот и бизань

  1510→    addLateenSail(1, 0.45, L * 0.5, mastDefs[1].h * 0.5, 0.85);
  1511→    addLateenSail(2, 0.8, L * 0.32, mastDefs[2].h * 0.5, 0.55);
  1512→  } else if (P.rig === 'lateen') {
  1513→    // шебека / галера: огромные латинские паруса на всех мачтах
  1514→    addLateenSail(0, 0.68, L * 0.34, mastDefs[0].h * 0.46, 0.5);
  1515→    addLateenSail(1, 0.45, L * 0.52, mastDefs[1].h * 0.52, 0.85);
  1516→    addLateenSail(2, 0.78, L * 0.3, mastDefs[2].h * 0.44, 0.5);
  1517→  } else if (P.masts === 2) {
  1518→    addSquareSail(0, 0.66, L * 0.3, L * 0.16, 0.62);   // фок
  1519→    addSquareSail(1, 0.72, L * 0.36, L * 0.2, 0.45);   // грот нижний — ставится первым
  1520→    addSquareSail(1, 0.45, L * 0.26, L * 0.14, 0.92);  // грот верхний (брамсель)
  1521→  } else {
  1522→    addSquareSail(0, 0.62, L * 0.24, L * 0.13, 0.60);

  1528→  // кливера (треугольные) от бушприта к фок-мачте: сетка с uv —

  1530→  const addJib = (tip: THREE.Vector3, foot: THREE.Vector3, headY: number, mastZ: number, setAt: number) => {
  1531→    const head = new THREE.Vector3(0, headY, mastZ - 0.3);
  1532→    // локальная система: X = −мирZ (продольная), Y = мирY, Z = мирX (пузо вбок)
  1533→    const a = { x: -(tip.z - head.z), y: tip.y - head.y };
  1534→    const b = { x: -(foot.z - head.z), y: foot.y - head.y };
  1535→    const NU = 8, NV = 5;
  1536→    const pos: number[] = [], uvs: number[] = [], idx: number[] = [];
  1537→    for (let i = 0; i <= NU; i++) {
  1538→      const u = i / NU;
  1539→      // точка на штаге (к галсовому углу) и на шкаторине
  1540→      const lx = a.x * u, ly = a.y * u;
  1541→      const rx = b.x * u, ry = b.y * u;
  1542→      for (let j = 0; j <= NV; j++) {
  1543→        const v = j / NV;
  1544→        pos.push(THREE.MathUtils.lerp(lx, rx, v), THREE.MathUtils.lerp(ly, ry, v), 0);
  1545→        uvs.push(u, v);
  1546→      }
  1547→    }
  1548→    const row = NV + 1;
  1549→    for (let i = 0; i < NU; i++) {
  1550→      for (let j = 0; j < NV; j++) {
  1551→        const p0 = i * row + j, p1 = p0 + 1, p2 = p0 + row, p3 = p2 + 1;
  1552→        idx.push(p0, p1, p2, p1, p3, p2);
  1553→      }
  1554→    }
  1555→    const g = new THREE.BufferGeometry();
  1556→    g.setAttribute('position', new THREE.Float32BufferAttribute(pos, 3));
  1557→    g.setAttribute('uv', new THREE.Float32BufferAttribute(uvs, 2));
  1558→    g.setIndex(idx);
  1559→    g.computeVertexNormals();
  1560→    const mat = sailMaterial(P.colors.sail);
  1561→    mat.uniforms.uAspect.value = Math.max(0.3, Math.hypot(a.x, a.y) / Math.max(1, Math.hypot(b.x, b.y)));
  1562→    const mesh = new THREE.Mesh(g, mat);
  1563→    mesh.position.set(head.x, head.y, head.z - mastZ);
  1564→    mesh.rotation.y = Math.PI / 2; // локальный Z (пузо) -> мир X (борт)
  1565→    mesh.frustumCulled = false;
  1566→    masts[0].add(mesh); // потомок фок-мачты — падает вместе с ней
  1567→    // зона прострела: треугольник голова-(конец штага)-(галсовый угол);
  1568→    // поворот меша +90° вокруг Y: (x, y, 0) -> (0, y, -x)
  1569→    const rotP = (x: number, y: number) => new THREE.Vector3(0, y, -x);
  1570→    const pv = new THREE.Vector3(head.x, head.y, head.z);
  1571→    const quad: SailHitQuad = {
  1572→      pivot: pv,
  1573→      corners: [rotP(0, 0), rotP(a.x, a.y), rotP(b.x, b.y), rotP(b.x, b.y)],
  1574→      uvs: [
  1575→        new THREE.Vector2(0, 0), new THREE.Vector2(1, 0),
  1576→        new THREE.Vector2(1, 1), new THREE.Vector2(1, 1),
  1577→      ],
  1578→      rotates: false,
  1579→    };
  1580→    sails.push({
  1581→      mesh, mat, setAt, furl: 1, square: false, braceAngle: 0, mastIndex: 0,
  1582→      quad, holeW: Math.hypot(a.x, a.y), holeH: Math.hypot(b.x, b.y),
  1583→    });
  1584→  };
  1585→  // кливера — только на «полных» вооружениях (у когга, каравеллы и шебеки их нет)
  1586→  const foreTop = mastDefs[0];
  1587→  if (P.rig !== 'lateenMix' && P.rig !== 'single' && P.rig !== 'lateen') {
  1588→    addJib(bowspritTip, new THREE.Vector3(0, P.freeboard * 0.72, L / 2 - 1.0), foreTop.h * 0.62 - P.draft * 0.3, foreTop.z, 0.72);
  1589→    addJib(new THREE.Vector3(bowspritTip.x * 0.55, bowspritTip.y - (bowspritTip.y - P.freeboard * 0.72) * 0.45, bowspritTip.z - L * 0.1),
  1590→      new THREE.Vector3(0, P.freeboard * 0.72, L / 2 - 2.0), foreTop.h * 0.78 - P.draft * 0.3, foreTop.z, 1.0);
  1591→  }

  1593→  // флаг на топе грот-мачты (потомок мачты — валится вместе с ней)
  1594→  const mainMastIdx = 1; // грот-мачта и у 2-, и у 3-мачтовых
  1595→  const mainMast = mastDefs[mainMastIdx];
  1596→  const { mesh: flag, mat: flagMat } = flagMesh(P.flagKind);

  1600→  // пушки по бортам — по раскладке gunLayout: ИТОГО ровно gunsPerSide

  1604→  const gunsStarboard: THREE.Object3D[] = [];
  1605→  for (const slot of gunLayout(P)) {

  1608→    const prof = hullProfile(tt, P);
  1609→    for (const side of [-1, 1]) {
  1610→      const gun = new THREE.Group();
  1611→      const barrel = new THREE.Mesh(new THREE.CylinderGeometry(0.11, 0.15, 1.15, 8),
  1612→        new THREE.MeshStandardMaterial({ color: 0x1f1f22, roughness: 0.5, metalness: 0.75 }));

  1632→  // штурвал: пьедестал (не вращается) + колесо, начало координат
  1633→  // вращающейся группы — в ступице, чтобы крутился вокруг СВОЕЙ оси
  1634→  const wheelAsm = new THREE.Group();
  1635→  const wheel = new THREE.Group();

  1637→  const hubY = 0.98; // высота оси над палубой
  1638→  const wheelTorus = new THREE.Mesh(new THREE.TorusGeometry(wheelR, 0.045, 6, 18), darkMat);

  1651→  const wheelHub = new THREE.Mesh(new THREE.CylinderGeometry(0.1, 0.1, 0.34, 8), woodMat);

  1656→  // стойка, подкосы и ось — вне вращающейся группы
  1657→  const post = new THREE.Mesh(new THREE.BoxGeometry(0.14, hubY, 0.12), darkMat);

  1659→  wheelAsm.add(post);
  1660→  for (const sgn of [-1, 1]) {

  1663→    brace.rotation.z = sgn * 0.55;
