   148→   148→  constructor(presetId: string, isPlayer: boolean, wind: WindState,
   149→   149→    terrainHeight: (x: number, z: number) => number, particles: Particles,
   150→   150→    opts?: ShipOpts) {
   151→   151→    this.preset = PRESETS[presetId];
   152→   152→    this.isPlayer = isPlayer;
   153→   153→    this.wind = wind;
   154→   154→    this.terrainHeight = terrainHeight;
   155→   155→    this.particles = particles;
   156→   156→    this.visual = buildShip(this.preset);
   157→   157→    // прочность по настройке игры (по умолчанию 1× — исторические ТТХ)
   158→   158→    const mul = opts?.durabilityMul ?? 1;
   159→   159→    this.hullHP = this.preset.hullHP * mul;
   160→   160→    this.hullMax = this.hullHP;
   161→   161→    this.sailHP = this.preset.sailHP * mul;
   162→   162→    this.sailMax = this.sailHP;
   163→   163→    this.mastMaxHP = 32 * mul;
   164→   164→    if (opts?.ammo) this.ammo = { ...opts.ammo };
   165→   165→    this.crew = Math.round(this.preset.maxCrew * 0.8);
   166→   166→    this.holdCap = this.preset.cargo * 0.8;
   167→   167→    this.mass = this.preset.mass;
   168→   168→
   169→   169→    const { length: L, beam: B, draft: D, freeboard: F } = this.preset;
   170→   170→    // моменты инерции (связные оси: x — крен? нет: x=право, y=верх, z=нос)
   171→   171→    // крен — вокруг z, дифферент — вокруг x, рыскание — вокруг y
   172→   172→    this.inertia.set(
   173→   173→      this.mass * (B * B + F * F) / 12 * 1.5,   // pitch (x)
   174→   174→      this.mass * (L * L + B * B) / 12 * 1.2,   // yaw (y)
   175→   175→      this.mass * (B * B + F * F) / 12 * 1.5,   // roll (z)
   176→   176→    );
   177→   177→
   178→   178→    // зонды плавучести: 9 шпангоутов × 3 точки (ниже ватерлинии),
   179→   179→    // вплоть до носа/кормы (±0.47 полудлины). Пружины жёсткие
   180→   180→    // (нейтраль 0.38 осадки, запас ×1.35) и демпфер ζ≈0.62 — корпус
   181→   181→    // БЫСТРО всходит на гребень и не проваливается внутрь волны.
   182→   182→    const NZ = 9;
   183→   183→    let n = 0;
   184→   184→    for (let i = 0; i < NZ; i++) {
   185→   185→      const t = i / (NZ - 1);
   186→   186→      const z = THREE.MathUtils.lerp(-L * 0.47, L * 0.47, t);
   187→   187→      const prof = this.halfWidthAt(z);
   188→   188→      const offs = [0, -prof * 0.75, prof * 0.75];
   189→   189→      for (const x of offs) {
   190→   190→        this.probes.push({ off: new THREE.Vector3(x, -D * 0.38, z), k: 0, c: 0 });
   191→   191→        n++;
   192→   192→      }
   193→   193→    }
   194→   194→    const neutral = D * 0.38;
   195→   195→    const kTotal = this.mass * G / neutral * 1.35;
   196→   196→    for (const p of this.probes) {
   197→   197→      p.k = kTotal / n;
   198→   198→      p.c = 2 * Math.sqrt(p.k * (this.mass / n)) * 0.62; // демпфер качки
   199→   199→    }
   200→   200→    // Резервные зонды по скулам борта чуть выше спокойной ватерлинии:
   201→   201→    // моделируют запас плавучести корпуса до планширя — гребень, накрывший
   202→   202→    // палубу, встречает сильное сопротивление и кренящий момент вправку,
   203→   203→    // «полностью по борт» корабль больше не погружается.
   204→   204→    let rn = 0;
   205→   205→    for (let i = 0; i < NZ; i++) {
   206→   206→      const t = i / (NZ - 1);
   207→   207→      const z = THREE.MathUtils.lerp(-L * 0.44, L * 0.44, t);
   208→   208→      const prof = this.halfWidthAt(z) * 0.85;
   209→   209→      for (const x of [-prof, prof]) {
   210→   210→        this.reserve.push({ off: new THREE.Vector3(x, F * 0.34, z), k: 0, c: 0 });
   211→   211→        rn++;
   212→   212→      }
   213→   213→    }
   214→   214→    // жёстче прежнего: полный вес запаса уже на 15 см погружения борта —
   215→   215→    // волна не может накрыть палубу, корпус «всплывает» на гребень
   216→   216→    const rkTotal = this.mass * G / 0.15;
   217→   217→    for (const p of this.reserve) {
   218→   218→      p.k = rkTotal / rn;
   219→   219→      p.c = 2 * Math.sqrt(p.k * (this.mass / rn)) * 1.3;
   220→   220→    }
   221→   221→
   222→   222→    this.foam = new WaterlineFoam(this.visual.outline);
   223→   223→    this.wake.setWaterHeight(waterHeightAt);
   224→   224→    if (isPlayer) enableLantern(this.visual);
   225→   225→
   226→   226→    // экипаж на палубе: процедурные типажи, сид фиксирован по модели
   227→   227→    const seed = Array.from(presetId).reduce((a, ch) => a + ch.charCodeAt(0) * 31, 11)
   228→   228→      + Math.round(this.preset.length * 7);
   229→   229→    this.crew3d = buildCrew(this.preset, seed);
   230→   230→    this.visual.group.add(this.crew3d.group);
   231→   231→    for (let i = 0; i < this.visual.masts.length; i++) {
   232→   232→      this.mastHP.push(this.mastMaxHP);
   233→   233→      this.mastFall.push(-1);
   234→   234→    }
   235→   235→  }

   237→   237→  halfWidthAt(z: number): number {
   238→   238→    const t = clamp((z + this.preset.length / 2) / this.preset.length, 0, 1);

   241→   241→    else w = THREE.MathUtils.lerp(1.0, 0.0, Math.pow(clamp((t - 0.45) / 0.55, 0, 1), 0.75));
   242→   242→    return Math.max(0.3, w * this.preset.beam * 0.5);

   255→   255→  speed(): number { return Math.hypot(this.velocity.x, this.velocity.z); }
   256→   256→  speedKnots(): number { return this.speed() * 1.9438; }

   258→   258→  /** Угол между направлением ОТКУДА дует (кажущийся) ветер и курсом, 0..PI */
   259→   259→  apparentWindAngle(): number {
   260→   260→    const wf = this.tmpV.set(Math.cos(this.wind.from), 0, Math.sin(this.wind.from));
   261→   261→    // кажущийся ветер = истинный - скорость корабля
   262→   262→    const av = this.tmpV2.set(-wf.x * this.wind.speed - this.velocity.x, 0, -wf.z * this.wind.speed - this.velocity.z);
   263→   263→    // вектор ОТКУДА дует кажущийся ветер:
   264→   264→    const aFromX = -av.x, aFromZ = -av.z;
   265→   265→    const len = Math.hypot(aFromX, aFromZ) || 1;
   266→   266→    const f = this.forward;
   267→   267→    const dot = (aFromX / len) * f.x + (aFromZ / len) * f.z;
   268→   268→    return Math.acos(clamp(dot, -1, 1));
   269→   269→  }

   271→   271→  /** Полярная эффективность парусов по углу (рад): лавировка/бейдевинд/галфвинд/фордевинд */
   272→   272→  static polar(awa: number): number {
   273→   273→    const d = awa * 180 / Math.PI;
   274→   274→    if (d < 25) return 0.02;                       // левентик — почти нет хода
   275→   275→    if (d < 60) return THREE.MathUtils.lerp(0.08, 0.7, (d - 25) / 35);
   276→   276→    if (d < 110) return THREE.MathUtils.lerp(0.7, 1.0, (d - 60) / 50);  // бейдевинд-галфвинд
   277→   277→    if (d < 150) return THREE.MathUtils.lerp(1.0, 0.82, (d - 110) / 40);
   278→   278→    return THREE.MathUtils.lerp(0.82, 0.68, (d - 150) / 30);            // фордевинд
   279→   279→  }

   282→   282→  fixedStep(dt: number, time: number) {
   283→   283→    if (this.sunk) return;
   284→   284→    // затопление — ОТДЕЛЬНАЯ кинематика (фазы крена/погружения/спирали):
   285→   285→    // пружины плавучести уже не нужны, корпус ведёт себя как тело,
   286→   286→    // теряющее остойчивость
   287→   287→    if (this.sinking) { this.stepSinking(dt, time); return; }
   288→   288→    this.force.set(0, 0, 0);
   289→   289→    this.torque.set(0, 0, 0);
   290→   290→
   291→   291→    const f = this._f.copy(this.forward);
   292→   292→    // штирборт (правый борт): forward × up = (−f.z, 0, f.x)
   293→   293→    const right = this._r.set(-f.z, 0, f.x).normalize();
   294→   294→    const e = this.euler;
   295→   295→    const spd = this.speed();
   296→   296→
   297→   297→    const m = this.mass + this.water * 1000; // вода в трюме утяжеляет
   298→   298→
   299→   299→    // --- гравитация ---
   300→   300→    this.force.y -= m * G;
   301→   301→
   302→   302→    // --- плавучесть по зондам (τ = r × F, F = (0, fl, 0)) ---
   303→   303→    for (const p of this.probes) {
   304→   304→      const wp = this._t1.copy(p.off).applyQuaternion(this.quaternion).add(this.position);
   305→   305→      const wy = waterHeightAt(wp.x, wp.z, time);
   306→   306→      const depth = wy - wp.y;
   307→   307→      if (depth > 0) {
   308→   308→        const rx = wp.x - this.position.x, rz = wp.z - this.position.z;
   309→   309→        // вертикальная скорость точки: v = ω × r → vy = ωz*rx − ωx*rz
   310→   310→        const vy = this.velocity.y + (this.angular.z * rx - this.angular.x * rz);
   311→   311→        let fl = p.k * Math.min(depth, 6) - p.c * vy;
   312→   312→        if (fl < 0) fl = 0;
   313→   313→        this.force.y += fl;
   314→   314→        this.torque.x += -rz * fl;
   315→   315→        this.torque.z += rx * fl;
   316→   316→      }
   317→   317→    }
   318→   318→    // --- запас плавучести бортов (пока не тонем) ---
   319→   319→    if (!this.sinking) {
   320→   320→      for (const p of this.reserve) {
   321→   321→        const wp = this._t1.copy(p.off).applyQuaternion(this.quaternion).add(this.position);
   322→   322→        const wy = waterHeightAt(wp.x, wp.z, time);
   323→   323→        const depth = wy - wp.y;
   324→   324→        if (depth > 0) {
   325→   325→          const rx = wp.x - this.position.x, rz = wp.z - this.position.z;
   326→   326→          const vy = this.velocity.y + (this.angular.z * rx - this.angular.x * rz);
   327→   327→          let fl = p.k * Math.min(depth, 2.5) - p.c * vy;
   328→   328→          if (fl < 0) fl = 0;
   329→   329→          this.force.y += fl;
   330→   330→          this.torque.x += -rz * fl;
   331→   331→          this.torque.z += rx * fl;
   332→   332→        }
   333→   333→      }
   334→   334→    }
   335→   335→
   336→   336→    // --- ветер на парусах ---
   337→   337→    this.sailFrac += (this.sailTarget / 3 - this.sailFrac) * Math.min(1, dt * 0.7);
   338→   338→    const awa = this.apparentWindAngle();
   339→   339→    const polar = Ship.polar(awa);
   340→   340→    const sailEffMul = this.sailHP / this.preset.sailHP;
   341→   341→    const damaged = sailEffMul < 0.5 ? 0.45 : sailEffMul < 0.85 ? 0.8 : 1.0;
   342→   342→    const windPower = Math.pow(this.wind.speed / 8.0, 1.6);
   343→   343→    let thrust = 0;
   344→   344→    if (!this.anchored && !this.sinking && this.mastsBroken < 2) {
   345→   345→      thrust = this.sailFrac * polar * windPower * this.preset.maxSpeed * 7.0 * damaged * (this.sailHP > 0 ? 1 : 0);
   346→   346→    }
   347→   347→    if (this.reverseThrust && !this.anchored) thrust = -this.preset.maxSpeed * 0.45; // задний ход на вёслах
   348→   348→    // ГРЕБНЫЕ суда (шебека, галера): вёсла дают ход даже в полный штиль;
   349→   349→    // под парусами вёсла помогают меньше — экипаж занят парусами
   350→   350→    if (this.preset.oars && !this.anchored && !this.sinking) {
   351→   351→      const crewFrac = clamp(this.crew / this.preset.maxCrew, 0.2, 1);
   352→   352→      const oarThrust = this.preset.maxSpeed * 0.34 * crewFrac * (1 - 0.55 * this.sailFrac);
   353→   353→      thrust += oarThrust;
   354→   354→    }
   355→   355→    // сила тяги вдоль forward
   356→   356→    this.force.x += f.x * thrust * m * 0.06;
   357→   357→    this.force.z += f.z * thrust * m * 0.06;
   358→   358→    // крен от ветра: боковая составляющая ветра на парусах
   359→   359→    // (умеренный: в шторм подветренный борт не зарывается в волну)
   360→   360→    const wf = new THREE.Vector3(Math.cos(this.wind.from), 0, Math.sin(this.wind.from));
   361→   361→    const windTravel = wf.clone().multiplyScalar(-this.wind.speed);
   362→   362→    const lateral = windTravel.dot(right);
   363→   363→    const heelF = lateral * this.sailFrac * windPower * m * 0.034;
   364→   364→    this.torque.z += heelF;
   365→   365→    // ... и небольшое боковое сносовое усилие
   366→   366→    this.force.x += right.x * lateral * this.sailFrac * m * 0.004;
   367→   367→    this.force.z += right.z * lateral * this.sailFrac * m * 0.004;
   368→   368→
   369→   369→    // --- руль ---
   370→   370→    this.rudder += (this.rudderTarget - this.rudder) * Math.min(1, dt * 1.8);
   371→   371→    const rudderAngle = this.rudder * 0.6; // рад, макс ~34°
   372→   372→    const yawAuth = clamp(spd / 4.0, 0, 1.15); // авторитеты руля от скорости
   373→   373→    // положительный руль (D) = поворот ВПРАВО (штирборт):
   374→   374→    // вращение вокруг −Y, т.к. +Y = поворот влево (к порту)
   375→   375→    this.torque.y -= rudderAngle * yawAuth * m * 12.0; // ~6°/с на полном ходу
   376→   376→    // крен на повороте (отцентробежный)
   377→   377→    this.torque.z += this.angular.y * spd * m * 0.35;
   378→   378→    // лёгкое зарывание носа на скорости (слабое: сильное топило борт
   379→   379→    // в волну на 3+ метра — см. жалобы «корабль погружается по борт»)
   380→   380→    this.torque.x += spd * spd * m * 0.0012;
   381→   381→
   382→   382→    // --- сопротивление (по связным осям) ---
   383→   383→    const vLocal = this.velocity.clone();
   384→   384→    const fwdSpd = vLocal.dot(f);
   385→   385→    const latSpd = vLocal.dot(right);
   386→   386→    const fwdDrag = -Math.sign(fwdSpd) * (18 * fwdSpd + 55 * fwdSpd * Math.abs(fwdSpd)) * m * 0.001;
   387→   387→    const latDrag = -latSpd * (3.2) * m * 0.1;
   388→   388→    this.force.x += f.x * fwdDrag + right.x * latDrag;
   389→   389→    this.force.z += f.z * fwdDrag + right.z * latDrag;
   390→   390→    this.force.y -= this.velocity.y * m * 0.12; // вертикальное демпфирование
   391→   391→    if (this.anchored) {
   392→   392→      this.force.x -= this.velocity.x * m * 1.2;
   393→   393→      this.force.z -= this.velocity.z * m * 1.2;
   394→   394→    }
   395→   395→    if (this.grounded) {
   396→   396→      this.force.x -= this.velocity.x * m * 2.5;
   397→   397→      this.force.z -= this.velocity.z * m * 2.5;
   398→   398→    }
   399→   399→
   400→   400→    // --- угловое демпфирование по связным осям ---
   401→   401→    const invQ = this.tmpQ.copy(this.quaternion).invert();
   402→   402→    const ab = this._t3.copy(this.angular).applyQuaternion(invQ); // связные
   403→   403→    ab.x *= Math.exp(-1.6 * dt);  // дифферент
   404→   404→    ab.y *= Math.exp(-0.9 * dt);  // рыскание
   405→   405→    ab.z *= Math.exp(-1.1 * dt);  // крен
   406→   406→    this.angular.copy(ab.applyQuaternion(this.quaternion));
   407→   407→
   408→   408→    // --- ограничение крена/дифферента (мягкие стопы) ---
   409→   409→    // дифферент 0.30 рад (17°) — нос всходит на передний склон
   410→   410→    // штормового гребня, а не прорезает его;
   411→   411→    // крен 0.36 рад (20°) — подветренный борт остаётся над водой
   412→   412→    const roll = e.z, pitch = e.x;
   413→   413→    const rollLim = 0.36, pitchLim = 0.30;
   414→   414→    if (Math.abs(roll) > rollLim) this.torque.z -= (roll - Math.sign(roll) * rollLim) * m * 9.0;
   415→   415→    if (Math.abs(pitch) > pitchLim) this.torque.x -= (pitch - Math.sign(pitch) * pitchLim) * m * 12.0;
   416→   416→    // затопление кренит корабль в сторону пробоин
   417→   417→    this.floodRoll(m);
   418→   418→
   419→   419→    // --- интеграция ---
   420→   420→    this.velocity.addScaledVector(this.force, dt / m);
   421→   421→    this.position.addScaledVector(this.velocity, dt);
   422→   422→    // α = I⁻¹ τ в связных осях
   423→   423→    const tb = this._t3.copy(this.torque).applyQuaternion(invQ);
   424→   424→    tb.x /= this.inertia.x; tb.y /= this.inertia.y; tb.z /= this.inertia.z;
   425→   425→    this.angular.addScaledVector(tb.applyQuaternion(this.quaternion), dt);
   426→   426→    // ограничим угловые скорости
   427→   427→    const wLen = this.angular.length();
   428→   428→    if (wLen > 1.1) this.angular.multiplyScalar(1.1 / wLen);
   429→   429→    // q̇ = 0.5 · ω⊗q → q_new = q + 0.5·dt·(ωx,ωy,ωz,0)⊗q, потом нормализация
   430→   430→    this.wq.set(this.angular.x, this.angular.y, this.angular.z, 0);
   431→   431→    this.dq.copy(this.wq).multiply(this.quaternion); // dq = ω ⊗ q
   432→   432→    const qs = 0.5 * dt;
   433→   433→    this.quaternion.set(
   434→   434→      this.quaternion.x + this.dq.x * qs,
   435→   435→      this.quaternion.y + this.dq.y * qs,
   436→   436→      this.quaternion.z + this.dq.z * qs,
   437→   437→      this.quaternion.w + this.dq.w * qs
   438→   438→    ).normalize();
   439→   439→
   440→   440→    // === КИНЕМАТИЧЕСКАЯ ПОСАДКА НА ВОЛНУ (плывёт ПОВЕРХНОСТЬЮ, без парения) ===
   441→   441→    // Итог четырёх итераций:
   442→   442→    //  1) среднее 4 точек — гребень проходил СКВОЗЬ корпус (жалоба);
   443→   443→    //  2) жёсткая огибающая (max по станциям) — корабль «повис над водой»;
   444→   444→    //  3) среднее + подъём ≤ 0.45D — в шторм гребень выше осадки, корпус
   445→   445→    //     снова резал волну (жалоба «уходит внутрь волны»);
   446→   446→    //  4) текущий баланс:
   447→   447→    //   - ВЕРТИКАЛЬ — средневзвешенный по ширине уровень + МЯГКАЯ
   448→   448→    //     огибающая с потолком 0.45D + 0.7·огибающая_волнения: на большом
   449→   449→    //     гребне потолок растёт вместе с волной — корпус ВЗХОДИТ на гребень;
   450→   450→    //     в штиле волны ниже 0.25D — подъём не включается вовсе (нет парения);
   451→   451→    //   - ДИФФЕРЕНТ — МНК-наклон поверхности (вес по ширине): нос задирается
   452→   452→    //     заранее, на переднем склоне приближающегося гребня.
   453→   453→    // Крен остаётся живым (ветер/течь/поворот) — пружины бортов.
   454→   454→    if (!this.sinking && !this.sunk && !this.grounded) {
   455→   455→      const L = this.preset.length, D = this.preset.draft;
   456→   456→      const FR = this._st;
   457→   457→      const sinP = Math.sin(e.x);
   458→   458→      const halfB = Math.max(0.5, this.preset.beam * 0.5);
   459→   459→      // проход 1: высоты воды и веса станций (ширина корпуса в точке)
   460→   460→      let wSum = 0, whSum = 0, whzSum = 0, wz2Sum = 0;
   461→   461→      for (let i = 0; i < FR.length; i++) {
   462→   462→        const zz = FR[i] * L;
   463→   463→        const h = waterHeightAt(this.position.x + f.x * zz, this.position.z + f.z * zz, time);
   464→   464→        // вес 0.25..1: оконечности узкие — их гребни корпус не поднимают всем весом
   465→   465→        const w = Math.max(0.25, this.halfWidthAt(zz) / halfB);
   466→   466→        this._hs[i] = h;
   467→   467→        wSum += w;
   468→   468→        whSum += w * h;
   469→   469→        whzSum += w * h * zz;
   470→   470→        wz2Sum += w * zz * zz;
   471→   471→      }
   472→   472→      // спокойная посадка: центр на 0.10·D над средневзвешенным уровнем;
   473→   473→      // затопленный трюм садит корпус ниже (до 30 см)
   474→   474→      const sink = clamp(this.water / Math.max(1, this.holdCap), 0, 1) * 0.30;
   475→   475→      const meanY = whSum / wSum + D * 0.10 - sink;
   476→   476→      // проход 2: мягкая огибающая против зарывания оконечностей.
   477→   477→      // Заглубление киля станции при среднем уровне: dig = h + z·sinP + 0.9D − meanY;
   478→   478→      // нормальная осадка даёт dig ≈ 0.8D — лишнее свыше 1.15D выталкиваем,
   479→   479→      // потолок подъёма растёт с волнением (шторм = корпус идёт по гребню)
   480→   480→      let lift = 0;
   481→   481→      for (let i = 0; i < FR.length; i++) {
   482→   482→        const zz = FR[i] * L;
   483→   483→        const dig = this._hs[i] + zz * sinP + 0.9 * D - meanY;
   484→   484→        const l = dig - 1.15 * D;
   485→   485→        if (l > lift) lift = l;
   486→   486→      }
   487→   487→      const liftCap = 0.45 * D + 0.70 * waveEnvelope();
   488→   488→      const targetY = meanY + clamp(lift, 0, liftCap);
   489→   489→      const kY = 1 - Math.exp(-dt * 6.0);
   490→   490→      const corrY = clamp((targetY - this.position.y) * kY, -17 * dt, 17 * dt);
   491→   491→      this.position.y += corrY;
   492→   492→      // согласованное затухание вертикальной скорости — пружины
   493→   493→      // не раскачивают корпус вокруг кинематической посадки
   494→   494→      this.velocity.y *= Math.exp(-dt * 3.0);
   495→   495→      // дифферент: МНК-наклон поверхности вдоль ДП (вода выше у носа → нос вверх)
   496→   496→      const wPitch = -Math.atan2(whzSum, wz2Sum);
   497→   497→      const e2 = this._e2.setFromQuaternion(this.quaternion, 'YXZ');
   498→   498→      const kP = 1 - Math.exp(-dt * 5.0);
   499→   499→      e2.x += clamp((wPitch - e2.x) * kP, -1.2 * dt, 1.2 * dt);
   500→   500→      this.quaternion.setFromEuler(e2);
   501→   501→    }
   502→   502→
   503→   503→    // --- посадка на мель / столкновение с землёй ---
   504→   504→    this.checkGrounding(dt);
   505→   505→
   506→   506→    // --- течение воды в трюме ---
   507→   507→    if (this.leakRate > 0) {
   508→   508→      this.water = Math.min(this.holdCap, this.water + this.leakRate * dt);
   509→   509→    }
   510→   510→    if (this.onFire) {
   511→   511→      this.fireT += dt;
   512→   512→      const I = this.fireIntensity;
   513→   513→      // рост: сухая палуба разгорается, сильный ветер раздувает;
   514→   514→      // после 40 с горения топливо прогорает — интенсивность падает
   515→   515→      let dI: number;
   516→   516→      if (this.fireT > 40) dI = -0.05;                             // догорело
   517→   517→      else dI = 0.008 + 0.022 * I + Math.min(0.012, this.wind.speed * 0.0009);
   518→   518→      // дождь тушит
   519→   519→      dI -= this.rainRate * 0.03;
   520→   520→      this.fireIntensity = clamp(this.fireIntensity + dI * dt, 0, 1);
   521→   521→      if (this.fireIntensity <= 0.01) {
   522→   522→        this.onFire = false; this.fireT = 0;
   523→   523→      } else {
   524→   524→        // урон: корпус горит всегда, паруса — от 0.4, команда — от 0.6,
   525→   525→        // такелаж вспыхивает на полном огне
   526→   526→        this.hullHP -= dt * (0.5 + 2.8 * I);
   527→   527→        if (I > 0.4) this.sailHP = Math.max(0, this.sailHP - dt * I * 2.2);
   528→   528→        if (I > 0.6) this.crew = Math.max(0, this.crew - dt * 0.22 * I);
   529→   529→        if (I > 0.9 && Math.random() < dt * 0.05) this.breakMast();
   530→   530→        // визуал: частота и размер пламени от интенсивности
   531→   531→        if (Math.random() < dt * (0.8 + 3.2 * I)) {
   532→   532→          const p = this.particles;
   533→   533→          const wp = this.position;
   534→   534→          const spread = 3 + I * 6;
   535→   535→          p.fire(wp.x + (Math.random() - 0.5) * spread, this.waterlineY() + 0.8 + Math.random() * 2.5,
   536→   536→            wp.z + (Math.random() - 0.5) * spread * 2.2);
   537→   537→          p.smoke(wp.x + (Math.random() - 0.5) * spread, this.waterlineY() + 3,
   538→   538→            wp.z + (Math.random() - 0.5) * spread * 2.2,
   539→   539→            this.wind ? Math.cos(this.wind.from + Math.PI) * 2 : 0, 2.2,
   540→   540→            this.wind ? Math.sin(this.wind.from + Math.PI) * 2 : 0,
   541→   541→            2.2 + I * 1.6, 2.6 + I);
   542→   542→        }
   543→   543→      }
   544→   544→    }
   545→   545→    // порог затопления
   546→   546→    if (!this.sinking && this.water > this.holdCap * 0.62) this.startSinking();
   547→   547→    if (this.hullHP <= 0 && !this.sinking) this.startSinking();
   548→   548→
   549→   549→    // --- сдался? ---
   550→   550→    if (!this.isPlayer && !this.surrendered && !this.sinking) {
   551→   551→      if (this.hullHP < this.hullMax * 0.34 || this.crew < this.preset.maxCrew * 0.25) {
   552→   552→        this.surrendered = true;
   553→   553→        this.sailTarget = 0;
   554→   554→        this.onSurrender?.(this);
   555→   555→      }
   556→   556→    }
   557→   557→    // (момент гибели считает stepSinking — фазовая машина затопления)
   558→   558→  }
   559→   559→
   560→   560→  private checkGrounding(dt: number) {
   561→   561→    if (this.sinking || this.sunk) return;
   562→   562→    const f = this.forward;
   563→   563→    const bow = this.tmpV.copy(this.position).addScaledVector(f, this.preset.length * 0.42);
   564→   564→    const stern = this.tmpV2.copy(this.position).addScaledVector(f, -this.preset.length * 0.42);
   565→   565→    let hit = false;
   566→   566→    for (const p of [bow, stern]) {
   567→   567→      const th = this.terrainHeight(p.x, p.z);
   568→   568→      if (th > this.position.y - this.preset.draft * 0.6) {
   569→   569→        hit = true;
   570→   570→        // нормаль склона
   571→   571→        const e = 2;
   572→   572→        const nx = this.terrainHeight(p.x - e, p.z) - this.terrainHeight(p.x + e, p.z);
   573→   573→        const nz = this.terrainHeight(p.x, p.z - e) - this.terrainHeight(p.x, p.z + e);
   574→   574→        const n = new THREE.Vector3(nx, 2 * e, nz).normalize();
   575→   575→        this.force.addScaledVector(n, this.mass * 22);
   576→   576→        this.velocity.multiplyScalar(0.94);
   577→   577→      }
   578→   578→    }
   579→   579→    if (hit) {
   580→   580→      if (!this.grounded && this.speed() > 3.5) {
   581→   581→        this.applyDamage({ kind: 'ground', amount: 5 + this.speed() });
   582→   582→      }
   583→   583→      this.grounded = true;
   584→   584→    } else this.grounded = false;
   585→   585→  }
   586→   586→
   587→   587→  waterlineY(): number {
   588→   588→    return this.position.y;
   589→   589→  }
   590→   590→
   591→   591→  // ---------- повреждения ----------
   592→   592→  applyDamage(d: DamageInput) {
   593→   593→    if (this.sunk || this.sinking) return;
   594→   594→    const rng = Math.random;
   595→   595→    switch (d.kind) {
   596→   596→      case 'ball': {
   597→   597→        this.hullHP -= d.amount;
   598→   598→        if (d.hitPoint && d.hitPoint.y < 0.45 + this.position.y) this.addLeak(0.06 + rng() * 0.05);
   599→   599→        // ЭКИПАЖ гибнет ТОЛЬКО от прямого прострела палубы (CombatSystem
   600→   600→        // → crew3d.killNearPath). Залп в борт ниже палубы людей не косит:
   601→   601→        // доска и переборки принимают удар на себя
   602→   602→        // случайный «рикошет в рангоут» — редкий, основную работу делают
   603→   603→        // прицельные попадания из CombatSystem (mast hits)
   604→   604→        if (rng() < 0.10) this.breakMast();
   605→   605→        break;
   606→   606→      }
   607→   607→      case 'chain': {
   608→   608→        this.sailHP = Math.max(0, this.sailHP - d.amount);
   609→   609→        // книппель рвёт такелаж над головами — люди гибнут только от
   610→   610→        // обрывов рангоута (breakMast) и прямых прострелов
   611→   611→        break;
   612→   612→      }
   613→   613→      case 'grape': {
   614→   614→        this.crew = Math.max(0, this.crew - Math.round(d.amount));
   615→   615→        this.hullHP -= d.amount * 0.4;
   616→   616→        break;
   617→   617→      }
   618→   618→      case 'fire': {
   619→   619→        this.hullHP -= d.amount;
   620→   620→        if (rng() < 0.35) this.ignite(0.3 + rng() * 0.25); // очаг на палубе
   621→   621→        if (rng() < 0.08) { this.addLeak(0.03); }
   622→   622→        break;
   623→   623→      }
   624→   624→      case 'ram': {
   625→   625→        this.hullHP -= d.amount;
   626→   626→        if (rng() < 0.5) this.addLeak(0.04);
   627→   627→        break;
   628→   628→      }
   629→   629→      case 'ground': this.hullHP -= d.amount * 0.5; break;
   630→   630→    }
   631→   631→    if (this.hullHP <= 0 && !this.sinking) this.startSinking();
   632→   632→    if (!this.isPlayer && !this.noSurrender && !this.surrendered && this.hullHP < this.hullMax * 0.34) {
   633→   633→      this.surrendered = true; this.sailTarget = 0; this.onSurrender?.(this);
   634→   634→    }
   635→   635→  }
   636→   636→
   637→   637→  /** Урон в КОНКРЕТНУЮ мачту (прицельное попадание ядром/книппелем) */
   638→   638→  damageMast(idx: number, amount: number): boolean {
   639→   639→    if (idx < 0 || idx >= this.mastHP.length) return false;
   640→   640→    if (this.mastFall[idx] >= 0) return false; // уже сломана
   641→   641→    this.mastHP[idx] -= amount;
   642→   642→    if (this.mastHP[idx] <= 0) { this.breakMast(idx); return true; }
   643→   643→    return false;
   644→   644→  }
   645→   645→
   646→   646→  addLeak(rate: number) {
   647→   647→    this.leakRate = Math.min(1.8, this.leakRate + rate);
   648→   648→    if (this.leakRate < 0) this.leakRate = 0;
   649→   649→  }
   650→   650→
   651→   651→  /** Зажечь палубу: strength — начальная сила очага 0..1.
   652→   652→   *  Существующий пожар не ослабляет — новое пламя добавляет силы */
   653→   653→  ignite(strength: number): boolean {
   654→   654→    if (this.sinking || this.sunk) return false;
   655→   655→    this.onFire = true;
   656→   656→    this.fireIntensity = clamp(Math.max(this.fireIntensity, strength), 0, 1);
   657→   657→    return true;
   658→   658→  }
   659→   659→
   660→   660→  /** Урон ПАРУСАМ: пробоины картечи/книппеля снижают тягу (sailHP).
   661→   661→   *  Дыры рисуются отдельно — см. combat.applySailHit + sailAddHole. */
   662→   662→  damageSails(n: number) {
   663→   663→    this.sailHP = Math.max(0, this.sailHP - n);
   664→   664→  }
   665→   665→
   666→   666→  /** Пробоина в борту: тёмная рваная дыра + торчащие щепки-доски
   667→   667→   *  вокруг + течь (ниже ватерлинии) + кренящий момент */
   668→   668→  addHole(localPos: THREE.Vector3, belowWL: boolean) {
   669→   669→    const side = localPos.x >= 0 ? 1 : -1;
   670→   670→    const w = Math.max(0.3, this.halfWidthAt(localPos.z) * 0.985);
   671→   671→    // группа: дыра + щепки (одна запись в hullHoles)
   672→   672→    const hole = new THREE.Group();
   673→   673→    const R = 0.36 + Math.random() * 0.18;
   674→   674→    const mat = new THREE.MeshStandardMaterial({ color: 0x120d08, roughness: 1 });
