    31→/** Процедурная генерация архипелага: детерминированно по сиду —
    32→ *  кольцом вокруг порта, с минимальными проливами между островами.
    33→ *  Базовые острова не трогаем (животные/порт привязаны к индексам). */
    34→function generateIslands(): IslandDef[] {
    35→  const rng = new RNG(WORLD_SEED + 9173);
    36→  const names = [
    37→    'Остров Черепа', 'Мыс Багровой Луны', 'Скалы Трёх Братьев', 'Лагуна Сирен',
    38→    'Кайманов ключ', 'Затерянный риф', 'Бухта Повешенного', 'Остров Гвоздей',
    39→    'Перьевой атолл', 'Голый риф', 'Отмель Мёртвого Якоря', 'Тихая банка',
    40→  ];
    41→  const pool = [...names];
    42→  const out: IslandDef[] = [];
    43→  let guard = 0;
    44→  while (out.length < 11 && guard++ < 600) {
    45→    const a = rng.range(0, Math.PI * 2);
    46→    const d = rng.range(1000, 2850);
    47→    const cx = Math.cos(a) * d;
    48→    const cz = Math.sin(a) * d * 0.82; // мир чуть шире по X — под карту 640×520
    49→    if (Math.abs(cx) > 2850 || Math.abs(cz) > 2250) continue;
    50→    const roll = rng.next();
    51→    const type: IslandDef['type'] = roll < 0.5 ? 'jungle' : roll < 0.78 ? 'rock' : 'atoll';
    52→    const r = rng.range(85, 230);
    53→    // проливы не уже 520 м между берегами
    54→    let ok = true;
    55→    for (const isl of [...BASE_ISLANDS, ...out]) {
    56→      if (Math.hypot(cx - isl.cx, cz - isl.cz) < isl.r + r + 520) { ok = false; break; }
    57→    }
    58→    if (!ok) continue;
    59→    const maxH = type === 'rock' ? rng.range(18, 34) : type === 'jungle' ? rng.range(10, 26) : rng.range(6, 11);
    60→    const name = pool.length ? pool.splice(Math.floor(rng.next() * pool.length), 1)[0] : `Безымянный риф ${out.length + 1}`;
    61→    out.push({ cx: Math.round(cx), cz: Math.round(cz), r: Math.round(r), maxH: Math.round(maxH), type, name, discovered: false });
    62→  }
    63→  return out;
    64→}

    68→/** Высота суши в мировой точке. Открытое море = -14 (глубина), суша > 0 */
    69→export function heightAt(x: number, z: number): number {
    70→  let h = -14; // дно открытого моря
    71→  for (const isl of ISLANDS) {
    72→    const dx = x - isl.cx, dz = z - isl.cz;
    73→    const d = Math.hypot(dx, dz);
    74→    if (d > isl.r * 2.0) continue;
    75→    // рваная береговая линия: радиус модулируется шумом
    76→    const coastNoise = fbm2(x * 0.008, z * 0.008, 3, WORLD_SEED + isl.r);
    77→    const rEff = isl.r * (0.78 + 0.5 * coastNoise);
    78→    if (d > rEff * 1.5) continue;
    79→    const fall = smoothstep(1.18, 0.22, d / rEff);
    80→    const detail = 0.55 + 0.45 * fbm2(x * 0.02, z * 0.02, 4, WORLD_SEED + isl.cx);
    81→    const ridge = isl.type === 'rock' ? Math.pow(fall, 1.6) : fall;
    82→    let hi = isl.maxH * ridge * detail - 2.6;
    83→    // плавный склон к морскому дну за отмелью
    84→    hi = THREE.MathUtils.lerp(hi, -14, smoothstep(1.12, 1.5, d / rEff));
    85→    if (hi > h) h = hi;
    86→  }
    87→  return h;
    88→}

    90→function slopeAt(x: number, z: number): number {
    91→  const e = 1.5;
    92→  const hx = heightAt(x + e, z) - heightAt(x - e, z);
    93→  const hz = heightAt(x, z + e) - heightAt(x, z - e);
    94→  return Math.hypot(hx, hz) / (2 * e);
    95→}

    97→// ---------- геометрия с цветом ----------
    98→function paint(g: THREE.BufferGeometry, color: number, jitter = 0.06): THREE.BufferGeometry {
    99→  const c = new THREE.Color(color);
   100→  const n = g.getAttribute('position').count;
   101→  const arr = new Float32Array(n * 3);
   102→  for (let i = 0; i < n; i++) {
   103→    const j = 1 + (Math.random() - 0.5) * 2 * jitter;
   104→    arr[i * 3] = c.r * j; arr[i * 3 + 1] = c.g * j; arr[i * 3 + 2] = c.b * j;
   105→  }
   106→  g.setAttribute('color', new THREE.BufferAttribute(arr, 3));
   107→  return g;
   108→}
   109→function tf(g: THREE.BufferGeometry, x: number, y: number, z: number, rx = 0, ry = 0, rz = 0, s = 1): THREE.BufferGeometry {
   110→  const m = new THREE.Matrix4().compose(
   111→    new THREE.Vector3(x, y, z),
   112→    new THREE.Quaternion().setFromEuler(new THREE.Euler(rx, ry, rz)),
   113→    new THREE.Vector3(s, s, s));
   114→  g.applyMatrix4(m);
   115→  return g;
   116→}

   118→// ветровая анимация крон (общий uniform)
   119→export const windTimeUniform = { value: 0 };
   120→function applySway(mat: THREE.Material, amp: number) {
   121→  mat.onBeforeCompile = (shader) => {
   122→    shader.uniforms.uTime = windTimeUniform;
   123→    shader.vertexShader = shader.vertexShader
   124→      .replace('#include <common>', '#include <common>\nuniform float uTime;')
   125→      .replace('#include <begin_vertex>', `#include <begin_vertex>
   126→        #ifdef USE_INSTANCING
   127→          vec2 ipos2 = vec2(instanceMatrix[3].x, instanceMatrix[3].z);
   128→          float swayA = ${amp.toFixed(3)} * sin(uTime * 1.35 + ipos2.x * 0.37 + ipos2.y * 0.43);
   129→          float hf2 = max(transformed.y - 0.6, 0.0);
   130→          transformed.x += swayA * hf2;
   131→          transformed.z += swayA * 0.55 * hf2;
   132→        #endif`);
   133→  };
   134→  mat.customProgramCacheKey = () => `sway-${amp}`;
   135→}

   242→  // ---------- острова ----------
   243→  private buildIslands() {
   244→    this.islands.forEach((isl, islIdx) => {
   245→      // размер квадрата: с запасом за всю зону влияния рельефа
   246→      // (rEff·1.5 до ~1.9r) — раньше 3.2r обрезали отмель НА КВАДРАТЕ,
   247→      // и под водой читался «квадрат под островом»; теперь кромка
   248→      // уходит на глубину -14 и не видна
   249→      const size = isl.r * 4.3;
   250→      const seg = 110;
   251→      // у каждого острова своё дно с крошечным сдвигом: перекрывающиеся
   252→      // плоскости не мерцают z-fight в открытом море
   253→      const deep = -14 - islIdx * 0.06;
   254→      const g = new THREE.PlaneGeometry(size, size, seg, seg);
   255→      g.rotateX(-Math.PI / 2);
   256→      const pos = g.getAttribute('position') as THREE.BufferAttribute;
   257→      const colors = new Float32Array(pos.count * 3);
   258→      const cUnder = new THREE.Color(0x9a8a66);
   259→      const cSand = new THREE.Color(0xd9c28f);
   260→      const cGrass = new THREE.Color(0x5f863d);
   261→      const cJungle = new THREE.Color(0x33642e);
   262→      const cRock = new THREE.Color(0x8a8378);
   263→      const c = new THREE.Color();
   264→      for (let i = 0; i < pos.count; i++) {
   265→        const wx = pos.getX(i) + isl.cx;
   266→        const wz = pos.getZ(i) + isl.cz;
   267→        let h = heightAt(wx, wz);
   268→        if (h < deep) h = deep;
   269→        pos.setY(i, h);
   270→        // цвет по высоте и уклону (уклон приблизим после)
   271→        if (h < -0.4) c.copy(cUnder).lerp(cSand, clamp((h + 3) / 2.6, 0, 1));
   272→        else if (h < 1.3) c.copy(cSand);
   273→        else if (h < isl.maxH * 0.62) c.copy(cGrass).lerp(cJungle, fbm2(wx * 0.05, wz * 0.05, 3, WORLD_SEED + 9));
   274→        else c.copy(cRock);
   275→        colors[i * 3] = c.r; colors[i * 3 + 1] = c.g; colors[i * 3 + 2] = c.b;
   276→      }
   277→      g.setAttribute('color', new THREE.BufferAttribute(colors, 3));
   278→      g.computeVertexNormals();
   279→      // скалы на крутых склонах
   280→      const nrm = g.getAttribute('normal') as THREE.BufferAttribute;
   281→      for (let i = 0; i < pos.count; i++) {
   282→        if (nrm.getY(i) < 0.62 && pos.getY(i) > 0.3) {
   283→          colors[i * 3] = cRock.r; colors[i * 3 + 1] = cRock.g; colors[i * 3 + 2] = cRock.b;
   284→        }
   285→      }
   286→      (g.getAttribute('color') as THREE.BufferAttribute).needsUpdate = true;
   287→      const mat = new THREE.MeshStandardMaterial({ vertexColors: true, roughness: 0.94, metalness: 0 });
   288→      const mesh = new THREE.Mesh(g, mat);
   289→      mesh.position.set(isl.cx, 0, isl.cz);
   290→      mesh.receiveShadow = true;
   291→      mesh.castShadow = true;
   292→      this.group.add(mesh);
   293→    });
   294→  }

   296→  // ---------- растительность ----------
   297→  private buildVegetation() {
   298→    interface Slot { geo: () => THREE.BufferGeometry; sway: number; }
   299→    const types: Record<string, Slot> = {
   300→      palm: { geo: palmGeo, sway: 0.05 },
   301→      ficus: { geo: ficusGeo, sway: 0.028 },
   302→      jungle: { geo: jungleGeo, sway: 0.035 },
   303→      bush: { geo: bushGeo, sway: 0.05 },
   304→      bamboo: { geo: bambooGeo, sway: 0.06 },
   305→      rock: { geo: rockGeo, sway: 0 },
   306→    };
   307→    const placements: Record<string, { m: THREE.Matrix4; islandType: string; h: number }[]> = {};
   308→    for (const k of Object.keys(types)) placements[k] = [];
   309→
   310→    for (const isl of this.islands) {
   311→      const density = isl.type === 'rock' ? 0.35 : isl.type === 'atoll' ? 1.0 : 1.25;
   312→      const area = Math.PI * isl.r * isl.r;
   313→      const target = Math.floor(area / 220 * density);
   314→      const rng = new RNG(WORLD_SEED + isl.cx * 31 + isl.cz);
   315→      let tries = 0;
   316→      let placed = 0;
   317→      while (placed < target && tries < target * 14) {
   318→        tries++;
   319→        const a = rng.range(0, Math.PI * 2);
   320→        const rr = Math.sqrt(rng.next()) * isl.r * 1.25;
   321→        const x = isl.cx + Math.cos(a) * rr;
   322→        const z = isl.cz + Math.sin(a) * rr;
   323→        const h = heightAt(x, z);
   324→        if (h < 0.75 || h > isl.maxH * 0.9) continue;
   325→        const sl = slopeAt(x, z);
   326→        if (sl > 0.62) continue;
   327→        // тип дерева по высоте и типу острова
   328→        let kind: string;
   329→        const roll = rng.next();
   330→        if (isl.type === 'rock') kind = roll < 0.7 ? 'rock' : 'bush';
   331→        else if (h < 4.2) kind = roll < 0.72 ? 'palm' : roll < 0.85 ? 'bush' : 'rock';
   332→        else if (h < isl.maxH * 0.55) kind = roll < 0.42 ? 'jungle' : roll < 0.62 ? 'palm' : roll < 0.82 ? 'bamboo' : roll < 0.94 ? 'bush' : 'ficus';
   333→        else kind = roll < 0.5 ? 'jungle' : roll < 0.75 ? 'bush' : 'rock';
   334→        if (kind === 'rock' && h < 1.0 && roll < 0.5) kind = 'palm';
   335→        const s = rng.range(0.7, 1.35);
   336→        const m = new THREE.Matrix4().compose(
   337→          new THREE.Vector3(x, h - 0.15, z),
   338→          new THREE.Quaternion().setFromEuler(new THREE.Euler(rng.range(-0.06, 0.06), rng.range(0, Math.PI * 2), rng.range(-0.06, 0.06))),
   339→          new THREE.Vector3(s, s * rng.range(0.85, 1.2), s));
   340→        placements[kind].push({ m, islandType: isl.type, h });
   341→        placed++;
   342→      }
   343→    }
   344→
   345→    for (const [kind, list] of Object.entries(placements)) {
   346→      if (!list.length) continue;
   347→      const geo = types[kind].geo();
   348→      const mat = new THREE.MeshStandardMaterial({ vertexColors: true, roughness: 0.9 });
   349→      if (types[kind].sway > 0) applySway(mat, types[kind].sway);
   350→      const inst = new THREE.InstancedMesh(geo, mat, list.length);
   351→      list.forEach((p, i) => inst.setMatrixAt(i, p.m));
   352→      inst.castShadow = true;
   353→      inst.receiveShadow = false;
   354→      inst.frustumCulled = false; // острова распределены широко — оставляем всегда видимыми
   355→      this.group.add(inst);
   356→      // позиции для ломания (камни не ломаются)
   357→      const positions = list.map((p) => {
   358→        const v = new THREE.Vector3();
   359→        v.setFromMatrixPosition(p.m);
   360→        return v;
   361→      });
   362→      this.vegSlots.push({
   363→        inst, positions,
   364→        broken: new Array(list.length).fill(false),
   365→        breakable: kind !== 'rock',
   366→      });
   367→    }
   368→  }

   370→  /** Ядро/книппель валят деревья: ломает экземпляры в радиусе,
   371→   *  возвращает количество поваленных (0 = промах/камни) */
   372→  breakTrees(x: number, y: number, z: number, radius: number,
   373→    fx?: { debris: (x: number, y: number, z: number, n?: number, power?: number) => void }): number {
   374→    let broke = 0;
   375→    const r2 = radius * radius;
   376→    for (const slot of this.vegSlots) {
   377→      if (!slot.breakable) continue;
   378→      for (let i = 0; i < slot.positions.length; i++) {
   379→        if (slot.broken[i]) continue;
   380→        const p = slot.positions[i];
   381→        const dx = p.x - x, dz = p.z - z, dy = p.y - y;
   382→        if (dx * dx + dz * dz > r2 || dy < -3 || dy > 9) continue;
   383→        slot.broken[i] = true;
   384→        // спрятать экземпляр: сдвинуть под дно + сжать в точку
   385→        this.dummy.position.set(p.x, -40, p.z);
   386→        this.dummy.quaternion.identity();
   387→        this.dummy.scale.setScalar(0.001);
   388→        this.dummy.updateMatrix();
   389→        slot.inst.setMatrixAt(i, this.dummy.matrix);
   390→        slot.inst.instanceMatrix.needsUpdate = true;
   391→        // щепа и листва из точки удара
   392→        fx?.debris(p.x, p.y + 1.6, p.z, 12, 1.2);
   393→        broke++;
   394→      }
   395→    }
   396→    this.treesBroken += broke;
   397→    return broke;
   398→  }

   622→  update(time: number, dt: number, playerPos: THREE.Vector3, waterHeight: (x: number, z: number, t: number) => number) {

   713→      this.dummy.quaternion.copy(this.tmpQ.setFromEuler(this.tmpE.set(0, Math.atan2(mk.to.x - mk.from.x, mk.to.z - mk.from.z), 0)));

   722→      const b = this.boars[i];
   723→      b.timer -= dt;
   724→      if (b.timer < 0) { b.timer = 2 + Math.random() * 4; b.dirA += (Math.random() - 0.5) * 1.8; }
   725→      const nx = b.x + Math.cos(b.dirA) * b.speed * dt;
