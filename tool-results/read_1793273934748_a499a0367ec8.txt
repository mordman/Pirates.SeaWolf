     7→// Экипаж на палубе: процедурные типажи «наугад» — бороды,
     8→// крюки вместо руки, деревянные ноги, повязки на глаз, тель-
     9→// нявки/банданы/треуголки. Роли: капитан (шкафут у штурвала),
    10→// боцман (у шпиля), канониры у пушек, вахтенные у мачт,
    11→// «ходоки» — бродят по палубе.
    12→// ЧИСЛЕННОСТЬ: одна фигура представляет CREW_PER_FIGURE единиц
    13→// экипажа; на палубе находится доля от общей численности
    14→// (малые суда 80%, средние 50%, крупные 20%) — см.
    15→// deckShareForPreset. При экипаже 0 палуба ПУСТА.
    16→// ПОТЕРИ: картечь ВАЛИТ фигуры (падают трупами на палубу),
    17→// ядро насквозь РАЗРЫВАЕТ на куски (гибы + кровь); каждая
    18→// убитая фигура = −CREW_PER_FIGURE единиц экипажа. Абстракт-
    19→// ные потери тоже обращаются в трупы. При затоплении экипаж
    20→// «покидает» корабль.
    21→// ============================================================

    26→/** Доля экипажа, находящаяся НА ПАЛУБЕ (остальные — в трюме,
    27→ *  на орудийных деках и на вахтах ниже): малые суда 80%,
    28→ *  средние 50%, крупные 20% */
    29→export function deckShareForPreset(P: ShipPreset): number {
    30→  if (P.maxCrew < 40) return 0.8;
    31→  if (P.maxCrew < 250) return 0.5;
    32→  return 0.2;
    33→}

    72→function mulberry32(seed: number) {
    73→  let a = seed >>> 0;
    74→  return () => {
    75→    a |= 0; a = (a + 0x6D2B79F5) | 0;
    76→    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    77→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    78→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    79→  };
    80→}

    93→interface Traits {
    94→  skin: number; shirt: number; pants: number;
    95→  beard: 0 | 1 | 2;        // нет / короткая / длинная
    96→  hook: -1 | 0 | 1;        // нет / правая / левая рука
    97→  pegLeg: -1 | 0 | 1;      // нет / правая / левая нога
    98→  eyePatch: boolean;
    99→  headwear: 'none' | 'bandana' | 'hat';
   100→  striped: boolean;
   101→  height: number;          // 0.92..1.08
   102→}

   104→function rollTraits(rnd: () => number, role: CrewRole): Traits {
   105→  const pick = <T,>(arr: T[]): T => arr[Math.floor(rnd() * arr.length)];
   106→  return {
   107→    skin: pick(SKIN),
   108→    shirt: pick(SHIRT),
   109→    pants: pick(PANTS),
   110→    beard: rnd() < 0.55 ? (rnd() < 0.4 ? 2 : 1) : 0,
   111→    hook: rnd() < 0.22 ? (rnd() < 0.5 ? 1 : -1) : 0,
   112→    pegLeg: rnd() < 0.18 ? (rnd() < 0.5 ? 1 : -1) : 0,
   113→    eyePatch: rnd() < 0.25,
   114→    headwear: role === 'captain' ? 'hat' : rnd() < 0.45 ? 'bandana' : rnd() < 0.35 ? 'hat' : 'none',
   115→    striped: role !== 'captain' && rnd() < 0.4,
   116→    height: 0.92 + rnd() * 0.16,
   117→  };
   118→}

   120→/** Запечь геометрию с цветом в общие массивы (как реквизит палубы) */
   121→function bake(target: { pos: number[]; col: number[] }, geo: THREE.BufferGeometry, color: number,
   122→  x: number, y: number, z: number, euler?: THREE.Euler, scale?: [number, number, number]) {
   123→  const g = geo.index ? geo.toNonIndexed() : geo;
   124→  const p = g.attributes.position as THREE.BufferAttribute;
   125→  const v = new THREE.Vector3();
   126→  const c = new THREE.Color(color);
   127→  for (let i = 0; i < p.count; i++) {
   128→    v.fromBufferAttribute(p, i);
   129→    if (scale) v.multiply(new THREE.Vector3(...scale));
   130→    if (euler) v.applyEuler(euler);
   131→    target.pos.push(v.x + x, v.y + y, v.z + z);
   132→    target.col.push(c.r, c.g, c.b);
   133→  }
   134→}
   135→
   136→function bakeGeom(t: { pos: number[]; col: number[] }): THREE.BufferGeometry {
   137→  const g = new THREE.BufferGeometry();
   138→  g.setAttribute('position', new THREE.Float32BufferAttribute(t.pos, 3));
   139→  g.setAttribute('color', new THREE.Float32BufferAttribute(t.col, 3));
   140→  g.computeVertexNormals();
   141→  return g;
   142→}

   144→export const CREW_MAT = new THREE.MeshStandardMaterial({ vertexColors: true, roughness: 0.9, metalness: 0.03 });

   146→/**
   147→ * Собрать фигурку. feetY — уровень палубы под ногами (в локальных
   148→ * координатах корабля). Возвращает группу с 1..3 мешами.
   149→ * Оси: +Z — нос (лицо по умолчанию к носу), +X — порт.
   150→ */
   151→function buildFigure(role: CrewRole, traits: Traits, seed: number): CrewFigure {
   152→  const rnd = mulberry32(seed);
   153→  const group = new THREE.Group();
   154→  const t: { pos: number[]; col: number[] } = { pos: [], col: [] };
   155→  const h = traits.height;
   156→  // размеры (м): фигура ~1.6 м
   157→  const legH = 0.62, torsoH = role === 'captain' ? 0.62 : 0.5, torsoW = role === 'captain' ? 0.4 : 0.36;
   158→  const hipY = legH;
   159→  const torsoY = hipY + torsoH / 2;
   160→  const headR = 0.13;
   161→  const headY = hipY + torsoH + headR + 0.02;
   162→
   163→  // --- торс (тельняшка — полосами) ---
   164→  if (traits.striped) {
   165→    const stripeC = [0xe8e2d4, traits.shirt];
   166→    const n = 5;
   167→    for (let i = 0; i < n; i++) {
   168→      bake(t, new THREE.BoxGeometry(torsoW, torsoH / n + 0.005, 0.22),
   169→        stripeC[i % 2], 0, hipY + torsoH * (i + 0.5) / n, 0);
   170→    }
   171→  } else {
   172→    bake(t, new THREE.BoxGeometry(torsoW, torsoH, 0.22), traits.shirt, 0, torsoY, 0);
   173→  }
   174→  // капитан: длинный кафтан + золотой кант
   175→  if (role === 'captain') {
   176→    bake(t, new THREE.BoxGeometry(torsoW + 0.06, 0.42, 0.26), rnd() < 0.5 ? COAT[0] : COAT[1], 0, hipY + 0.05, 0);
   177→    bake(t, new THREE.BoxGeometry(0.05, torsoH * 0.9, 0.015), 0xb8912f, 0, torsoY, 0.135);
   178→  }
   179→
   180→  // --- руки: плечо→кисть, у боцмана/канонира правая рука к борту ---
   181→  const armLen = 0.48;
   182→  for (const side of [-1, 1] as const) {
   183→    const isHook = traits.hook === side;
   184→    const sh = new THREE.Vector3(side * (torsoW / 2 + 0.05), hipY + torsoH - 0.06, 0);
   185→    if (role === 'helmsman') {
   186→      // рулевой: руки вытянуты ВПЕРЁД к штурвалу (+Z — нос)
   187→      const rot = new THREE.Euler(-1.15, 0, side * -0.22);
   188→      bake(t, new THREE.BoxGeometry(0.1, armLen * 0.55, 0.11), traits.shirt,
   189→        sh.x * 0.8, sh.y - 0.06, 0.16, rot);
   190→      if (!isHook) {
   191→        bake(t, new THREE.BoxGeometry(0.09, 0.24, 0.1), traits.skin,
   192→          sh.x * 1.0, sh.y - 0.02, 0.42, rot);
   193→      } else {
   194→        bake(t, new THREE.BoxGeometry(0.085, 0.2, 0.09), WOOD, sh.x * 1.0, sh.y - 0.02, 0.4, rot);
   195→      }
   196→      continue;
   197→    }
   198→    const rot = new THREE.Euler(0, 0, side * -0.35);
   199→    bake(t, new THREE.BoxGeometry(0.1, armLen * 0.55, 0.11), role === 'captain' ? COAT[0] : traits.shirt,
   200→      sh.x * 0.75, sh.y - armLen * 0.26, 0.03, rot);
   201→    // предплечье/кисть
   202→    if (isHook) {
   203→      // деревянная культа + крюк
   204→      bake(t, new THREE.BoxGeometry(0.085, 0.2, 0.09), WOOD, sh.x * 1.25, sh.y - armLen * 0.62, 0.08, rot);
   205→      const hookE = new THREE.Euler(0, 0, side * 1.57);
   206→      bake(t, new THREE.TorusGeometry(0.06, 0.018, 5, 10, Math.PI * 1.2), METAL,
   207→        sh.x * 1.32, sh.y - armLen * 0.78, 0.1, hookE);
   208→    } else {
   209→      bake(t, new THREE.BoxGeometry(0.09, 0.22, 0.1), traits.skin,
   210→        sh.x * 1.2, sh.y - armLen * 0.72, 0.06, rot);
   211→    }
   212→  }
   213→
   214→  // --- голова ---
   215→  bake(t, new THREE.SphereGeometry(headR, 10, 8), traits.skin, 0, headY, 0);
   216→  // волосы (если нет головного убора или под ним)
   217→  bake(t, new THREE.SphereGeometry(headR * 1.04, 10, 6, 0, Math.PI * 2, 0, Math.PI * 0.55),
   218→    BEARD[Math.floor(rnd() * BEARD.length)], 0, headY + 0.01, 0);
   219→  // повязка на глаз: наглазник + ремешок через лоб
   220→  if (traits.eyePatch) {
   221→    const eyeX = rnd() < 0.5 ? -0.052 : 0.052;
   222→    bake(t, new THREE.BoxGeometry(0.075, 0.07, 0.02), 0x141414, eyeX, headY + 0.01, headR - 0.008);
   223→    bake(t, new THREE.BoxGeometry(headR * 2.05, 0.024, 0.02), 0x141414, 0, headY + 0.045, headR - 0.006,
   224→      new THREE.Euler(0, 0, 0.16));
   225→  }
   226→  // борода
   227→  if (traits.beard > 0) {
   228→    const bl = traits.beard === 2 ? 0.3 : 0.14;
   229→    bake(t, new THREE.BoxGeometry(0.15, bl, 0.08),
   230→      BEARD[Math.floor(rnd() * BEARD.length)], 0, headY - headR * 0.55 - bl * 0.25, headR - 0.045);
   231→    if (traits.beard === 2) {
   232→      bake(t, new THREE.BoxGeometry(0.1, 0.14, 0.06), 0x1d1712, 0, headY - headR * 0.55 - bl - 0.05, headR - 0.05);
   233→    }
   234→  }
   235→  // головной убор
   236→  if (traits.headwear === 'bandana') {
   237→    const bc = BANDANA[Math.floor(rnd() * BANDANA.length)];
   238→    bake(t, new THREE.SphereGeometry(headR * 1.06, 10, 5, 0, Math.PI * 2, 0, Math.PI * 0.5), bc, 0, headY + 0.02, 0);
   239→    bake(t, new THREE.BoxGeometry(0.05, 0.05, 0.12), bc, 0.06, headY + 0.06, -headR - 0.03);
   240→  } else if (traits.headwear === 'hat') {
   241→    // треуголка: поля + тулья
   242→    const brim = role === 'captain' ? 0.34 : 0.28;
   243→    bake(t, new THREE.BoxGeometry(brim, 0.025, brim * 0.8), HAT, 0, headY + headR * 0.82, 0,
   244→      new THREE.Euler(0, 0.35, 0));
   245→    bake(t, new THREE.BoxGeometry(brim * 0.62, 0.12, brim * 0.5), HAT, 0, headY + headR * 0.95, 0,
   246→      new THREE.Euler(0, 0.35, 0));
   247→    if (role === 'captain') bake(t, new THREE.BoxGeometry(0.06, 0.02, 0.1), 0xb8912f, 0, headY + headR * 0.86, brim * 0.28);
   248→  }
   249→
   250→  // --- предметы ролей ---
   251→  if (role === 'captain') {
   252→    // подзорная труба в правой руке
   253→    bake(t, new THREE.CylinderGeometry(0.035, 0.05, 0.34, 6), 0xb8912f, torsoW / 2 + 0.14, hipY + 0.18, 0.14,
   254→      new THREE.Euler(1.1, 0, -0.5));
   255→  }
   256→  if (role === 'boatswain') {
   257→    // свисток боцмана на груди
   258→    bake(t, new THREE.CylinderGeometry(0.018, 0.018, 0.1, 5), 0xb8912f, 0.07, torsoY + 0.04, 0.125,
   259→      new THREE.Euler(1.57, 0, 0.2));
   260→  }
   261→
   262→  const body = new THREE.Mesh(bakeGeom(t), CREW_MAT);
   263→  group.add(body);
   264→
   265→  // --- ноги: у ходоков отдельные (шаг), у остальных в общий меш ---
   266→  const mkLeg = (side: number, wood: boolean): THREE.Mesh => {
   267→    const lt: { pos: number[]; col: number[] } = { pos: [], col: [] };
   268→    if (wood) {
   269→      bake(lt, new THREE.CylinderGeometry(0.05, 0.062, legH + 0.06, 6), WOOD, 0, (legH + 0.06) / 2, 0);
   270→    } else {
   271→      bake(lt, new THREE.BoxGeometry(0.11, legH, 0.13), traits.pants, 0, legH / 2, 0);
   272→      bake(lt, new THREE.BoxGeometry(0.115, 0.07, 0.2), 0x1f1a14, 0, 0.035, 0.03);
   273→    }
   274→    const m = new THREE.Mesh(bakeGeom(lt), CREW_MAT);
   275→    m.position.set(side * 0.095, legH, 0);
   276→    return m;
   277→  };
   278→
   279→  if (role === 'walker') {
   280→    const legL = mkLeg(-1, traits.pegLeg === -1);
   281→    const legR = mkLeg(1, traits.pegLeg === 1);
   282→    group.add(legL, legR);
   283→    // центр вращения ноги — бедро
   284→    for (const leg of [legL, legR]) {
   285→      const off = leg.position.y;
   286→      leg.geometry.translate(0, -off, 0);
   287→      leg.position.y = off;
   288→    }
   289→    return {
   290→      group, role, legL, legR, seed, walkT: rnd(), walkDir: rnd() < 0.5 ? 1 : -1,
   291→      walkSpeed: 0.55 + rnd() * 0.5, phase: rnd() * 6.28,
   292→      dead: false, deadT: 0, deathMode: 'fall', corpseLife: 0, fallDir: 1, fallRoll: 0,
   293→      skin: traits.skin, shirt: traits.shirt, pants: traits.pants,
   294→      hasHook: traits.hook !== 0, hasPeg: traits.pegLeg !== 0,
   295→    };
   296→  }
   297→  // статичные: ноги вторым мешом (деревянная нога заменяет одну из них)
   298→  const legsStatic = new THREE.Mesh(bakeGeom(t2legs(traits, legH)), CREW_MAT);
   299→  group.add(legsStatic);
   300→  return {
   301→    group, role, seed, walkT: 0, walkDir: 1, walkSpeed: 0, phase: rnd() * 6.28,
   302→    dead: false, deadT: 0, deathMode: 'fall', corpseLife: 0, fallDir: 1, fallRoll: 0,
   303→    skin: traits.skin, shirt: traits.shirt, pants: traits.pants,
   304→    hasHook: traits.hook !== 0, hasPeg: traits.pegLeg !== 0,
   305→  };
   306→}

   308→/** ноги для статичной фигуры отдельным мешом */
   309→function t2legs(traits: Traits, legH: number): { pos: number[]; col: number[] } {
   310→  const t: { pos: number[]; col: number[] } = { pos: [], col: [] };
   311→  for (const side of [-1, 1]) {
   312→    if (traits.pegLeg === side) {
   313→      bake(t, new THREE.CylinderGeometry(0.05, 0.062, legH + 0.06, 6), WOOD, side * 0.095, (legH + 0.06) / 2, 0);
   314→    } else {
   315→      bake(t, new THREE.BoxGeometry(0.11, legH, 0.13), traits.pants, side * 0.095, legH / 2, 0);
   316→      bake(t, new THREE.BoxGeometry(0.115, 0.07, 0.2), 0x1f1a14, side * 0.095, 0.035, 0.03);
   317→    }
   318→  }
   319→  return t;
   320→}

   322→/** Расстояние между отрезками AB и CD в 3D (для капсул фигур) */
   323→function segSegDist3D(a: THREE.Vector3, b: THREE.Vector3, c: THREE.Vector3, d: THREE.Vector3): number {
   324→  const u = b.clone().sub(a), v = d.clone().sub(c), w = a.clone().sub(c);
   325→  const uu = u.dot(u), vv = v.dot(v), uv = u.dot(v);
   326→  const wu = w.dot(u), wv = w.dot(v);
   327→  const den = uu * vv - uv * uv;
   328→  let s: number, t: number;
   329→  if (den < 1e-9) { s = 0; t = vv > 1e-9 ? clampN(wv / vv, 0, 1) : 0; }
   330→  else { s = clampN((uv * wv - vv * wu) / den, 0, 1); t = clampN((uv * wu - uu * wv) / den, 0, 1); }
   331→  // одна итерация уточнения (границы)
   332→  const t2 = clampN((uv * s + wv) / (vv > 1e-9 ? vv : 1), 0, 1);
   333→  const s2 = clampN((uv * t2 - wu) / (uu > 1e-9 ? uu : 1), 0, 1);
   334→  const p1 = a.clone().addScaledVector(u, s2);
   335→  const p2 = c.clone().addScaledVector(v, t2);
   336→  return p1.distanceTo(p2);
   337→}
   338→function clampN(v: number, lo: number, hi: number) { return v < lo ? lo : v > hi ? hi : v; }

   355→  /** Поражение экипажа вдоль пути снаряда (a→b в ЛОКАЛЬНЫХ координатах
   356→   *  корабля). grape — фигуры падают; ball — разрывает на куски.
   357→   *  Возвращает число убитых. */
   358→  killNearPath(a: THREE.Vector3, b: THREE.Vector3, mode: KillMode,
   359→    maxCount: number, radius: number,
   360→    shipPos: THREE.Vector3, shipQuat: THREE.Quaternion): number;
   361→  /** Оживить всех (полигон/сброс повреждений) */
   362→  reset(): void;
   363→}
   364→
   365→/** Собрать экипаж для корабля (local coords: +Z нос, +X порт).
   366→ *  Число фигур = maxCrew × доля на палубе ÷ CREW_PER_FIGURE —
   367→ *  от 1 фигуры на шлюпе до ~21 на линейном корабле */
   368→export function buildCrew(P: ShipPreset, seed: number): CrewVisual {
   369→  const group = new THREE.Group();
   370→  const L = P.length;
   371→  const deckYAt = (z: number) => {
   372→    const t = Math.min(1, Math.max(0, (z + L / 2) / L));
   373→    return hullProfile(t, P).bulwark - 1.0;
   374→  };
   375→  const halfW = (z: number) => {
   376→    const t = Math.min(1, Math.max(0, (z + L / 2) / L));
   377→    return hullProfile(t, P).w;
   378→  };
   379→  const figures: CrewFigure[] = [];
   380→  const place = (f: CrewFigure, x: number, z: number, faceDir: number) => {
   381→    f.group.position.set(x, deckYAt(z), z);
   382→    f.group.rotation.y = faceDir;
   383→    group.add(f.group);
   384→    figures.push(f);
   385→  };
   386→  const rot = mulberry32(seed * 7919 + 13);
   387→  let s = seed * 104729 + 7;
   388→  const nextSeed = () => { s = (s * 16807) % 2147483647; return s % 100000; };
   389→
   390→  // === СКОЛЬКО ЛЮДЕЙ НА ПАЛУБЕ: одна фигура = CREW_PER_FIGURE
   391→  // единиц; доля от численности зависит от размера судна ===
   392→  const deckShare = deckShareForPreset(P);
   393→  const nDeck = Math.max(1, Math.min(26, Math.round(P.maxCrew * deckShare / CREW_PER_FIGURE)));
   394→
   395→  const mastZ = P.masts === 2 ? [L * 0.16, -L * 0.10] : [L * 0.18, -L * 0.05, -L * 0.28];
   396→  const deckGuns = gunLayout(P).filter(sl => sl.tier === 0).map(sl => sl.z);
   397→  const wheelT = P.castleH ? 0.29 : 0.2;
   398→  const wheelZ = THREE.MathUtils.lerp(-L / 2, L / 2, wheelT);
   399→
   400→  // план ролей по приоритету: капитан → рулевой → боцман →
   401→  // канониры у орудий → вахтенные у мачт → ходоки
   402→  const plan: CrewRole[] = ['captain'];
   403→  if (nDeck >= 2) plan.push('helmsman');
   404→  if (nDeck >= 4) plan.push('boatswain');
   405→  const nRest = nDeck - plan.length;
   406→  const nGunners = Math.min(Math.floor(nRest * 0.5), deckGuns.length * 2, 8);
   407→  const nWatch = Math.min(Math.max(0, nRest - nGunners), P.masts, 3);
   408→  const nWalk = Math.max(0, nRest - nGunners - nWatch);
   409→  for (let i = 0; i < nGunners; i++) plan.push('gunner');
   410→  for (let i = 0; i < nWatch; i++) plan.push('watch');
   411→  for (let i = 0; i < nWalk; i++) plan.push('walker');
   412→
   413→  let gi = 0, wi = 0, ki = 0;
   414→  for (const role of plan) {
   415→    const f = buildFigure(role, rollTraits(rot, role), nextSeed());
   416→    if (role === 'captain') {
   417→      place(f, 0.55, -L * 0.245, 0.3);
   418→    } else if (role === 'helmsman') {
   419→      // рулевой — за штурвалом (руки на ободе), лицом к носу
   420→      place(f, 0, wheelZ - 0.52, 0);
   421→    } else if (role === 'boatswain') {
   422→      // боцман — у шпиля (шпиль на -L*0.16), лицом к носу-борту
   423→      place(f, -0.75, -L * 0.115, 0.5);
   424→    } else if (role === 'gunner') {
   425→      // канонир — за ПАЛУБНЫМ орудием (через одно), борт чередуется
   426→      const z = deckGuns[Math.floor(gi / 2) % Math.max(1, deckGuns.length)];
   427→      const side = gi % 2 === 0 ? -1 : 1;
   428→      gi++;
   429→      const w = halfW(z);
   430→      place(f, side * (w * 0.56), z + 0.55, side > 0 ? Math.PI / 2 : -Math.PI / 2);
   431→    } else if (role === 'watch') {
   432→      // вахтенный — у мачты, борт чередуется
   433→      const mz = mastZ[wi % mastZ.length];
   434→      const side = wi % 2 === 0 ? -1 : 1;
   435→      wi++;
   436→      place(f, side * Math.max(0.7, halfW(mz) * 0.42), mz + 0.9, side * (Math.PI / 2));
   437→    } else {
   438→      // ходок — гуляет по палубе туда-обратно
   439→      f.xPath = (ki % 2 ? 1 : -1) * Math.max(0.55, halfW(0) * 0.34);
   440→      f.walkA = P.castleH ? -L * 0.22 : -L * 0.3;
   441→      f.walkB = P.castleH ? L * 0.2 : L * 0.28;
   442→      f.walkT = rot();
   443→      ki++;
   444→      figures.push(f);
   445→      group.add(f.group);
   446→    }
   447→  }
   448→
   449→  group.traverse(o => { o.castShadow = false; o.receiveShadow = false; });
   450→  // масштаб по высоте применяем к группе (высота фигурки внутри)
   451→  for (const f of figures) f.group.scale.setScalar(1);
   452→
   453→  // --- убийство фигуры: картечь валит трупом, ядро разрывает ---
   454→  const killFigure = (f: CrewFigure, mode: KillMode,
   455→    a: THREE.Vector3, b: THREE.Vector3,
   456→    shipPos: THREE.Vector3, shipQuat: THREE.Quaternion) => {
   457→    f.dead = true;
   458→    f.deadT = 0;
   459→    f.deathMode = mode === 'grape' ? 'fall' : 'gib';
   460→    // мировые координаты центра тела
   461→    const lp = f.group.position;
   462→    const wpos = new THREE.Vector3(lp.x, lp.y + 0.75, lp.z).applyQuaternion(shipQuat).add(shipPos);
   463→    // мировое направление пути снаряда
   464→    const wdir = new THREE.Vector3(b.x - a.x, b.y - a.y, b.z - a.z);
   465→    if (wdir.lengthSq() < 1e-6) wdir.set(0, 0, 1);
   466→    wdir.applyQuaternion(shipQuat).normalize();
   467→
   468→    if (mode === 'grape') {
   469→      // картечь: осел — падает на палубу трупом
   470→      f.corpseLife = 28 + Math.random() * 27;
   471→      f.fallDir = Math.random() < 0.5 ? 1 : -1;
   472→      f.fallRoll = (Math.random() - 0.5) * 0.8;
   473→      FX.particles?.blood(wpos.x, wpos.y - 0.35, wpos.z, wdir.x * 1.5, 0.4, wdir.z * 1.5, 7, 0.7);
   474→    } else {
   475→      // ядро насквозь: фигуры больше нет — куски летят вдоль пути
   476→      f.group.visible = false;
   477→      f.corpseLife = 0;
   478→      const gibs = FX.gibs;
   479→      if (gibs) {
   480→        const mk = (sx: number, sy: number, sz: number, hex: number, extra = 1) => {
   481→          const push = (2.6 + Math.random() * 3.6) * extra;
   482→          gibs.spawnPart(
   483→            new THREE.Vector3(
   484→              wpos.x + (Math.random() - 0.5) * 0.35,
   485→              wpos.y + (Math.random() - 0.5) * 0.7,
   486→              wpos.z + (Math.random() - 0.5) * 0.35),
   487→            new THREE.Vector3(
   488→              wdir.x * push + (Math.random() - 0.5) * 3.4,
   489→              1.5 + Math.random() * 3.0,
   490→              wdir.z * push + (Math.random() - 0.5) * 3.4),
   491→            sx, sy, sz, hex, 1.5 + Math.random() * 1.1);
   492→        };
   493→        mk(0.14, 0.14, 0.14, f.skin);        // голова
   494→        mk(0.24, 0.2, 0.16, f.shirt, 1.15);  // торс
   495→        mk(0.09, 0.3, 0.09, f.shirt, 0.9);   // рука
   496→        mk(0.09, 0.28, 0.09, f.skin, 0.85);  // рука
   497→        mk(0.11, 0.44, 0.12, f.pants, 0.8);  // нога
   498→        if (f.hasPeg) mk(0.09, 0.48, 0.09, 0x8a6a42, 0.75); // деревянная нога
   499→        if (f.hasHook) mk(0.09, 0.09, 0.09, 0x3a3f46, 1.3); // крюк
   500→      }
   501→      FX.particles?.blood(wpos.x, wpos.y, wpos.z, wdir.x * 4, 1.1, wdir.z * 4, 16, 1.4);
   502→    }
   503→  };
   504→
   505→  const vis: CrewVisual = {
   506→    group, total: figures.length, shown: figures.length, dead: 0, deckShare,
   507→    setVisible(v: boolean) { group.visible = v; },
   508→    killNearPath(a, b, mode, maxCount, radius, shipPos, shipQuat) {
   509→      // ТОЧНОЕ поражение: путь снаряда (a→b, локальные координаты корабля)
   510→      // против вертикальной капсулы КАЖДОЙ фигуры (ноги → голова).
   511→      // Ядро/пуля картечи убивают только фигуру, которую ПРОШЁЛ снаряд —
   512→      // залп в борт рядом с матросом его не трогает.
   513→      const A = new THREE.Vector3(a.x, a.y, a.z);
   514→      const B = new THREE.Vector3(b.x, b.y, b.z);
   515→      const cands: { f: CrewFigure; d: number }[] = [];
   516→      for (const f of figures) {
   517→        if (f.dead || !f.group.visible) continue;
   518→        const p = f.group.position;
   519→        const lo = new THREE.Vector3(p.x, p.y + 0.12, p.z);
   520→        const hi = new THREE.Vector3(p.x, p.y + 1.62, p.z);
   521→        const d = segSegDist3D(A, B, lo, hi);
   522→        if (d > radius) continue;
   523→        cands.push({ f, d });
   524→      }
   525→      if (!cands.length) return 0;
   526→      cands.sort((u, v) => u.d - v.d);
   527→      const n = Math.min(maxCount, cands.length);
   528→      for (let i = 0; i < n; i++) killFigure(cands[i].f, mode, a, b, shipPos, shipQuat);
   529→      vis.dead = figures.filter(f => f.dead).length;
   530→      vis.shown = figures.filter(f => !f.dead && f.group.visible).length;
   531→      return n;
   532→    },
   533→    reset() {
   534→      for (const f of figures) {
   535→        f.dead = false;
   536→        f.deadT = 0;
   537→        f.corpseLife = 0;
   538→        f.deathMode = 'fall';
   539→        f.group.visible = true;
   540→        f.group.rotation.x = 0;
   541→        f.group.rotation.z = 0;
   542→        f.group.scale.setScalar(1);
   543→      }
   544→      vis.dead = 0;
   545→      vis.shown = figures.length;
   546→    },
   547→    update(dt, time, ship) {
   548→      if (ship.sunk) { group.visible = false; return; }
   549→      group.visible = !ship.sinking;
   550→      if (ship.sinking) return;
   551→
   552→      // --- трупы: падение и истлевание ---
   553→      for (const f of figures) {
   554→        if (!f.dead || f.deathMode !== 'fall') continue;
   555→        f.deadT += dt;
   556→        const k = Math.min(1, f.deadT / 0.5);
   557→        const e = 1 - (1 - k) * (1 - k); // easeOut
   558→        f.group.rotation.x = f.fallDir * (Math.PI / 2 - 0.07) * e;
   559→        f.group.rotation.z = f.fallRoll * e;
   560→        if (f.deadT > f.corpseLife) {
   561→          // истлел: усох за 1.2 с
   562→          const fade = Math.max(0, 1 - (f.deadT - f.corpseLife) / 1.2);
   563→          f.group.scale.setScalar(Math.max(0.01, fade));
   564→          if (fade <= 0) f.group.visible = false;
   565→        }
   566→      }
   567→
   568→      // --- потери команды: живых фигур РОВНО столько, сколько
   569→      // людей на палубе (crew × доля ÷ CREW_PER_FIGURE).
   570→      // crew = 0 → палуба ПУСТАЕТ (все фигуры погибают).
   571→      // Избыток умирает по одной за кадр — трупы падают естественно
   572→      const target = Math.max(0, Math.min(figures.length,
   573→        Math.round(Math.max(ship.crew, 0) * vis.deckShare / CREW_PER_FIGURE)));
   574→      const nAlive = figures.reduce((a, f) => a + (f.dead ? 0 : 1), 0);
   575→      const order: CrewRole[] = ['walker', 'watch', 'gunner', 'boatswain', 'helmsman', 'captain'];
   576→      let shownAlive = 0;
   577→      let excess: CrewFigure | null = null; // одна «потеря» за кадр
   578→      if (nAlive > target) {
   579→        outer:
   580→        for (const role of order) {
   581→          for (const f of figures) {
   582→            if (f.role !== role || f.dead) continue;
   583→            if (shownAlive < target) shownAlive++;
   584→            else { excess = f; break outer; }
   585→          }
   586→        }
   587→      } else shownAlive = nAlive;
   588→      if (excess) killFigure(excess, 'grape', excess.group.position,
   589→        excess.group.position.clone().add(new THREE.Vector3(0, 0, 1)),
   590→        ship.position, ship.quaternion);
   591→      vis.dead = figures.reduce((a, f) => a + (f.dead ? 1 : 0), 0);
   592→      vis.shown = shownAlive;
   593→
   594→      // --- анимация живых ---
   595→      for (const f of figures) {
   596→        if (!f.group.visible || f.dead) continue;
   597→        if (f.role === 'walker' && f.xPath !== undefined && f.walkA !== undefined && f.walkB !== undefined) {
   598→          const span = f.walkB - f.walkA;
   599→          f.walkT += (f.walkDir * f.walkSpeed * dt) / span;
   600→          if (f.walkT > 1) { f.walkT = 1; f.walkDir = -1; }
   601→          if (f.walkT < 0) { f.walkT = 0; f.walkDir = 1; }
   602→          const z = THREE.MathUtils.lerp(f.walkA, f.walkB, f.walkT);
   603→          f.group.position.set(f.xPath, deckYAt(z), z);
   604→          f.group.rotation.y = f.walkDir > 0 ? 0 : Math.PI;
   605→          const step = time * f.walkSpeed * 7;
   606→          if (f.legL && f.legR) {
   607→            f.legL.rotation.x = Math.sin(step) * 0.55;
   608→            f.legR.rotation.x = -Math.sin(step) * 0.55;
   609→          }
   610→          f.group.position.y += Math.abs(Math.sin(step)) * 0.03;
   611→        } else {
   612→          // idle: лёгкое покачивание, боцман «осматривается»
   613→          const sway = Math.sin(time * 1.3 + f.phase) * 0.035;
   614→          f.group.rotation.z = sway;
   615→          if (f.role === 'boatswain') {
   616→            f.group.rotation.y = 0.5 + Math.sin(time * 0.4 + f.phase) * 0.8;
   617→          } else if (f.role === 'gunner') {
   618→            f.group.rotation.z = sway * 0.5;
   619→          }
   620→        }
   621→      }
   622→    },
   623→  };
   624→  return vis;
   625→}
