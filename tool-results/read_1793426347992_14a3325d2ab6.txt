    12→
    13→export interface ShipPreset {
    14→  id: string;
    15→  name: string;
    16→  length: number; beam: number; draft: number; freeboard: number;
    17→  masts: 2 | 3;
    18→  gunsPerSide: number;
    19→  mass: number;          // кг
    20→  maxSpeed: number;      // м/с
    21→  hullHP: number; sailHP: number; maxCrew: number;
    22→  cargo: number;         // грузоподъёмность, т
    23→  colors: { hull: number; deck: number; trim: number; sail: number };
    24→  flagKind: FlagKind;
    25→  // --- классификация (энциклопедия, мини-карта) ---
    26→  category: 'war' | 'merchant';          // военный / торговый
    27→  shipClass: string;                     // класс/ранг судна
    28→  year?: number;                         // год постройки
    29→  country?: string;                      // страна
    30→  // --- архитектура XVI–XVIII веков ---
    31→  rig?: 'full' | 'single' | 'lateenMix' | 'lateen'; // полное / один парус на мачту (когг) / фок прямой + латинские (каравелла) / все латинские (шебека, галера)
    32→  gunportRows?: number;  // рядов пушечных портов по борту (1–4)
    33→  castleH?: number;      // высота надстроек-«замков» (когг)
    34→  flatBow?: boolean;     // пинасс: почти плоская носовая переборка
    35→  beakhead?: boolean;    // галеон: низкий бак с гальюном
    36→  gallery?: boolean;     // кормовая галерея с окнами
    37→  oars?: number;         // вёсел на борт (гребные суда: шебека, галера)
    38→  stripes?: { y0: number; y1: number; color: number }[]; // цветные полосы борта (доли свободного борта): «Victory stripe»
    39→}

   424→export function sailAddHole(s: SailInfo, u: number, v: number, r: number) {
   425→  const arr = s.mat.uniforms.uHoles.value as THREE.Vector3[];
   426→  const n = s.mat.uniforms.uHoleN.value as number;
   427→  if (n < arr.length) {
   428→    arr[n].set(u, v, r);
   429→    s.mat.uniforms.uHoleN.value = n + 1;
   430→  } else {
   431→    // парус уже изрешечён — рвём существующюю дыру шире
   432→    const h = arr[(Math.random() * arr.length) | 0];
   433→    h.z = Math.min(0.34, h.z + r * 0.6);
   434→  }

   437→// ---------- шейдер паруса ----------
   438→function sailMaterial(baseColor: number): THREE.ShaderMaterial {
   439→  return new THREE.ShaderMaterial({
   440→    uniforms: {
   441→      uBelly: { value: 0.5 },
   442→      uFlutter: { value: 0.3 },
   443→      uTime: { value: 0 },
   444→      uDamage: { value: 0 },
   445→      uColor: { value: new THREE.Color(baseColor) },
   446→      uSunDir: { value: new THREE.Vector3(0.4, 0.8, 0.3) },
   447→      uFurl: { value: 0 },
   448→      // пробоины от картечи/ядер/книппеля: (u, v, радиус в долях высоты)
   449→      uHoles: { value: Array.from({ length: 8 }, () => new THREE.Vector3(0, 0, 0)) },
   450→      uHoleN: { value: 0 },
   451→      uAspect: { value: 1 },      // ширина/высота — дыры круглые в метрах
   452→    },
   453→    vertexShader: /* glsl */`
   454→      uniform float uBelly, uTime, uFurl, uFlutter;
   455→      varying vec2 vUv;
   456→      varying vec3 vN;
   457→      varying float vDmgU;
   458→      uniform float uDamage;
   459→      void main() {
   460→        vUv = uv;
   461→        vDmgU = uDamage;
   462→        vec3 p = position;
   463→        // «пузо» паруса: синус по ширине и высоте, кромки закреплены.
   464→        // асимметрия по высоте: максимум ниже середины — наполненное
   465→        // полотно не выглядит плоским мешком. uBelly задан на CPU с
   466→        // учётом ветра и степени постановки
   467→        float belly = uBelly * sin(uv.x * 3.14159) * sin(uv.y * 3.14159)
   468→                    * (0.72 + 0.28 * (1.0 - uv.y));
   469→        // трепет свободной кромки: две волны, сильнее при слабом пузе (флюктер)
   470→        float flut = uFlutter * (
   471→          sin(uv.x * 9.0 + uTime * 7.3) * 0.6 +
   472→          sin(uv.x * 17.0 - uTime * 11.7) * 0.4
   473→        ) * sin(uv.y * 3.14159) * (1.0 - uv.y * 0.75);
   474→        float dz = belly + flut;
   475→        p.z += dz;
   476→        // нормаль аналитически (по пуза) — с полным наклоном, чтобы
   477→        // выпуклость читалась и в освещении, а не только в силуэте
   478→        float dzu = uBelly * cos(uv.x * 3.14159) * 3.14159 * sin(uv.y * 3.14159);
   479→        float dzv = uBelly * sin(uv.x * 3.14159) * 3.14159 * cos(uv.y * 3.14159);
   480→        vN = normalize(vec3(-dzu * 0.8, -dzv * 0.8, 1.0));
   481→        // рифление: при uFurl парус собирается к рее/штагу
   482→        p.y *= mix(1.0, 0.10, uFurl);
   483→        p.x *= mix(1.0, 0.55, uFurl);
   484→        p.z *= mix(1.0, 0.08, uFurl);
   485→        gl_Position = projectionMatrix * modelViewMatrix * vec4(p, 1.0);
   486→      }
   487→    `,
   488→    fragmentShader: /* glsl */`
   489→      varying vec2 vUv;
   490→      varying vec3 vN;
   491→      varying float vDmgU;
   492→      uniform vec3 uColor;
   493→      uniform vec3 uSunDir;
   494→      uniform vec3 uHoles[8];
   495→      uniform float uHoleN;
   496→      uniform float uAspect;
   497→      float hash12(vec2 p){ vec3 p3 = fract(vec3(p.xyx)*0.1031); p3+=dot(p3,p3.yzx+33.33); return fract((p3.x+p3.y)*p3.z); }
   498→      float vnoise(vec2 p){ vec2 i=floor(p), f=fract(p); vec2 u=f*f*(3.0-2.0*f);
   499→        return mix(mix(hash12(i),hash12(i+vec2(1,0)),u.x), mix(hash12(i+vec2(0,1)),hash12(i+vec2(1,1)),u.x), u.y); }
   500→      void main() {
   501→        vec3 col = uColor;
   502→        // швы (горизонтальные полотнища)
   503→        float seams = smoothstep(0.035, 0.0, abs(fract(vUv.y * 5.0) - 0.5) - 0.46);
   504→        col *= 1.0 - seams * 0.16;
   505→        // заплатки и рваные края от повреждений
   506→        float dmg = vDmgU;
   507→        float patches = step(0.55, vnoise(vUv * 4.7 + 7.3)) * dmg;
   508→        col = mix(col, col * 0.55, patches);
   509→        float holes = step(0.82 - dmg * 0.15, vnoise(vUv * 7.1 + 2.9)) * dmg * 1.4;
   510→        // ===== ПРОБОИНЫ от снарядов (картечь/ядро/книппель) =====
   511→        // рваный тёмный край + сквозная дыра (discard);
   512→        // uAspect приводит UV к метрам — дыры круглые
   513→        for (int i = 0; i < 8; i++) {
   514→          if (float(i) >= uHoleN) break;
   515→          vec2 hd = (vUv - uHoles[i].xy) * vec2(uAspect, 1.0);
   516→          float hdist = length(hd);
   517→          float n = vnoise(vUv * 17.0 + float(i) * 7.31);
   518→          float rr = uHoles[i].z * (0.78 + 0.5 * n);
   519→          if (hdist < rr) {
   520→            col *= 0.42;
   521→            if (hdist < rr * 0.58) discard;   // рваный сквозной проём
   522→          }
   523→        }
   524→        // простое освещение
   525→        float ndl = max(dot(normalize(vN), normalize(uSunDir)), 0.0) * 0.5 + 0.55;
   526→        col *= ndl;
   527→        float a = 1.0 - clamp(holes, 0.0, 0.9);
   528→        if (a < 0.55) discard; // дыры — без полупрозрачности (нет проблем сортировки)
   529→        gl_FragColor = vec4(col, 1.0);
   530→      }
   531→    `,
   532→    side: THREE.DoubleSide,
   533→  });
   534→}

   623→// ---------- корпус по шпангоутам ----------
   624→export function hullProfile(t: number, P: ShipPreset) {
   625→  // t: 0 корма -> 1 нос. Полуширина, осадка, высота фальшборта
   626→  const B = P.beam / 2;
   627→  let w: number;
   628→  if (t < 0.45) w = THREE.MathUtils.lerp(0.58, 1.0, smooth01(t / 0.45));
   629→  else w = THREE.MathUtils.lerp(1.0, 0.0, Math.pow(smooth01((t - 0.45) / 0.55), 0.75));
   630→  // пинасс: сужение только в самом конце — почти вертикальный плоский нос
   631→  if (P.flatBow && t > 0.82) w = Math.max(w, THREE.MathUtils.lerp(0.74, 0.42, smooth01((t - 0.82) / 0.18)));
   632→  const draft = P.draft * (0.72 + 0.28 * Math.sin(Math.PI * Math.min(1, t * 1.15)));
   633→  let bulwark: number;
   634→  if (P.beakhead) {
   635→    // галеон: высокая корма, низкий бак — характерный силуэт с гальюном
   636→    bulwark = P.freeboard * (0.85 + 0.36 * Math.pow(1 - t, 2.4) + 0.08 * Math.pow(Math.max(0, t - 0.7) / 0.3, 2));
   637→  } else {
   638→    bulwark = P.freeboard * (0.82 + 0.35 * Math.pow(Math.max(0, t - 0.75) / 0.25, 1.4) + 0.5 * Math.pow(1 - t, 2.2) * 0.6);
   639→  }
   640→  return { w: w * B, draft, bulwark };
   641→}
   642→function smooth01(x: number) { const t = Math.min(1, Math.max(0, x)); return t * t * (3 - 2 * t); }
   643→
   644→// ---------- раскладка орудий по палубам ----------
   645→/** Слот орудия: z — вдоль корпуса (м, локально), tier 0 = открытая
   646→ *  (верхнепалубная) батарея с лафетами, 1.. — закрытые орудийные деки,
   647→ *  чьи стволы смотрят наружу из пушечных портов. */
   648→export interface GunSlot { z: number; tier: number; y: number; }
   649→
   650→/** Раскладка ВСЕХ орудий борта (число всегда = gunsPerSide — отрисовка
   651→ *  соответствует количеству). Без gunportRows всё на палубе (шлюп/флейт);
   652→ *  1 ряд портов — палуба + одна закрытая батарея; 2–4 ряда — палуба +
   653→ *  (rows−1) деков, порт-ряды идут сверху вниз (0.62F … 0.28F). */
   654→export function gunLayout(P: ShipPreset): GunSlot[] {
   655→  const N = Math.max(0, Math.round(P.gunsPerSide));
   656→  if (N === 0) return [];
   657→  const L = P.length;
   658→  const rows = Math.max(0, Math.min(4, P.gunportRows ?? 0));
   659→  const tiers = rows === 0 ? 1 : Math.max(2, rows);
   660→  // распределение стволов по ярусам: поровну, остаток — верхним ярусам
   661→  const counts = new Array<number>(tiers).fill(0);
   662→  for (let i = 0; i < N; i++) counts[i % tiers]++;
   663→  const slots: GunSlot[] = [];
   664→  for (let tier = 0; tier < tiers; tier++) {
   665→    const n = counts[tier];
   666→    if (n <= 0) continue;
   667→    if (tier === 0) {
   668→      // открытая палуба: от кормы (не задевая надстройки) до полубака
   669→      const z0 = -L * 0.22, z1 = P.castleH ? L * 0.18 : L * 0.24;
   670→      for (let i = 0; i < n; i++) {
   671→        const z = THREE.MathUtils.lerp(z0, z1, n === 1 ? 0.5 : i / (n - 1));
   672→        slots.push({ z, tier: 0, y: 0 }); // y палубы считает потребитель
   673→      }
   674→    } else {
   675→      // закрытая батарея: высота порт-ряда (tier−1), чуть уже по длине —
   676→      // корпус в оконечностях сужается, порты там не ставили
   677→      const y = P.freeboard * (rows === 1 ? 0.5 : 0.62 - (tier - 1) * (0.34 / (rows - 1)));
   678→      const z0 = -L * 0.24, z1 = L * 0.18;
   679→      for (let i = 0; i < n; i++) {
   680→        const z = THREE.MathUtils.lerp(z0, z1, n === 1 ? 0.5 : i / (n - 1));
   681→        slots.push({ z, tier, y });
   682→      }
   683→    }
   684→  }
   685→  return slots;
   686→}

   688→function buildHull(P: ShipPreset): { geometry: THREE.BufferGeometry; outline: THREE.Vector2[] } {
   689→  const S = 26, R = 8;
   690→  const L = P.length;
   691→  const pos: number[] = [];
   692→  const col: number[] = [];
   693→  const uvs: number[] = [];
   694→  const cHull = new THREE.Color(P.colors.hull);
   695→  const cBelow = new THREE.Color(0x33402a); // подводная часть — обрастание
   696→  const cBoot = new THREE.Color(0xd8d2c2); // пояс ватерлинии
   697→  const cWale = new THREE.Color(P.colors.trim);
   698→  const tmp = new THREE.Color();
   699→  // цветные полосы борта (доли свободного борта) — «Victory stripe»
   700→  const F = P.freeboard;
   701→  const stripes = (P.stripes ?? []).map(st => ({
   702→    y0: st.y0 * F, y1: st.y1 * F, c: new THREE.Color(st.color),
   703→  }));
   704→  // высотные зоны МАСШТАБИРУЮТСЯ под высоту борта (у линейных кораблей
   705→  // ватерлинионный пояс и вельс не должны лепиться у самой палубы)
   706→  const bootTop = Math.max(0.42, F * 0.13);        // верх ватерлинионного пояса
   707→  const waleY0 = Math.max(0.9, F * 0.22);          // вельс (дополнительный пояс прочности)
   708→  const waleY1 = waleY0 + Math.max(0.45, F * 0.09);
   709→
   710→  const stations: { z: number; w: number; draft: number; bulwark: number }[] = [];
   711→  for (let s = 0; s <= S; s++) {
   712→    const t = s / S;
   713→    const p = hullProfile(t, P);
   714→    stations.push({ z: THREE.MathUtils.lerp(-L / 2, L / 2, t), w: p.w, draft: p.draft, bulwark: p.bulwark });
   715→  }
   716→
   717→  const pushPoint = (x: number, y: number, z: number) => {
   718→    pos.push(x, y, z);
   719→    // UV для PBR: u — вдоль корабля (планки по длине), v — по высоте;
   720→    // тайл = ширина корпуса — как у палубы, масштаб рельефа совпадает
   721→    uvs.push(z / P.beam, y / P.beam);
   722→    // раскраска по высоте: подводная часть -> ватерлинионный пояс ->
   723→    // (полосы пресета) -> вельс -> основной цвет борта
   724→    let stripeC: THREE.Color | null = null;
   725→    if (y > 0.1) {
   726→      for (const st of stripes) {
   727→        if (y >= st.y0 && y <= st.y1) { stripeC = st.c; break; }
   728→      }
   729→    }
   730→    if (y < -0.15) tmp.copy(cHull).lerp(cBelow, Math.min(1, (-y + 0.15) / P.draft));
   731→    else if (y < bootTop) tmp.copy(cBoot);
   732→    else if (stripeC) tmp.copy(stripeC);
   733→    else if (y > waleY0 && y < waleY1) tmp.copy(cWale);
   734→    else tmp.copy(cHull);
   735→    // лёгкая полосатость обшивки
   736→    const stripe = 1.0 - 0.06 * (Math.sin(y * 9.0) * 0.5 + 0.5);
   737→    col.push(tmp.r * stripe, tmp.g * stripe, tmp.b * stripe);
   738→  };
   739→
   740→  for (const st of stations) {
   741→    for (let r = 0; r <= R; r++) {
   742→      const u = r / R;
   743→      const y = THREE.MathUtils.lerp(-st.draft, st.bulwark, u);
   744→      const flare = Math.pow(u, 0.6);
   745→      const halfW = st.w * (0.16 + 0.84 * flare);
   746→      pushPoint(halfW, y, st.z);
   747→      pushPoint(-halfW, y, st.z);
   748→    }
   749→  }
   750→  const idx: number[] = [];
   751→  const rowLen = (R + 1) * 2;
   752→  for (let s = 0; s < S; s++) {
   753→    for (let r = 0; r < R; r++) {
   754→      const a = s * rowLen + r * 2, b = a + 1, c = a + 2, d = a + 3;
   755→      const e = a + rowLen, f = e + 1, g = e + 2, h = e + 3;
   756→      idx.push(a, c, e, c, g, e);
   757→      // зеркальная сторона (обратный порядок): два РАЗНЫХ треугольника квада —
   758→      // раньше (b,f,d) пушлся дважды (e+1===f!), второй (f,h,d) отсутствовал →
   759→      // левый борт был решетом из чередующихся треугольников
   760→      idx.push(b, f, d, f, h, d);
   761→    }
   762→  }
   763→  // транцевая корма: веер от центра к ПОСЛЕДОВАТЕЛЬНЫМ граничным вершинам
   764→  // (правые снизу вверх, затем левые сверху вниз) — без дыр между хордами.
   765→  // ВАЖНО: контур ЗАМКНУТ последним треугольником (левая-верх → правая-верх):
   766→  // раньше верхний клин транца не закрывался — сквозь корму «просвечивало»
   767→  const st0 = stations[0];
   768→  const centerIdx = pos.length / 3;
   769→  pushPoint(0, -st0.draft * 0.55, st0.z - 0.01);
   770→  const boundary: number[] = [];
   771→  for (let r = R; r >= 0; r--) boundary.push(r * 2);      // правые: верх → низ
   772→  for (let r = 0; r <= R; r++) boundary.push(r * 2 + 1);  // левые: низ → верх
   773→  boundary.push(boundary[0]);                             // замыкание контура
   774→  for (let i = 0; i < boundary.length - 1; i++) {
   775→    idx.push(centerIdx, boundary[i], boundary[i + 1]); // нормаль наружу (−Z)
   776→  }
   777→  // днище: замыкаем корпус снизу (r=0 — нижняя кромка шпангоутов),
   778→  // нормали вниз — при крене/под водой не «просвечивает»
   779→  for (let s = 0; s < S; s++) {
   780→    const a = s * rowLen + 1;        // левый борт, низ
   781→    const b = s * rowLen;            // правый борт, низ
   782→    const c = (s + 1) * rowLen + 1;
   783→    const d = (s + 1) * rowLen;
   784→    idx.push(a, b, c, b, d, c);
   785→  }
   786→  const g = new THREE.BufferGeometry();
   787→  g.setAttribute('position', new THREE.Float32BufferAttribute(pos, 3));
   788→  g.setAttribute('color', new THREE.Float32BufferAttribute(col, 3));
   789→  g.setAttribute('uv', new THREE.Float32BufferAttribute(uvs, 2));
   790→  g.setIndex(idx);
   791→  g.computeVertexNormals();
   792→
   793→  // контур ватерлинии
   794→  const outline: THREE.Vector2[] = [];
   795→  for (let s = 0; s <= S; s += 2) {
   796→    const st = stations[s];
   797→    const u = (st.draft) / (st.draft + st.bulwark); // u при y=0
   798→    const flare = Math.pow(u, 0.6);
   799→    const halfW = Math.max(0.25, st.w * (0.16 + 0.84 * flare));
   800→    outline.push(new THREE.Vector2(halfW, st.z));
   801→  }
   802→  for (let s = S; s >= 0; s -= 2) {
   803→    const st = stations[s];
   804→    const u = st.draft / (st.draft + st.bulwark);
   805→    const flare = Math.pow(u, 0.6);
   806→    const halfW = Math.max(0.25, st.w * (0.16 + 0.84 * flare));
   807→    outline.push(new THREE.Vector2(-halfW, st.z));
   808→  }
   809→  return { geometry: g, outline };
   810→}

   812→function buildDeck(P: ShipPreset): THREE.BufferGeometry {
   813→  // палуба: полоса по длине чуть ниже фальшборта, вплотную к обшивке.
   814→  // ВАЖНО: порядок вершин (a,b,c)(b,d,c) даёт нормали ВВЕРХ —
   815→  // иначе палуба отсекается backface-куллингом и «просвечивает».
   816→  // UV: u — поперёк (0..1 = полная ширина), v — вдоль (тайл = ширина),
   817→  // доски идут вдоль корпуса, у краёв — тёмный водосток (в текстуре).
   818→  const S = 20, L = P.length;
   819→  const pos: number[] = [], uv: number[] = [], idx: number[] = [];
   820→  for (let s = 0; s <= S; s++) {
   821→    const t = s / S;
   822→    const p = hullProfile(t, P);
   823→    const z = THREE.MathUtils.lerp(-L / 2, L / 2, t);
   824→    const deckY = p.bulwark - 1.0;
   825→    // полуширина корпуса на высоте палубы — та же формула, что в buildHull,
   826→    // чтобы палуба доходила до борта без щели
   827→    const u = Math.min(1, Math.max(0, (deckY + p.draft) / (p.draft + p.bulwark)));
   828→    const flare = Math.pow(u, 0.6);
   829→    const w = Math.max(0.15, p.w * (0.16 + 0.84 * flare) - 0.05);
   830→    pos.push(w, deckY, z, -w, deckY, z);
   831→    const vv = z / P.beam;
   832→    uv.push(0.5 + w / P.beam, vv, 0.5 - w / P.beam, vv);
   833→  }
   834→  for (let s = 0; s < S; s++) {
   835→    const a = s * 2, b = a + 1, c = a + 2, d = a + 3;
   836→    idx.push(a, b, c, b, d, c);
   837→  }
   838→  const g = new THREE.BufferGeometry();
   839→  g.setAttribute('position', new THREE.Float32BufferAttribute(pos, 3));
   840→  g.setAttribute('uv', new THREE.Float32BufferAttribute(uv, 2));
   841→  g.setIndex(idx);
   842→  g.computeVertexNormals();
   843→  return g;
   844→}

   846→/** Текстура палубных досок: конопаченые швы, разбежка стыков, гвозди,
   847→ *  волокна, потёртости; тёмные водосточные планки по краям (u≈0 и u≈1) */
   848→function makeDeckTexture(baseColor: number, seed: number): THREE.Texture {
   849→  const S = 512;
   850→  const c = document.createElement('canvas');
   851→  c.width = S; c.height = S;
   852→  const x = c.getContext('2d')!;
   853→  const base = new THREE.Color(baseColor);
   854→  let rs = seed * 2654435761 % 2147483647;
   855→  const rnd = () => { rs = (rs * 16807) % 2147483647; return rs / 2147483647; };
   856→  const N = 30, pw = S / N;
   857→  for (let i = 0; i < N; i++) {
   858→    const v = 0.84 + rnd() * 0.26;
   859→    x.fillStyle = `rgb(${Math.round(base.r * 255 * v)},${Math.round(base.g * 255 * v)},${Math.round(base.b * 255 * v)})`;
   860→    x.fillRect(i * pw, 0, pw + 1, S);
   861→    // волокна дерева
   862→    x.strokeStyle = 'rgba(30,18,8,0.10)';
   863→    x.lineWidth = 1;
   864→    for (let k = 0; k < 3; k++) {
   865→      const gx = i * pw + pw * (0.2 + 0.6 * rnd());
   866→      x.beginPath();
   867→      x.moveTo(gx, 0);
   868→      for (let y = 0; y <= S; y += 32) x.lineTo(gx + Math.sin(y * 0.045 + i * 7 + k * 3) * 2.2, y);
   869→      x.stroke();
   870→    }
   871→    // стыки досок с разбежкой + гвозди
   872→    const b1 = (i * 137 + 61) % S, b2 = (b1 + Math.round(S * 0.55)) % S;
   873→    for (const by of [b1, b2]) {
   874→      x.fillStyle = 'rgba(22,14,7,0.6)';
   875→      x.fillRect(i * pw, by, pw + 1, 2);
   876→      x.fillStyle = 'rgba(32,32,36,0.85)';
   877→      x.beginPath(); x.arc(i * pw + pw * 0.3, by + 5, 1.4, 0, 7); x.fill();
   878→      x.beginPath(); x.arc(i * pw + pw * 0.7, by - 3, 1.4, 0, 7); x.fill();
   879→    }
   880→    // конопаченый шов между досками
   881→    x.fillStyle = 'rgba(20,13,7,0.72)';
   882→    x.fillRect(i * pw, 0, 1.6, S);
   883→  }
   884→  // водосточный жёлоб по краям палубы (тёмные продольные планки)
   885→  for (const [x0, wW] of [[0, S * 0.075], [S * 0.925, S * 0.075]] as [number, number][]) {
   886→    const gr = x.createLinearGradient(x0, 0, x0 + wW, 0);
   887→    gr.addColorStop(0, 'rgba(24,15,7,0.5)');
   888→    gr.addColorStop(1, 'rgba(24,15,7,0.12)');
   889→    x.fillStyle = gr;
   890→    x.fillRect(x0, 0, wW, S);
   891→  }
   892→  // потёртости и разводы
   893→  for (let k = 0; k < 26; k++) {
   894→    const rx = rnd() * S, ry = rnd() * S, rr = 18 + rnd() * 55;
   895→    const rg = x.createRadialGradient(rx, ry, 0, rx, ry, rr);
   896→    rg.addColorStop(0, 'rgba(28,18,8,0.09)');
   897→    rg.addColorStop(1, 'rgba(0,0,0,0)');
   898→    x.fillStyle = rg;
   899→    x.fillRect(rx - rr, ry - rr, rr * 2, rr * 2);
   900→  }
   901→  const t = new THREE.CanvasTexture(c);
   902→  t.wrapS = t.wrapT = THREE.RepeatWrapping;
   903→  t.colorSpace = THREE.SRGBColorSpace;
   904→  t.anisotropy = 4;
   905→  return t;
   906→}

   908→/** Планширь: планка, закрывающая верхнюю кромку фальшборта */
   909→function buildCapRail(P: ShipPreset): THREE.BufferGeometry {
   910→  const S = 26, L = P.length;
   911→  const pos: number[] = [], col: number[] = [], idx: number[] = [], uvs: number[] = [];
   912→  const cRail = new THREE.Color(P.colors.trim).multiplyScalar(1.25);
   913→  const cRail2 = new THREE.Color(P.colors.trim);
   914→  const push = (px: number, py: number, z: number, c: THREE.Color) => {
   915→    pos.push(px, py, z); col.push(c.r, c.g, c.b);
   916→    // UV как у корпуса: планки по длине (u вдоль корабля)
   917→    uvs.push(z / P.beam, py / P.beam);
   918→  };
   919→  for (let s = 0; s <= S; s++) {
   920→    const t = s / S;
   921→    const p = hullProfile(t, P);
   922→    const z = THREE.MathUtils.lerp(-L / 2, L / 2, t);
   923→    const y = p.bulwark;
   924→    // внутренняя и внешняя кромки планки (лёгкий свес наружу)
   925→    push(p.w - 0.26, y + 0.07, z, cRail);
   926→    push(p.w + 0.13, y + 0.015, z, cRail2);
   927→    push(-p.w + 0.26, y + 0.07, z, cRail);
   928→    push(-p.w - 0.13, y + 0.015, z, cRail2);
   929→  }
   930→  for (let s = 0; s < S; s++) {
   931→    const a = s * 4;
   932→    // правый борт
   933→    idx.push(a, a + 4, a + 1, a + 1, a + 4, a + 5);
   934→    // левый борт
   935→    idx.push(a + 2, a + 3, a + 6, a + 3, a + 7, a + 6);

   937→  const g = new THREE.BufferGeometry();
   938→  g.setAttribute('position', new THREE.Float32BufferAttribute(pos, 3));
   939→  g.setAttribute('color', new THREE.Float32BufferAttribute(col, 3));
   940→  g.setAttribute('uv', new THREE.Float32BufferAttribute(uvs, 2));
   941→  g.setIndex(idx);
   942→  g.computeVertexNormals();
   943→  return g;
   944→}

   950→/** Запечь геометрию в общие массивы (для слияния реквизита в один меш) */
   951→function bakeInto(target: BakedTarget, geo: THREE.BufferGeometry,
   952→  color: THREE.Color, x: number, y: number, z: number, euler?: THREE.Euler) {
   953→  const g = geo.index ? geo.toNonIndexed() : geo;
   954→  const p = g.attributes.position as THREE.BufferAttribute;
   955→  const v = new THREE.Vector3();
   956→  for (let i = 0; i < p.count; i++) {
   957→    v.fromBufferAttribute(p, i);
   958→    if (euler) v.applyEuler(euler);
   959→    target.pos.push(v.x + x, v.y + y, v.z + z);
   960→    target.col.push(color.r, color.g, color.b);
   961→    // тайл ~1.5 м — нормали металла/дерева без пикселизации
   962→    target.uv.push((v.x + x + v.z + z) * 0.33, (v.y + y + v.z + z) * 0.33);
   963→  }
   964→}

   966→/** Палубный реквизит: решётчатый люк, бочки, канаты, кнехты, колокол,
   967→ *  ядра, сундук, ведро — всё сливается в ОДИН меш с vertex colors */
   968→function buildDeckProps(P: ShipPreset): THREE.BufferGeometry {
   969→  const L = P.length;
   970→  const t: BakedTarget = { pos: [], col: [], uv: [] };
   971→  const deckYAt = (tt: number) => hullProfile(tt, P).bulwark - 1.0;
   972→  const cWood = new THREE.Color(0x7a5c38);
   973→  const cWoodD = new THREE.Color(0x4a3420);
   974→  const cDark = new THREE.Color(P.colors.trim);
   975→  const cIron = new THREE.Color(0x24262b);
   976→  const cBrass = new THREE.Color(0xb8912f);
   977→  const cRope = new THREE.Color(0xa08656);
   978→  const cBarrel = new THREE.Color(0x8a6a42);
   979→
   980→  // --- решётчатый люк (гратинг) с коомингом ---
   981→  const hx = L * 0.08, hz = L * 0.05; // пол-размера (между мачтами)
   982→  const hy = deckYAt(0.48);
   983→  // кооминг (рамка)
   984→  bakeInto(t, new THREE.BoxGeometry(hx * 2 + 0.24, 0.22, 0.12), cWoodD, 0, hy + 0.11, hz);
   985→  bakeInto(t, new THREE.BoxGeometry(hx * 2 + 0.24, 0.22, 0.12), cWoodD, 0, hy + 0.11, -hz);
   986→  bakeInto(t, new THREE.BoxGeometry(0.12, 0.22, hz * 2), cWoodD, hx, hy + 0.11, 0);
   987→  bakeInto(t, new THREE.BoxGeometry(0.12, 0.22, hz * 2), cWoodD, -hx, hy + 0.11, 0);
   988→  // продольные планки
   989→  const nL = 9;
   990→  for (let i = 0; i < nL; i++) {
   991→    const px = -hx + 0.12 + (i / (nL - 1)) * (hx * 2 - 0.24);
   992→    bakeInto(t, new THREE.BoxGeometry(0.09, 0.07, hz * 2 - 0.1), cWood, px, hy + 0.24, 0);
   993→  }
   994→  // поперечные планки
   995→  const nT = 5;
   996→  for (let i = 0; i < nT; i++) {
   997→    const pz = -hz + 0.1 + (i / (nT - 1)) * (hz * 2 - 0.2);
   998→    bakeInto(t, new THREE.BoxGeometry(hx * 2 - 0.16, 0.07, 0.09), cWood, 0, hy + 0.30, pz);
   999→  }
  1000→
  1001→  // --- бочки (2 стоя, 1 лёжа) ---
  1002→  const barrelG = new THREE.CylinderGeometry(0.34, 0.34, 0.8, 10);
  1003→  const hoopG = new THREE.TorusGeometry(0.345, 0.022, 5, 12);
  1004→  const hoopE = new THREE.Euler(Math.PI / 2, 0, 0);
  1005→  const mkBarrel = (x: number, z: number, lying = false) => {
  1006→    const y = deckYAt(0.5 + z / L) + (lying ? 0.34 : 0.4);
  1007→    const e = lying ? new THREE.Euler(0, lying ? 0.7 : 0, Math.PI / 2) : undefined;
  1008→    bakeInto(t, barrelG, cBarrel, x, y, z, e);
  1009→    if (lying) {
  1010→      bakeInto(t, hoopG, cIron, x, y - 0.25, z, e);
  1011→      bakeInto(t, hoopG, cIron, x, y + 0.25, z, e);
  1012→    } else {
  1013→      bakeInto(t, hoopG, cIron, x, y - 0.28, z, hoopE);
  1014→      bakeInto(t, hoopG, cIron, x, y + 0.28, z, hoopE);
  1015→    }
  1016→  };
  1017→  mkBarrel(P.beam * 0.24, L * 0.10);
  1018→  mkBarrel(-P.beam * 0.26, L * 0.04);
  1019→  mkBarrel(P.beam * 0.22, -L * 0.12, true);
  1020→  // --- бухты канатов ---
  1021→  const coilG = new THREE.TorusGeometry(0.30, 0.10, 6, 14);
  1022→  const coilE = new THREE.Euler(Math.PI / 2, 0, 0);
  1023→  bakeInto(t, coilG, cRope, P.beam * 0.20, deckYAt(0.75) + 0.10, P.castleH ? L * 0.2 : L * 0.30, coilE);
  1024→  bakeInto(t, coilG, cRope, -P.beam * 0.18, deckYAt(0.62) + 0.10, L * 0.18, coilE);
  1025→  bakeInto(t, coilG, cRope, P.beam * 0.34, deckYAt(0.36) + 0.10, -L * 0.15, coilE);
  1026→
  1027→  // --- кнехты (битенги) с поперечиной у носа (на кораблях с «замками» нос занят надстройкой) ---
  1028→  const bittG = new THREE.BoxGeometry(0.15, 0.6, 0.15);
  1029→  if (!P.castleH) for (const side of [-1, 1]) {
  1030→    bakeInto(t, bittG, cWoodD, side * 0.42, deckYAt(0.85) + 0.3, L * 0.36);
  1031→    bakeInto(t, bittG, cWoodD, side * 0.42, deckYAt(0.85) + 0.3, L * 0.36 - 0.5);
  1032→    bakeInto(t, new THREE.BoxGeometry(0.13, 0.13, 0.75), cWoodD, side * 0.42, deckYAt(0.85) + 0.55, L * 0.36 - 0.25);
  1033→  }
  1034→
  1035→  // --- корабельный колокол у штурвала ---
  1036→  const bellZ = THREE.MathUtils.lerp(-L / 2, L / 2, 0.2) + 1.6;
  1037→  bakeInto(t, new THREE.BoxGeometry(0.12, 0.8, 0.12), cWoodD, 0.8, deckYAt(0.22) + 0.4, bellZ);
  1038→  bakeInto(t, new THREE.CylinderGeometry(0.05, 0.13, 0.2, 8), cBrass, 0.8, deckYAt(0.22) + 0.85, bellZ);
  1039→
  1040→  // --- пирамиды ядер у мачты ---
  1041→  const ballG = new THREE.SphereGeometry(0.105, 7, 6);
  1042→  const pile = (x: number, z: number) => {
  1043→    const y0 = deckYAt(0.5 + z / L);
  1044→    for (const [dx, dz] of [[-0.115, 0], [0, 0], [0.115, 0], [-0.058, 0.1], [0.058, 0.1], [0, 0.05]]) {
  1045→      bakeInto(t, ballG, cIron, x + dx, y0 + 0.105, z + dz);
  1046→    }
  1047→  };
  1048→  pile(0.85, -L * 0.16);
  1049→  pile(-P.beam * 0.20, L * 0.09);
  1050→
  1051→  // --- сундук с инструментом ---
  1052→  bakeInto(t, new THREE.BoxGeometry(0.72, 0.42, 0.5), cWoodD, -P.beam * 0.26, deckYAt(0.42) + 0.21, -L * 0.2);
  1053→  bakeInto(t, new THREE.BoxGeometry(0.78, 0.1, 0.56), cDark, -P.beam * 0.26, deckYAt(0.42) + 0.46, -L * 0.2);
  1054→
  1055→  // --- ведро ---
  1056→  bakeInto(t, new THREE.CylinderGeometry(0.17, 0.13, 0.3, 8, 1, true), cWood, -P.beam * 0.2, deckYAt(0.72) + 0.15, L * 0.24);
  1057→
  1058→  const g = new THREE.BufferGeometry();
  1059→  g.setAttribute('position', new THREE.Float32BufferAttribute(t.pos, 3));
  1060→  g.setAttribute('color', new THREE.Float32BufferAttribute(t.col, 3));
  1061→  g.computeVertexNormals();
  1062→  return g;
  1063→}

  1065→/** Полуширина корпуса на высоте y в шпации t (та же формула развала бортов, что в buildHull) */
  1066→function hullHalfWidthAt(P: ShipPreset, t: number, y: number): number {
  1067→  const p = hullProfile(t, P);
  1068→  const u = Math.min(1, Math.max(0, (y + p.draft) / (p.draft + p.bulwark)));
  1069→  const flare = Math.pow(u, 0.6);
  1070→  return p.w * (0.16 + 0.84 * flare);
  1071→}

  1083→/** Надстройки-«замки» когга: массивые кормовый и носовой — ганзейский силуэт */
  1084→function buildCastles(P: ShipPreset): THREE.BufferGeometry {
  1085→  const L = P.length;
  1086→  const t: BakedTarget = { pos: [], col: [], uv: [] };
  1087→  const cWall = new THREE.Color(P.colors.trim).multiplyScalar(1.15);
  1088→  const cWoodD = new THREE.Color(0x4a3420);
  1089→  const cRail = new THREE.Color(P.colors.trim).multiplyScalar(1.35);
  1090→  const ch = P.castleH || 0;
  1091→
  1092→  const mkCastle = (t0: number, t1: number, extra: number) => {
  1093→    const zA = THREE.MathUtils.lerp(-L / 2, L / 2, t0);
  1094→    const zB = THREE.MathUtils.lerp(-L / 2, L / 2, t1);
  1095→    const zc = (zA + zB) / 2, d = Math.abs(zB - zA);
  1096→    const prof = hullProfile((t0 + t1) / 2, P);
  1097→    const w = Math.max(1.2, prof.w * 0.92);
  1098→    const deckY = prof.bulwark - 1.0;
  1099→    const topY = deckY + extra;
  1100→    // массивная надстройка от палубы до площадки
  1101→    bakeInto(t, new THREE.BoxGeometry(w * 2, topY - deckY, d), cWall, 0, (deckY + topY) / 2, zc);
  1102→    // площадка чуть шире стен
  1103→    bakeInto(t, new THREE.BoxGeometry(w * 2 + 0.3, 0.22, d + 0.35), cWoodD, 0, topY + 0.11, zc);
  1104→    // стойки ограждения + продольный поручень
  1105→    for (const side of [-1, 1]) {
  1106→      for (let k = 0; k <= 4; k++) {
  1107→        const zz = zA + (k / 4) * (zB - zA);
  1108→        bakeInto(t, new THREE.BoxGeometry(0.12, 0.55, 0.12), cRail, side * (w + 0.05), topY + 0.5, zz);
  1109→      }
  1110→      bakeInto(t, new THREE.BoxGeometry(0.08, 0.08, d), cRail, side * (w + 0.05), topY + 0.76, zc);
  1111→    }
  1112→  };
  1113→  mkCastle(0.08, 0.24, ch);          // кормовый (ахтеркастель) — выше
  1114→  mkCastle(0.74, 0.88, ch * 0.72);   // носовой (бак) — ниже
  1115→  return bakedGeo(t);
  1116→}

  1118→/** Наружный декор корпуса: пушечные порты с вельсовыми полосами
  1119→ *  (галеон/фрегат), кормовая галерея с окнами, плоская носовая
  1120→ *  переборка пинасса, гальюн галеона */
  1121→function buildHullOrnaments(P: ShipPreset): THREE.BufferGeometry | null {
  1122→  const L = P.length;
  1123→  const t: BakedTarget = { pos: [], col: [], uv: [] };
  1124→  const cPort = new THREE.Color(0x17100d);
  1125→  const cGold = new THREE.Color(P.colors.trim).lerp(new THREE.Color(0xd8b45a), 0.55);
  1126→  const cGlass = new THREE.Color(0x1d2a33);
  1127→  const cWoodD = new THREE.Color(0x4a3420);
  1128→  let any = false;
  1129→
  1130→  // пушечные порты: РОВНО под стволами закрытых батарей (gunLayout) —
  1131→  // число портов соответствует числу орудий деку + золочёный вельс над батареей
  1132→  const rows = P.gunportRows || 0;
  1133→  if (rows > 0) {
  1134→    const layout = gunLayout(P).filter(s => s.tier > 0);
  1135→    const decks = [...new Set(layout.map(s => s.tier))];
  1136→    for (const tier of decks) {
  1137→      const y = layout.find(s => s.tier === tier)!.y;
  1138→      for (const side of [-1, 1]) {
  1139→        for (const slot of layout.filter(s => s.tier === tier)) {
  1140→          const tt = clamp2((slot.z + L / 2) / L, 0, 1);
  1141→          if (tt < 0.12 || tt > 0.86) continue;
  1142→          const hw = hullHalfWidthAt(P, tt, y);
  1143→          bakeInto(t, new THREE.BoxGeometry(0.06, Math.min(0.5, P.freeboard * 0.09), 0.62), cPort, side * (hw + 0.02), y, slot.z);
  1144→        }
  1145→        const S2 = 26;
  1146→        for (let s = 0; s < S2; s++) {
  1147→          const zA = THREE.MathUtils.lerp(-L * 0.32, L * 0.32, s / S2);
  1148→          const zB = THREE.MathUtils.lerp(-L * 0.32, L * 0.32, (s + 1) / S2);
  1149→          const tt = ((zA + zB) / 2 + L / 2) / L;
  1150→          if (tt < 0.1 || tt > 0.88) continue;
  1151→          const hw = hullHalfWidthAt(P, tt, y + 0.42);
  1152→          bakeInto(t, new THREE.BoxGeometry(0.05, 0.1, (zB - zA) * 1.05), cGold,
  1153→            side * (hw + 0.015), y + 0.42, (zA + zB) / 2);
  1154→        }
  1155→      }
  1156→    }
  1157→    any = true;
  1158→  }
  1159→
  1160→  // кормовая галерея: два яруса окон с золочёными рамами
  1161→  if (P.gallery) {
  1162→    for (const y of [P.freeboard * 0.3, P.freeboard * 0.64]) {
  1163→      for (const xoff of [-P.beam * 0.24, 0, P.beam * 0.24]) {
  1164→        bakeInto(t, new THREE.BoxGeometry(0.62, 0.54, 0.1), cGold, xoff, y, -L / 2 + 0.02);
  1165→        bakeInto(t, new THREE.BoxGeometry(0.46, 0.38, 0.12), cGlass, xoff, y, -L / 2 + 0.03);
  1166→      }
  1167→    }
  1168→    any = true;
  1169→  }
  1170→
  1171→  // пинасс: почти плоская носовая переборка с планкирём
  1172→  if (P.flatBow) {
  1173→    const tt = 0.93;
  1174→    const prof = hullProfile(tt, P);
  1175→    const w = prof.w * 0.96;
  1176→    const z = THREE.MathUtils.lerp(-L / 2, L / 2, tt);
  1177→    bakeInto(t, new THREE.BoxGeometry(w * 2, prof.bulwark + P.draft * 0.5, 0.16), cWoodD,
  1178→      0, (prof.bulwark - P.draft * 0.5) / 2, z);
  1179→    bakeInto(t, new THREE.BoxGeometry(w * 2 + 0.14, 0.14, 0.2),
  1180→      new THREE.Color(P.colors.trim).multiplyScalar(1.3), 0, prof.bulwark + 0.05, z);
  1181→    any = true;
  1182→  }
  1183→
  1184→  // галеон: гальюн — решётчатая площадка от низкого бака к бушприту
  1185→  if (P.beakhead) {
  1186→    const y = hullProfile(0.92, P).bulwark * 0.55;
  1187→    const z0 = L / 2 - 1.2, z1 = L / 2 + L * 0.1;
  1188→    for (let k = 0; k < 3; k++) {
  1189→      bakeInto(t, new THREE.BoxGeometry(1.7 - k * 0.5, 0.09, 0.09), cWoodD, 0, y,
  1190→        THREE.MathUtils.lerp(z0, z1, k / 2));
  1191→    }
  1192→    bakeInto(t, new THREE.BoxGeometry(0.09, 0.09, z1 - z0), cWoodD, -0.55, y, (z0 + z1) / 2);
  1193→    bakeInto(t, new THREE.BoxGeometry(0.09, 0.09, z1 - z0), cWoodD, 0.55, y, (z0 + z1) / 2);
  1194→    any = true;
  1195→  }
  1196→
  1197→  if (!any) return null;
  1198→  return bakedGeo(t);
  1199→}

  1201→/** Мелкие детали, видимые только вблизи (LOD): выбленки на вантах,
  1202→ *  шлюпка вверх дном, настоящие якоря у скул, крышки пушечных портов,
  1203→ *  камбуз с трубой, кранцы, стол экипажа, блоки у мачт */
  1204→function buildShipDetail(P: ShipPreset, mastDefs: { z: number; h: number }[]): THREE.Group {
  1205→  const L = P.length;
  1206→  const group = new THREE.Group();
  1207→  const deckYAt = (tt: number) => hullProfile(tt, P).bulwark - 1.0;
  1208→  const cWood = new THREE.Color(0x7a5c38);
  1209→  const cWoodD = new THREE.Color(0x4a3420);
  1210→  const cRope = new THREE.Color(0xa08656);
  1211→
  1212→  // --- ванты с выбленками (верёвочная лестница) — линиями ---
  1213→  const pts: number[] = [];
  1214→  const addLine = (a: THREE.Vector3, b: THREE.Vector3) => { pts.push(a.x, a.y, a.z, b.x, b.y, b.z); };
  1215→  for (const m of mastDefs) {
  1216→    for (const side of [-1, 1]) {
  1217→      const N = 5;
  1218→      const tops: THREE.Vector3[] = [], bases: THREE.Vector3[] = [];
  1219→      for (let k = 0; k < N; k++) {
  1220→        tops.push(new THREE.Vector3(0, m.h * (0.55 + k * 0.09), m.z));
  1221→        bases.push(new THREE.Vector3(side * 2.6, 1.6, m.z - 2.0 + k * 1.0));
  1222→      }
  1223→      for (let k = 0; k < N; k++) addLine(tops[k], bases[k]); // плотный веер вант
  1224→      for (let k = 0; k < N - 1; k++) {
  1225→        for (let f = 0.15; f <= 0.78; f += 0.09) {
  1226→          const a = new THREE.Vector3().lerpVectors(bases[k], tops[k], f);
  1227→          const b = new THREE.Vector3().lerpVectors(bases[k + 1], tops[k + 1], f);
  1228→          addLine(a, b); // выбленки
  1229→        }
  1230→      }
  1231→    }
  1232→  }
  1233→
  1234→  const t: BakedTarget = { pos: [], col: [], uv: [] };
  1235→  // металлическая выпечка (якоря): отдельный меш с PBR-металлом —
  1236→  // отражает небо, кованая фактура картой нормалей
  1237→  const tm: BakedTarget = { pos: [], col: [], uv: [] };
  1238→  const cAnchor = new THREE.Color(0x454a52); // светлый gunmetal (тёмный в metalness-пайплайне не бликует)
  1239→
  1240→  // --- шлюпка вверх дном у борта ---
  1241→  const boatG = new THREE.CapsuleGeometry(0.45, L * 0.14, 3, 8);
  1242→  boatG.rotateX(Math.PI / 2);
  1243→  boatG.scale(1, 0.5, 1);
  1244→  bakeInto(t, boatG, new THREE.Color(0x7d6446), P.beam * 0.16, deckYAt(0.63) + 0.22, L * 0.13);
  1245→  bakeInto(t, new THREE.BoxGeometry(0.08, 0.1, L * 0.3), cWoodD, P.beam * 0.16, deckYAt(0.63) + 0.44, L * 0.13);
  1246→
  1247→  // --- настоящие якоря на фалах у скул носа ---
  1248→  for (const side of [-1, 1]) {
  1249→    const z = L / 2 - L * 0.1;
  1250→    const prof = hullProfile(0.85, P);
  1251→    const ax = side * (prof.w + 0.3), ay = P.freeboard * 0.55;
  1252→    bakeInto(tm, new THREE.BoxGeometry(0.1, 1.35, 0.1), cAnchor, ax, ay, z);          // веретено
  1253→    bakeInto(t, new THREE.BoxGeometry(0.9, 0.11, 0.11), cWoodD, ax, ay + 0.6, z);  // шток
  1254→    const armG = new THREE.TorusGeometry(0.4, 0.05, 5, 10, Math.PI);
  1255→    armG.rotateZ(Math.PI); // нижняя полуокружность — рога
  1256→    bakeInto(tm, armG, cAnchor, ax, ay - 0.6, z);
  1257→    for (const s2 of [-1, 1]) { // лапы
  1258→      const fl = new THREE.ConeGeometry(0.15, 0.36, 4);
  1259→      fl.rotateZ(s2 * Math.PI / 2);
  1260→      bakeInto(tm, fl, cAnchor, ax + s2 * 0.42, ay - 0.55, z);
  1261→    }
  1262→    const ring = new THREE.TorusGeometry(0.14, 0.03, 5, 10);
  1263→    bakeInto(tm, ring, cAnchor, ax, ay + 0.75, z);                                    // рым
  1264→    addLine(new THREE.Vector3(side * prof.w * 0.7, prof.bulwark, z - 1.2),
  1265→      new THREE.Vector3(ax, ay + 0.75, z));                                        // фал
  1266→  }
  1267→
  1268→  // --- крышки пушечных портов (открытые, на верхних петлях) — только
  1269→  //     под ПАЛУБНЫМИ орудиями (закрытые деки стреляют из своих портов) ---
  1270→  for (const slot of gunLayout(P)) {
  1271→    if (slot.tier !== 0) continue;
  1272→    const tt = clamp2((slot.z + L / 2) / L, 0, 1);
  1273→    const y = hullProfile(tt, P).bulwark - 0.72;
  1274→    for (const side of [-1, 1]) {
  1275→      const hw = hullHalfWidthAt(P, tt, y);
  1276→      bakeInto(t, new THREE.BoxGeometry(0.06, 0.55, 0.6), cWoodD,
  1277→        side * (hw + 0.14), y + 0.3, slot.z, new THREE.Euler(0, 0, side * 0.55));
  1278→    }
  1279→  }
  1280→
  1281→  // --- камбуз (печь) с трубой ---
  1282→  bakeInto(t, new THREE.BoxGeometry(0.85, 0.75, 0.85), new THREE.Color(0x5c4034),
  1283→    -P.beam * 0.26, deckYAt(0.56) + 0.38, L * 0.06);
  1284→  bakeInto(t, new THREE.CylinderGeometry(0.09, 0.12, 1.5, 7), new THREE.Color(0x2b2d31),
  1285→    -P.beam * 0.26, deckYAt(0.56) + 1.35, L * 0.06);
  1286→
  1287→  // --- стол экипажа со скамьёй ---
  1288→  bakeInto(t, new THREE.BoxGeometry(1.5, 0.08, 0.7), cWood, P.beam * 0.12, deckYAt(0.4) + 0.62, -L * 0.1);
  1289→  for (const [lx, lz] of [[-0.65, -0.28], [0.65, -0.28], [-0.65, 0.28], [0.65, 0.28]] as [number, number][]) {
  1290→    bakeInto(t, new THREE.CylinderGeometry(0.045, 0.045, 0.62, 5), cWoodD,
  1291→      P.beam * 0.12 + lx, deckYAt(0.4) + 0.31, -L * 0.1 + lz);
  1292→  }
  1293→  bakeInto(t, new THREE.BoxGeometry(1.7, 0.09, 0.34), cWood, P.beam * 0.12, deckYAt(0.4) + 0.28, -L * 0.1 + 0.75);
  1294→
  1295→  // --- кранцы: канатные шары на фалинях у планширя ---
  1296→  for (let k = 0; k < 3; k++) {
  1297→    const z = L * (0.08 + k * 0.2);
  1298→    const prof = hullProfile(0.5 + z / L, P);
  1299→    for (const side of [-1, 1]) {
  1300→      const hx = side * (prof.w + 0.06), hy = prof.bulwark + 0.28;
  1301→      bakeInto(t, new THREE.SphereGeometry(0.13, 6, 5), cRope, hx, hy, z);
  1302→      addLine(new THREE.Vector3(side * prof.w * 0.92, prof.bulwark + 0.05, z),
  1303→        new THREE.Vector3(hx, hy, z));
  1304→    }
  1305→  }
  1306→
  1307→  // --- блоки у мачт ---
  1308→  for (const md of mastDefs) {
  1309→    bakeInto(t, new THREE.CylinderGeometry(0.09, 0.09, 0.14, 6), cWoodD, 0.35, deckYAt(0.5 + md.z / L) + 0.35, md.z - 0.6);
  1310→    bakeInto(t, new THREE.CylinderGeometry(0.09, 0.09, 0.14, 6), cWoodD, -0.35, deckYAt(0.5 + md.z / L) + 0.35, md.z - 0.6);
  1311→  }
  1312→
  1313→  const lg = new THREE.BufferGeometry();
  1314→  lg.setAttribute('position', new THREE.Float32BufferAttribute(pts, 3));
  1315→  const lines = new THREE.LineSegments(lg, new THREE.LineBasicMaterial({ color: 0x241c12, transparent: true, opacity: 0.9 }));
  1316→  group.add(lines);
  1317→  const mesh = new THREE.Mesh(bakedGeo(t),
  1318→    new THREE.MeshStandardMaterial({
  1319→      vertexColors: true, metalness: 0.2, side: THREE.DoubleSide,
  1320→      normalMap: woodNormalTex('spar'), roughnessMap: woodRoughTex('spar'),
  1321→      normalScale: new THREE.Vector2(0.35, 0.35), roughness: 1,
  1322→    }));
  1323→  group.add(mesh);
  1324→  // якоря — кованое железо: PBR-металл с картой окружения неба
  1325→  if (tm.pos.length) {
  1326→    const mm = new THREE.Mesh(bakedGeo(tm),
  1327→      new THREE.MeshStandardMaterial({
  1328→        vertexColors: true, metalness: 0.88,
  1329→        normalMap: metalNormalTex(), roughnessMap: metalRoughTex(),
  1330→        normalScale: new THREE.Vector2(0.5, 0.5), roughness: 1,
  1331→      }));
  1332→    group.add(mm);
  1333→  }
  1334→  return group;
  1335→}

  1337→/** Вёсла гребных судов (шебека, галера): вальные порты по борту +
  1338→ *  вёсла с лопастями. ВАЖНО: вал строится ВНАРУЖУ от своего борта
  1339→ *  (dir = side) — раньше оба борта строились в +X и левые вёсла
  1340→ *  торчали СКВОЗЬ корпус на правый край.
  1341→ *  УБРАНОЕ положение (stow): вал лежит ВДОЛЬ борта валом к корме
  1342→ *  (как на стоящей галере) — НЕ вертикально вверх («торчащие
  1343→ *  вверх палки» из фидбека). Разворот задан кватернионом:
  1344→ *  Ship.updateVisual slerp'ит stow → развёрнутое (валом к воде) */
  1345→function buildOars(P: ShipPreset): { group: THREE.Group; oars: THREE.Object3D[] } {
  1346→  const group = new THREE.Group();
  1347→  const oars: THREE.Object3D[] = [];
  1348→  const n = P.oars || 0;
  1349→  if (!n) return { group, oars };
  1350→  const L = P.length;
  1351→  const shaftMat = new THREE.MeshStandardMaterial({
  1352→    color: 0x8a6a42,
  1353→    normalMap: woodNormalTex('spar'), roughnessMap: woodRoughTex('spar'),
  1354→    normalScale: new THREE.Vector2(0.45, 0.45), roughness: 1,
  1355→  });
  1356→  const portMat = new THREE.MeshStandardMaterial({ color: 0x14100d, roughness: 0.9 });
  1357→  const oarLen = clamp2(L * 0.17, 4.5, 11);
  1358→  const z0 = -L * 0.26, z1 = L * 0.30;
  1359→  for (let i = 0; i < n; i++) {
  1360→    const t = n === 1 ? 0.5 : i / (n - 1);
  1361→    const z = THREE.MathUtils.lerp(z0, z1, t);
  1362→    const tt = clamp2((z + L / 2) / L, 0, 1);
  1363→    const y = P.freeboard * 0.30; // вальная палуба — ниже орудийной
  1364→    const hw = hullHalfWidthAt(P, tt, y);
  1365→    for (const side of [-1, 1]) {
  1366→      // тёмный вальный порт
  1367→      const port = new THREE.Mesh(new THREE.BoxGeometry(0.06, 0.26, 0.3), portMat);
  1368→      port.position.set(side * (hw + 0.01), y, z);
  1369→      group.add(port);
  1370→      // весло: вал + лопасть — ВНАРУЖУ борта (side), наклон к воде
  1371→      const dir = side;
  1372→      const oar = new THREE.Group();
  1373→      const shaft = new THREE.Mesh(new THREE.CylinderGeometry(0.05, 0.065, oarLen, 5), shaftMat);
  1374→      shaft.rotation.z = Math.PI / 2;
  1375→      shaft.position.x = dir * oarLen * 0.42;
  1376→      oar.add(shaft);
  1377→      const blade = new THREE.Mesh(new THREE.BoxGeometry(oarLen * 0.3, 0.03, 0.2), shaftMat);
  1378→      blade.position.x = dir * oarLen * 0.92;
  1379→      oar.add(blade);
  1380→      oar.position.set(side * (hw + 0.04), y, z);
  1381→      // УБРАНО: вал вдоль борта к корме (yaw ~80°), чуть приподнят.
  1382→      // Раньше rotation.z = side*1.52 ставил вёсла ВЕРТИКАЛЬНО вверх
  1383→      const stowQ = new THREE.Quaternion().setFromEuler(
  1384→        new THREE.Euler(0, side * 1.40, side * 0.12));
  1385→      oar.quaternion.copy(stowQ);
  1386→      oar.userData.phase = t * Math.PI * 2 + (side > 0 ? Math.PI : 0);
  1387→      oar.userData.side = side;
  1388→      oar.userData.base = side * -0.34; // развёрнуто: вал наружу, к воде
  1389→      oar.userData.stowQ = stowQ;
  1390→      group.add(oar);
  1391→      oars.push(oar);
  1392→    }
  1393→  }
  1394→  return { group, oars };
  1395→}
  1396→function clamp2(v: number, a: number, b: number) { return v < a ? a : v > b ? b : v; }

  1399→/** Одна снасть обвязки: ванта/штаг/бакштаг. Принадлежит своей мачте —
  1400→ *  при её ломке обвязка ОБРЫВАЕТСЯ (Ship.snapRigging) */
  1401→export interface RigLine {
  1402→  mast: number;                        // индекс мачты-владельца
  1403→  a: THREE.Vector3; b: THREE.Vector3;  // оригинальные концы (для восстановления)
  1404→}
  1405→export interface Rigging {
  1406→  lines: THREE.LineSegments;
  1407→  meta: RigLine[];
  1408→}
  1409→function buildRigging(P: ShipPreset, masts: { z: number; h: number }[], bowspritTip: THREE.Vector3): Rigging {
  1410→  const meta: RigLine[] = [];
  1411→  const L = P.length;
  1412→  const addLine = (mast: number, a: THREE.Vector3, b: THREE.Vector3) => {
  1413→    meta.push({ mast, a, b });
  1414→  };
  1415→  for (let mi = 0; mi < masts.length; mi++) {
  1416→    const m = masts[mi];
  1417→    const top = new THREE.Vector3(0, m.h, m.z);
  1418→    // штаги к носу и корме
  1419→    addLine(mi, top.clone(), bowspritTip.clone());
  1420→    addLine(mi, top.clone(), new THREE.Vector3(0, 1.2, -L / 2 + 0.4));
  1421→    // ванты к бортам
  1422→    for (const side of [-1, 1]) {
  1423→      for (let k = 0; k < 3; k++) {
  1424→        const zz = m.z - 1.5 + k * 1.5;
  1425→        addLine(mi, new THREE.Vector3(side * 2.6, 1.6, zz),
  1426→          new THREE.Vector3(0, m.h * (0.55 + k * 0.15), m.z));
  1427→      }
  1428→    }
  1429→  }
  1430→  // бакштаги к корме от грот-мачты
  1431→  if (masts.length >= 2) {
  1432→    const mainIdx = masts.length - 1 > 1 ? masts.length - 2 : 0;
  1433→    const main = masts[mainIdx];
  1434→    addLine(mainIdx, new THREE.Vector3(0, main.h * 0.8, main.z), new THREE.Vector3(2.4, 1.4, -L / 2 + 0.3));
  1435→    addLine(mainIdx, new THREE.Vector3(0, main.h * 0.8, main.z), new THREE.Vector3(-2.4, 1.4, -L / 2 + 0.3));
  1436→  }
  1437→  const pts: number[] = [];
  1438→  for (const l of meta) pts.push(l.a.x, l.a.y, l.a.z, l.b.x, l.b.y, l.b.z);
  1439→  const g = new THREE.BufferGeometry();
  1440→  g.setAttribute('position', new THREE.Float32BufferAttribute(pts, 3));
  1441→  const mat = new THREE.LineBasicMaterial({ color: 0x241c12, transparent: true, opacity: 0.85 });
  1442→  return { lines: new THREE.LineSegments(g, mat), meta };
  1443→}

  1446→export function buildShip(P: ShipPreset): ShipVisual {
  1447→  const group = new THREE.Group();
  1448→  const L = P.length;
  1449→
  1450→  // корпус
  1451→  const { geometry: hullGeo, outline } = buildHull(P);
  1452→  // DoubleSide: даже если камера (или волна) заглянет под корпус/за корму —
  1453→  // стенки видны, «прозрачной кормы» нет (изнутри односторонние грани куллятся)
  1454→  // PBR: пояс обшивки + волокна — нормали, крашеное дерево — шероховатость
  1455→  const hullMat = new THREE.MeshStandardMaterial({
  1456→    vertexColors: true, metalness: 0.05, side: THREE.DoubleSide,
  1457→    normalMap: woodNormalTex('hull'), roughnessMap: woodRoughTex('hull'),
  1458→    normalScale: new THREE.Vector2(0.6, 0.6), roughness: 1,
  1459→  });

  1461→  hull.castShadow = true;
  1462→  group.add(hull);
  1463→
  1464→  // палуба: текстура досок с конопачением (DoubleSide — страховка от куллинга);
  1465→  // нормали канавок/стыков/гвоздей СОВПАДАЮТ с нарисованными досками
  1466→  const deckMat = new THREE.MeshStandardMaterial({
  1467→    map: makeDeckTexture(P.colors.deck, P.colors.deck + P.length),
  1468→    normalMap: woodNormalTex('deck'), roughnessMap: woodRoughTex('deck'),
  1469→    normalScale: new THREE.Vector2(0.5, 0.5), roughness: 1,
  1470→    side: THREE.DoubleSide,
  1471→  });
  1472→  const deck = new THREE.Mesh(buildDeck(P), deckMat);
  1473→  group.add(deck);
  1474→
  1475→  // планширь по верхней кромке фальшборта
  1476→  const capRail = new THREE.Mesh(buildCapRail(P),
  1477→    new THREE.MeshStandardMaterial({
  1478→      vertexColors: true, metalness: 0.05,
  1479→      normalMap: woodNormalTex('hull'), roughnessMap: woodRoughTex('hull'),
  1480→      normalScale: new THREE.Vector2(0.4, 0.4), roughness: 1,
  1481→    }));
  1482→  group.add(capRail);
  1483→
  1484→  // рангоут — оструганные брёвна: вертикальные волокна;
  1485→  // тёмная окраска (darkMat) — крашеное дерево: рельеф обшивки мягче
  1486→  const woodMat = new THREE.MeshStandardMaterial({
  1487→    color: 0x6e5433,
  1488→    normalMap: woodNormalTex('spar'), roughnessMap: woodRoughTex('spar'),
  1489→    normalScale: new THREE.Vector2(0.5, 0.5), roughness: 1,
  1490→  });
  1491→  const darkMat = new THREE.MeshStandardMaterial({
  1492→    color: P.colors.trim,
  1493→    normalMap: woodNormalTex('hull'), roughnessMap: woodRoughTex('hull'),
  1494→    normalScale: new THREE.Vector2(0.35, 0.35), roughness: 1,
  1495→  });
  1496→
  1497→  // бушприт
  1498→  const bowsprit = new THREE.Mesh(new THREE.CylinderGeometry(0.09, 0.16, L * 0.32, 6), woodMat);
  1499→  bowsprit.rotation.x = Math.PI / 2 - 0.18;
  1500→  bowsprit.position.set(0, P.freeboard * 0.75, L / 2 + L * 0.12);
  1501→  group.add(bowsprit);
  1502→  const bowspritTip = new THREE.Vector3(0, P.freeboard * 0.75 + L * 0.32 * 0.18, L / 2 + L * 0.12 + L * 0.32 * Math.cos(0.18) * 0.98);
  1503→
  1504→  // мачты
  1505→  const mastDefs: { z: number; h: number }[] = [];
  1506→  const H = P.freeboard + L * 0.78;
  1507→  if (P.masts === 2) {
  1508→    mastDefs.push({ z: L * 0.16, h: H * 0.88 });
  1509→    mastDefs.push({ z: -L * 0.10, h: H });
  1510→  } else {
  1511→    mastDefs.push({ z: L * 0.18, h: H * 0.8 });
  1512→    mastDefs.push({ z: -L * 0.05, h: H });
  1513→    mastDefs.push({ z: -L * 0.28, h: H * 0.62 });
  1514→  }
  1515→  const masts: THREE.Object3D[] = [];
  1516→  for (const md of mastDefs) {
  1517→    const mast = new THREE.Group();
  1518→    const pole = new THREE.Mesh(new THREE.CylinderGeometry(0.11, 0.24, md.h, 7), woodMat);
  1519→    pole.position.y = md.h / 2 - P.draft * 0.3;
  1520→    pole.castShadow = true;
  1521→    mast.add(pole);
  1522→    // марс (платформа)
  1523→    const top = new THREE.Mesh(new THREE.CylinderGeometry(0.75, 0.6, 0.28, 8), darkMat);
  1524→    top.position.y = md.h * 0.72 - P.draft * 0.3;
  1525→    mast.add(top);
  1526→    mast.position.set(0, -P.draft * 0.3 > 0 ? 0 : 0, md.z);
  1527→    mast.userData.height = md.h;
  1528→    masts.push(mast);
  1529→    group.add(mast);
  1530→  }
  1531→
  1532→  // паруса: каждый со своим материалом (своё рифление), квадратные —
  1533→  // на поворотной группе «рея+парус» для брасопки по ветру
  1534→  const sails: SailInfo[] = [];
  1535→  const addSquareSail = (mastIndex: number, level: number, width: number, height: number, setAt: number) => {
  1536→    const md = mastDefs[mastIndex];
  1537→    const g = new THREE.PlaneGeometry(width, height, 10, 7);
  1538→    g.translate(0, -height / 2, 0); // верхняя кромка у реи
  1539→    const mat = sailMaterial(P.colors.sail);
  1540→    mat.uniforms.uAspect.value = width / height;
  1541→    const mesh = new THREE.Mesh(g, mat);
  1542→    const y = md.h * level - P.draft * 0.3;
  1543→    // поворотная группа: ось вращения — мачта (брасопка рея по ветру)
  1544→    // рея+парус — потомок МАЧТЫ: сломанная мачта валится вместе с ними
  1545→    const pivot = new THREE.Group();
  1546→    pivot.position.set(0, y + 0.05, 0);
  1547→    // рея
  1548→    const yard = new THREE.Mesh(new THREE.CylinderGeometry(0.07, 0.07, width * 1.04, 5), woodMat);
  1549→    yard.rotation.z = Math.PI / 2;
  1550→    pivot.add(yard);
  1551→    mesh.position.set(0, 0, -0.25);
  1552→    pivot.add(mesh);
  1553→    masts[mastIndex].add(pivot);
  1554→    // зона прострела: прямоугольник полотна в координатах корабля (brace=0);
  1555→    // при проверке поворачивается на текущий угол брасопки вокруг мачты
  1556→    const quad: SailHitQuad = {
  1557→      pivot: new THREE.Vector3(0, y + 0.05, md.z),
  1558→      corners: [
  1559→        new THREE.Vector3(-width / 2, -height, -0.25), // uv(0,0) — нижний-лево
  1560→        new THREE.Vector3(width / 2, -height, -0.25),  // uv(1,0)
  1561→        new THREE.Vector3(width / 2, 0, -0.25),        // uv(1,1)
  1562→        new THREE.Vector3(-width / 2, 0, -0.25),       // uv(0,1)
  1563→      ],
  1564→      uvs: [
  1565→        new THREE.Vector2(0, 0), new THREE.Vector2(1, 0),
  1566→        new THREE.Vector2(1, 1), new THREE.Vector2(0, 1),
  1567→      ],
  1568→      rotates: true,
  1569→    };
  1570→    sails.push({ mesh, mat, brace: pivot, setAt, furl: 1, square: true, braceAngle: 0, mastIndex, quad, holeW: width, holeH: height });
  1571→  };
  1572→  // латинский парус (каравелла): длинная рейка от мачты к носу,
  1573→  // треугольное полотно в диаметральной плоскости, «пузо» — на борт
  1574→  const addLateenSail = (mastIndex: number, setAt: number, yLen: number, drop: number, aftFrac: number) => {
  1575→    const md = mastDefs[mastIndex];
  1576→    // локальная плоскость XY: +X -> нос, +Y вверх; поворот меша
  1577→    // развернёт её в диаметральную, «пузо» (локальный +Z) — на борт
  1578→    const H = { x: 0, y: 0 };
  1579→    const T = { x: yLen, y: -drop * 0.62 };        // нижний конец рея — вперёд и вниз
  1580→    const C = { x: -yLen * aftFrac, y: -drop };     // шкотовый угол — к корме
  1581→    const NU = 9, NV = 6;
  1582→    const pos: number[] = [], uvs: number[] = [], idx: number[] = [];
  1583→    for (let i = 0; i <= NU; i++) {
  1584→      const u = i / NU;
  1585→      const ex = THREE.MathUtils.lerp(H.x, T.x, u), ey = THREE.MathUtils.lerp(H.y, T.y, u);
  1586→      const fx = THREE.MathUtils.lerp(H.x, C.x, u), fy = THREE.MathUtils.lerp(H.y, C.y, u);
  1587→      for (let j = 0; j <= NV; j++) {
  1588→        const v = j / NV;
  1589→        pos.push(THREE.MathUtils.lerp(ex, fx, v), THREE.MathUtils.lerp(ey, fy, v), 0);
  1590→        uvs.push(u, v);
  1591→      }
  1592→    }
  1593→    const row = NV + 1;
  1594→    for (let i = 0; i < NU; i++) {
  1595→      for (let j = 0; j < NV; j++) {
  1596→        const p0 = i * row + j, p1 = p0 + 1, p2 = p0 + row, p3 = p2 + 1;
  1597→        idx.push(p0, p1, p2, p1, p3, p2);
  1598→      }
  1599→    }
  1600→    const g = new THREE.BufferGeometry();
  1601→    g.setAttribute('position', new THREE.Float32BufferAttribute(pos, 3));
  1602→    g.setAttribute('uv', new THREE.Float32BufferAttribute(uvs, 2));
  1603→    g.setIndex(idx);
  1604→    g.computeVertexNormals();
  1605→    const mat = sailMaterial(P.colors.sail);
  1606→    // зона прострела латинского паруса — треугольник H(голова)-T(конец реи)-C(шкот):
  1607→    // в локалях меша +X -> нос после поворота меша на -90° вокруг Y
  1608→    const yardLenM = Math.hypot(T.x - H.x, T.y - H.y);
  1609→    mat.uniforms.uAspect.value = Math.max(0.3, (yardLenM + Math.abs(C.x)) / Math.max(1, drop));
  1610→    const mesh = new THREE.Mesh(g, mat);
  1611→    mesh.rotation.y = -Math.PI / 2; // локальный +X -> нос, «пузо» -> борт
  1612→    mesh.frustumCulled = false;
  1613→    const pivot = new THREE.Group();
  1614→    pivot.position.set(0, md.h * 0.72 - P.draft * 0.3, 0);
  1615→    // рея: от головы у мачты к носу-вниз
  1616→    const yardLen = Math.hypot(T.x - H.x, T.y - H.y) * 1.06;
  1617→    const yard = new THREE.Mesh(new THREE.CylinderGeometry(0.055, 0.085, yardLen, 5), woodMat);
  1618→    yard.rotation.z = Math.atan2(T.y - H.y, T.x - H.x) - Math.PI / 2;
  1619→    yard.position.set(T.x / 2, T.y / 2, 0);
  1620→    pivot.add(yard);
  1621→    pivot.add(mesh);
  1622→    masts[mastIndex].add(pivot);
  1623→    // поворот меша -90° вокруг Y: (x, y, 0) -> (0, y, x)
  1624→    const rotP = (x: number, y: number) => new THREE.Vector3(0, y, x);
  1625→    const pv = new THREE.Vector3(0, md.h * 0.72 - P.draft * 0.3, md.z);
  1626→    const quad: SailHitQuad = {
  1627→      pivot: pv,
  1628→      corners: [rotP(H.x, H.y), rotP(T.x, T.y), rotP(C.x, C.y), rotP(C.x, C.y)],
  1629→      uvs: [
  1630→        new THREE.Vector2(0, 0), new THREE.Vector2(1, 0),
  1631→        new THREE.Vector2(1, 1), new THREE.Vector2(1, 1),
  1632→      ],
  1633→      rotates: false,
  1634→    };
  1635→    sails.push({
  1636→      mesh, mat, setAt, furl: 1, square: false, braceAngle: 0, mastIndex,
  1637→      quad, holeW: yardLenM + Math.abs(C.x), holeH: drop,
  1638→    });
  1639→  };
  1640→  if (P.rig === 'single') {
  1641→    // когг: один большой прямой парус на мачту
  1642→    addSquareSail(0, 0.7, L * 0.42, L * 0.24, 0.5);
  1643→    addSquareSail(1, 0.76, L * 0.4, L * 0.22, 0.72);
  1644→  } else if (P.rig === 'lateenMix') {
  1645→    // каравелла редонда: прямой фок + латинские грот и бизань
  1646→    addSquareSail(0, 0.6, L * 0.28, L * 0.13, 0.55);
  1647→    addLateenSail(1, 0.45, L * 0.5, mastDefs[1].h * 0.5, 0.85);
  1648→    addLateenSail(2, 0.8, L * 0.32, mastDefs[2].h * 0.5, 0.55);
  1649→  } else if (P.rig === 'lateen') {
  1650→    // шебека / галера: огромные латинские паруса на всех мачтах
  1651→    addLateenSail(0, 0.68, L * 0.34, mastDefs[0].h * 0.46, 0.5);
  1652→    addLateenSail(1, 0.45, L * 0.52, mastDefs[1].h * 0.52, 0.85);
  1653→    addLateenSail(2, 0.78, L * 0.3, mastDefs[2].h * 0.44, 0.5);
  1654→  } else if (P.masts === 2) {
  1655→    addSquareSail(0, 0.66, L * 0.3, L * 0.16, 0.62);   // фок
  1656→    addSquareSail(1, 0.72, L * 0.36, L * 0.2, 0.45);   // грот нижний — ставится первым
  1657→    addSquareSail(1, 0.45, L * 0.26, L * 0.14, 0.92);  // грот верхний (брамсель)
  1658→  } else {
  1659→    addSquareSail(0, 0.62, L * 0.24, L * 0.13, 0.60);
  1660→    addSquareSail(1, 0.7, L * 0.34, L * 0.18, 0.45);
  1661→    addSquareSail(1, 0.44, L * 0.24, L * 0.13, 0.92);
  1662→    addSquareSail(2, 0.5, L * 0.18, L * 0.11, 0.78);
  1663→  }
  1664→
  1665→  // кливера (треугольные) от бушприта к фок-мачте: сетка с uv —
  1666→  // есть «пузо» и трепет; при рифлении собираются к голове
  1667→  const addJib = (tip: THREE.Vector3, foot: THREE.Vector3, headY: number, mastZ: number, setAt: number) => {
  1668→    const head = new THREE.Vector3(0, headY, mastZ - 0.3);
  1669→    // локальная система: X = −мирZ (продольная), Y = мирY, Z = мирX (пузо вбок)
  1670→    const a = { x: -(tip.z - head.z), y: tip.y - head.y };
  1671→    const b = { x: -(foot.z - head.z), y: foot.y - head.y };
  1672→    const NU = 8, NV = 5;
  1673→    const pos: number[] = [], uvs: number[] = [], idx: number[] = [];
  1674→    for (let i = 0; i <= NU; i++) {
  1675→      const u = i / NU;
  1676→      // точка на штаге (к галсовому углу) и на шкаторине
  1677→      const lx = a.x * u, ly = a.y * u;
  1678→      const rx = b.x * u, ry = b.y * u;
  1679→      for (let j = 0; j <= NV; j++) {
  1680→        const v = j / NV;
  1681→        pos.push(THREE.MathUtils.lerp(lx, rx, v), THREE.MathUtils.lerp(ly, ry, v), 0);
  1682→        uvs.push(u, v);
  1683→      }
  1684→    }
  1685→    const row = NV + 1;
  1686→    for (let i = 0; i < NU; i++) {
  1687→      for (let j = 0; j < NV; j++) {
  1688→        const p0 = i * row + j, p1 = p0 + 1, p2 = p0 + row, p3 = p2 + 1;
  1689→        idx.push(p0, p1, p2, p1, p3, p2);
  1690→      }
  1691→    }
  1692→    const g = new THREE.BufferGeometry();
  1693→    g.setAttribute('position', new THREE.Float32BufferAttribute(pos, 3));
  1694→    g.setAttribute('uv', new THREE.Float32BufferAttribute(uvs, 2));
  1695→    g.setIndex(idx);
  1696→    g.computeVertexNormals();
  1697→    const mat = sailMaterial(P.colors.sail);
  1698→    mat.uniforms.uAspect.value = Math.max(0.3, Math.hypot(a.x, a.y) / Math.max(1, Math.hypot(b.x, b.y)));
  1699→    const mesh = new THREE.Mesh(g, mat);
  1700→    mesh.position.set(head.x, head.y, head.z - mastZ);
  1701→    mesh.rotation.y = Math.PI / 2; // локальный Z (пузо) -> мир X (борт)
  1702→    mesh.frustumCulled = false;
  1703→    masts[0].add(mesh); // потомок фок-мачты — падает вместе с ней
  1704→    // зона прострела: треугольник голова-(конец штага)-(галсовый угол);
  1705→    // поворот меша +90° вокруг Y: (x, y, 0) -> (0, y, -x)
  1706→    const rotP = (x: number, y: number) => new THREE.Vector3(0, y, -x);
  1707→    const pv = new THREE.Vector3(head.x, head.y, head.z);
  1708→    const quad: SailHitQuad = {
  1709→      pivot: pv,
  1710→      corners: [rotP(0, 0), rotP(a.x, a.y), rotP(b.x, b.y), rotP(b.x, b.y)],
  1711→      uvs: [
  1712→        new THREE.Vector2(0, 0), new THREE.Vector2(1, 0),
  1713→        new THREE.Vector2(1, 1), new THREE.Vector2(1, 1),
  1714→      ],
  1715→      rotates: false,
  1716→    };
  1717→    sails.push({
  1718→      mesh, mat, setAt, furl: 1, square: false, braceAngle: 0, mastIndex: 0,
  1719→      quad, holeW: Math.hypot(a.x, a.y), holeH: Math.hypot(b.x, b.y),
  1720→    });
  1721→  };
  1722→  // кливера — только на «полных» вооружениях (у когга, каравеллы и шебеки их нет)
  1723→  const foreTop = mastDefs[0];
  1724→  if (P.rig !== 'lateenMix' && P.rig !== 'single' && P.rig !== 'lateen') {
  1725→    addJib(bowspritTip, new THREE.Vector3(0, P.freeboard * 0.72, L / 2 - 1.0), foreTop.h * 0.62 - P.draft * 0.3, foreTop.z, 0.72);
  1726→    addJib(new THREE.Vector3(bowspritTip.x * 0.55, bowspritTip.y - (bowspritTip.y - P.freeboard * 0.72) * 0.45, bowspritTip.z - L * 0.1),
  1727→      new THREE.Vector3(0, P.freeboard * 0.72, L / 2 - 2.0), foreTop.h * 0.78 - P.draft * 0.3, foreTop.z, 1.0);
  1728→  }
  1729→
  1730→  // флаг на топе грот-мачты (потомок мачты — валится вместе с ней)
  1731→  const mainMastIdx = 1; // грот-мачта и у 2-, и у 3-мачтовых
  1732→  const mainMast = mastDefs[mainMastIdx];
  1733→  const { mesh: flag, mat: flagMat } = flagMesh(P.flagKind);
  1734→  flag.position.set(0.05, mainMast.h - P.draft * 0.3 - 0.4, 0);
  1735→  masts[mainMastIdx].add(flag);
  1736→
  1737→  // пушки по бортам — по раскладке gunLayout: ИТОГО ровно gunsPerSide
  1738→  // на борт. Верхняя палуба — полные лафеты, закрытые деки — стволы,
  1739→  // высунутые из портов (лафет внутри корпуса не виден)
  1740→  const gunsPort: THREE.Object3D[] = [];
  1741→  const gunsStarboard: THREE.Object3D[] = [];
  1742→  for (const slot of gunLayout(P)) {
  1743→    const tt = clamp2((slot.z + L / 2) / L, 0, 1);
  1744→    const isDeck = slot.tier === 0;
  1745→    const prof = hullProfile(tt, P);
  1746→    for (const side of [-1, 1]) {
  1747→      const gun = new THREE.Group();
  1748→      // ствол: СВЕТЛЫЙ gunmetal — тёмное железо в metalness-пайплайне
  1749→      // отражает ~12% неба и выглядит «картонкой»; светлее + глянец +
  1750→      // усиленные отражения неба = видимый солнечный блик по стволу
  1751→      const barrel = new THREE.Mesh(new THREE.CylinderGeometry(0.11, 0.15, 1.15, 12),
  1752→        new THREE.MeshStandardMaterial({
  1753→          color: 0x4a4e57, metalness: 0.9, envMapIntensity: 1.6,
  1754→          normalMap: metalNormalTex(), roughnessMap: metalRoughTex(),
  1755→          normalScale: new THREE.Vector2(0.55, 0.55), roughness: 0.72,
  1756→        }));
  1757→      barrel.rotation.z = Math.PI / 2;
  1758→      barrel.position.set(side * 0.62, 0.08, 0);
  1759→      gun.add(barrel);
  1760→      const carriage = new THREE.Mesh(new THREE.BoxGeometry(0.5, 0.34, 0.55), darkMat);
  1761→      gun.add(carriage);
  1762→      if (isDeck) {
  1763→        gun.position.set(side * (prof.w * 0.9 - 0.15), prof.bulwark - 0.72, slot.z);
  1764→      } else {
  1765→        // закрытая батарея: дуло высунуто из порта за борт
  1766→        const hw = hullHalfWidthAt(P, tt, slot.y);
  1767→        gun.position.set(side * (hw - 0.85), slot.y, slot.z);
  1768→      }
  1769→      gun.userData.side = side;
  1770→      group.add(gun);
  1771→      // локальный −X = штирборт (правый борт), локальный +X = порт (левый)
  1772→      (side < 0 ? gunsStarboard : gunsPort).push(gun);
  1773→    }
  1774→  }
  1775→
  1776→  // штурвал: пьедестал (не вращается) + колесо, начало координат
  1777→  // вращающейся группы — в ступице, чтобы крутился вокруг СВОЕЙ оси
  1778→  const wheelAsm = new THREE.Group();
  1779→  const wheel = new THREE.Group();
  1780→  const wheelR = 0.55;
  1781→  const hubY = 0.98; // высота оси над палубой
  1782→  const wheelTorus = new THREE.Mesh(new THREE.TorusGeometry(wheelR, 0.045, 6, 18), darkMat);
  1783→  wheel.add(wheelTorus);
  1784→  for (let i = 0; i < 4; i++) {
  1785→    const ang = (i / 4) * Math.PI;
  1786→    const spoke = new THREE.Mesh(new THREE.CylinderGeometry(0.028, 0.028, wheelR * 2.3, 4), darkMat);
  1787→    spoke.rotation.z = ang;
  1788→    wheel.add(spoke);
  1789→    // рукоятки наружу, к рулевому (в сторону кормы, -Z)
  1790→    const handle = new THREE.Mesh(new THREE.CylinderGeometry(0.035, 0.035, 0.26, 5), woodMat);
  1791→    handle.rotation.x = Math.PI / 2;
  1792→    handle.position.set(-Math.sin(ang) * (wheelR + 0.02), Math.cos(ang) * (wheelR + 0.02), -0.14);
  1793→    wheel.add(handle);
  1794→  }
  1795→  const wheelHub = new THREE.Mesh(new THREE.CylinderGeometry(0.1, 0.1, 0.34, 8), woodMat);
  1796→  wheelHub.rotation.x = Math.PI / 2;
  1797→  wheel.add(wheelHub);
  1798→  wheel.position.set(0, hubY, 0);
  1799→  wheelAsm.add(wheel);
  1800→  // стойка, подкосы и ось — вне вращающейся группы
  1801→  const post = new THREE.Mesh(new THREE.BoxGeometry(0.14, hubY, 0.12), darkMat);
  1802→  post.position.set(0, hubY / 2, 0);
  1803→  wheelAsm.add(post);
  1804→  for (const sgn of [-1, 1]) {
  1805→    const brace = new THREE.Mesh(new THREE.BoxGeometry(0.08, hubY * 1.2, 0.08), darkMat);
  1806→    brace.position.set(sgn * 0.24, hubY * 0.5, 0);
  1807→    brace.rotation.z = sgn * 0.55;
  1808→    wheelAsm.add(brace);
  1809→  }
  1810→  const wheelAxle = new THREE.Mesh(new THREE.CylinderGeometry(0.04, 0.04, 0.55, 6), darkMat);
  1811→  wheelAxle.rotation.x = Math.PI / 2;
  1812→  wheelAxle.position.set(0, hubY, 0.12);
  1813→  wheelAsm.add(wheelAxle);
  1814→  const wheelT = P.castleH ? 0.29 : 0.2;
  1815→  wheelAsm.position.set(0, hullProfile(wheelT, P).bulwark - 1.0, THREE.MathUtils.lerp(-L / 2, L / 2, wheelT));
  1816→  group.add(wheelAsm);
  1817→
  1818→  // палубный реквизит: гратинг-люк, бочки, канаты, кнехты, колокол, ядра,
  1819→  // сундук, ведро — одним мешом
  1820→  const props = new THREE.Mesh(buildDeckProps(P),
  1821→    new THREE.MeshStandardMaterial({
  1822→      vertexColors: true, side: THREE.DoubleSide,
  1823→      normalMap: woodNormalTex('spar'), roughnessMap: woodRoughTex('spar'),
  1824→      normalScale: new THREE.Vector2(0.35, 0.35), roughness: 1,
  1825→    }));
  1826→  props.castShadow = true;
  1827→  group.add(props);
  1828→
  1829→  // шпиль, фонарь, якоря — строго на палубе
  1830→  const deckYAt = (t: number) => hullProfile(t, P).bulwark - 1.0;
  1831→  const capstan = new THREE.Mesh(new THREE.CylinderGeometry(0.4, 0.5, 0.7, 8), woodMat);
  1832→  capstan.position.set(0, deckYAt(0.36) + 0.35, -L * 0.16);
  1833→  group.add(capstan);
  1834→  const sternLantern = new THREE.Mesh(new THREE.SphereGeometry(0.22, 8, 8),
  1835→    new THREE.MeshStandardMaterial({
  1836→      color: 0xffd88a, emissive: 0xffaa33, emissiveIntensity: 1.6,
  1837→      metalness: 0.9,
  1838→      normalMap: metalNormalTex(), roughnessMap: metalRoughTex(),
  1839→      normalScale: new THREE.Vector2(0.4, 0.4), roughness: 1,
  1840→    }));
  1841→  sternLantern.position.set(0, P.freeboard * 1.15, -L / 2 - 0.15);
  1842→  group.add(sternLantern);
  1843→  let lantern: THREE.PointLight | undefined;
  1844→  // фигура на носу
  1845→  const figure = new THREE.Mesh(new THREE.ConeGeometry(0.22, 0.8, 6),
  1846→    new THREE.MeshStandardMaterial({
  1847→      color: 0xc9a24a, metalness: 0.6,
  1848→      normalMap: metalNormalTex(), roughnessMap: metalRoughTex(),
  1849→      normalScale: new THREE.Vector2(0.45, 0.45), roughness: 1,
  1850→    }));
  1851→  figure.position.set(0, P.freeboard * 1.12, L / 2 - 0.2);
  1852→  figure.rotation.x = -0.5;
  1853→  group.add(figure);
  1854→  // (простые якоря-коробки заменены настоящими якорями в группе мелких деталей)
  1855→
  1856→  // такелаж
  1857→  const rigging = buildRigging(P, mastDefs, bowspritTip);
  1858→  group.add(rigging.lines);
  1859→
  1860→  // надстройки-«замки» (когг) — силуэт виден издалека
  1861→  if (P.castleH) {
  1862→    const castles = new THREE.Mesh(buildCastles(P),
  1863→      new THREE.MeshStandardMaterial({
  1864→        vertexColors: true,
  1865→        normalMap: woodNormalTex('spar'), roughnessMap: woodRoughTex('spar'),
  1866→        normalScale: new THREE.Vector2(0.35, 0.35), roughness: 1,
  1867→      }));
  1868→    castles.castShadow = true;
  1869→    group.add(castles);
  1870→  }
  1871→  // наружный декор: пушечные порты, вельсы, кормовая галерея, нос/гальюн
  1872→  const ornGeo = buildHullOrnaments(P);
  1873→  if (ornGeo) {
  1874→    group.add(new THREE.Mesh(ornGeo,
  1875→      new THREE.MeshStandardMaterial({
  1876→        vertexColors: true, metalness: 0.15,
  1877→        normalMap: woodNormalTex('spar'), roughnessMap: woodRoughTex('spar'),
  1878→        normalScale: new THREE.Vector2(0.35, 0.35), roughness: 1,
  1879→      })));
  1880→  }
  1881→  // мелкие детали — только вблизи (LOD, переключает game.ts)
  1882→  const detail = buildShipDetail(P, mastDefs);
  1883→  detail.visible = false;
  1884→  group.add(detail);
  1885→
  1886→  // вёсла (шебека, галера)
  1887→  let oarsVis: THREE.Object3D[] | undefined;
  1888→  if (P.oars) {
  1889→    const o = buildOars(P);
  1890→    group.add(o.group);
  1891→    oarsVis = o.oars;
  1892→  }
  1893→
  1894→  // руль за транцем: перо на петлях, поворачивается вместе с рулём
  1895→  const rudLen = Math.min(2.6, Math.max(1.4, L * 0.045));
  1896→  const rudGeo = new THREE.BoxGeometry(0.15, Math.max(1.2, P.draft * 0.72), rudLen);
  1897→  rudGeo.translate(0, 0, -rudLen / 2); // ось вращения — передняя кромка
  1898→  const rudderMesh = new THREE.Mesh(rudGeo, darkMat);
  1899→  rudderMesh.position.set(0, -P.draft * 0.40, -L / 2 + 0.03);
  1900→  rudderMesh.castShadow = true;
  1901→  group.add(rudderMesh);
  1902→
  1903→  const rig: SailRig = { sails };
  1904→
  1905→  // ломаемые детали палубы: срываются попаданиями / давлением воды
  1906→  const breakables: THREE.Object3D[] = [wheelAsm, capstan, sternLantern, figure, rudderMesh];
  1907→
  1908→  // материалы дерева для скина «цвет корабля» (собираем ДО того, как
  1909→  // в группу добавится экипаж с общим CREW_MAT — его тонировать нельзя)
  1910→  const tintSet = new Set<THREE.MeshStandardMaterial>();
  1911→  group.traverse(o => {
  1912→    const m = (o as THREE.Mesh).material;
  1913→    if (!m) return;
  1914→    const arr = Array.isArray(m) ? m : [m];
  1915→    for (const mm of arr) {
  1916→      if ((mm as THREE.MeshStandardMaterial).isMeshStandardMaterial) tintSet.add(mm as THREE.MeshStandardMaterial);
  1917→    }
  1918→  });
  1919→
  1920→  return {
  1921→    group, sails: rig, gunsPort, gunsStarboard, flag, flagMat,
  1922→    dims: { length: L, beam: P.beam, draft: P.draft }, outline,
  1923→    masts, lantern, wheel, detail, breakables, rigging, tintMats: [...tintSet],
  1924→    oars: oarsVis, rudderMesh,
  1925→  };
  1926→}

  1956→/** Цена судна на верфи (золото) — от ТТХ; легендарные суда дороже */
  1957→export function shipPrice(P: ShipPreset): number {
  1958→  if (P.id === 'blackpearl') return 35000; // легенда не продаётся дёшево
  1959→  const p = P.hullHP * 3 + Math.pow(P.gunsPerSide, 1.6) * 90
  1960→    + P.maxCrew * 3 + P.cargo * 0.8 + P.maxSpeed * 120;
  1961→  return Math.round(p / 10) * 10;
  1962→}
