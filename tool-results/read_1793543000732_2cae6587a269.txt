    13→// человек. На палубе видно deckCrewFor(экипаж):
    14→//   500 человек → 50 фигур; 50 → 10; ≥10 всегда минимум 10;
    15→//   осталось ≤10 — показываем ВСЕХ (экипаж 10 → 10 фигур).
    16→// Степенная интерполяция 10·(N/50)^0.7 между опорными точками.
    17→// ПОТЕРИ: картечь ВАЛИТ фигуры (падают трупами на палубу),
    18→// ядро насквозь РАЗРЫВАЕТ на куски (гибы + кровь); каждая
    19→// убитая фигура = −1 человек экипажа. Абстрактные потери
    20→// (пожар/сдача в плен) тоже обращаются в трупы. При затоплении
    21→// экипаж «покидает» корабль.
    22→// buildFigure/rollTraits экспортируются — те же пираты живут
    23→// на острове логова (world/islandFolk.ts).

    26→/** Живая ёмкость палубы (перф-потолок числа фигур) */
    27→export const MAX_DECK_FIGURES = 80;

    29→/** Сколько человек видно на палубе (1 фигура = 1 человек).
    30→ *  Опоры от пользователя: 500→50, 50→10, всегда ≥10, при N≤10 —
    31→ *  все; между точками — степень 10·(N/50)^0.7; 0 → палуба пуста */
    32→export function deckCrewFor(crew: number): number {
    33→  const n = Math.max(0, Math.round(crew));
    34→  if (n <= 0) return 0;
    35→  if (n <= 10) return n;                    // осталось мало — показываем всех
    36→  const scaled = Math.round(10 * Math.pow(n / 50, 0.7));
    37→  return Math.min(n, Math.max(10, scaled)); // никогда < 10 (если есть хоть 10)
    38→}

    84→export function mulberry32(seed: number) {
    85→  let a = seed >>> 0;
    86→  return () => {
    87→    a |= 0; a = (a + 0x6D2B79F5) | 0;
    88→    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    89→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    90→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    91→  };
    92→}

   116→export function rollTraits(rnd: () => number, role: CrewRole): Traits {
   117→  const pick = <T,>(arr: T[]): T => arr[Math.floor(rnd() * arr.length)];
   118→  return {
   119→    skin: pick(SKIN),
   120→    shirt: pick(SHIRT),
   121→    pants: pick(PANTS),
   122→    beard: rnd() < 0.55 ? (rnd() < 0.4 ? 2 : 1) : 0,
   123→    hook: rnd() < 0.22 ? (rnd() < 0.5 ? 1 : -1) : 0,
   124→    pegLeg: rnd() < 0.18 ? (rnd() < 0.5 ? 1 : -1) : 0,
   125→    eyePatch: rnd() < 0.25,
   126→    headwear: role === 'captain' ? 'hat' : rnd() < 0.45 ? 'bandana' : rnd() < 0.35 ? 'hat' : 'none',
   127→    striped: role !== 'captain' && rnd() < 0.4,
   128→    height: 0.92 + rnd() * 0.16,
   129→  };
   130→}

   132→/** Запечь геометрию с цветом в общие массивы (как реквизит палубы) */
   133→function bake(target: { pos: number[]; col: number[] }, geo: THREE.BufferGeometry, color: number,
   134→  x: number, y: number, z: number, euler?: THREE.Euler, scale?: [number, number, number]) {
   135→  const g = geo.index ? geo.toNonIndexed() : geo;
   136→  const p = g.attributes.position as THREE.BufferAttribute;
   137→  const v = new THREE.Vector3();
   138→  const c = new THREE.Color(color);
   139→  for (let i = 0; i < p.count; i++) {
   140→    v.fromBufferAttribute(p, i);
   141→    if (scale) v.multiply(new THREE.Vector3(...scale));
   142→    if (euler) v.applyEuler(euler);
   143→    target.pos.push(v.x + x, v.y + y, v.z + z);
   144→    target.col.push(c.r, c.g, c.b);
   145→  }
   146→}
   147→
   148→function bakeGeom(t: { pos: number[]; col: number[] }): THREE.BufferGeometry {
   149→  const g = new THREE.BufferGeometry();
   150→  g.setAttribute('position', new THREE.Float32BufferAttribute(t.pos, 3));
   151→  g.setAttribute('color', new THREE.Float32BufferAttribute(t.col, 3));
   152→  g.computeVertexNormals();
   153→  return g;
   154→}

   156→export const CREW_MAT = new THREE.MeshStandardMaterial({ vertexColors: true, roughness: 0.9, metalness: 0.03 });

   158→/** Собрать фигурку (экспорт для островных пиратов islandFolk.ts).
   159→ *  feetY задаётся позиционированием группы: фигура ~1.6 м.
   160→ *  Оси: +Z — лицо (по умолчанию к носу корабля), +X — порт. */
   161→export function buildFigure(role: CrewRole, traits: Traits, seed: number): CrewFigure {
   162→  const rnd = mulberry32(seed);
   163→  const group = new THREE.Group();
   164→  const t: { pos: number[]; col: number[] } = { pos: [], col: [] };
   165→  const h = traits.height;
   166→  // размеры (м): фигура ~1.6 м
   167→  const legH = 0.62, torsoH = role === 'captain' ? 0.62 : 0.5, torsoW = role === 'captain' ? 0.4 : 0.36;
   168→  const hipY = legH;
   169→  const torsoY = hipY + torsoH / 2;
   170→  const headR = 0.13;
   171→  const headY = hipY + torsoH + headR + 0.02;
   172→
   173→  // --- торс (тельняшка — полосами) ---
   174→  if (traits.striped) {
   175→    const stripeC = [0xe8e2d4, traits.shirt];
   176→    const n = 5;
   177→    for (let i = 0; i < n; i++) {
   178→      bake(t, new THREE.BoxGeometry(torsoW, torsoH / n + 0.005, 0.22),
   179→        stripeC[i % 2], 0, hipY + torsoH * (i + 0.5) / n, 0);
   180→    }
   181→  } else {
   182→    bake(t, new THREE.BoxGeometry(torsoW, torsoH, 0.22), traits.shirt, 0, torsoY, 0);
   183→  }
   184→  // капитан: длинный кафтан + золотой кант
   185→  if (role === 'captain') {
   186→    bake(t, new THREE.BoxGeometry(torsoW + 0.06, 0.42, 0.26), rnd() < 0.5 ? COAT[0] : COAT[1], 0, hipY + 0.05, 0);
   187→    bake(t, new THREE.BoxGeometry(0.05, torsoH * 0.9, 0.015), 0xb8912f, 0, torsoY, 0.135);
   188→  }
   189→
   190→  // --- руки: плечо→кисть, у боцмана/канонира правая рука к борту ---
   191→  const armLen = 0.48;
   192→  for (const side of [-1, 1] as const) {
   193→    const isHook = traits.hook === side;
   194→    const sh = new THREE.Vector3(side * (torsoW / 2 + 0.05), hipY + torsoH - 0.06, 0);
   195→    if (role === 'helmsman') {
   196→      // рулевой: руки вытянуты ВПЕРЁД к штурвалу (+Z — нос)
   197→      const rot = new THREE.Euler(-1.15, 0, side * -0.22);
   198→      bake(t, new THREE.BoxGeometry(0.1, armLen * 0.55, 0.11), traits.shirt,
   199→        sh.x * 0.8, sh.y - 0.06, 0.16, rot);
   200→      if (!isHook) {
   201→        bake(t, new THREE.BoxGeometry(0.09, 0.24, 0.1), traits.skin,
   202→          sh.x * 1.0, sh.y - 0.02, 0.42, rot);
   203→      } else {
   204→        bake(t, new THREE.BoxGeometry(0.085, 0.2, 0.09), WOOD, sh.x * 1.0, sh.y - 0.02, 0.4, rot);
   205→      }
   206→      continue;
   207→    }
   208→    const rot = new THREE.Euler(0, 0, side * -0.35);
   209→    bake(t, new THREE.BoxGeometry(0.1, armLen * 0.55, 0.11), role === 'captain' ? COAT[0] : traits.shirt,
   210→      sh.x * 0.75, sh.y - armLen * 0.26, 0.03, rot);
   211→    // предплечье/кисть
   212→    if (isHook) {
   213→      // деревянная культа + крюк
   214→      bake(t, new THREE.BoxGeometry(0.085, 0.2, 0.09), WOOD, sh.x * 1.25, sh.y - armLen * 0.62, 0.08, rot);
   215→      const hookE = new THREE.Euler(0, 0, side * 1.57);
   216→      bake(t, new THREE.TorusGeometry(0.06, 0.018, 5, 10, Math.PI * 1.2), METAL,
   217→        sh.x * 1.32, sh.y - armLen * 0.78, 0.1, hookE);
   218→    } else {
   219→      bake(t, new THREE.BoxGeometry(0.09, 0.22, 0.1), traits.skin,
   220→        sh.x * 1.2, sh.y - armLen * 0.72, 0.06, rot);
   221→    }
   222→  }
   223→
   224→  // --- голова ---
   225→  bake(t, new THREE.SphereGeometry(headR, 10, 8), traits.skin, 0, headY, 0);
   226→  // волосы (если нет головного убора или под ним)
   227→  bake(t, new THREE.SphereGeometry(headR * 1.04, 10, 6, 0, Math.PI * 2, 0, Math.PI * 0.55),
   228→    BEARD[Math.floor(rnd() * BEARD.length)], 0, headY + 0.01, 0);
   229→  // повязка на глаз: наглазник + ремешок через лоб
   230→  if (traits.eyePatch) {
   231→    const eyeX = rnd() < 0.5 ? -0.052 : 0.052;
   232→    bake(t, new THREE.BoxGeometry(0.075, 0.07, 0.02), 0x141414, eyeX, headY + 0.01, headR - 0.008);
   233→    bake(t, new THREE.BoxGeometry(headR * 2.05, 0.024, 0.02), 0x141414, 0, headY + 0.045, headR - 0.006,
   234→      new THREE.Euler(0, 0, 0.16));
   235→  }
   236→  // борода
   237→  if (traits.beard > 0) {
   238→    const bl = traits.beard === 2 ? 0.3 : 0.14;
   239→    bake(t, new THREE.BoxGeometry(0.15, bl, 0.08),
   240→      BEARD[Math.floor(rnd() * BEARD.length)], 0, headY - headR * 0.55 - bl * 0.25, headR - 0.045);
   241→    if (traits.beard === 2) {
   242→      bake(t, new THREE.BoxGeometry(0.1, 0.14, 0.06), 0x1d1712, 0, headY - headR * 0.55 - bl - 0.05, headR - 0.05);
   243→    }
   244→  }
   245→  // головной убор
   246→  if (traits.headwear === 'bandana') {
   247→    const bc = BANDANA[Math.floor(rnd() * BANDANA.length)];
   248→    bake(t, new THREE.SphereGeometry(headR * 1.06, 10, 5, 0, Math.PI * 2, 0, Math.PI * 0.5), bc, 0, headY + 0.02, 0);
   249→    bake(t, new THREE.BoxGeometry(0.05, 0.05, 0.12), bc, 0.06, headY + 0.06, -headR - 0.03);
   250→  } else if (traits.headwear === 'hat') {
   251→    // треуголка: поля + тулья
   252→    const brim = role === 'captain' ? 0.34 : 0.28;
   253→    bake(t, new THREE.BoxGeometry(brim, 0.025, brim * 0.8), HAT, 0, headY + headR * 0.82, 0,
   254→      new THREE.Euler(0, 0.35, 0));
   255→    bake(t, new THREE.BoxGeometry(brim * 0.62, 0.12, brim * 0.5), HAT, 0, headY + headR * 0.95, 0,
   256→      new THREE.Euler(0, 0.35, 0));
   257→    if (role === 'captain') bake(t, new THREE.BoxGeometry(0.06, 0.02, 0.1), 0xb8912f, 0, headY + headR * 0.86, brim * 0.28);
   258→  }
   259→
   260→  // --- предметы ролей ---
   261→  if (role === 'captain') {
   262→    // подзорная труба в правой руке
   263→    bake(t, new THREE.CylinderGeometry(0.035, 0.05, 0.34, 6), 0xb8912f, torsoW / 2 + 0.14, hipY + 0.18, 0.14,
   264→      new THREE.Euler(1.1, 0, -0.5));
   265→  }
   266→  if (role === 'boatswain') {
   267→    // свисток боцмана на груди
   268→    bake(t, new THREE.CylinderGeometry(0.018, 0.018, 0.1, 5), 0xb8912f, 0.07, torsoY + 0.04, 0.125,
   269→      new THREE.Euler(1.57, 0, 0.2));
   270→  }
   271→
   272→  const body = new THREE.Mesh(bakeGeom(t), CREW_MAT);
   273→  group.add(body);
   274→
   275→  // --- ноги: у ходоков отдельные (шаг), у остальных в общий меш ---
   276→  const mkLeg = (side: number, wood: boolean): THREE.Mesh => {
   277→    const lt: { pos: number[]; col: number[] } = { pos: [], col: [] };
   278→    if (wood) {
   279→      bake(lt, new THREE.CylinderGeometry(0.05, 0.062, legH + 0.06, 6), WOOD, 0, (legH + 0.06) / 2, 0);
   280→    } else {
   281→      bake(lt, new THREE.BoxGeometry(0.11, legH, 0.13), traits.pants, 0, legH / 2, 0);
   282→      bake(lt, new THREE.BoxGeometry(0.115, 0.07, 0.2), 0x1f1a14, 0, 0.035, 0.03);
   283→    }
   284→    const m = new THREE.Mesh(bakeGeom(lt), CREW_MAT);
   285→    m.position.set(side * 0.095, legH, 0);
   286→    return m;
   287→  };
   288→
   289→  if (role === 'walker') {
   290→    const legL = mkLeg(-1, traits.pegLeg === -1);
   291→    const legR = mkLeg(1, traits.pegLeg === 1);
   292→    group.add(legL, legR);
   293→    // центр вращения ноги — бедро
   294→    for (const leg of [legL, legR]) {
   295→      const off = leg.position.y;
   296→      leg.geometry.translate(0, -off, 0);
   297→      leg.position.y = off;
   298→    }
   299→    return {
   300→      group, role, legL, legR, seed, walkT: rnd(), walkDir: rnd() < 0.5 ? 1 : -1, baseScaleY: 1,
   301→      walkSpeed: 0.55 + rnd() * 0.5, phase: rnd() * 6.28,
   302→      dead: false, deadT: 0, deathMode: 'fall', corpseLife: 0, fallDir: 1, fallRoll: 0,
   303→      skin: traits.skin, shirt: traits.shirt, pants: traits.pants,
   304→      hasHook: traits.hook !== 0, hasPeg: traits.pegLeg !== 0,
   305→    };
   306→  }
   307→  // статичные: ноги вторым мешом (деревянная нога заменяет одну из них)
   308→  const legsStatic = new THREE.Mesh(bakeGeom(t2legs(traits, legH)), CREW_MAT);
   309→  group.add(legsStatic);
   310→  return {
   311→    group, role, seed, walkT: 0, walkDir: 1, walkSpeed: 0, phase: rnd() * 6.28, baseScaleY: 1,
   312→    dead: false, deadT: 0, deathMode: 'fall', corpseLife: 0, fallDir: 1, fallRoll: 0,
   313→    skin: traits.skin, shirt: traits.shirt, pants: traits.pants,
   314→    hasHook: traits.hook !== 0, hasPeg: traits.pegLeg !== 0,
   315→  };
   316→}

   318→/** ноги для статичной фигуры отдельным мешом */
   319→function t2legs(traits: Traits, legH: number): { pos: number[]; col: number[] } {
   320→  const t: { pos: number[]; col: number[] } = { pos: [], col: [] };
   321→  for (const side of [-1, 1]) {
   322→    if (traits.pegLeg === side) {
   323→      bake(t, new THREE.CylinderGeometry(0.05, 0.062, legH + 0.06, 6), WOOD, side * 0.095, (legH + 0.06) / 2, 0);
   324→    } else {
   325→      bake(t, new THREE.BoxGeometry(0.11, legH, 0.13), traits.pants, side * 0.095, legH / 2, 0);
   326→      bake(t, new THREE.BoxGeometry(0.115, 0.07, 0.2), 0x1f1a14, side * 0.095, 0.035, 0.03);
   327→    }
   328→  }
   329→  return t;
   330→}

   332→/** Расстояние между отрезками AB и CD в 3D (для капсул фигур);
   333→ *  при c=d — расстояние от точки до отрезка */
   334→export function segSegDist3D(a: THREE.Vector3, b: THREE.Vector3, c: THREE.Vector3, d: THREE.Vector3): number {
   335→  const u = b.clone().sub(a), v = d.clone().sub(c), w = a.clone().sub(c);
   336→  const uu = u.dot(u), vv = v.dot(v), uv = u.dot(v);
   337→  const wu = w.dot(u), wv = w.dot(v);
   338→  const den = uu * vv - uv * uv;
   339→  let s: number, t: number;
   340→  if (den < 1e-9) { s = 0; t = vv > 1e-9 ? clampN(wv / vv, 0, 1) : 0; }
   341→  else { s = clampN((uv * wv - vv * wu) / den, 0, 1); t = clampN((uv * wu - uu * wv) / den, 0, 1); }
   342→  // одна итерация уточнения (границы)
   343→  const t2 = clampN((uv * s + wv) / (vv > 1e-9 ? vv : 1), 0, 1);
   344→  const s2 = clampN((uv * t2 - wu) / (uu > 1e-9 ? uu : 1), 0, 1);
   345→  const p1 = a.clone().addScaledVector(u, s2);
   346→  const p2 = c.clone().addScaledVector(v, t2);
   347→  return p1.distanceTo(p2);
   348→}
   349→function clampN(v: number, lo: number, hi: number) { return v < lo ? lo : v > hi ? hi : v; }
   350→
   351→/** Состояние корабля для анимации экипажа.
   352→ *  stroke — фаза гребка для гребцов (гребные суда): phase (рад),
   353→ *  active (0/1 — гребут сейчас), dir (+1 вперёд / −1 назад) */
   354→export interface CrewShipState {
   355→  crew: number; maxCrew: number; sinking: boolean; sunk: boolean;
   356→  position: THREE.Vector3; quaternion: THREE.Quaternion;
   357→  stroke?: { phase: number; active: number; dir: number };
   358→}
   359→
   360→export interface CrewVisual {
   361→  group: THREE.Group;
   362→  total: number;
   363→  shown: number;          // живые видимые фигуры
   364→  dead: number;           // потери (трупы + разорванные)
   365→  update(dt: number, time: number, ship: CrewShipState): void;
   366→  setVisible(v: boolean): void;
   367→  /** Поражение экипажа вдоль пути снаряда (a→b в ЛОКАЛЬНЫХ координатах
   368→   *  корабля). grape — фигуры падают; ball — разрывает на куски.
   369→   *  Возвращает число убитых (1 фигура = 1 человек). */
   370→  killNearPath(a: THREE.Vector3, b: THREE.Vector3, mode: KillMode,
   371→    maxCount: number, radius: number,
   372→    shipPos: THREE.Vector3, shipQuat: THREE.Quaternion): number;
   373→  /** Оживить всех (полигон/сброс повреждений) */
   374→  reset(): void;
   375→}
   376→
   377→/** Собрать экипаж для корабля (local coords: +Z нос, +X порт).
   378→ *  Число фигур = deckCrewFor(maxCrew): 1 фигура = 1 человек;
   379→ *  500-экипаж → 50 фигур, минимум 10 — от 10 на шлюпе до ~73
   380→ *  на линейном корабле */
   381→export function buildCrew(P: ShipPreset, seed: number): CrewVisual {
   382→  const group = new THREE.Group();
   383→  const L = P.length;
   384→  // палуба под ногами: на квартердеке — приподнята; корма судов
   385→  // без квартердека — обычная палуба
   386→  const qFront = P.quarter ? quarterFrontZ(P) : -Infinity;
   387→  const deckYAt = (z: number) => {
   388→    const t = Math.min(1, Math.max(0, (z + L / 2) / L));
   389→    return hullProfile(t, P).bulwark - 1.0 + (z < qFront ? (P.quarter ?? 0) : 0);
   390→  };
   391→  const halfW = (z: number) => {
   392→    const t = Math.min(1, Math.max(0, (z + L / 2) / L));
   393→    return hullProfile(t, P).w;
   394→  };
   395→  const figures: CrewFigure[] = [];
   396→  const place = (f: CrewFigure, x: number, z: number, faceDir: number) => {
   397→    f.group.position.set(x, deckYAt(z), z);
   398→    f.group.rotation.y = faceDir;
   399→    group.add(f.group);
   400→    figures.push(f);
   401→  };
   402→  const rot = mulberry32(seed * 7919 + 13);
   403→  let s = seed * 104729 + 7;
   404→  const nextSeed = () => { s = (s * 16807) % 2147483647; return s % 100000; };
   405→
   406→  // === СКОЛЬКО ЛЮДЕЙ НА ПАЛУБЕ: 1 фигура = 1 человек;
   407→  // deckCrewFor по формуле пользователя (500→50, 50→10,
   408→  // всегда ≥10, при N≤10 — все) ===
   409→  const nDeck = Math.max(1, Math.min(MAX_DECK_FIGURES, deckCrewFor(P.maxCrew)));
   410→
   411→  const mastZ = P.masts === 2 ? [L * 0.16, -L * 0.10] : [L * 0.18, -L * 0.05, -L * 0.28];
   412→  const deckGuns = gunLayout(P).filter(sl => sl.tier === 0).map(sl => sl.z);
   413→  const wheelZ = wheelZFor(P);
   414→
   415→  // план ролей по приоритету: капитан → рулевой → боцман →
   416→  // ГРЕБЦЫ (гребные суда, на банках у вёсел) → канониры у орудий
   417→  // (до 2 на ствол) → вахтенные у мачт (2 на мачту) → ходоки
   418→  const plan: CrewRole[] = ['captain'];
   419→  if (nDeck >= 2) plan.push('helmsman');
   420→  if (nDeck >= 4) plan.push('boatswain');
   421→  const nRest = nDeck - plan.length;
   422→  const nRowers = P.oars ? Math.min(Math.max(2, nRest * 0.45), 14) : 0;
   423→  const nGunners = Math.min(Math.floor((nRest - nRowers) * 0.5), deckGuns.length * 2, 16);
   424→  const nWatch = Math.min(Math.max(0, nRest - nRowers - nGunners), P.masts * 2, 6);
   425→  const nWalk = Math.max(0, nRest - nRowers - nGunners - nWatch);
   426→  for (let i = 0; i < nRowers; i++) plan.push('rower');
   427→  for (let i = 0; i < nGunners; i++) plan.push('gunner');
   428→  for (let i = 0; i < nWatch; i++) plan.push('watch');
   429→  for (let i = 0; i < nWalk; i++) plan.push('walker');
   430→
   431→  let gi = 0, wi = 0, ki = 0, ri = 0;
   432→  // банки гребцов: позиции вёсел (z от buildOars) — гребцы сидят
   433→  // лицом к корме (как на галерах), по двое на банку с обоих бортов
   434→  const oarN = P.oars ?? 0;
   435→  const oarZ = oarN
   436→    ? Array.from({ length: oarN }, (_, i) =>
   437→        THREE.MathUtils.lerp(-L * 0.26, L * 0.30, oarN === 1 ? 0.5 : i / (oarN - 1)))
   438→    : [];
   439→  for (const role of plan) {
   440→    const f = buildFigure(role, rollTraits(rot, role), nextSeed());
   441→    if (role === 'captain') {
   442→      // капитан — на квартердеке (если есть), у штурвала
   443→      place(f, 0.55, wheelZ - 1.6, 0.3);
   444→    } else if (role === 'helmsman') {
   445→      // рулевой — за штурвалом (руки на ободе), лицом к носу
   446→      place(f, 0, wheelZ - 0.52, 0);
   447→    } else if (role === 'boatswain') {
   448→      // боцман — у шпиля (шпиль перед квартердеком), лицом к носу-борту
   449→      place(f, -0.75, wheelZ + 2.4, 0.5);
   450→    } else if (role === 'rower') {
   451→      // гребец — на банке у своего весла, лицом к корме (сидя)
   452→      const z = oarZ.length ? oarZ[(ri * 2) % oarZ.length] : L * 0.1;
   453→      const side = ri % 2 === 0 ? -1 : 1;
   454→      ri++;
   455→      place(f, side * Math.max(0.55, halfW(z) * 0.5), z + 0.35, Math.PI);
   456→      // «сидит»: приземнее и приземистее стоящего матроса
   457→      f.group.position.y -= 0.30;
   458→      f.group.scale.y = 0.78;
   459→      f.baseScaleY = 0.78;
   460→    } else if (role === 'gunner') {
   461→      // канонир — за ПАЛУБНЫМ орудием (через одно), борт чередуется;
   462→      // при 16 канонирах — лёгкий разброс вдоль ствола (без слипаний)
   463→      const z = deckGuns[Math.floor(gi / 2) % Math.max(1, deckGuns.length)];
   464→      const side = gi % 2 === 0 ? -1 : 1;
   465→      gi++;
   466→      const w = halfW(z);
   467→      const jz = (rot() - 0.5) * 1.3;
   468→      place(f, side * (w * 0.56), z + 0.55 + jz, side > 0 ? Math.PI / 2 : -Math.PI / 2);
   469→    } else if (role === 'watch') {
   470→      // вахтенный — у мачты, борт чередуется
   471→      const mz = mastZ[wi % mastZ.length];
   472→      const side = wi % 2 === 0 ? -1 : 1;
   473→      wi++;
   474→      place(f, side * Math.max(0.7, halfW(mz) * 0.42), mz + 0.9, side * (Math.PI / 2));
   475→    } else {
   476→      // ходок — гуляет по палубе туда-обратно; полосы движения +
   477→      // живой дрейф поперёк (каждый со своим периодом), паузы у
   478→      // поворотов и разный темп — толпа не марширует строем
   479→      // (по замечанию игрока: экипаж ходит по палубе живее)
   480→      const LANES = [-0.31, -0.13, 0.13, 0.31];
   481→      f.xPath = LANES[ki % 4] * Math.max(0.55, halfW(0)) + (rot() - 0.5) * 0.3;
   482→      f.walkA = Math.max(P.castleH ? -L * 0.22 : -L * 0.3, qFront + 1.2);
   483→      f.walkB = P.castleH ? L * 0.2 : L * 0.28;
   484→      f.walkT = rot();
   485→      f.walkSpeed = 0.5 + rot() * 0.45;                    // разный темп
   486→      f.driftAmp = 0.1 + rot() * 0.22;                     // дрейф полосы
   487→      f.driftPeriod = 7 + rot() * 9;
   488→      f.walkPause = 0;
   489→      ki++;
   490→      figures.push(f);
   491→      group.add(f.group);
   492→    }
   493→  }
   494→
   495→  group.traverse(o => { o.castShadow = false; o.receiveShadow = false; });
   496→  // масштаб по высоте применяем к группе (высота фигурки внутри)
   497→  for (const f of figures) f.group.scale.setScalar(1);
   498→
   499→  // --- убийство фигуры: картечь валит трупом, ядро разрывает ---
   500→  const killFigure = (f: CrewFigure, mode: KillMode,
   501→    a: THREE.Vector3, b: THREE.Vector3,
   502→    shipPos: THREE.Vector3, shipQuat: THREE.Quaternion) => {
   503→    f.dead = true;
   504→    f.deadT = 0;
   505→    f.deathMode = mode === 'grape' ? 'fall' : 'gib';
   506→    // мировые координаты центра тела
   507→    const lp = f.group.position;
   508→    const wpos = new THREE.Vector3(lp.x, lp.y + 0.75, lp.z).applyQuaternion(shipQuat).add(shipPos);
   509→    // мировое направление пути снаряда
   510→    const wdir = new THREE.Vector3(b.x - a.x, b.y - a.y, b.z - a.z);
   511→    if (wdir.lengthSq() < 1e-6) wdir.set(0, 0, 1);
   512→    wdir.applyQuaternion(shipQuat).normalize();
   513→
   514→    if (mode === 'grape') {
   515→      // картечь: осел — падает на палубу трупом
   516→      f.corpseLife = 28 + Math.random() * 27;
   517→      f.fallDir = Math.random() < 0.5 ? 1 : -1;
   518→      f.fallRoll = (Math.random() - 0.5) * 0.8;
   519→      FX.particles?.blood(wpos.x, wpos.y - 0.35, wpos.z, wdir.x * 1.5, 0.4, wdir.z * 1.5, 7, 0.7);
   520→    } else {
   521→      // ядро насквозь: фигуры больше нет — куски летят вдоль пути
   522→      f.group.visible = false;
   523→      f.corpseLife = 0;
   524→      const gibs = FX.gibs;
   525→      if (gibs) {
   526→        const mk = (sx: number, sy: number, sz: number, hex: number, extra = 1) => {
   527→          const push = (2.6 + Math.random() * 3.6) * extra;
   528→          gibs.spawnPart(
   529→            new THREE.Vector3(
   530→              wpos.x + (Math.random() - 0.5) * 0.35,
   531→              wpos.y + (Math.random() - 0.5) * 0.7,
   532→              wpos.z + (Math.random() - 0.5) * 0.35),
   533→            new THREE.Vector3(
   534→              wdir.x * push + (Math.random() - 0.5) * 3.4,
   535→              1.5 + Math.random() * 3.0,
   536→              wdir.z * push + (Math.random() - 0.5) * 3.4),
   537→            sx, sy, sz, hex, 1.5 + Math.random() * 1.1);
   538→        };
   539→        mk(0.14, 0.14, 0.14, f.skin);        // голова
   540→        mk(0.24, 0.2, 0.16, f.shirt, 1.15);  // торс
   541→        mk(0.09, 0.3, 0.09, f.shirt, 0.9);   // рука
   542→        mk(0.09, 0.28, 0.09, f.skin, 0.85);  // рука
   543→        mk(0.11, 0.44, 0.12, f.pants, 0.8);  // нога
   544→        if (f.hasPeg) mk(0.09, 0.48, 0.09, 0x8a6a42, 0.75); // деревянная нога
   545→        if (f.hasHook) mk(0.09, 0.09, 0.09, 0x3a3f46, 1.3); // крюк
   546→      }
   547→      FX.particles?.blood(wpos.x, wpos.y, wpos.z, wdir.x * 4, 1.1, wdir.z * 4, 16, 1.4);
   548→    }
   549→  };
   550→
   551→  const vis: CrewVisual = {
   552→    group, total: figures.length, shown: figures.length, dead: 0,
   553→    setVisible(v: boolean) { group.visible = v; },
   554→    killNearPath(a, b, mode, maxCount, radius, shipPos, shipQuat) {
   555→      // ТОЧНОЕ поражение: путь снаряда (a→b, локальные координаты корабля)
   556→      // против вертикальной капсулы КАЖДОЙ фигуры (ноги → голова).
   557→      // Ядро/пуля картечи убивают только фигуру, которую ПРОШЁЛ снаряд —
   558→      // залп в борт рядом с матросом его не трогает.
   559→      const A = new THREE.Vector3(a.x, a.y, a.z);
   560→      const B = new THREE.Vector3(b.x, b.y, b.z);
   561→      const cands: { f: CrewFigure; d: number }[] = [];
   562→      for (const f of figures) {
   563→        if (f.dead || !f.group.visible) continue;
   564→        const p = f.group.position;
   565→        const lo = new THREE.Vector3(p.x, p.y + 0.12, p.z);
   566→        const hi = new THREE.Vector3(p.x, p.y + 1.62, p.z);
   567→        const d = segSegDist3D(A, B, lo, hi);
   568→        if (d > radius) continue;
   569→        cands.push({ f, d });
   570→      }
   571→      if (!cands.length) return 0;
   572→      cands.sort((u, v) => u.d - v.d);
   573→      const n = Math.min(maxCount, cands.length);
   574→      for (let i = 0; i < n; i++) killFigure(cands[i].f, mode, a, b, shipPos, shipQuat);
   575→      vis.dead = figures.filter(f => f.dead).length;
   576→      vis.shown = figures.filter(f => !f.dead && f.group.visible).length;
   577→      return n;
   578→    },
   579→    reset() {
   580→      for (const f of figures) {
   581→        f.dead = false;
   582→        f.deadT = 0;
   583→        f.corpseLife = 0;
   584→        f.deathMode = 'fall';
   585→        f.group.visible = true;
   586→        f.group.rotation.x = 0;
   587→        f.group.rotation.z = 0;
   588→        f.group.scale.set(1, f.baseScaleY, 1);
   589→      }
   590→      vis.dead = 0;
   591→      vis.shown = figures.length;
   592→    },
   593→    update(dt, time, ship) {
   594→      if (ship.sunk) { group.visible = false; return; }
   595→      group.visible = !ship.sinking;
   596→      if (ship.sinking) return;
   597→
   598→      // --- трупы: падение и истлевание ---
   599→      for (const f of figures) {
   600→        if (!f.dead || f.deathMode !== 'fall') continue;
   601→        f.deadT += dt;
   602→        const k = Math.min(1, f.deadT / 0.5);
   603→        const e = 1 - (1 - k) * (1 - k); // easeOut
   604→        f.group.rotation.x = f.fallDir * (Math.PI / 2 - 0.07) * e;
   605→        f.group.rotation.z = f.fallRoll * e;
   606→        if (f.deadT > f.corpseLife) {
   607→          // истлел: усох за 1.2 с
   608→          const fade = Math.max(0, 1 - (f.deadT - f.corpseLife) / 1.2);
   609→          f.group.scale.setScalar(Math.max(0.01, fade));
   610→          if (fade <= 0) f.group.visible = false;
   611→        }
   612→      }
   613→
   614→      // --- потери команды: живых фигур РОВНО столько, сколько
   615→      // людей на палубе — deckCrewFor(crew): 1 фигура = 1 человек.
   616→      // crew = 0 → палуба ПУСТАЕТ (все фигуры погибают).
   617→      // Избыток умирает по одной за кадр — трупы падают естественно
   618→      const target = Math.min(figures.length, deckCrewFor(Math.max(ship.crew, 0)));
   619→      const nAlive = figures.reduce((a, f) => a + (f.dead ? 0 : 1), 0);
   620→      const order: CrewRole[] = ['walker', 'rower', 'watch', 'gunner', 'boatswain', 'helmsman', 'captain'];
   621→      let shownAlive = 0;
   622→      let excess: CrewFigure | null = null; // одна «потеря» за кадр
   623→      if (nAlive > target) {
   624→        outer:
   625→        for (const role of order) {
   626→          for (const f of figures) {
   627→            if (f.role !== role || f.dead) continue;
   628→            if (shownAlive < target) shownAlive++;
   629→            else { excess = f; break outer; }
   630→          }
   631→        }
   632→      } else shownAlive = nAlive;
   633→      if (excess) killFigure(excess, 'grape', excess.group.position,
   634→        excess.group.position.clone().add(new THREE.Vector3(0, 0, 1)),
   635→        ship.position, ship.quaternion);
   636→      vis.dead = figures.reduce((a, f) => a + (f.dead ? 1 : 0), 0);
   637→      vis.shown = shownAlive;
   638→
   639→      // --- анимация живых ---
   640→      const stroke = ship.stroke;
   641→      for (const f of figures) {
   642→        if (!f.group.visible || f.dead) continue;
   643→        if (f.role === 'rower' && stroke && stroke.active) {
   644→          // гребец отклоняется в такт: замах на переносе, наклон
   645→          // корпуса на рабочем ходу (лёгкая асимметрия от фазы)
   646→          const lean = Math.sin(stroke.phase) * stroke.dir;
   647→          f.group.rotation.x = -0.26 * lean;
   648→          f.group.rotation.z = Math.sin(stroke.phase * 2) * 0.03;
   649→        } else if (f.role === 'walker' && f.xPath !== undefined && f.walkA !== undefined && f.walkB !== undefined) {
   650→          const span = f.walkB - f.walkA;
   651→          if ((f.walkPause ?? 0) > 0) {
   652→            // пауза у планширя: осматривается, ноги стоят
   653→            f.walkPause = (f.walkPause ?? 0) - dt;
   654→            const swayP = Math.sin(time * 1.1 + f.phase) * 0.05;
   655→            f.group.rotation.z = swayP;
   656→            f.group.rotation.y = (f.walkDir > 0 ? 0 : Math.PI) + Math.sin(time * 0.5 + f.phase) * 0.7;
   657→            if (f.legL && f.legR) {
   658→              f.legL.rotation.x *= 0.8; f.legR.rotation.x *= 0.8;
   659→            }
   660→          } else {
   661→            f.walkT += (f.walkDir * f.walkSpeed * dt) / span;
   662→            if (f.walkT > 1) { f.walkT = 1; f.walkDir = -1; f.walkPause = 0.7 + f.phase * 2.2; }
   663→            if (f.walkT < 0) { f.walkT = 0; f.walkDir = 1; f.walkPause = 0.7 + (1 - f.phase) * 2.2; }
   664→            const z = THREE.MathUtils.lerp(f.walkA, f.walkB, f.walkT);
   665→            // живой дрейф полосы поперёк палубы (у каждого свой)
   666→            const x = f.xPath + Math.sin(time / (f.driftPeriod ?? 10) + f.phase * 6) * (f.driftAmp ?? 0.15);
   667→            f.group.position.set(x, deckYAt(z), z);
   668→            f.group.rotation.y = f.walkDir > 0 ? 0 : Math.PI;
   669→            f.group.rotation.z = 0;
   670→            const step = time * f.walkSpeed * 7;
   671→            if (f.legL && f.legR) {
   672→              f.legL.rotation.x = Math.sin(step) * 0.55;
   673→              f.legR.rotation.x = -Math.sin(step) * 0.55;
   674→            }
   675→            f.group.position.y += Math.abs(Math.sin(step)) * 0.03;
   676→          }
   677→        } else {
   678→          // idle: лёгкое покачивание, боцман «осматривается»
   679→          const sway = Math.sin(time * 1.3 + f.phase) * 0.035;
   680→          f.group.rotation.z = sway;
   681→          if (f.role === 'boatswain') {
   682→            f.group.rotation.y = 0.5 + Math.sin(time * 0.4 + f.phase) * 0.8;
   683→          } else if (f.role === 'gunner') {
   684→            f.group.rotation.z = sway * 0.5;
   685→          }
   686→        }
   687→      }
   688→    },
   689→  };
   690→  return vis;
   691→}
   692→
