   178→  constructor(presetId: string, isPlayer: boolean, wind: WindState,
   179→    terrainHeight: (x: number, z: number) => number, particles: Particles,
   180→    opts?: ShipOpts) {
   181→    this.preset = PRESETS[presetId];
   182→    this.isPlayer = isPlayer;
   183→    this.wind = wind;
   184→    this.terrainHeight = terrainHeight;
   185→    this.particles = particles;
   186→    this.visual = buildShip(this.preset);
   187→    // прочность по настройке игры (по умолчанию 1× — исторические ТТХ)
   188→    const mul = opts?.durabilityMul ?? 1;
   189→    this.hullHP = this.preset.hullHP * mul;
   190→    this.hullMax = this.hullHP;
   191→    this.sailHP = this.preset.sailHP * mul;
   192→    this.sailMax = this.sailHP;
   193→    this.mastMaxHP = 32 * mul;
   194→    if (opts?.ammo) this.ammo = { ...opts.ammo };
   195→    this.crew = Math.round(this.preset.maxCrew * 0.8);
   196→    this.holdCap = this.preset.cargo * 0.8;
   197→    this.mass = this.preset.mass;
   198→
   199→    const { length: L, beam: B, draft: D, freeboard: F } = this.preset;
   200→    // моменты инерции (связные оси: x — крен? нет: x=право, y=верх, z=нос)
   201→    // крен — вокруг z, дифферент — вокруг x, рыскание — вокруг y
   202→    this.inertia.set(
   203→      this.mass * (B * B + F * F) / 12 * 1.5,   // pitch (x)
   204→      this.mass * (L * L + B * B) / 12 * 1.2,   // yaw (y)
   205→      this.mass * (B * B + F * F) / 12 * 1.5,   // roll (z)
   206→    );
   207→
   208→    // зонды плавучести: 9 шпангоутов × 3 точки (ниже ватерлинии),
   209→    // вплоть до носа/кормы (±0.47 полудлины). Пружины жёсткие
   210→    // (нейтраль 0.38 осадки, запас ×1.35) и демпфер ζ≈0.62 — корпус
   211→    // БЫСТРО всходит на гребень и не проваливается внутрь волны.
   212→    const NZ = 9;
   213→    let n = 0;
   214→    for (let i = 0; i < NZ; i++) {
   215→      const t = i / (NZ - 1);
   216→      const z = THREE.MathUtils.lerp(-L * 0.47, L * 0.47, t);
   217→      const prof = this.halfWidthAt(z);
   218→      const offs = [0, -prof * 0.75, prof * 0.75];
   219→      for (const x of offs) {
   220→        this.probes.push({ off: new THREE.Vector3(x, -D * 0.38, z), k: 0, c: 0 });
   221→        n++;
   222→      }
   223→    }
   224→    const neutral = D * 0.38;
   225→    const kTotal = this.mass * G / neutral * 1.35;
   226→    for (const p of this.probes) {
   227→      p.k = kTotal / n;
   228→      p.c = 2 * Math.sqrt(p.k * (this.mass / n)) * 0.62; // демпфер качки
   229→    }
   230→    // Резервные зонды по скулам борта чуть выше спокойной ватерлинии:
   231→    // моделируют запас плавучести корпуса до планширя — гребень, накрывший
   232→    // палубу, встречает сильное сопротивление и кренящий момент вправку,
   233→    // «полностью по борт» корабль больше не погружается.
   234→    let rn = 0;
   235→    for (let i = 0; i < NZ; i++) {
   236→      const t = i / (NZ - 1);
   237→      const z = THREE.MathUtils.lerp(-L * 0.44, L * 0.44, t);
   238→      const prof = this.halfWidthAt(z) * 0.85;
   239→      for (const x of [-prof, prof]) {
   240→        this.reserve.push({ off: new THREE.Vector3(x, F * 0.34, z), k: 0, c: 0 });
   241→        rn++;
   242→      }
   243→    }
   244→    // жёстче прежнего: полный вес запаса уже на 15 см погружения борта —
   245→    // волна не может накрыть палубу, корпус «всплывает» на гребень
   246→    const rkTotal = this.mass * G / 0.15;
   247→    for (const p of this.reserve) {
   248→      p.k = rkTotal / rn;
   249→      p.c = 2 * Math.sqrt(p.k * (this.mass / rn)) * 1.3;
   250→    }
   251→
   252→    this.foam = new WaterlineFoam(this.visual.outline);
   253→    this.wake.setWaterHeight(waterHeightAt);
   254→    if (isPlayer) enableLantern(this.visual);
   255→
   256→    // экипаж на палубе: процедурные типажи, сид фиксирован по модели
   257→    const seed = Array.from(presetId).reduce((a, ch) => a + ch.charCodeAt(0) * 31, 11)
   258→      + Math.round(this.preset.length * 7);
   259→    this.crew3d = buildCrew(this.preset, seed);
   260→    this.visual.group.add(this.crew3d.group);
   261→    // ИИ-гребцы ходят на вёслах постоянно (у игрока вёсла вываливаются
   262→    // клавишей или командой гребли Alt+W/S)
   263→    if (!isPlayer && this.preset.oars) {
   264→      this.oarsOut = true;
   265→      this.rowing = 1;
   266→    }
   267→    for (let i = 0; i < this.visual.masts.length; i++) {
   268→      this.mastHP.push(this.mastMaxHP);
   269→      this.mastFall.push(-1);
   270→      this.mastSplashed.push(false);
   271→    }
   272→  }

   274→  halfWidthAt(z: number): number {
   275→    const t = clamp((z + this.preset.length / 2) / this.preset.length, 0, 1);
   276→    let w: number;
   277→    if (t < 0.45) w = THREE.MathUtils.lerp(0.58, 1.0, t / 0.45);
   278→    else w = THREE.MathUtils.lerp(1.0, 0.0, Math.pow(clamp((t - 0.45) / 0.55, 0, 1), 0.75));
   279→    return Math.max(0.3, w * this.preset.beam * 0.5);
   280→  }

   295→  /** Угол между направлением ОТКУДА дует (кажущийся) ветер и курсом, 0..PI */
   296→  apparentWindAngle(): number {
   297→    const wf = this.tmpV.set(Math.cos(this.wind.from), 0, Math.sin(this.wind.from));
   298→    // кажущийся ветер = истинный - скорость корабля
   299→    const av = this.tmpV2.set(-wf.x * this.wind.speed - this.velocity.x, 0, -wf.z * this.wind.speed - this.velocity.z);
   300→    // вектор ОТКУДА дует кажущийся ветер:
   301→    const aFromX = -av.x, aFromZ = -av.z;
   302→    const len = Math.hypot(aFromX, aFromZ) || 1;
   303→    const f = this.forward;
   304→    const dot = (aFromX / len) * f.x + (aFromZ / len) * f.z;
   305→    return Math.acos(clamp(dot, -1, 1));
   306→  }

   308→  /** Полярная эффективность парусов по углу (рад): лавировка/бейдевинд/галфвинд/фордевинд */
   309→  static polar(awa: number): number {
   310→    const d = awa * 180 / Math.PI;
   311→    if (d < 25) return 0.02;                       // левентик — почти нет хода
   312→    if (d < 60) return THREE.MathUtils.lerp(0.08, 0.7, (d - 25) / 35);
   313→    if (d < 110) return THREE.MathUtils.lerp(0.7, 1.0, (d - 60) / 50);  // бейдевинд-галфвинд
   314→    if (d < 150) return THREE.MathUtils.lerp(1.0, 0.82, (d - 110) / 40);
   315→    return THREE.MathUtils.lerp(0.82, 0.68, (d - 150) / 30);            // фордевинд
   316→  }

   318→  // ---------- фиксированный шаг физики ----------
   319→  fixedStep(dt: number, time: number) {
   320→    if (this.sunk) return;
   321→    // затопление — ОТДЕЛЬНАЯ кинематика (фазы крена/погружения/спирали):
   322→    // пружины плавучести уже не нужны, корпус ведёт себя как тело,
   323→    // теряющее остойчивость
   324→    if (this.sinking) { this.stepSinking(dt, time); return; }
   325→    this.force.set(0, 0, 0);
   326→    this.torque.set(0, 0, 0);
   327→
   328→    const f = this._f.copy(this.forward);
   329→    // штирборт (правый борт): forward × up = (−f.z, 0, f.x)
   330→    const right = this._r.set(-f.z, 0, f.x).normalize();
   331→    const e = this.euler;
   332→    const spd = this.speed();
   333→
   334→    const m = this.mass + this.water * 1000; // вода в трюме утяжеляет
   335→
   336→    // --- гравитация ---
   337→    this.force.y -= m * G;
   338→
   339→    // --- плавучесть по зондам (τ = r × F, F = (0, fl, 0)) ---
   340→    for (const p of this.probes) {
   341→      const wp = this._t1.copy(p.off).applyQuaternion(this.quaternion).add(this.position);
   342→      const wy = waterHeightAt(wp.x, wp.z, time);
   343→      const depth = wy - wp.y;
   344→      if (depth > 0) {
   345→        const rx = wp.x - this.position.x, rz = wp.z - this.position.z;
   346→        // вертикальная скорость точки: v = ω × r → vy = ωz*rx − ωx*rz
   347→        const vy = this.velocity.y + (this.angular.z * rx - this.angular.x * rz);
   348→        let fl = p.k * Math.min(depth, 6) - p.c * vy;
   349→        if (fl < 0) fl = 0;
   350→        this.force.y += fl;
   351→        this.torque.x += -rz * fl;
   352→        this.torque.z += rx * fl;
   353→      }
   354→    }
   355→    // --- запас плавучести бортов (пока не тонем) ---
   356→    if (!this.sinking) {
   357→      for (const p of this.reserve) {
   358→        const wp = this._t1.copy(p.off).applyQuaternion(this.quaternion).add(this.position);
   359→        const wy = waterHeightAt(wp.x, wp.z, time);
   360→        const depth = wy - wp.y;
   361→        if (depth > 0) {
   362→          const rx = wp.x - this.position.x, rz = wp.z - this.position.z;
   363→          const vy = this.velocity.y + (this.angular.z * rx - this.angular.x * rz);
   364→          let fl = p.k * Math.min(depth, 2.5) - p.c * vy;
   365→          if (fl < 0) fl = 0;
   366→          this.force.y += fl;
   367→          this.torque.x += -rz * fl;
   368→          this.torque.z += rx * fl;
   369→        }
   370→      }
   371→    }
   372→
   373→    // --- ветер на парусах ---
   374→    this.sailFrac += (this.sailTarget / 3 - this.sailFrac) * Math.min(1, dt * 0.7);
   375→    const awa = this.apparentWindAngle();
   376→    const polar = Ship.polar(awa);
   377→    const sailEffMul = this.sailHP / this.preset.sailHP;
   378→    const damaged = sailEffMul < 0.5 ? 0.45 : sailEffMul < 0.85 ? 0.8 : 1.0;
   379→    const windPower = Math.pow(this.wind.speed / 8.0, 1.6);
   380→    let thrust = 0;
   381→    if (!this.anchored && !this.sinking && this.mastsBroken < 2) {
   382→      thrust = this.sailFrac * polar * windPower * this.preset.maxSpeed * 7.0 * damaged * (this.sailHP > 0 ? 1 : 0);
   383→    }
   384→    if (this.reverseThrust && !this.anchored) thrust = -this.preset.maxSpeed * 0.45; // задний ход на вёслах
   385→    // ГРЕБЛЯ (Alt+W/S): вёсла дают ход даже в полный штиль — но только
   386→    // ВЫВАЛЕННЫЕ (oarsDeployT); под парусами экипаж занят парусами —
   387→    // вёсла помогают меньше
   388→    if (this.preset.oars && !this.anchored && !this.sinking) {
   389→      const crewFrac = clamp(this.crew / this.preset.maxCrew, 0.2, 1);
   390→      const power = this.oarsDeployT * (1 - 0.55 * this.sailFrac) * crewFrac;
   391→      if (this.rowing > 0) thrust += this.preset.maxSpeed * 0.40 * power;
   392→      else if (this.rowing < 0) thrust -= this.preset.maxSpeed * 0.28 * power;
   393→    }
   394→    // сила тяги вдоль forward
   395→    this.force.x += f.x * thrust * m * 0.06;
   396→    this.force.z += f.z * thrust * m * 0.06;
   397→    // крен от ветра: боковая составляющая ветра на парусах
   398→    // (умеренный: в шторм подветренный борт не зарывается в волну)
   399→    const wf = new THREE.Vector3(Math.cos(this.wind.from), 0, Math.sin(this.wind.from));
   400→    const windTravel = wf.clone().multiplyScalar(-this.wind.speed);
   401→    const lateral = windTravel.dot(right);
   402→    const heelF = lateral * this.sailFrac * windPower * m * 0.034;
   403→    this.torque.z += heelF;
   404→    // ... и небольшое боковое сносовое усилие
   405→    this.force.x += right.x * lateral * this.sailFrac * m * 0.004;
   406→    this.force.z += right.z * lateral * this.sailFrac * m * 0.004;
   407→
   408→    // --- руль ---
   409→    this.rudder += (this.rudderTarget - this.rudder) * Math.min(1, dt * 1.8);
   410→    const rudderAngle = this.rudder * 0.6; // рад, макс ~34°
   411→    const yawAuth = clamp(spd / 4.0, 0, 1.15); // авторитеты руля от скорости
   412→    // положительный руль (D) = поворот ВПРАВО (штирборт):
   413→    // вращение вокруг −Y, т.к. +Y = поворот влево (к порту)
   414→    this.torque.y -= rudderAngle * yawAuth * m * 12.0; // ~6°/с на полном ходу
   415→    // крен на повороте (отцентробежный)
   416→    this.torque.z += this.angular.y * spd * m * 0.35;
   417→    // лёгкое зарывание носа на скорости (слабое: сильное топило борт
   418→    // в волну на 3+ метра — см. жалобы «корабль погружается по борт»)
   419→    this.torque.x += spd * spd * m * 0.0012;
   420→
   421→    // --- сопротивление (по связным осям) ---
   422→    const vLocal = this.velocity.clone();
   423→    const fwdSpd = vLocal.dot(f);
   424→    const latSpd = vLocal.dot(right);
   425→    // КВАДРАТИЧНОЕ сопротивление: полином (18v + 55v|v|) сам нечётен по v,
   426→    // умножение на −sign(v) ИНВЕРТИРОВАЛО его при заднем ходе — попятное
   427→    // движение разгонялось сопротивлением до 20+ уз и NaN (задний ход
   428→    // вёслами/парусами). Теперь сопротивление всегда ПРОТИВ скорости.
   429→    const fwdDrag = -(18 * fwdSpd + 55 * fwdSpd * Math.abs(fwdSpd)) * m * 0.001;
   430→    const latDrag = -latSpd * (3.2) * m * 0.1;
   431→    this.force.x += f.x * fwdDrag + right.x * latDrag;
   432→    this.force.z += f.z * fwdDrag + right.z * latDrag;
   433→    this.force.y -= this.velocity.y * m * 0.12; // вертикальное демпфирование
   434→    if (this.anchored) {
   435→      this.force.x -= this.velocity.x * m * 1.2;
   436→      this.force.z -= this.velocity.z * m * 1.2;
   437→    }
   438→    if (this.grounded) {
   439→      this.force.x -= this.velocity.x * m * 2.5;
   440→      this.force.z -= this.velocity.z * m * 2.5;
   441→    }
   442→
   443→    // --- угловое демпфирование по связным осям ---
   444→    const invQ = this.tmpQ.copy(this.quaternion).invert();
   445→    const ab = this._t3.copy(this.angular).applyQuaternion(invQ); // связные
   446→    ab.x *= Math.exp(-1.6 * dt);  // дифферент
   447→    ab.y *= Math.exp(-0.9 * dt);  // рыскание
   448→    ab.z *= Math.exp(-1.1 * dt);  // крен
   449→    this.angular.copy(ab.applyQuaternion(this.quaternion));
   450→
   451→    // --- ограничение крена/дифферента (мягкие стопы) ---
   452→    // дифферент 0.30 рад (17°) — нос всходит на передний склон
   453→    // штормового гребня, а не прорезает его;
   454→    // крен 0.36 рад (20°) — подветренный борт остаётся над водой
   455→    const roll = e.z, pitch = e.x;
   456→    const rollLim = 0.36, pitchLim = 0.30;
   457→    if (Math.abs(roll) > rollLim) this.torque.z -= (roll - Math.sign(roll) * rollLim) * m * 9.0;
   458→    if (Math.abs(pitch) > pitchLim) this.torque.x -= (pitch - Math.sign(pitch) * pitchLim) * m * 12.0;
   459→    // затопление кренит корабль в сторону пробоин
   460→    this.floodRoll(m);
   461→
   462→    // --- интеграция ---
   463→    this.velocity.addScaledVector(this.force, dt / m);
   464→    this.position.addScaledVector(this.velocity, dt);
   465→    // α = I⁻¹ τ в связных осях
   466→    const tb = this._t3.copy(this.torque).applyQuaternion(invQ);
   467→    tb.x /= this.inertia.x; tb.y /= this.inertia.y; tb.z /= this.inertia.z;
   468→    this.angular.addScaledVector(tb.applyQuaternion(this.quaternion), dt);
   469→    // ограничим угловые скорости
   470→    const wLen = this.angular.length();
   471→    if (wLen > 1.1) this.angular.multiplyScalar(1.1 / wLen);
   472→    // q̇ = 0.5 · ω⊗q → q_new = q + 0.5·dt·(ωx,ωy,ωz,0)⊗q, потом нормализация
   473→    this.wq.set(this.angular.x, this.angular.y, this.angular.z, 0);
   474→    this.dq.copy(this.wq).multiply(this.quaternion); // dq = ω ⊗ q
   475→    const qs = 0.5 * dt;
   476→    this.quaternion.set(
   477→      this.quaternion.x + this.dq.x * qs,
   478→      this.quaternion.y + this.dq.y * qs,
   479→      this.quaternion.z + this.dq.z * qs,
   480→      this.quaternion.w + this.dq.w * qs
   481→    ).normalize();
   482→
   483→    // === КИНЕМАТИЧЕСКАЯ ПОСАДКА НА ВОЛНУ (плывёт ПОВЕРХНОСТЬЮ, без парения) ===
   484→    // Итог четырёх итераций:
   485→    //  1) среднее 4 точек — гребень проходил СКВОЗЬ корпус (жалоба);
   486→    //  2) жёсткая огибающая (max по станциям) — корабль «повис над водой»;
   487→    //  3) среднее + подъём ≤ 0.45D — в шторм гребень выше осадки, корпус
   488→    //     снова резал волну (жалоба «уходит внутрь волны»);
   489→    //  4) текущий баланс:
   490→    //   - ВЕРТИКАЛЬ — средневзвешенный по ширине уровень + МЯГКАЯ
   491→    //     огибающая с потолком 0.45D + 0.7·огибающая_волнения: на большом
   492→    //     гребне потолок растёт вместе с волной — корпус ВЗХОДИТ на гребень;
   493→    //     в штиле волны ниже 0.25D — подъём не включается вовсе (нет парения);
   494→    //   - ДИФФЕРЕНТ — МНК-наклон поверхности (вес по ширине): нос задирается
   495→    //     заранее, на переднем склоне приближающегося гребня.
   496→    // Крен остаётся живым (ветер/течь/поворот) — пружины бортов.
   497→    if (!this.sinking && !this.sunk && !this.grounded) {
   498→      const L = this.preset.length, D = this.preset.draft;
   499→      const FR = this._st;
   500→      const sinP = Math.sin(e.x);
   501→      const halfB = Math.max(0.5, this.preset.beam * 0.5);
   502→      // проход 1: высоты воды и веса станций (ширина корпуса в точке)
   503→      let wSum = 0, whSum = 0, whzSum = 0, wz2Sum = 0;
   504→      for (let i = 0; i < FR.length; i++) {
   505→        const zz = FR[i] * L;
   506→        const h = waterHeightAt(this.position.x + f.x * zz, this.position.z + f.z * zz, time);
   507→        // вес 0.25..1: оконечности узкие — их гребни корпус не поднимают всем весом
   508→        const w = Math.max(0.25, this.halfWidthAt(zz) / halfB);
   509→        this._hs[i] = h;
   510→        wSum += w;
   511→        whSum += w * h;
   512→        whzSum += w * h * zz;
   513→        wz2Sum += w * zz * zz;
   514→      }
   515→      // спокойная посадка: центр на 0.10·D над средневзвешенным уровнем;
   516→      // затопленный трюм садит корпус ниже (до 30 см)
   517→      const sink = clamp(this.water / Math.max(1, this.holdCap), 0, 1) * 0.30;
   518→      const meanY = whSum / wSum + D * 0.10 - sink;
   519→      // проход 2: мягкая огибающая против зарывания оконечностей.
   520→      // Заглубление киля станции при среднем уровне: dig = h + z·sinP + 0.9D − meanY;
   521→      // нормальная осадка даёт dig ≈ 0.8D — лишнее свыше 1.15D выталкиваем,
   522→      // потолок подъёма растёт с волнением (шторм = корпус идёт по гребню)
   523→      let lift = 0;
   524→      for (let i = 0; i < FR.length; i++) {
   525→        const zz = FR[i] * L;
   526→        const dig = this._hs[i] + zz * sinP + 0.9 * D - meanY;
   527→        const l = dig - 1.15 * D;
   528→        if (l > lift) lift = l;
   529→      }
   530→      const liftCap = 0.45 * D + 0.70 * waveEnvelope();
   531→      const targetY = meanY + clamp(lift, 0, liftCap);
   532→      const kY = 1 - Math.exp(-dt * 6.0);
   533→      const corrY = clamp((targetY - this.position.y) * kY, -17 * dt, 17 * dt);
   534→      this.position.y += corrY;
   535→      // согласованное затухание вертикальной скорости — пружины
   536→      // не раскачивают корпус вокруг кинематической посадки
   537→      this.velocity.y *= Math.exp(-dt * 3.0);
   538→      // дифферент: МНК-наклон поверхности вдоль ДП (вода выше у носа → нос вверх)
   539→      const wPitch = -Math.atan2(whzSum, wz2Sum);
   540→      const e2 = this._e2.setFromQuaternion(this.quaternion, 'YXZ');
   541→      const kP = 1 - Math.exp(-dt * 5.0);
   542→      e2.x += clamp((wPitch - e2.x) * kP, -1.2 * dt, 1.2 * dt);
   543→      this.quaternion.setFromEuler(e2);
   544→    }
   545→
   546→    // --- посадка на мель / столкновение с землёй ---
   547→    this.checkGrounding(dt);
   548→
   549→    // --- течение воды в трюме ---
   550→    if (this.leakRate > 0) {
   551→      this.water = Math.min(this.holdCap, this.water + this.leakRate * dt);
   552→    }
   553→    if (this.onFire) {
   554→      this.fireT += dt;
   555→      const I = this.fireIntensity;
   556→      // рост: сухая палуба разгорается, сильный ветер раздувает;
   557→      // после 40 с горения топливо прогорает — интенсивность падает
   558→      let dI: number;
   559→      if (this.fireT > 40) dI = -0.05;                             // догорело
   560→      else dI = 0.008 + 0.022 * I + Math.min(0.012, this.wind.speed * 0.0009);
   561→      // дождь тушит
   562→      dI -= this.rainRate * 0.03;
   563→      this.fireIntensity = clamp(this.fireIntensity + dI * dt, 0, 1);
   564→      if (this.fireIntensity <= 0.01) {
   565→        this.onFire = false; this.fireT = 0;
   566→      } else {
   567→        // урон: корпус горит всегда, паруса — от 0.4, команда — от 0.6,
   568→        // такелаж вспыхивает на полном огне
   569→        this.hullHP -= dt * (0.5 + 2.8 * I);
   570→        if (I > 0.4) this.sailHP = Math.max(0, this.sailHP - dt * I * 2.2);
   571→        if (I > 0.6) this.crew = Math.max(0, this.crew - dt * 0.22 * I);
   572→        if (I > 0.9 && Math.random() < dt * 0.05) this.breakMast();
   573→        // визуал: частота и размер пламени от интенсивности
   574→        if (Math.random() < dt * (0.8 + 3.2 * I)) {
   575→          const p = this.particles;
   576→          const wp = this.position;
   577→          const spread = 3 + I * 6;
   578→          p.fire(wp.x + (Math.random() - 0.5) * spread, this.waterlineY() + 0.8 + Math.random() * 2.5,
   579→            wp.z + (Math.random() - 0.5) * spread * 2.2);
   580→          p.smoke(wp.x + (Math.random() - 0.5) * spread, this.waterlineY() + 3,
   581→            wp.z + (Math.random() - 0.5) * spread * 2.2,
   582→            this.wind ? Math.cos(this.wind.from + Math.PI) * 2 : 0, 2.2,
   583→            this.wind ? Math.sin(this.wind.from + Math.PI) * 2 : 0,
   584→            2.2 + I * 1.6, 2.6 + I);
   585→        }
   586→      }
   587→    }
   588→    // порог затопления
   589→    if (!this.sinking && this.water > this.holdCap * 0.62) this.startSinking();
   590→    if (this.hullHP <= 0 && !this.sinking) this.startSinking();
   591→
   592→    // --- сдался? ---
   593→    if (!this.isPlayer && !this.surrendered && !this.sinking) {
   594→      if (this.hullHP < this.hullMax * 0.34 || this.crew < this.preset.maxCrew * 0.25) {
   595→        this.surrendered = true;
   596→        this.sailTarget = 0;
   597→        this.onSurrender?.(this);
   598→      }
   599→    }
   600→    // (момент гибели считает stepSinking — фазовая машина затопления)
   601→  }
   602→
   603→  private checkGrounding(dt: number) {
   604→    if (this.sinking || this.sunk) return;
   605→    const f = this.forward;
   606→    const bow = this.tmpV.copy(this.position).addScaledVector(f, this.preset.length * 0.42);
   607→    const stern = this.tmpV2.copy(this.position).addScaledVector(f, -this.preset.length * 0.42);
   608→    let hit = false;
   609→    for (const p of [bow, stern]) {
   610→      const th = this.terrainHeight(p.x, p.z);
   611→      if (th > this.position.y - this.preset.draft * 0.6) {
   612→        hit = true;
   613→        // нормаль склона
   614→        const e = 2;
   615→        const nx = this.terrainHeight(p.x - e, p.z) - this.terrainHeight(p.x + e, p.z);
   616→        const nz = this.terrainHeight(p.x, p.z - e) - this.terrainHeight(p.x, p.z + e);
   617→        const n = new THREE.Vector3(nx, 2 * e, nz).normalize();
   618→        this.force.addScaledVector(n, this.mass * 22);
   619→        this.velocity.multiplyScalar(0.94);
   620→      }
   621→    }
   622→    if (hit) {
   623→      if (!this.grounded && this.speed() > 3.5) {
   624→        this.applyDamage({ kind: 'ground', amount: 5 + this.speed() });
   625→      }
   626→      this.grounded = true;
   627→    } else this.grounded = false;
   628→  }
   629→
   630→  waterlineY(): number {
   631→    return this.position.y;
   632→  }
   633→
   634→  // ---------- повреждения ----------
   635→  applyDamage(d: DamageInput) {
   636→    if (this.sunk || this.sinking) return;
   637→    const rng = Math.random;
   638→    switch (d.kind) {
   639→      case 'ball': {
   640→        this.hullHP -= d.amount;
   641→        if (d.hitPoint && d.hitPoint.y < 0.45 + this.position.y) this.addLeak(0.06 + rng() * 0.05);
   642→        // ЭКИПАЖ гибнет ТОЛЬКО от прямого прострела палубы (CombatSystem
   643→        // → crew3d.killNearPath). Залп в борт ниже палубы людей не косит:
   644→        // доска и переборки принимают удар на себя
   645→        // случайный «рикошет в рангоут» — редкий, основную работу делают
   646→        // прицельные попадания из CombatSystem (mast hits)
   647→        if (rng() < 0.10) this.breakMast();
   648→        break;
   649→      }
   650→      case 'chain': {
   651→        this.sailHP = Math.max(0, this.sailHP - d.amount);
   652→        // книппель рвёт такелаж над головами — люди гибнут только от
   653→        // обрывов рангоута (breakMast) и прямых прострелов
   654→        break;
   655→      }
   656→      case 'grape': {
   657→        // экипаж гибнет ТОЛЬКО фигурами на палубе (combat вычитает
   658→        // 1 за каждую убитую) — абстрактного списания
   659→        // нет: залп в борт людей не косит
   660→        this.hullHP -= d.amount * 0.4;
   661→        break;
   662→      }
   663→      case 'fire': {
   664→        this.hullHP -= d.amount;
   665→        if (rng() < 0.35) this.ignite(0.3 + rng() * 0.25); // очаг на палубе
   666→        if (rng() < 0.08) { this.addLeak(0.03); }
   667→        break;
   668→      }
   669→      case 'ram': {
   670→        this.hullHP -= d.amount;
   671→        if (rng() < 0.5) this.addLeak(0.04);
   672→        break;
   673→      }
   674→      case 'ground': this.hullHP -= d.amount * 0.5; break;
   675→    }
   676→    if (this.hullHP <= 0 && !this.sinking) this.startSinking();
   677→    if (!this.isPlayer && !this.noSurrender && !this.surrendered && this.hullHP < this.hullMax * 0.34) {
   678→      this.surrendered = true; this.sailTarget = 0; this.onSurrender?.(this);
   679→    }
   680→  }

   682→  /** Урон в КОНКРЕТНУЮ мачту (прицельное попадание ядром/книппелем) */
   683→  damageMast(idx: number, amount: number): boolean {
   684→    if (idx < 0 || idx >= this.mastHP.length) return false;
   685→    if (this.mastFall[idx] >= 0) return false; // уже сломана
   686→    this.mastHP[idx] -= amount;
   687→    if (this.mastHP[idx] <= 0) { this.breakMast(idx); return true; }
   688→    return false;
   689→  }
   690→
   691→  addLeak(rate: number) {
   692→    this.leakRate = Math.min(1.8, this.leakRate + rate);
   693→    if (this.leakRate < 0) this.leakRate = 0;
   694→  }
   695→
   696→  /** Зажечь палубу: strength — начальная сила очага 0..1.
   697→   *  Существующий пожар не ослабляет — новое пламя добавляет силы */
   698→  ignite(strength: number): boolean {
   699→    if (this.sinking || this.sunk) return false;
   700→    this.onFire = true;
   701→    this.fireIntensity = clamp(Math.max(this.fireIntensity, strength), 0, 1);
   702→    return true;
   703→  }
   704→
   705→  /** Урон ПАРУСАМ: пробоины картечи/книппеля снижают тягу (sailHP).
   706→   *  Дыры рисуются отдельно — см. combat.applySailHit + sailAddHole. */
   707→  damageSails(n: number) {
   708→    this.sailHP = Math.max(0, this.sailHP - n);
   709→  }
   710→
   711→  /** Пробоина в борту: тёмная рваная дыра + торчащие щепки-доски
   712→   *  вокруг + течь (ниже ватерлинии) + кренящий момент */
   713→  addHole(localPos: THREE.Vector3, belowWL: boolean) {
   714→    const side = localPos.x >= 0 ? 1 : -1;
   715→    const w = Math.max(0.3, this.halfWidthAt(localPos.z) * 0.985);
   716→    // группа: дыра + щепки (одна запись в hullHoles)
   717→    const hole = new THREE.Group();
   718→    const R = 0.36 + Math.random() * 0.18;
   719→    const mat = new THREE.MeshStandardMaterial({ color: 0x120d08, roughness: 1 });
   720→    const mesh = new THREE.Mesh(new THREE.CircleGeometry(R, 9), mat);
   721→    mesh.rotation.y = side * Math.PI / 2;
   722→    mesh.rotation.x = (Math.random() - 0.5) * 0.5;
   723→    hole.add(mesh);
   724→    // рваные доски по краям — вывернуло обшивку
   725→    const nSp = 3 + Math.floor(Math.random() * 3);
   726→    for (let i = 0; i < nSp; i++) {
   727→      const a = Math.random() * Math.PI * 2;
   728→      const r = R * (0.75 + Math.random() * 0.5);
   729→      const sp = new THREE.Mesh(
   730→        new THREE.BoxGeometry(0.05, 0.07, 0.35 + Math.random() * 0.5),
   731→        new THREE.MeshStandardMaterial({ color: 0x4a3420, roughness: 0.95 }));
   732→      sp.position.set(side * 0.06, Math.sin(a) * r, Math.cos(a) * r);
   733→      sp.rotation.set((Math.random() - 0.5) * 0.9, side * Math.PI / 2 + (Math.random() - 0.5) * 0.8, (Math.random() - 0.5) * 0.9);
   734→      hole.add(sp);
   735→    }
   736→    hole.position.set(side * w, localPos.y, localPos.z);
   737→    this.visual.group.add(hole);
   738→    this.hullHoles.push({ side, belowWL, mesh: hole as unknown as THREE.Mesh });
   739→    if (belowWL) {
   740→      this.addLeak(0.16 + Math.random() * 0.10);
   741→      this.leakBias = clamp(this.leakBias + side * 0.35, -1, 1);
   742→    }
   743→  }

   745→  /** Сорвать ломаемую деталь палубы (штурвал/шпиль/фонарь/фигура/руль)
   746→   *  прямым попаданием: деталь исчезает, щепа летит. Возвращает true,
   747→   *  если что-то сорвали */
   748→  damageProp(localHit: THREE.Vector3, power: number): boolean {
   749→    if (power < 6) return false; // лёгкие снаряды детали не срывают
   750→    let best: THREE.Object3D | null = null;
   751→    let bestD = 4.2; // радиус срыва
   752→    for (const obj of this.visual.breakables) {
   753→      if (!obj.visible) continue;
   754→      const d = Math.hypot(obj.position.x - localHit.x, (obj.position.y - localHit.y) * 0.7,
   755→        obj.position.z - localHit.z);
   756→      if (d < bestD) { bestD = d; best = obj; }
   757→    }
   758→    if (!best) return false;
   759→    best.visible = false;
   760→    const wp = this.tmpV.copy(best.position).applyQuaternion(this.quaternion).add(this.position);
   761→    const gibs = Ship.gibs;
   762→    // шлюпка разлетается ЩЕПОЙ ПОКРУПНЕЕ (доски вразлёт)
   763→    const isBoat = !!(best.userData as { deckBoat?: boolean }).deckBoat;
   764→    if (gibs) {
   765→      const n = isBoat ? 14 : 6;
   766→      for (let i = 0; i < n; i++) {
   767→        gibs.spawnPart(
   768→          new THREE.Vector3(wp.x + (Math.random() - 0.5) * (isBoat ? 1.8 : 0.6), wp.y + Math.random() * (isBoat ? 1.0 : 0.8), wp.z + (Math.random() - 0.5) * (isBoat ? 1.8 : 0.6)),
   769→          new THREE.Vector3((Math.random() - 0.5) * (isBoat ? 10 : 7), 2.5 + Math.random() * (isBoat ? 5 : 4), (Math.random() - 0.5) * (isBoat ? 10 : 7)),
   770→          isBoat ? 0.24 : 0.16, 0.1, 0.55, isBoat && Math.random() < 0.55 ? 0x8a6a44 : 0x5c4430, 1.8 + Math.random());
   771→      }
   772→    }
   773→    this.particles.smoke(wp.x, wp.y, wp.z, 0, isBoat ? 1.1 : 0.6, 0, 1.0, isBoat ? 1.6 : 1.1);
   774→    return true;
   775→  }

   779→    if (this.water <= 0 || this.sinking) return;

   784→  breakMast(idx?: number) {
   785→    let i = idx;
   786→    if (i === undefined || i < 0 || i >= this.visual.masts.length || this.mastFall[i] >= 0) {
   787→      // без указания — случайная целая мачта
   788→      const intact = this.mastFall.map((f, k) => f < 0 ? k : -1).filter(k => k >= 0);
   789→      if (!intact.length) return;
   790→      i = intact[Math.floor(Math.random() * intact.length)];
   791→    }
   792→    this.mastFall[i] = 0;
   793→    this.mastsBroken = this.mastFall.filter(f => f >= 0).length;
   794→    this.sailHP = Math.max(0, this.sailHP - 30);
   795→    if (this.sailTarget > 0) this.sailTarget = Math.max(0, this.sailTarget - 1);
   796→    // === КАСКАД: вместе с мачтой гибнут ВСЕ её элементы ===
   797→    // паруса этой мачты РВЁТСЯ: полотно изрешечено (uDamage=1 + дыры),
   798→    // обвисает и больше не даёт тягу; обрывки парусины летят вниз
   799→    let tornSails = 0;
   800→    for (const s of this.visual.sails.sails) {
   801→      if (s.mastIndex !== i) continue;
   802→      tornSails++;
   803→      s.mat.uniforms.uDamage.value = 1;
   804→      sailAddHole(s, 0.3 + Math.random() * 0.4, 0.25 + Math.random() * 0.5, 0.22);
   805→      sailAddHole(s, 0.15 + Math.random() * 0.7, 0.5 + Math.random() * 0.4, 0.18);
   806→      sailAddHole(s, 0.4 + Math.random() * 0.3, 0.1 + Math.random() * 0.3, 0.15);
   807→      s.furl = 1;
   808→    }
   809→    // снасть рушится с высоты — обрывки парусины и щепа рея
   810→    const mast = this.visual.masts[i];
   811→    const mid = this.tmpV.set(0, ((mast.userData.height as number) || 20) * 0.55, mast.position.z)
   812→      .applyQuaternion(this.quaternion).add(this.position);
   813→    if (tornSails > 0 && this.particles) {
   814→      for (let k = 0; k < 5; k++) {
   815→        this.particles.spawn(mid.x + (Math.random() - 0.5) * 6, mid.y - Math.random() * 4,

   819→      }
   820→      this.particles.smoke(mid.x, mid.y, mid.z, 0, 0.4, 0, 0.8, 1.4);
   821→    }
   822→    // === ОБВЯЗКА МАЧТЫ ЛОПАЕТСЯ ВМЕСТЕ С НЕЙ: все её ванты/
   823→    // штаги/бакштаги схлопываются + обрывки троса летят гибами ===
   824→    this.snapRigging(i);
   825→  }

   827→  /** Обрыв обвязки мачты idx: её снасти (ванты/штаги) вырождаются
   828→   *  в точку (линия исчезает), обрывки троса летят из середины
   829→   *  снасти. Возвращает число оборванных снастей */
   830→  private snapRigging(idx: number): number {
   831→    const rig = this.visual.rigging;
   832→    if (!rig || !rig.meta.length) return 0;
   833→    const posAttr = rig.lines.geometry.attributes.position as THREE.BufferAttribute;
   834→    const mast = this.visual.masts[idx];
   835→    const bx = 0, by = 0.4, bz = mast ? mast.position.z : 0;
   836→    let n = 0;
   837→    const snappedAt: THREE.Vector3[] = [];
   838→    for (let li = 0; li < rig.meta.length; li++) {
   839→      const l = rig.meta[li];
   840→      if (l.mast !== idx) continue;
   841→      // вырожденный отрезок (ноль длины) — линия не рисуется
   842→      posAttr.setXYZ(li * 2, bx, by, bz);
   843→      posAttr.setXYZ(li * 2 + 1, bx, by, bz);
   844→      n++;
   845→      if (snappedAt.length < 6) snappedAt.push(l.a.clone().lerp(l.b, 0.45));
   846→    }
   847→    if (n > 0) {
   848→      posAttr.needsUpdate = true;
   849→      this.rigSnapped = Math.min(this.rigSnapped + n, rig.meta.length);
   850→    }
   851→    // ДЕРЖАЩИЕ ЭЛЕМЕНТЫ падают вместе с мачтой: ванты с выбленками
   852→    // из группы мелких деталей — своя пара линий на мачту — исчезает
   853→    // (жалоба «тросы не упали вместе с мачтой»: обрывалась только
   854→    // обвязка, а веер вант с выбленками висел в воздухе)
   855→    if (this.visual.detailRigs?.length) {
   856→      for (const dr of this.visual.detailRigs) {
   857→        if (dr.mast !== idx || !dr.lines.visible) continue;
   858→        dr.lines.visible = false;
   859→        n++;
   860→        // обрывки троса из середины вант
   861→        const g = Ship.gibs;
   862→        for (let k = 0; k < 3 && g; k++) {
   863→          const sp = this.tmpV.set((Math.random() - 0.5) * 4, 2 + Math.random() * 3,
   864→            this.visual.masts[idx].position.z + (Math.random() - 0.5) * 4)
   865→            .applyQuaternion(this.quaternion).add(this.position);
   866→          g.spawnPart(
   867→            new THREE.Vector3(sp.x, sp.y, sp.z),
   868→            new THREE.Vector3((Math.random() - 0.5) * 3, 1 + Math.random() * 2, (Math.random() - 0.5) * 3),
   869→            0.05, 0.05, 0.7 + Math.random() * 0.5, 0x241c12, 1.5 + Math.random());
   870→        }
   871→      }
   872→    }
   873→    // обрывки троса + пыль от лопнувших вант
   874→    const gibs = Ship.gibs;
   875→    for (const sp of snappedAt) {
   876→      const wp = this.tmpV.copy(sp).applyQuaternion(this.quaternion).add(this.position);
   877→      if (gibs) {
   878→        for (let k = 0; k < 2; k++) {
   879→          gibs.spawnPart(
   880→            new THREE.Vector3(wp.x, wp.y, wp.z),
   881→            new THREE.Vector3((Math.random() - 0.5) * 3.5, 1.2 + Math.random() * 2.2,
   882→              (Math.random() - 0.5) * 3.5),
   883→            0.05, 0.05, 0.7 + Math.random() * 0.5, 0x241c12, 1.6 + Math.random());
   884→        }
   885→      }
   886→      this.particles?.smoke(wp.x, wp.y, wp.z, 0, 0.3, 0, 0.5, 0.8);
   887→    }
   888→    return n;
   889→  }

   964→  startSinking() {

   967→    this.sinkT = 0;

   970→    // дотягивает пламя до фазы 2 (палуба под водой — там вода его гасит)
   971→    this.sailTarget = 0;
   972→    this.rudderTarget = 0;
   973→    // сценарий гибели: валиться в сторону пробоин (или наугад),

   975→    this.sinkSide = Math.abs(this.leakBias) > 0.15 ? Math.sign(this.leakBias) : (Math.random() < 0.5 ? -1 : 1);
   976→    this.sinkTrim = Math.random() < 0.62 ? 1 : -1; // чаще — носом
   977→    this.sinkSpin = Math.random() < 0.5 ? -1 : 1;
   978→    this.sinkRollT = this.sinkSide * (0.85 + Math.random() * 0.45);   // 49..74°

   980→    this.sinkDebrisDone = false;

   984→  // ПОЭТАПНОЕ ЗАТОПЛЕНИЕ (по сценарию гибели парусника):
   985→  //  фаза 1 — потеря остойчивости: крен на борт, дифферент носом/
   986→  //    кормой, медленное погружение; вода в трюме добирается
   987→  //  фаза 2 — палуба уходит под воду: паруса и мачты работают
   988→  //    ГИДРОТОРМОЗОМ (вращение замедляется), дерево намокает
   989→  //    (темнеет, roughness → 1), с палубы срываются обломки
   990→  //    (доски/бочки всплывают рядом), надстройки рушатся
   991→  //  фаза 3 — финальный уход: заполненные отсеки тянут вниз
   992→  //    быстрее, корпус ЗАКРУЧИВАЕТСЯ по спирали (yaw), пузыри
   993→  //    воздуха рвутся из люков и пушечных портов
   994→  // ============================================================
   995→  private stepSinking(dt: number, time: number) {
   996→    this.sinkT += dt;
   997→    const P = this.preset;
   998→    const L = P.length, B = P.beam, F = P.freeboard;
   999→    const e2 = this._e2.setFromQuaternion(this.quaternion, 'YXZ');
  1000→
  1001→    // вода в трюме прибывает неотвратимо (пробоины + открытые люки)
  1002→    this.water = Math.min(this.holdCap, this.water + Math.max(this.leakRate, 0.5) * dt);
  1003→
  1004→    // палуба низкого борта: высота над ТЕКУЩЕЙ водой у корпуса
  1005→    const wy = waterHeightAt(this.position.x, this.position.z, time);
  1006→    const lowDeck = this.position.y + F * 0.82 * Math.cos(e2.z) - (B * 0.5) * Math.abs(Math.sin(e2.z));
  1007→    const subFrac = clamp((wy - lowDeck) / Math.max(1, F), 0, 1); // 0 сухая .. 1 над палубой
  1008→
  1009→    // --- переходы фаз ---
  1010→    if (this.sinkPhase === 1 && (subFrac > 0.22 || this.sinkT > 9)) this.enterSinkPhase2();
  1011→    if (this.sinkPhase === 2 && (subFrac > 0.88 || this.sinkT > 17)) this.sinkPhase = 3;
  1012→
  1013→    // --- пожар фазы 1: надводная часть ещё горит (картина гибели
  1014→    //     брандера — пламя + дым, пока палуба не ушла под воду) ---
  1015→    if (this.onFire && this.sinkPhase === 1) {
  1016→      const I = this.fireIntensity;
  1017→      if (Math.random() < dt * (1.2 + 2.2 * I)) {
  1018→        this.particles.fire(
  1019→          this.position.x + (Math.random() - 0.5) * B * 0.6,
  1020→          this.position.y + F * 0.5 + Math.random() * 2,
  1021→          this.position.z + (Math.random() - 0.5) * L * 0.5);
  1022→        this.particles.smoke(
  1023→          this.position.x + (Math.random() - 0.5) * B,
  1024→          this.position.y + F + 2,
  1025→          this.position.z + (Math.random() - 0.5) * L * 0.6,
  1026→          (Math.random() - 0.5) * 1.5, 2.6, (Math.random() - 0.5) * 1.5,
  1027→          2.8 + I * 1.4, 3.6);
  1028→      }
  1029→    }
  1030→
  1031→    // --- вертикаль: погружение ускоряется по мере заполнения отсеков ---
  1032→    const vSink = this.sinkPhase === 1 ? 0.20
  1033→      : this.sinkPhase === 2 ? 0.42
  1034→        : 0.85 + Math.min(0.9, (this.sinkT - 17) * 0.05);
  1035→    this.position.y -= vSink * dt;
  1036→    this.velocity.set(this.velocity.x * 0.9, -vSink, this.velocity.z * 0.9);
  1037→
  1038→    // --- вращение: гидротормоз мачт/парусов в воде ---
  1039→    // чем глубже села палуба, тем сильнее вода давит на рангоут —
  1040→    // корпус «не может» докрутиться (как настоящие паруса-гидротормоз)
  1041→    const mastBrake = 1 / (1 + subFrac * 4.5 + (this.mastsBroken > 0 ? 0.7 : 0));
  1042→    const rollRate = 0.34 * mastBrake;
  1043→    const pitchRate = 0.13 * mastBrake;
  1044→    e2.z += clamp(this.sinkRollT - e2.z, -rollRate * dt, rollRate * dt);
  1045→    e2.x += clamp(this.sinkPitchT - e2.x, -pitchRate * dt, pitchRate * dt);
  1046→    // спираль: едва заметное рыскание с начала, бодрое — в финале
  1047→    const spinRate = this.sinkPhase === 3 ? 0.16 + Math.min(0.22, this.sinkT * 0.008) : 0.02;
  1048→    e2.y += this.sinkSpin * spinRate * dt;
  1049→    this.quaternion.setFromEuler(e2);
  1050→    this.angular.set(0, this.sinkSpin * spinRate, 0);
  1051→
  1052→    // --- намокание дерева: темнее и матовее с каждой секундой ---
  1053→    if (this.sinkPhase >= 2) this.applyWetness(dt);
  1054→
  1055→    // --- эффекты: пена у борта, пузыри из портов, водоворот ---
  1056→    this.sinkFx(dt, time, subFrac);
  1057→
  1058→    // --- конец: надстройки глубоко под водой — корабль ушёл ---
  1059→    if (this.position.y < -(F + L * 0.30) || this.sinkT > 26) {

  1061→      this.sinkPhase = 3;
  1062→      this.visual.group.visible = false;
  1063→      this.onSunk?.(this);
  1064→    }
  1065→  }

  1067→  /** Фаза 2: мачты валятся за борт, обломки и бочки срываются
  1068→   *  с палубы и всплывают рядом, надстройки рушатся давлением */
  1069→  private enterSinkPhase2() {

  1117→    // щепа от рушащихся надстроек

  1133→  /** Собрать материалы корпуса (один раз) и плавно «мочить» их */
  1134→  private applyWetness(dt: number) {
  1135→    if (!this.wetMats) {
  1136→      this.wetMats = [];
  1137→      this.visual.group.traverse((o) => {
  1138→        const mesh = o as THREE.Mesh;
  1139→        const mat = mesh.material as THREE.MeshStandardMaterial | undefined;
  1140→        // только СТАНДАРТНЫЕ материалы ЭТОГО корабля (общий CREW_MAT
  1141→        // экипажа и шейдеры парусов/флага не трогаем)
  1142→        if (mat && (mat as THREE.MeshStandardMaterial).isMeshStandardMaterial
  1143→          && mat !== CREW_MAT && !(mat as unknown as THREE.ShaderMaterial).uniforms) {
  1144→          this.wetMats!.push({
  1145→            mat: mat as THREE.MeshStandardMaterial,
  1146→            base: (mat as THREE.MeshStandardMaterial).color.clone(),
  1147→            rough: (mat as THREE.MeshStandardMaterial).roughness,
  1148→            metal: (mat as THREE.MeshStandardMaterial).metalness,
  1149→          });
  1150→        }
  1151→      });
  1152→    }
  1153→    this.wetness = Math.min(1, this.wetness + dt * 0.22);
  1154→    const dark = 0.62 - 0.38 * this.wetness; // 1 → 0.62 → 0.24 яркости
  1155→    for (const w of this.wetMats) {
  1156→      w.mat.color.copy(w.base).multiplyScalar(dark);
  1157→      w.mat.roughness = clamp(w.rough + (1 - w.rough) * this.wetness, 0, 1);
  1158→      w.mat.metalness = w.metal * (1 - this.wetness * 0.8);
  1159→    }
  1160→  }

  1162→  /** Восстановить сухие материалы (после утопления/сброса полигона) */
  1163→  private resetWetness() {
  1164→    if (this.wetMats) {
  1165→      for (const w of this.wetMats) {
  1166→        w.mat.color.copy(w.base);
  1167→        w.mat.roughness = w.rough;
  1168→        w.mat.metalness = w.metal;
  1169→      }
  1170→    }
  1171→    this.wetMats = null;
  1172→    this.wetness = 0;
  1173→  }

  1175→  /** Эффекты гибели: пена вдоль борта (вортекс), пузыри из люков,
  1176→   *  брызги при погружении палубы */
  1177→  private sinkFx(dt: number, time: number, subFrac: number) {
  1178→    this.sinkFxT += dt;
  1179→    const P = this.preset;
  1180→    const per = this.sinkFxT > 0.10; // 10 Гц — не заливаем пул частиц
  1181→    if (!per) return;
  1182→    this.sinkFxT = 0;
  1183→    const p = this.particles;
  1184→    // 1) пенный воротник кромки воды вдоль корпуса (зона водоворота)
  1185→    const nFoam = this.sinkPhase >= 2 ? 3 : 2;
  1186→    for (let i = 0; i < nFoam; i++) {
  1187→      const zz = (Math.random() - 0.5) * P.length * 0.9;
  1188→      const side = Math.random() < 0.5 ? -1 : 1;
  1189→      const lp = this.tmpV.set(side * P.beam * 0.55, 0, zz)
  1190→        .applyQuaternion(this.quaternion).add(this.position);
  1191→      const wy = waterHeightAt(lp.x, lp.z, time);
  1192→      if (Math.abs(lp.y - wy) < P.freeboard) {
  1193→        p.spawn(lp.x, wy + 0.15, lp.z,
  1194→          (Math.random() - 0.5) * 2.5, 0.6 + Math.random() * 1.6, (Math.random() - 0.5) * 2.5,
  1195→          0.5 + Math.random() * 0.5, 0.45 + Math.random() * 0.4, 0, 0.6, 10);
  1196→      }
  1197→    }
  1198→    // 2) пузыри воздуха: из пушечных портов и люков — рвутся вверх
  1199→    if (subFrac > 0.1) {
  1200→      const nBub = this.sinkPhase === 3 ? 3 : 1;
  1201→      for (let i = 0; i < nBub; i++) {
  1202→        const zz = (Math.random() - 0.5) * P.length * 0.7;
  1203→        const side = Math.random() < 0.5 ? -1 : 1;
  1204→        const lp = this.tmpV.set(side * P.beam * 0.42, P.freeboard * 0.5, zz)
  1205→          .applyQuaternion(this.quaternion).add(this.position);
  1206→        const wy = waterHeightAt(lp.x, lp.z, time);
  1207→        if (lp.y < wy - 0.4) {
  1208→          // время жизни = дойти до поверхности при всплытии ~1.6 м/с
  1209→          const life = clamp((wy - lp.y) / 1.6, 0.25, 2.2);
  1210→          p.spawn(lp.x, lp.y, lp.z,
  1211→            (Math.random() - 0.5) * 0.5, 1.3 + Math.random() * 0.9, (Math.random() - 0.5) * 0.5,
  1212→            life, 0.1 + Math.random() * 0.12, 0, 0.15, -1.5);
  1213→        }
  1214→      }
  1215→    }
  1216→    // 3) мощный всплеск-водоворот, когда палуба захлёбывается
  1217→    if (this.sinkPhase === 2 && subFrac > 0.3 && Math.random() < 0.5) {
  1218→      p.splash(this.position.x, waterHeightAt(this.position.x, this.position.z, time), this.position.z, 1.4);
  1219→    }
  1220→  }

  1222→  /** Действие команды (R): пожар -> течь -> ремонт.
  1223→   *  Тушение — БОРЬБА с огнём: интенсивность падает тем быстрее,
  1224→   *  чем больше живых людей; наверху огонь всё равно разгорается */
  1225→  crewAction(dt: number): string | null {
  1226→    if (this.onFire) {
  1227→      const crewFrac = clamp(this.crew / Math.max(1, this.preset.maxCrew), 0.05, 1);
  1228→      const pump = 0.07 + 0.17 * crewFrac;               // ведёрные цепи
  1229→      const grow = 0.008 + 0.022 * this.fireIntensity;   // огонь не ждёт
  1230→      this.fireIntensity = clamp(this.fireIntensity - (pump - grow) * dt, 0, 1);
  1231→      if (this.fireIntensity <= 0.01) {
  1232→        this.onFire = false; this.fireT = 0;
  1233→        return 'Пожар потушен!';
  1234→      }
  1235→      return `Тушение пожара... ${Math.round(this.fireIntensity * 100)}%`;
  1236→    }
  1237→    if (this.water > 0.01) {
  1238→      const pump = 0.05 + this.crew * 0.004;
  1239→      this.water = Math.max(0, this.water - pump * dt);
  1240→      this.leakRate = Math.max(0, this.leakRate - 0.004 * dt);
  1241→      return this.water > 0.05 ? 'Откачка воды...' : 'Трюм осушён';
  1242→    }
  1243→    if (this.hullHP < this.hullMax && this.planks > 0) {
  1244→      this.hullHP = Math.min(this.hullMax, this.hullHP + 0.9 * dt);
  1245→      this.planks -= dt * 0.15;
  1246→      return 'Ремонт корпуса...';
  1247→    }
  1248→    return null;
  1249→  }

  1251→  // ---------- визуал ----------
  1252→  updateVisual(time: number, dt: number, particles: Particles, negShift?: THREE.Vector3) {
  1253→    const v = this.visual;
  1254→    v.group.position.copy(this.position);
  1255→    v.group.quaternion.copy(this.quaternion);
  1256→
  1257→    // паруса: количество поставленных парусов соответствует ФАКТИЧЕСКОЙ
  1258→    // СКОРОСТИ — экипаж ставит паруса по мере разгона и убирает по мере
  1259→    // остановки; уровень W/S задаёт максимум. Рея брасопится по ветру,
  1260→    // «пузо» — по кажущемуся ветру. Паруса СЛОМАННОЙ мачты обвисают.
  1261→    const awa = this.apparentWindAngle();
  1262→    const polar = Ship.polar(awa);
  1263→    const windClamp = clamp(this.wind.speed / 9, 0.3, 1.4);
  1264→    // скорость, достижимая под полным парусным вооружением в этот ветер
  1265→    // (корень из тяги — сопротивление квадратично): ~0.85·maxSpeed
  1266→    const windPow = Math.pow(clamp(this.wind.speed / 8.0, 0.05, 1.6), 1.6);
  1267→    const attainable = this.preset.maxSpeed * 0.85 * Math.sqrt(clamp(polar * windPow, 0, 1.5));
  1268→    // в левентик/штиль паруса остаются на мачтах и трепещут (не убираются
  1269→    // механически) — убираются только когда есть чем идти, но нет хода
  1270→    const spdSetFrac = attainable > 1.5 ? clamp(this.speed() / attainable, 0, 1) : 1;
  1271→    const setTotal = this.sailOverride ?? Math.min(this.sailFrac, spdSetFrac);
  1272→    // локальный вектор «откуда дует» (x — борт, z — нос): ψ = φ + курс
  1273→    const psi = this.wind.from + this.heading;
  1274→    const wlX = Math.cos(psi); // >0 — ветер с порта (левый борт)
  1275→    // цель брасопки: рея поперёк ветра (нормаль паруса по ветру), ±55°
  1276→    let bt = Math.atan2(wlX, Math.sin(psi)) - Math.PI;
  1277→    bt = Math.atan2(Math.sin(bt), Math.cos(bt)); // кратчайшая дуга
  1278→    const braceTarget = clamp(bt, -0.96, 0.96);
  1279→    const bellyBase = (0.35 + 0.95 * windClamp) * (0.15 + 0.85 * polar);
  1280→    const flutterBase = 0.012 * windClamp * (1 + (1 - polar) * 2.2);
  1281→    for (const s of v.sails.sails) {
  1282→      // мачта-носитель сломана → парус безжизненно обвисает
  1283→      const mastDead = s.mastIndex !== undefined && this.mastFall[s.mastIndex] >= 0;
  1284→      // доля постановки этого паруса при текущей скорости хода
  1285→      const setFrac = mastDead ? 0 : smoothstep(s.setAt - 0.38, s.setAt, setTotal);
  1286→      s.furl += ((1 - setFrac) - s.furl) * Math.min(1, dt * (mastDead ? 2.2 : 1.1));
  1287→      const sf = 1 - s.furl;
  1288→      s.mat.uniforms.uTime.value = time;
  1289→      s.mat.uniforms.uFurl.value = s.furl;
  1290→      s.mat.uniforms.uDamage.value = 1 - this.sailHP / this.preset.sailHP;
  1291→      s.mat.uniforms.uFlutter.value = flutterBase * (0.25 + 0.75 * sf);
  1292→      if (s.square && s.brace) {
  1293→        // реи разворачиваются медленно — как будто работают брашпили
  1294→        s.braceAngle += ((mastDead ? 0 : braceTarget) - s.braceAngle) * Math.min(1, dt * 0.5);
  1295→        s.brace.rotation.y = s.braceAngle;
  1296→        s.mat.uniforms.uBelly.value = mastDead ? 0 : bellyBase * sf;
  1297→      } else {
  1298→        // кливер: пузо в подветренную сторону (локальный Z → борт)
  1299→        s.mat.uniforms.uBelly.value = mastDead ? 0 : bellyBase * sf * (-wlX);
  1300→      }
  1301→    }
  1302→
  1303→    // флаг — по ветру
  1304→    const wf = Math.atan2(-Math.cos(this.wind.from), -Math.sin(this.wind.from));
  1305→    v.flag.rotation.y = wf - this.heading + Math.PI;
  1306→    v.flagMat.uniforms.uTime.value = time;
  1307→
  1308→    // штурвал крутится от руля: D (руль вправо) — по часовой от рулевого
  1309→    v.wheel.rotation.z = this.rudder * 2.2;
  1310→
  1311→    // руль за кормой следует за рулём
  1312→    if (v.rudderMesh) v.rudderMesh.rotation.y = this.rudder * 0.45;
  1313→
  1314→    // вёсла: УБРАНЫ по умолчанию — вал вдоль борта к корме
  1315→    // (кватернион stowQ из buildShip). ГРЕБКА — синхронная модель:
  1316→    //   yaw  вокруг вертикали — лопасть МЕТЁТ от носа к корме
  1317→    //         (задний ход — зеркально, к носу);
  1318→    //   dip   вокруг диаметрали — лопасть ПОГРУЖАЕТСЯ в рабочей
  1319→    //         фазе (sin φ > 0) и выходит на переносе.
  1320→    // Фаза ОДНА на все вёсла и оба борта — «гребут в ногу»;
  1321→    // темп растёт со скоростью хода.
  1322→    if (v.oars && v.oars.length) {
  1323→      this.oarsDeployT += ((this.oarsOut ? 1 : 0) - this.oarsDeployT) * Math.min(1, dt * 1.5);
  1324→      const rowing = this.rowing !== 0 && this.oarsDeployT > 0.25;
  1325→      const rowDir = this.rowing < 0 ? -1 : 1;
  1326→      if (rowing) {
  1327→        const spdFrac = clamp(this.speed() / Math.max(1, this.preset.maxSpeed), 0, 1);
  1328→        const rate = 0.55 + spdFrac * 0.5;                 // гребков/с
  1329→        this.strokePhase += dt * rate * Math.PI * 2;
  1330→      }
  1331→      const SWEEP = 0.42;                                  // размах гребка, рад
  1332→      const DIP = 0.30;                                    // погружение лопасти, рад
  1333→      const DIP0 = 0.14;                                   // статичный наклон к воде
  1334→      // знак гребка РАЗВЁРНУТ (замечание игрока): в подводной фазе
  1335→      // (sin φ > 0) лопасть теперь метётся К КОРМЕ — как на настоящей
  1336→      // галере; раньше выходил «задом наперёд»
  1337→      const sweep = rowing ? -Math.cos(this.strokePhase) * SWEEP * rowDir : 0;
  1338→      const dip = rowing ? Math.max(0, Math.sin(this.strokePhase)) * DIP : 0;
  1339→      for (const o of v.oars) {
  1340→        const side = o.userData.side as number;
  1341→        const stowQ = o.userData.stowQ as THREE.Quaternion;
  1342→        // развёрнутое: гребок (yaw) в осях корабля, затем наклон (dip)
  1343→        this._oarYaw.setFromAxisAngle(this._yAxis, side * sweep);
  1344→        this._oarDip.setFromAxisAngle(this._zAxis, -side * (DIP0 + dip));
  1345→        this._oarQ.copy(this._oarDip).multiply(this._oarYaw);
  1346→        o.quaternion.copy(stowQ).slerp(this._oarQ, this.oarsDeployT);
  1347→      }
  1348→    }
  1349→
  1350→    // пушечные порты: крышка откидывается вверх-наружу (петля вверху),
  1351→    // орудие ВЫЕЗЖАЕТ наружу; закрыты — не стреляют (гейт в combat.ts)
  1352→    if (v.gunPorts.length) {
  1353→      this.portsT += ((this.portsOpen ? 1 : 0) - this.portsT) * Math.min(1, dt * 1.6);
  1354→      const lidA = smoothstep(0, 1, this.portsT);
  1355→      const slide = clamp((this.portsT - 0.25) / 0.75, 0, 1);
  1356→      for (const gp of v.gunPorts) {
  1357→        gp.open = this.portsT;
  1358→        gp.lid.rotation.z = gp.side * 2.6 * lidA;
  1359→        const baseX = gp.gun.userData.baseX as number;
  1360→        const slideX = gp.gun.userData.slideX as number;
  1361→        gp.gun.position.x = baseX + slideX * slide;
  1362→      }
  1363→    }
  1364→
  1365→    // магнитная стрелка нактоуза: плывёт в картушке и всегда смотрит
  1366→    // на СЕВЕР (мир −Z) — рулевому виден истинный курс; лёгкая
  1367→    // дрожь картушки от хода
  1368→    if (v.binnacleNeedle) {
  1369→      v.binnacleNeedle.rotation.y = -this.heading + Math.PI +
  1370→        Math.sin(time * 3.1) * 0.03 + Math.sin(time * 7.7) * 0.015;
  1371→    }
  1372→
  1373→    // ЯКОРЯ (отдача/подъём, клавиша F): якорь вываливается за борт
  1374→    // по дуге и опускается под воду на цепи; жёлтый буй-кольцо
  1375→    // всплывает у цепи и покачивается на волне. Убран — лежит
  1376→    // вдоль борта у скулы (просьба игрока: с анимацией отдачи).
  1377→    if (v.anchors?.length) {
  1378→      const target = this.anchored ? 1 : 0;
  1379→      const prev = this.anchorT;
  1380→      if (this.anchorT !== target) {
  1381→        this.anchorT += Math.sign(target - this.anchorT) * Math.min(Math.abs(target - this.anchorT), dt * 0.4);
  1382→        // якоря входят в воду — всплеск под обоими
  1383→        if (prev < 0.42 && this.anchorT >= 0.42 && target === 1) {
  1384→          for (const A of v.anchors) {
  1385→            this.tmpV.copy(A.ctrlPos).lerp(A.dropPos, 0.3)
  1386→              .applyQuaternion(this.quaternion).add(this.position);
  1387→            this.particles?.splash(this.tmpV.x, this.position.y + 0.1, this.tmpV.z, 1.0);
  1388→          }
  1389→        }
  1390→      }
  1391→      const sa = smoothstep(0, 1, this.anchorT);
  1392→      const p1 = this.tmpV, p2 = this.tmpV2;
  1393→      for (const A of v.anchors) {
  1394→        // позиция по квадратичной дуге «вывалился — упал»
  1395→        p1.lerpVectors(A.stowPos, A.ctrlPos, sa);
  1396→        p2.lerpVectors(A.ctrlPos, A.dropPos, sa);
  1397→        A.anchor.position.lerpVectors(p1, p2, sa);
  1398→        A.anchor.quaternion.slerpQuaternions(A.stowQuat, A.dropQuat, sa);
  1399→        // цепь: клюз -> рым якоря, с провисом наружу
  1400→        p1.copy(A.ringOffset).applyQuaternion(A.anchor.quaternion).add(A.anchor.position);
  1401→        const pos = A.chain.geometry.getAttribute('position') as THREE.BufferAttribute;
  1402→        const n = pos.count;
  1403→        for (let i = 0; i < n; i++) {
  1404→          const u = i / (n - 1);
  1405→          const sag = Math.sin(Math.PI * u) * (0.12 + sa * 0.5);
  1406→          p2.lerpVectors(A.cathead, p1, u);
  1407→          pos.setXYZ(i, p2.x + A.side * sag, p2.y - sag * 0.45, p2.z);
  1408→        }
  1409→        pos.needsUpdate = true;
  1410→        // буй: у планширя -> плавает у цепи
  1411→        A.buoy.position.lerpVectors(A.buoyStow, A.buoyFloat, sa);
  1412→        A.buoy.position.y += Math.sin(time * 1.6 + A.side * 2.1) * 0.045 * sa;
  1413→        A.buoy.rotation.z = Math.sin(time * 1.2 + A.side) * 0.14 * sa;
  1414→      }
  1415→    }
  1416→
  1417→    // падение мачт: валится с нарастающим ускорением (ease-in),
  1418→    // заваливаясь за борт; реи и паруса едут вместе с мачтой (они
  1419→    // её потомки), обвязка уже оборвана (breakMast → snapRigging).
  1420→    // Когда топ мачты достаёт воду — фонтан брызг
  1421→    for (let i = 0; i < v.masts.length; i++) {
  1422→      const f = this.mastFall[i];
  1423→      if (f >= 0 && f < 1) {
  1424→        // ускорение падения: 0.30 → 0.62 (мачта «набирает ход»)
  1425→        this.mastFall[i] = Math.min(1, f + dt * (0.30 + f * 0.32));
  1426→        const k = this.mastFall[i];
  1427→        const ke = k * k; // ease-in: поворот отстаёт от прогресса
  1428→        const lean = (i % 2 === 0 ? 1 : -1) * 0.52;
  1429→        v.masts[i].rotation.x = ke * 1.38;
  1430→        v.masts[i].rotation.z = ke * lean;
  1431→        v.masts[i].position.y = -ke * 2.1;
  1432→        // топ мачты доходит до воды — всплеск
  1433→        if (k > 0.62 && !this.mastSplashed[i]) {
  1434→          this.mastSplashed[i] = true;
  1435→          const h = ((v.masts[i].userData.height as number) || 20);
  1436→          this.tmpV.set(0, 0, h * 0.88 * Math.sin(Math.min(ke * 1.38, 1.2)))
  1437→            .applyQuaternion(this.quaternion).add(this.position);
  1438→          this.particles?.splash(this.tmpV.x, this.position.y + 0.1, this.tmpV.z, 1.25);
  1439→        }
  1440→      }
  1441→    }
  1442→
  1443→    // пенный воротник по ватерлинии: высоту волн добавляет ВЕРШИННЫЙ
  1444→    // шейдер (тот же Gerstner, что у океана) — кольцо лежит на гребнях;
  1445→    // на CPU остаётся только горизонтальная привязка к кораблю
  1446→    if (!this.sunk) {
  1447→      this.foam.mesh.position.set(this.position.x, 0.30, this.position.z);
  1448→      this.foam.mesh.rotation.y = this.heading;
  1449→      const spdFrac = clamp(this.speed() / this.preset.maxSpeed, 0, 1);
  1450→      this.foam.update(time, spdFrac, this.euler.z, negShift);
  1451→    } else {
  1452→      this.foam.mesh.visible = false;
  1453→    }
  1454→
  1455→    // кильватерный след
  1456→    const sternW = this.tmpV.set(0, 0, -this.preset.length * 0.48)
  1457→      .applyQuaternion(this.quaternion).add(this.position);
  1458→    const spdFrac = clamp(this.speed() / this.preset.maxSpeed, 0, 1);
  1459→    if (!this.sunk && !this.grounded) {
  1460→      this.wake.push(sternW, spdFrac, dt, time);
  1461→    }
  1462→    this.wake.update(time);
  1463→
  1464→    // носовой бурун: брызги при ударе носа о волну
  1465→    if (!this.sunk) {
  1466→      const bowW = this.tmpV2.set(0, -this.preset.draft * 0.3, this.preset.length * 0.46)
  1467→        .applyQuaternion(this.quaternion).add(this.position);
  1468→      const waveY = waterHeightAt(bowW.x, bowW.z, time);
  1469→      const relV = this.velocity.y - (waveY - this.lastStern.y) * 0; // грубая оценка
  1470→      const impact = (waveY - bowW.y) * (0.3 + spdFrac);
  1471→      if (impact > 0.55 && Math.random() < 0.6) {
  1472→        particles.splash(bowW.x, waveY, bowW.z, 0.5 + spdFrac * 0.9);
  1473→      }
  1474→      if (spdFrac > 0.35 && Math.random() < spdFrac * 0.5) {
  1475→        particles.spawn(bowW.x + (Math.random() - 0.5) * 2, waveY + 0.1, bowW.z + (Math.random() - 0.5) * 2,
  1476→          (Math.random() - 0.5) * 2, 1.5 + Math.random() * 2.5 * spdFrac, (Math.random() - 0.5) * 2,
  1477→          0.5, 0.3 + Math.random() * 0.35, 0, 0.4, 10);
  1478→      }
  1479→      // крен: всплески у борта
  1480→      const roll = this.euler.z;
  1481→      this.rollSmooth += (roll - this.rollSmooth) * Math.min(1, dt * 3);
  1482→      if (Math.abs(this.rollSmooth) > 0.28 && Math.random() < 0.3) {
  1483→        const side = Math.sign(this.rollSmooth) * this.preset.beam * 0.5;
  1484→        const px = this.position.x + Math.cos(this.heading) * side;
  1485→        const pz = this.position.z - Math.sin(this.heading) * side;
  1486→        particles.splash(px, waterHeightAt(px, pz, time), pz, 0.5);
  1487→      }
  1488→      // пробоины ниже ватерлинии: пузыри воздуха + пенная кромка
  1489→      for (const h of this.hullHoles) {
  1490→        if (!h.belowWL || Math.random() > dt * 1.6) continue;
  1491→        const wp = this.tmpV2.set(h.mesh.position.x, h.mesh.position.y, h.mesh.position.z)
  1492→          .applyQuaternion(this.quaternion).add(this.position);
  1493→        const wy = waterHeightAt(wp.x, wp.z, time);
  1494→        if (wp.y < wy - 0.15) {
  1495→          // пузырь идёт вверх от дыры и умирает у поверхности
  1496→          particles.spawn(wp.x, wp.y, wp.z, 0, 0.8 + Math.random() * 0.7, 0,
  1497→            clamp((wy - wp.y) / 1.1, 0.3, 1.6), 0.07 + Math.random() * 0.08, 0, 0.1, -1.2);
  1498→          particles.spawn(wp.x, wy, wp.z, (Math.random() - 0.5) * 0.8, 0.5 + Math.random(),
  1499→            (Math.random() - 0.5) * 0.8, 0.5, 0.18, 0, 0.5, 9);
  1500→        }
  1501→      }
  1502→    }
  1503→
  1504→    // экипаж: анимация, потери (трупы/разрывы), эвакуация при потоплении;
  1505→    // гребным судам передаём фазу гребка — гребцы отклоняются в такт
  1506→    this.crew3d.update(dt, time, {
  1507→      crew: this.crew, maxCrew: this.preset.maxCrew,
  1508→      sinking: this.sinking, sunk: this.sunk,
  1509→      position: this.position, quaternion: this.quaternion,
  1510→      stroke: {
  1511→        phase: this.strokePhase,
  1512→        active: (this.rowing !== 0 && this.oarsDeployT > 0.5 && !this.sinking) ? 1 : 0,
  1513→        dir: this.rowing < 0 ? -1 : 1,
  1514→      },
  1515→    });
  1516→  }
