   322→  /** Применить настройки эффектов дыма/огня */
   323→  private applyFx(fx: FxSettings) {
   324→    // прозрачность ↑ → непрозрачность ↓ (пик альфы клуба)
   325→    const alpha = 0.55 * (1 - fx.smokeTransparency);
   326→    this.particles.applyFx({
   327→      smokeVisible: fx.smokeVisible,
   328→      smokeSize: fx.smokeSize,
   329→      smokeDensity: fx.smokeDensity,
   330→      smokeAlpha: Math.max(0.03, alpha),
   331→    });
   332→  }

   334→  /** Переключение модели моря (Task 27): физическая (Архимед по
   335→   *  точкам корпуса) или «волна-опора» (по файлу игрока «Перун» —
   336→   *  корабль идёт по поверхности, нос всходит на волну). Волны
   337→   *  живут в uniform-массивах шейдера — переключение мгновенное */
   338→  applySeaModel(model: 'physical' | 'riding', announce = true) {
   339→    if (SEA.model === model) return;
   340→    SEA.model = model;
   341→    // сброс вертикальной динамики кораблей (пружина riding-режима
   342→    // подхватит посадку сама; у физической — Архимед)
   343→    if (this.player) {
   344→      this.player.velocity.y = 0;
   345→      for (const a of this.aiShips) a.ship.velocity.y = 0;
   346→    }
   347→    if (announce) {
   348→      this.pushMessage(model === 'riding'
   349→        ? 'Море: волна-опора — корабль идёт по поверхности волн'
   350→        : 'Море: физическое — Архимед по точкам корпуса');
   351→    }
   352→  }

   354→  applySettings(s: GameSettings) {
   355→    const prev = this.settings;
   356→    this.settings = s;
   357→    const g = s.graphics;
   358→    this.renderer.setPixelRatio(Math.min(window.devicePixelRatio, g.pixelRatio));
   359→    if (prev.graphics.shadows !== g.shadows) this.setShadows(g.shadows);
   360→    if (prev.graphics.shadowRes !== g.shadowRes) this.sky.setShadowRes(g.shadowRes);
   361→    if (prev.graphics.waterDetail !== g.waterDetail || prev.graphics.drawDistance !== g.drawDistance
   362→      || prev.graphics.waterChunkSize !== g.waterChunkSize || prev.graphics.waterLodStep !== g.waterLodStep) {
   363→      this.water.configure(g.waterDetail, g.waterChunkSize, g.drawDistance, g.waterLodStep);
   364→    }
   365→    // дистанция мишени на полигоне — переставить цель без сброса повреждений
   366→    if (prev.gameplay.provingDistance !== s.gameplay.provingDistance
   367→      && this.proving && this.pgTarget) {
   368→      const t = this.pgTarget.ship;
   369→      t.position.copy(this.pgTargetPos());
   370→      t.position.y = 0;
   371→      this.pushMessage(`Мишень переставлена: ${s.gameplay.provingDistance} м`);
   372→    }
   373→    // боезапас: изменение настройки сразу выдается игроку (песочница)
   374→    const prevAmmo = prev.gameplay.startingAmmo;
   375→    const nextAmmo = s.gameplay.startingAmmo;
   376→    if (prevAmmo.ball !== nextAmmo.ball || prevAmmo.chain !== nextAmmo.chain
   377→      || prevAmmo.grape !== nextAmmo.grape || prevAmmo.fire !== nextAmmo.fire) {
   378→      this.player.ammo = { ...nextAmmo };

   685→  // ===== Карта окружения для PBR-металла =====
   686→  // Панорама 64×32 из цветов неба (sky/horizon/fog) + пятно солнца
   687→  // по sunDir → PMREM → scene.environment: орудия, фурнитура и золото
   688→  // отражают небо, а не выглядят чёрными «картонками». Дереву (metalness
   689→  // 0.05) отражения почти не видны — только лёгкий подсвет.
   690→  private pmrem: THREE.PMREMGenerator | null = null;
   691→  private envTex: THREE.DataTexture | null = null;
   692→  private envRT: THREE.WebGLRenderTarget | null = null;
   693→  private envNextUpd = 2;
   694→  private updateEnvMap(dayF: number, env: { sky: THREE.Color; horizon: THREE.Color; fog: THREE.Color; sun: THREE.Color }) {
   695→    if (this.envNextUpd > 0) { this.envNextUpd -= 1; return; }
   696→    this.envNextUpd = 150; // ~2.5 с при 60 к/с
   697→    const W = 64, H = 32;
   698→    if (!this.envTex) {
   699→      this.envTex = new THREE.DataTexture(new Uint8Array(W * H * 4), W, H,
   700→        THREE.RGBAFormat, THREE.UnsignedByteType);
   701→      this.envTex.mapping = THREE.EquirectangularReflectionMapping;
   702→      this.envTex.minFilter = THREE.LinearFilter;
   703→      this.envTex.magFilter = THREE.LinearFilter;
   704→      this.envTex.generateMipmaps = false;
   705→      this.pmrem = new THREE.PMREMGenerator(this.renderer);
   706→    }
   707→    const d = this.envTex.image.data as Uint8Array;
   708→    const sd = this.sky.sunDir;
   709→    const sunI = Math.max(0.15, dayF);
   710→    const water = env.fog.clone().multiplyScalar(0.4);
   711→    const c = new THREE.Color();
   712→    for (let y = 0; y < H; y++) {
   713→      const v = (y + 0.5) / H;                 // v=0 низ (вода), v=1 зенит
   714→      const dy = Math.sin((v - 0.5) * Math.PI);
   715→      const rh = Math.sqrt(Math.max(0, 1 - dy * dy));
   716→      for (let x = 0; x < W; x++) {
   717→        const u = (x + 0.5) / W;
   718→        const phi = (u - 0.5) * Math.PI * 2;
   719→        const dx = Math.cos(phi) * rh, dz = Math.sin(phi) * rh;
   720→        // небо над горизонтом, тёмная вода под ним
   721→        if (dy >= 0) c.copy(env.horizon).lerp(env.sky, Math.pow(Math.min(1, dy * 1.6), 0.6));
   722→        else c.copy(env.horizon).lerp(water, Math.min(1, -dy * 2.2));
   723→        // солнце: яркое ядро + мягкий ореол
   724→        const s = Math.max(0, dx * sd.x + dy * sd.y + dz * sd.z);
   725→        const core = Math.pow(s, 380) * 255 * sunI;
   726→        const halo = Math.pow(s, 14) * 60 * sunI;
   727→        const k = Math.min(1, core + halo);
