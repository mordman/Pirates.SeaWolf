    63→/** Процедурная генерация архипелага: детерминированно по сиду —
    64→ *  кольцом вокруг порта, с минимальными проливами между островами
    65→ *  (увеличенная карта: дистанции 1700–4200 м, радиусы 120–330 м,
    66→ *  проливы не уже 800 м).
    67→ *  Базовые острова не трогаем (животные/порт привязаны к индексам). */
    68→function generateIslands(): IslandDef[] {
    69→  const rng = new RNG(WORLD_SEED + 9173);
    70→  const names = [
    71→    'Остров Черепа', 'Мыс Багровой Луны', 'Скалы Трёх Братьев', 'Лагуна Сирен',
    72→    'Кайманов ключ', 'Затерянный риф', 'Бухта Повешенного', 'Остров Гвоздей',
    73→    'Перьевой атолл', 'Голый риф', 'Отмель Мёртвого Якоря', 'Тихая банка',
    74→  ];
    75→  const pool = [...names];
    76→  const out: IslandDef[] = [];
    77→  let guard = 0;
    78→  while (out.length < 11 && guard++ < 900) {
    79→    const a = rng.range(0, Math.PI * 2);
    80→    const d = rng.range(1700, 4200);
    81→    const cx = Math.cos(a) * d;
    82→    const cz = Math.sin(a) * d * 0.82; // мир чуть шире по X — под карту 640×520
    83→    if (Math.abs(cx) > 4500 || Math.abs(cz) > 3600) continue;
    84→    const roll = rng.next();
    85→    const type: IslandDef['type'] = roll < 0.5 ? 'jungle' : roll < 0.78 ? 'rock' : 'atoll';
    86→    const r = rng.range(120, 330);
    87→    // проливы не уже 800 м между берегами
    88→    let ok = true;
    89→    for (const isl of [...BASE_ISLANDS, ...out]) {
    90→      if (Math.hypot(cx - isl.cx, cz - isl.cz) < isl.r + r + 800) { ok = false; break; }
    91→    }
    92→    if (!ok) continue;
    93→    const maxH = type === 'rock' ? rng.range(22, 40) : type === 'jungle' ? rng.range(12, 30) : rng.range(7, 13);
    94→    const name = pool.length ? pool.splice(Math.floor(rng.next() * pool.length), 1)[0] : `Безымянный риф ${out.length + 1}`;
    95→    out.push({ cx: Math.round(cx), cz: Math.round(cz), r: Math.round(r), maxH: Math.round(maxH), type, name, discovered: false });
    96→  }
    97→  return out;
    98→}

   102→/** Высота суши в мировой точке. Открытое море = -14 (глубина), суша > 0 */
   103→export function heightAt(x: number, z: number): number {
   104→  let h = -14; // дно открытого моря
   105→  for (const isl of ISLANDS) {
   106→    const dx = x - isl.cx, dz = z - isl.cz;
   107→    const d = Math.hypot(dx, dz);
   108→    if (d > isl.r * 2.0) continue;
   109→    // рваная береговая линия: радиус модулируется шумом
   110→    const coastNoise = fbm2(x * 0.008, z * 0.008, 3, WORLD_SEED + isl.r);
   111→    const rEff = isl.r * (0.78 + 0.5 * coastNoise);
   112→    if (d > rEff * 1.5) continue;
   113→    const fall = smoothstep(1.18, 0.22, d / rEff);
   114→    const detail = 0.55 + 0.45 * fbm2(x * 0.02, z * 0.02, 4, WORLD_SEED + isl.cx);
   115→    const ridge = isl.type === 'rock' ? Math.pow(fall, 1.6) : fall;
   116→    let hi = isl.maxH * ridge * detail - 2.6;
   117→    // плавный склон к морскому дну за отмелью
   118→    hi = THREE.MathUtils.lerp(hi, -14, smoothstep(1.12, 1.5, d / rEff));
   119→    if (hi > h) h = hi;
   120→  }
   121→  return h;
   122→}

   124→function slopeAt(x: number, z: number): number {
   125→  const e = 1.5;
   126→  const hx = heightAt(x + e, z) - heightAt(x - e, z);
   127→  const hz = heightAt(x, z + e) - heightAt(x, z - e);
   128→  return Math.hypot(hx, hz) / (2 * e);
   129→}

   131→// ---------- геометрия с цветом ----------
   132→function paint(g: THREE.BufferGeometry, color: number, jitter = 0.06): THREE.BufferGeometry {
   133→  const c = new THREE.Color(color);
   134→  const n = g.getAttribute('position').count;
   135→  const arr = new Float32Array(n * 3);
   136→  for (let i = 0; i < n; i++) {
   137→    const j = 1 + (Math.random() - 0.5) * 2 * jitter;
   138→    arr[i * 3] = c.r * j; arr[i * 3 + 1] = c.g * j; arr[i * 3 + 2] = c.b * j;
   139→  }
   140→  g.setAttribute('color', new THREE.BufferAttribute(arr, 3));
   141→  return g;
   142→}
   143→function tf(g: THREE.BufferGeometry, x: number, y: number, z: number, rx = 0, ry = 0, rz = 0, s = 1): THREE.BufferGeometry {
   144→  const m = new THREE.Matrix4().compose(
   145→    new THREE.Vector3(x, y, z),
   146→    new THREE.Quaternion().setFromEuler(new THREE.Euler(rx, ry, rz)),
   147→    new THREE.Vector3(s, s, s));
   148→  g.applyMatrix4(m);
   149→  return g;
   150→}

   152→// ветровая анимация крон (общий uniform)
   153→export const windTimeUniform = { value: 0 };
   154→function applySway(mat: THREE.Material, amp: number) {
   155→  mat.onBeforeCompile = (shader) => {
   156→    shader.uniforms.uTime = windTimeUniform;
   157→    shader.vertexShader = shader.vertexShader
   158→      .replace('#include <common>', '#include <common>\nuniform float uTime;')
   159→      .replace('#include <begin_vertex>', `#include <begin_vertex>
   160→        #ifdef USE_INSTANCING
   161→          vec2 ipos2 = vec2(instanceMatrix[3].x, instanceMatrix[3].z);
   162→          float swayA = ${amp.toFixed(3)} * sin(uTime * 1.35 + ipos2.x * 0.37 + ipos2.y * 0.43);
   163→          float hf2 = max(transformed.y - 0.6, 0.0);
   164→          transformed.x += swayA * hf2;
   165→          transformed.z += swayA * 0.55 * hf2;
   166→        #endif`);
   167→  };
   168→  mat.customProgramCacheKey = () => `sway-${amp}`;
   169→}

   279→  // ---------- острова ----------
   280→  private buildIslands() {
   281→    this.islands.forEach((isl, islIdx) => {
   282→      // размер квадрата: с запасом за всю зону влияния рельефа
   283→      // (rEff·1.5 до ~1.9r) — раньше 3.2r обрезали отмель НА КВАДРАТЕ,
   284→      // и под водой читался «квадрат под островом»; теперь кромка
   285→      // уходит на глубину -14 и не видна
   286→      const size = isl.r * 4.3;
   287→      const seg = 110;
   288→      // у каждого острова своё дно с крошечным сдвигом: перекрывающиеся
   289→      // плоскости не мерцают z-fight в открытом море
   290→      const deep = -14 - islIdx * 0.06;
   291→      const g = new THREE.PlaneGeometry(size, size, seg, seg);
   292→      g.rotateX(-Math.PI / 2);
   293→      const pos = g.getAttribute('position') as THREE.BufferAttribute;
   294→      const colors = new Float32Array(pos.count * 3);
   295→      const cUnder = new THREE.Color(0x9a8a66);
   296→      const cSand = new THREE.Color(0xd9c28f);
   297→      const cGrass = new THREE.Color(0x5f863d);
   298→      const cJungle = new THREE.Color(0x33642e);
   299→      const cRock = new THREE.Color(0x8a8378);
   300→      const c = new THREE.Color();
   301→      for (let i = 0; i < pos.count; i++) {
   302→        const wx = pos.getX(i) + isl.cx;
   303→        const wz = pos.getZ(i) + isl.cz;
   304→        let h = heightAt(wx, wz);
   305→        if (h < deep) h = deep;
   306→        pos.setY(i, h);
   307→        // цвет по высоте и уклону (уклон приблизим после)
   308→        if (h < -0.4) c.copy(cUnder).lerp(cSand, clamp((h + 3) / 2.6, 0, 1));
   309→        else if (h < 1.3) c.copy(cSand);
   310→        else if (h < isl.maxH * 0.62) c.copy(cGrass).lerp(cJungle, fbm2(wx * 0.05, wz * 0.05, 3, WORLD_SEED + 9));
   311→        else c.copy(cRock);
   312→        colors[i * 3] = c.r; colors[i * 3 + 1] = c.g; colors[i * 3 + 2] = c.b;
   313→      }
   314→      g.setAttribute('color', new THREE.BufferAttribute(colors, 3));
   315→      g.computeVertexNormals();
   316→      // скалы на крутых склонах
   317→      const nrm = g.getAttribute('normal') as THREE.BufferAttribute;
   318→      for (let i = 0; i < pos.count; i++) {
   319→        if (nrm.getY(i) < 0.62 && pos.getY(i) > 0.3) {
   320→          colors[i * 3] = cRock.r; colors[i * 3 + 1] = cRock.g; colors[i * 3 + 2] = cRock.b;
   321→        }
   322→      }
   323→      (g.getAttribute('color') as THREE.BufferAttribute).needsUpdate = true;
   324→      const mat = new THREE.MeshStandardMaterial({ vertexColors: true, roughness: 0.94, metalness: 0 });
   325→      const mesh = new THREE.Mesh(g, mat);
   326→      mesh.position.set(isl.cx, 0, isl.cz);
   327→      mesh.receiveShadow = true;
   328→      mesh.castShadow = true;
   329→      this.group.add(mesh);
   330→    });
   331→  }

   333→  // ---------- растительность ----------
   334→  // ДЕРЕВЬЯ ПО ОСТРОВУ: раньше все деревья всех островов сидели в
   335→  // ОДНОМ InstancedMesh с frustumCulled=false (мелкий мир — ок);
   336→  // с увеличенными островами это стало ~2× треугольников В КАЖДОМ
   337→  // кадре (даже за спиной). Теперь инстансы ГО на остров — и
   338→  // отсекаются по фрустуму: за бортом рисуются только близкие острова
   339→  private buildVegetation() {
   340→    interface Slot { geo: () => THREE.BufferGeometry; sway: number; }
   341→    const types: Record<string, Slot> = {
   342→      palm: { geo: palmGeo, sway: 0.05 },
   343→      ficus: { geo: ficusGeo, sway: 0.028 },
   344→      jungle: { geo: jungleGeo, sway: 0.035 },
   345→      bush: { geo: bushGeo, sway: 0.05 },
   346→      bamboo: { geo: bambooGeo, sway: 0.06 },
   347→      rock: { geo: rockGeo, sway: 0 },
   348→    };
   349→    // placements[остров][порода] — отдельный инстанс-меш на пару
   350→    const perIsland: { isl: IslandDef; byKind: Record<string, { m: THREE.Matrix4; islandType: string; h: number }[]> }[] = [];
   351→
   352→    for (const isl of this.islands) {
   353→      const density = isl.type === 'rock' ? 0.35 : isl.type === 'atoll' ? 1.0 : 1.25;
   354→      const area = Math.PI * isl.r * isl.r;
   355→      const target = Math.floor(area / 220 * density);
   356→      const rng = new RNG(WORLD_SEED + isl.cx * 31 + isl.cz);
   357→      const byKind: Record<string, { m: THREE.Matrix4; islandType: string; h: number }[]> = {};
   358→      for (const k of Object.keys(types)) byKind[k] = [];
   359→      let tries = 0;
   360→      let placed = 0;
   361→      while (placed < target && tries < target * 14) {
   362→        tries++;
   363→        const a = rng.range(0, Math.PI * 2);
   364→        const rr = Math.sqrt(rng.next()) * isl.r * 1.25;
   365→        const x = isl.cx + Math.cos(a) * rr;
   366→        const z = isl.cz + Math.sin(a) * rr;
   367→        const h = heightAt(x, z);
   368→        if (h < 0.75 || h > isl.maxH * 0.9) continue;
   369→        const sl = slopeAt(x, z);
   370→        if (sl > 0.62) continue;
   371→        // тип дерева по высоте и типу острова
   372→        let kind: string;
   373→        const roll = rng.next();
   374→        if (isl.type === 'rock') kind = roll < 0.7 ? 'rock' : 'bush';
   375→        else if (h < 4.2) kind = roll < 0.72 ? 'palm' : roll < 0.85 ? 'bush' : 'rock';
   376→        else if (h < isl.maxH * 0.55) kind = roll < 0.42 ? 'jungle' : roll < 0.62 ? 'palm' : roll < 0.82 ? 'bamboo' : roll < 0.94 ? 'bush' : 'ficus';
   377→        else kind = roll < 0.5 ? 'jungle' : roll < 0.75 ? 'bush' : 'rock';
   378→        if (kind === 'rock' && h < 1.0 && roll < 0.5) kind = 'palm';
   379→        const s = rng.range(0.7, 1.35);
   380→        const m = new THREE.Matrix4().compose(
   381→          new THREE.Vector3(x, h - 0.15, z),
   382→          new THREE.Quaternion().setFromEuler(new THREE.Euler(rng.range(-0.06, 0.06), rng.range(0, Math.PI * 2), rng.range(-0.06, 0.06))),
   383→          new THREE.Vector3(s, s * rng.range(0.85, 1.2), s));
   384→        byKind[kind].push({ m, islandType: isl.type, h });
   385→        placed++;
   386→      }
   387→      perIsland.push({ isl, byKind });
   388→    }
   389→
   390→    for (const { byKind } of perIsland) {
   391→      for (const [kind, list] of Object.entries(byKind)) {
   392→        if (!list.length) continue;
   393→        const geo = types[kind].geo();
   394→        const mat = new THREE.MeshStandardMaterial({ vertexColors: true, roughness: 0.9 });
   395→        if (types[kind].sway > 0) applySway(mat, types[kind].sway);
   396→        const inst = new THREE.InstancedMesh(geo, mat, list.length);
   397→        list.forEach((p, i) => inst.setMatrixAt(i, p.m));
   398→        inst.castShadow = true;
   399→        inst.receiveShadow = false;
   400→        // по острову — фрустум отсекает далёкие/за спиной
   401→        inst.frustumCulled = true;
   402→        inst.computeBoundingSphere();
   403→        this.group.add(inst);
   404→        // позиции для ломания (камни не ломаются)
   405→        const positions = list.map((p) => {
   406→          const v = new THREE.Vector3();
   407→          v.setFromMatrixPosition(p.m);
   408→          return v;
   409→        });
   410→        this.vegSlots.push({
   411→          inst, positions,
   412→          broken: new Array(list.length).fill(false),
   413→          breakable: kind !== 'rock',
   414→          kind,
   415→          scales: list.map((p) => Math.max(0.5, new THREE.Vector3().setFromMatrixScale(p.m).y)),
   416→        });
   417→      }
   418→    }
   419→  }

   421→  /** Ядро/книппель валят деревья: ломает экземпляры в радиусе
   422→   *  (дерево разлетается на ТРИ части: пень + ствол + крона —
   423→   *  гибы с гравитацией), возвращает количество поваленных */
   424→  breakTrees(x: number, y: number, z: number, radius: number,
   425→    fx?: { debris: (x: number, y: number, z: number, n?: number, power?: number) => void },
   426→    gibs?: { spawnPart: (pos: THREE.Vector3, vel: THREE.Vector3, sx: number, sy: number, sz: number, hex: number, life?: number) => void }): number {
   427→    let broke = 0;
   428→    const r2 = radius * radius;
   429→    for (const slot of this.vegSlots) {
   430→      if (!slot.breakable) continue;
   431→      for (let i = 0; i < slot.positions.length; i++) {
   432→        if (slot.broken[i]) continue;
   433→        const p = slot.positions[i];
   434→        const dx = p.x - x, dz = p.z - z, dy = p.y - y;
   435→        if (dx * dx + dz * dz > r2 || dy < -3 || dy > 9) continue;
   436→        slot.broken[i] = true;
   437→        // спрятать экземпляр: сдвинуть под дно + сжать в точку
   438→        this.dummy.position.set(p.x, -40, p.z);
   439→        this.dummy.quaternion.identity();
   440→        this.dummy.scale.setScalar(0.001);
   441→        this.dummy.updateMatrix();
   442→        slot.inst.setMatrixAt(i, this.dummy.matrix);
   443→        slot.inst.instanceMatrix.needsUpdate = true;
   444→        // щепа и листва из точки удара
   445→        fx?.debris(p.x, p.y + 1.6, p.z, 12, 1.2);
   446→        // разлом на ТРИ части — куски летят от точки попадания
   447→        if (gibs) this.gibTreePieces(gibs, slot.kind, p, slot.scales[i], { x, y, z });
   448→        broke++;
   449→      }
   450→    }
   451→    this.treesBroken += broke;
   452→    return broke;
   453→  }

   456→  private static readonly TREE_PIECES: Record<string, {

   467→  /** Три куска вместо дерева: пень у корня, ствол по полёту,
   468→   *  крона вверх; куст целиком — три кома листвы */
   469→  private gibTreePieces(
   470→    gibs: { spawnPart: (pos: THREE.Vector3, vel: THREE.Vector3, sx: number, sy: number, sz: number, hex: number, life?: number) => void },
   471→    kind: string, p: THREE.Vector3, s: number,
   472→    hit: { x: number; y: number; z: number }): void {
   473→    const spec = World.TREE_PIECES[kind] ?? World.TREE_PIECES.jungle;
   474→    const H = Math.max(0.8, spec.baseH * s);      // визуальная высота дерева
   475→    // направление разлёта: от точки попадания к дереву
   476→    const dir = new THREE.Vector3(p.x - hit.x, 0, p.z - hit.z);
   477→    if (dir.lengthSq() < 0.04) dir.set(Math.random() - 0.5, 0, Math.random() - 0.5);
   478→    dir.normalize();
   479→    if (kind === 'bush') {
   480→      // куст без ствола — три кома листвы разного размера
   481→      for (let k = 0; k < 3; k++) {
   482→        const w = (0.5 + k * 0.16) * s;
   483→        gibs.spawnPart(
   484→          new THREE.Vector3(p.x + (Math.random() - 0.5), p.y + 0.35 + k * 0.2, p.z + (Math.random() - 0.5)),
   485→          new THREE.Vector3(dir.x * (2.2 + k * 0.7) + (Math.random() - 0.5) * 1.6,
   486→            1.5 + Math.random() * 1.4, dir.z * (2.2 + k * 0.7) + (Math.random() - 0.5) * 1.6),
   487→          w, w * 0.8, w, k === 1 ? spec.crownHex : spec.trunkHex, 2.4 + Math.random() * 0.8);
   488→      }
   489→      return;
   490→    }
   491→    // 1) ПЕНЬ: нижняя часть ствола оседает у корня
   492→    gibs.spawnPart(
   493→      new THREE.Vector3(p.x, p.y + H * 0.12, p.z),
   494→      new THREE.Vector3(dir.x * 1.5, 0.5, dir.z * 1.5),
   495→      Math.max(0.16, 0.24 * s), Math.max(0.5, H * 0.2), Math.max(0.16, 0.24 * s),
   496→      spec.trunkHex, 3.2);
   497→    // 2) СТВОЛ: средняя часть — по траектории снаряда
   498→    gibs.spawnPart(
   499→      new THREE.Vector3(p.x, p.y + H * 0.55, p.z),
   500→      new THREE.Vector3(dir.x * 4.2 + (Math.random() - 0.5) * 1.6, 2.0 + Math.random(),
   501→        dir.z * 4.2 + (Math.random() - 0.5) * 1.6),
   502→      Math.max(0.13, 0.19 * s), Math.max(1.2, H * 0.33), Math.max(0.13, 0.19 * s),
   503→      spec.trunkHex, 2.8);
   504→    // 3) КРОНА: зелёная шапка вылетает вверх
   505→    const cw = Math.max(0.5, spec.crownW * s);
   506→    gibs.spawnPart(
   507→      new THREE.Vector3(p.x, p.y + H * 0.92, p.z),
   508→      new THREE.Vector3(dir.x * 3.0 + (Math.random() - 0.5) * 2, 3.1 + Math.random() * 1.2,
   509→        dir.z * 3.0 + (Math.random() - 0.5) * 2),
   510→      cw, Math.max(0.4, cw * 0.45), cw, spec.crownHex, 2.3);
   511→  }

   513→  // ---------- порты: по одному на каждом острове, размер по острову ----------
   514→  private buildPorts() {
   515→    const rng = new RNG(WORLD_SEED + 4242);
   516→    this.islands.forEach((isl, idx) => {
   517→      const tier = tierForIsland(isl);
   518→      // остров 0: прежнее направление пирса (гавань не двигаем — там
   519→      // спавнится игрок); остальные — пирс в сторону открытого моря
   520→      const dir = idx === 0
   521→        ? new THREE.Vector3(-1, 0, 1).normalize()
   522→        : new THREE.Vector3(isl.cx, 0, isl.cz).normalize();
   523→      let shore = new THREE.Vector3(isl.cx, 0, isl.cz);
   524→      for (let d = 0; d < isl.r * 1.4; d += 4) {
   525→        const p = new THREE.Vector3(isl.cx, 0, isl.cz).addScaledVector(dir, d);
   526→        if (heightAt(p.x, p.z) < 0.4) { shore = p; break; }
   527→      }
   528→      const dockLen = [26, 36, 46, 52][tier];
   529→      const dockEnd = shore.clone().addScaledVector(dir, dockLen - 4);
   530→      dockEnd.y = 0;
   531→      // детерминированный множитель цен из имени острова (0.78..1.32)
   532→      const hash = Array.from(isl.name).reduce((a, ch) => a + ch.charCodeAt(0) * 7, 3);
   533→      const priceMul = +(0.78 + ((hash * 2654435761) % 1000) / 1000 * 0.54).toFixed(2);
   534→      const population = [
   535→        8 + Math.floor(rng.range(0, 18)),
   536→        40 + Math.floor(rng.range(0, 120)),
   537→        250 + Math.floor(rng.range(0, 450)),
   538→        900 + Math.floor(rng.range(0, 1700)),
   539→      ][tier];
   540→      this.ports.push({
   541→        island: isl, name: isl.name, tier, tierName: TIER_NAMES[tier],
   542→        population, dockEnd, dir, priceMul,
   543→      });
   544→      this.buildPortStructs(this.ports[idx], rng);
   545→      // рыбацкий декор: шлюпки у пирса, краболовки, вешала, поплавки
   546→      this.fishing.push(buildFishingVillage(this.ports[idx], rng, heightAt));

   949→  update(time: number, dt: number, playerPos: THREE.Vector3, waterHeight: (x: number, z: number, t: number) => number) {
   950→    windTimeUniform.value = time;
   951→    this.gullMat.uniforms.uTime.value = time;
   952→
   953→    // рыбацкие деревни: лодки качаются на волнах / тонут, поплавки,
   954→    // рыбаки покачиваются. Далёкие гавани (за туманом) — скрываем:
   955→    // не тратим ни апдейт, ни draw calls
   956→    for (const fv of this.fishing) {
   957→      const anchor = fv.boats[0];
   958→      const d = anchor
   959→        ? Math.hypot(anchor.x - playerPos.x, anchor.z - playerPos.z)
   960→        : 0;
   961→      const vis = d < 4000;
   962→      if (fv.group.visible !== vis) fv.group.visible = vis;
   963→      if (vis) fv.update(time, dt, waterHeight);
   964→    }
   965→
   966→    // чайки
   967→    for (let i = 0; i < this.gulls.length; i++) {
   968→      const g = this.gulls[i];
   969→      g.a += g.speed * dt;
   970→      const x = g.island.cx + Math.cos(g.a) * g.r;
   971→      const z = g.island.cz + Math.sin(g.a) * g.r;
   972→      const y = g.h + Math.sin(time * 1.7 + g.phase) * 2.2;
   973→      this.dummy.position.set(x, y, z);
   974→      const heading = g.a + (g.speed > 0 ? Math.PI / 2 : -Math.PI / 2);
   975→      this.dummy.quaternion.copy(this.tmpQ.setFromEuler(this.tmpE.set(0, heading, Math.sin(time * 9 + g.phase) * 0.14)));
   976→      this.dummy.scale.setScalar(1.35);
   977→      this.dummy.updateMatrix();
   978→      this.gullMesh.setMatrixAt(i, this.dummy.matrix);
   979→    }
   980→    this.gullMesh.instanceMatrix.needsUpdate = true;
   981→
   982→    // крабы: суета по пляжу
   983→    for (let i = 0; i < this.crabs.length; i++) {
   984→      const c = this.crabs[i];
   985→      c.timer -= dt;
   986→      if (c.timer < 0) { c.timer = 1 + Math.random() * 3; c.dirA += (Math.random() - 0.5) * 2.4; }
   987→      const nx = c.x + Math.cos(c.dirA) * c.speed * dt;
   988→      const nz = c.z + Math.sin(c.dirA) * c.speed * dt;
   989→      const h = heightAt(nx, nz);
   990→      if (h > 0.15 && h < 1.8 && Math.hypot(nx - c.island.cx, nz - c.island.cz) < c.island.r * 1.15) { c.x = nx; c.z = nz; }
   991→      else c.dirA += Math.PI * 0.6;
   992→      this.dummy.position.set(c.x, h, c.z);
   993→      this.dummy.quaternion.copy(this.tmpQ.setFromEuler(this.tmpE.set(0, -c.dirA, 0))); // боком
   994→      this.dummy.scale.setScalar(0.9 + Math.sin(time * 8 + c.phase) * 0.05);
   995→      this.dummy.updateMatrix();
   996→      this.crabMesh.setMatrixAt(i, this.dummy.matrix);
   997→    }
   998→    this.crabMesh.instanceMatrix.needsUpdate = true;
   999→
  1000→    // черепахи и акулы — медленные круги у берега
  1001→    for (let i = 0; i < this.turtles.length; i++) {
  1002→      const t = this.turtles[i];
  1003→      t.a += t.speed * dt;
  1004→      const x = t.island.cx + Math.cos(t.a) * t.r;
  1005→      const z = t.island.cz + Math.sin(t.a) * t.r;
  1006→      const wy = waterHeight(x, z, time);
  1007→      this.dummy.position.set(x, wy - 0.12, z);
  1008→      this.dummy.quaternion.copy(this.tmpQ.setFromEuler(this.tmpE.set(0, -(t.a + Math.PI / 2) * Math.sign(t.speed), 0)));
  1009→      this.dummy.scale.setScalar(1.1);
  1010→      this.dummy.updateMatrix();
  1011→      this.turtleMesh.setMatrixAt(i, this.dummy.matrix);
  1012→    }
  1013→    this.turtleMesh.instanceMatrix.needsUpdate = true;
  1014→
  1015→    for (let i = 0; i < this.sharks.length; i++) {
  1016→      const s = this.sharks[i];
  1017→      s.a += s.speed * dt;
  1018→      const x = s.island.cx + Math.cos(s.a) * s.r;
  1019→      const z = s.island.cz + Math.sin(s.a) * s.r;
  1020→      const wy = waterHeight(x, z, time);
  1021→      this.dummy.position.set(x, wy + 0.28, z);
  1022→      this.dummy.quaternion.copy(this.tmpQ.setFromEuler(this.tmpE.set(0, -(s.a + Math.PI / 2), 0)));
  1023→      this.dummy.scale.setScalar(1);
  1024→      this.dummy.updateMatrix();
  1025→      this.sharkMesh.setMatrixAt(i, this.dummy.matrix);
  1026→    }
  1027→    this.sharkMesh.instanceMatrix.needsUpdate = true;
  1028→
  1029→    // обезьяны: прыжки
  1030→    for (let i = 0; i < this.monkeys.length; i++) {
  1031→      const mk = this.monkeys[i];
  1032→      mk.t += dt / mk.dur;
  1033→      if (mk.t >= 1) {
  1034→        mk.t = 0;
  1035→        mk.dur = 0.9 + Math.random() * 0.8;
  1036→        mk.from.copy(mk.to);
  1037→        let ok = false;
  1038→        for (let t = 0; t < 20 && !ok; t++) {
  1039→          const a = Math.random() * Math.PI * 2, rr = 2 + Math.random() * 7;
  1040→          const x = mk.from.x + Math.cos(a) * rr, z = mk.from.z + Math.sin(a) * rr;
  1041→          const h = heightAt(x, z);
  1042→          if (h > 2.0 && h < mk.island.maxH * 0.85 && Math.hypot(x - mk.island.cx, z - mk.island.cz) < mk.island.r * 0.9) {
  1043→            mk.to.set(x, h, z); ok = true;
  1044→          }
  1045→        }
  1046→        if (!ok) mk.to.copy(mk.from);
  1047→      }
  1048→      const k = Math.min(1, mk.t);
  1049→      const x = THREE.MathUtils.lerp(mk.from.x, mk.to.x, k);
  1050→      const z = THREE.MathUtils.lerp(mk.from.z, mk.to.z, k);
  1051→      const y = THREE.MathUtils.lerp(mk.from.y, mk.to.y, k) + Math.sin(k * Math.PI) * 1.35;
  1052→      this.dummy.position.set(x, y, z);
