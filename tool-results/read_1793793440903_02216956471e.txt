    10→// ============================================================
    11→// СПУСК ШЛЮПКИ НА ВОДУ (по замечанию игрока: «можно спустить
    12→// шлюпку на воду и управлять ей, пока не вернёшься обратно на
    13→// борт корабля; добавь анимацию спуска и подъёма»).
    14→//
    15→// Фазы:
    16→//   stowed  — ял висит на БОРТОВЫХ шлюпбалках ВДОЛЬ борта —
    17→//             БОРТОМ К БОРТУ с корпусом, носом к носу корабля (Task 26)
    18→//   lowering— ФАЛЫ ТРАВЯТСЯ: шлюпка опускается за борт к воде
    19→//             (лёгкое покачивание на талях, дрейф от обшивки)
    20→//   afloat  — на воде: отдельная сущность в мире; W/S — вёсла
    21→//             (гребки синхронные, импульс в рабочей фазе),
    22→//             A/D — поворот вёслами (руля нет); качка на волнах
    23→//             Gerstner, вёсла развёрнуты на воду, гребцы качаются
    24→//             в такт; за транцем — пенный след
    25→//   raising — подъём: подтяжка под шлюпбалки (мир) → репарент →
    26→//             выборка талов до исходного положения. Доворот курса —
    27→//             ВСЕГДА по кратчайшей дуге с гашением крена/дифферента:
    28→//             старый slerp на ~180° ПЕРЕВОРАЧИВАЛ шлюпку через борт
    29→//             (замечание игрока Task 25).
    30→//
    31→// Управление шлюпкой — ПОКА игрок не вернётся на борт (B у борта
    32→// у шлюпбалок). Условие спуска: малый ход/якорь, шлюпка цела, есть
    33→// экипаж.
    34→// ============================================================
    35→
    36→export type BoatPhase = 'stowed' | 'lowering' | 'afloat' | 'raising';
    37→
    38→const LOWER_DUR = 7.0;     // спуск/подъём, с
    39→const PULL_DUR = 2.8;      // подтяжка к шлюпбалкам, с
    40→const MAX_FWD = 1.9;       // м/с под вёслами (~3.7 уз)
    41→const MAX_REV = 0.85;
    42→const SWEEP = 0.5;         // размах гребка, рад
    43→const DIP = 0.34;          // погружение лопасти, рад
    44→const DIP0 = 0.12;         // статичный наклон к воде
    45→
    46→const sm = (x: number) => { const c = Math.min(1, Math.max(0, x)); return c * c * (3 - 2 * c); };
    47→/** приведение угла к [−π, π] — повороты ТОЛЬКО по кратчайшей дуге */
    48→const wrapPi = (a: number) => {
    49→  a = (a + Math.PI) % (Math.PI * 2);
    50→  if (a < 0) a += Math.PI * 2;
    51→  return a - Math.PI;
    52→};

    97→  // дорожка за транцем яла на ходу под вёслами — та же лента, что и
    98→  // кильватер корабля, но в масштабе шлюпки; пена Заметная (широкая
    99→  // полоса с турбулентным клином — как след от вёсел на воде)

   104→  });

   106→  // лодку, а не резать её»): та же кромка, что у корабля, в масштабе
   107→  // лодки — кольцо лежит на волнах (шейдерный Gerstner), у носа
   108→  // бурун на ходу; без него плоскость воды прорезала корпус линией

   112→  private pullQuat = new THREE.Quaternion();
   113→  private pullY = 0;          // локальная высота начала выборки талов
   114→  private pullPos = new THREE.Vector3();   // старт выборки талов (локалы корабля)
   115→  private splashed = false;
   116→  private groundCd = 0;       // кулдаун сообщения об отмели
   117→  // floating origin: сцена → мир (для шейдера воротника)

   121→  private _v2 = new THREE.Vector3();
   122→  private _q = new THREE.Quaternion();
   123→  private _qi = new THREE.Quaternion();
   124→  private _e = new THREE.Euler();
   125→  private _n: [number, number, number] = [0, 1, 0];

   127→  constructor(d: LaunchDeps) {
   128→    this.ship = d.ship;
   129→    this.worldGroup = d.worldGroup;
   130→    this.particles = d.particles;
   131→    this.islands = d.islands;
   132→    this.msg = d.msg;
   133→    this.onSplash = d.onSplash;
   134→    this.vis = d.ship.visual.boat ?? null;
   135→    // след пены за шлюпкой — в контейнер мира (координаты те же, что pos)
   136→    this.wake.setWaterHeight(waterHeightAt);
   137→    this.worldGroup.add(this.wake.mesh);
   138→    // пенный воротник по ватерлинии (масштаб лодки)
   139→    if (this.vis) {
   140→      this.foam = new WaterlineFoam(this.vis.outline, {
   141→        width: Math.max(0.4, this.vis.beam * 0.34),
   142→        size: this.vis.len,
   143→      });
   144→      this.foam.mesh.visible = false;
   145→      this.worldGroup.add(this.foam.mesh);
   146→    }
   147→    if (this.vis) {
   148→      // два гребца на банке (появляются при спуске): сидят лицом
   149→      // к корме, качаются в такт гребков
   150→      const rnd = mulberry32(20240);
   151→      for (const side of [-1, 1] as const) {
   152→        const f = buildFigure('rower', rollTraits(rnd, 'rower'), Math.floor(rnd() * 99999));
   153→        const thY = this.vis.gunY - 0.235;
   154→        f.group.position.set(side * this.vis.beam * 0.17, thY - 0.26, this.vis.rowZ + 0.3);
   155→        f.group.rotation.y = Math.PI;      // лицом к корме
   156→        f.group.scale.y = 0.78;
   157→        f.group.visible = false;
   158→        f.group.userData.isBoatRower = true;
   159→        this.vis.group.add(f.group);
   160→        this.rowers.push(f.group);
   161→      }
   162→    }
   163→  }
   164→
   165→  /** игрок в шлюпке (управление кораблём отключено)? */
   166→  inBoat(): boolean { return this.phase !== 'stowed'; }
   167→
   168→  /** убрать за собой: след пены и воротник снимаются со сцены (смена корабля) */
   169→  dispose() {
   170→    this.wake.clear();
   171→    this.wake.mesh.parent?.remove(this.wake.mesh);
   172→    if (this.foam) this.foam.mesh.parent?.remove(this.foam.mesh);
   173→  }
   174→
   175→  /** расстояние от шлюпки до шлюпбалок борта (для HUD/подъёма) */
   176→  distToHoist(): number {
   177→    if (!this.vis) return Infinity;
   178→    if (this.phase === 'afloat') {
   179→      this._v.copy(this.vis.stowPos)
   180→        .applyQuaternion(this.ship.quaternion).add(this.ship.position);
   181→      return Math.hypot(this.pos.x - this._v.x, this.pos.z - this._v.z);
   182→    }
   183→    return 0;
   184→  }
   185→
   186→  /** ПОПЫТКА СПУСКА. Возвращает текст ошибки или null (успех) */
   187→  tryLaunch(docked: boolean): string | null {
   188→    if (!this.vis) return 'На этом судне нет шлюпки';
   189→    if (this.phase !== 'stowed') return null;
   190→    if (!this.vis.group.visible) return 'Шлюпка разбита — в море не починить';
   191→    if (this.ship.sinking || this.ship.sunk) return 'Корабль тонет — не до шлюпки!';
   192→    if (docked) return 'В порту шлюпка не нужна — причал рядом';
   193→    if (this.ship.crew < 1) return 'Некому грести — нужен хоть один человек команды';
   194→    if (!this.ship.anchored && this.ship.speedKnots() > 2.6) {
   195→      return 'Слишком большой ход — уберите паруса или отдайте якорь (F)';
   196→    }
   197→    this.phase = 'lowering';
   198→    this.t = 0;
   199→    this.splashed = false;
   200→    this.setRowersVisible(true);   // команда заняла места на банках
   201→    return null;
   202→  }
   203→
   204→  /** ПОПЫТКА ПОДЪЁМА (когда шлюпка на воде). Ошибка или null */
   205→  tryHoist(): string | null {
   206→    if (this.phase !== 'afloat' || !this.vis) return null;
   207→    if (this.ship.sunk) return 'Корабль ушёл под воду — подниматься некуда…';
   208→    const d = this.distToHoist();
   209→    if (d > 15) return `Далеко от шлюпбалок (${Math.round(d)} м) — подойдите к борту`;
   210→    if (Math.abs(this.vel) > 1.25) return 'Замрите ход перед подъёмом';
   211→    // старт подтяжки
   212→    this.pullFrom.copy(this.vis.group.position);
   213→    this.pullQuat.copy(this.vis.group.quaternion);
   214→    this.phase = 'raising';
   215→    this.raiseStage = 0;
   216→    this.t = 0;
   217→    return null;
   218→  }
   219→
   220→  /** Немедленно вернуть шлюпку на шлюпбалки (смена корабля/смерть) */
   221→  abort() {
   222→    if (!this.vis || this.phase === 'stowed') return;
   223→    const b = this.vis.group;
   224→    if (b.parent !== this.vis.falls.parent) {
   225→      // шлюпка была в мире — вернуть в группу мелких деталей
   226→      this.vis.falls.parent?.add(b);
   227→    }
   228→    b.position.copy(this.vis.stowPos);
   229→    b.rotation.set(0, this.vis.stowRotY, 0);
   230→    b.quaternion.setFromEuler(new THREE.Euler(0, this.vis.stowRotY, 0));
   231→    this.setRowersVisible(false);
   232→    this.oarDeploy = 0;
   233→    this.stowOars();
   234→    this.phase = 'stowed';
   235→    this.t = 0;
   236→    this.vel = 0;
   237→    this.yawRate = 0;
   238→    this.row = 0;
   239→    this.steer = 0;
   240→    updateBoatFalls(this.vis);
   241→  }
   242→
   243→  private setRowersVisible(v: boolean) {
   244→    for (const r of this.rowers) r.visible = v;
   245→  }
   246→
   247→  private stowOars() {
   248→    if (!this.vis) return;
   249→    for (const o of this.vis.oars) {
   250→      const sq = o.userData.stowQ as THREE.Quaternion;
   251→      if (sq) o.quaternion.copy(sq);
   252→    }
   253→  }
   254→
   255→  update(dt: number, time: number, negShift?: THREE.Vector3) {
   256→    this._negShift = negShift;
   257→    if (!this.vis) return;
   258→    const b = this.vis.group;
   259→    // шлюпку разбило прямо во время спуска/на воде — подобрать игрока
   260→    if (!b.visible && this.phase !== 'stowed') {
   261→      this.abort();
   262→      this.msg?.('Шлюпку разнесло щепой! Вас подобрали на борт');
   263→      return;
   264→    }
   265→    // воротник показывается ТОЛЬКО на воде (не на талях/шлюпбалках)
   266→    if (this.foam && this.phase !== 'afloat') this.foam.mesh.visible = false;
   267→    switch (this.phase) {
   268→      case 'stowed': {
   269→        // фалы слегка тянутся от качки корабля
   270→        if (Math.random() < dt * 4) updateBoatFalls(this.vis);
   271→        break;
   272→      }
   273→      case 'lowering': {
   274→        this.t += dt / LOWER_DUR;
   275→        const k = sm(this.t);
   276→        // цель: планширь шлюпки на уровень воды (в локалах корабля)
   277→        this._v.copy(b.position).applyQuaternion(this.ship.quaternion).add(this.ship.position);
   278→        const wy = waterHeightAt(this._v.x, this._v.z, time);
   279→        const targetY = this.vis.stowPos.y + (wy + 0.12 - this._v.y);
   280→        b.position.y = THREE.MathUtils.lerp(this.vis.stowPos.y, targetY, k);
   281→        // лёгкий дрейф ОТ обшивки — чистый спуск за борт
   282→        b.position.x = THREE.MathUtils.lerp(this.vis.stowPos.x, this.vis.stowPos.x + Math.sign(this.vis.stowPos.x || 1) * 0.35, k);
   283→        b.rotation.z = Math.sin(k * Math.PI) * 0.05;   // покачивание на талях
   284→        updateBoatFalls(this.vis);
   285→        if (this.t >= 1) this.detachToWater(time);
   286→        break;
   287→      }
   288→      case 'afloat': {
   289→        this.floatUpdate(dt, time);
   290→        break;
   291→      }
   292→      case 'raising': {
   293→        if (this.raiseStage === 0) {
   294→          // подтяжка: шлюпка в МИРЕ подтягивается под шлюпбалки борта
   295→          this.t += dt / PULL_DUR;
   296→          const k = sm(this.t);
   297→          this._v.set(this.vis.stowPos.x * 0.8, this.vis.stowPos.y * 0.55, this.vis.stowPos.z)
   298→            .applyQuaternion(this.ship.quaternion).add(this.ship.position);
   299→          this._v.y = Math.max(this._v.y, waterHeightAt(this._v.x, this._v.z, time) + 0.9);
   300→          b.position.lerpVectors(this.pullFrom, this._v, k);
   301→          // доворот к курсу корабля — по кратчайшей дуге, крен/дифферент
   302→          // гаснут (никаких переворотов через борт на ~180°)
   303→          this._e.setFromQuaternion(b.quaternion, 'YXZ');
   304→          const dy = wrapPi(this.ship.heading - this._e.y);
   305→          this._e.set(this._e.x * (1 - k), this._e.y + dy * k, this._e.z * (1 - k), 'YXZ');
   306→          b.quaternion.setFromEuler(this._e);
   307→          // фалы тянутся к поднимаемой шлюпке (мировые → локальные корабля)
   308→          this.fallsToBoat();
   309→          if (this.t >= 1) {
   310→            // репарент в группу мелких деталей (мировая поза сохранится)
   311→            this.vis.falls.parent?.attach(b);
   312→            this.raiseStage = 1;
   313→            this.t = 0;
   314→            this.pullY = b.position.y;
   315→            this.pullPos.copy(b.position);
   316→            this.setRowersVisible(false);
   317→          }
   318→        } else {
   319→          // выборка талов: подъём от борта до шлюпбалок
   320→          this.t += dt / LOWER_DUR;
   321→          const k = sm(this.t);
   322→          b.position.lerpVectors(this.pullPos, this.vis.stowPos, k);
   323→          b.rotation.z = Math.sin(k * Math.PI) * 0.05;
   324→          // курс — к stowRotY по кратчайшей дуге, качка гаснет
   325→          this._e.setFromQuaternion(b.quaternion, 'YXZ');
   326→          const k2 = Math.min(1, dt * 2.5);
   327→          const dy = wrapPi(this.vis.stowRotY - this._e.y);
   328→          this._e.set(this._e.x * (1 - k2), this._e.y + dy * k2, this._e.z * (1 - k2), 'YXZ');
   329→          b.quaternion.setFromEuler(this._e);
   330→          // вёсла убираются на планширь по мере подъёма
   331→          this.oarDeploy = Math.max(0, 1 - k * 1.6);
   332→          this.animateOars(0, false);
   333→          updateBoatFalls(this.vis);
   334→          if (this.t >= 1) {
   335→            b.position.copy(this.vis.stowPos);
   336→            this._q.setFromEuler(this._e.set(0, this.vis.stowRotY, 0));
   337→            b.quaternion.copy(this._q);
   338→            b.rotation.set(0, this.vis.stowRotY, 0);
   339→            this.stowOars();
   340→            this.phase = 'stowed';
   341→            this.oarDeploy = 0;
   342→            updateBoatFalls(this.vis);
   343→          }
   344→        }
   345→        break;
   346→      }
   347→    }
   348→    // лента следа живёт всегда: точки стареют и тают и после подъёма
   349→    this.wake.update(time);
   350→  }
   351→
   352→  /** фалы к шлюпке в МИРЕ (подтяжка): концы — позиция шлюпки,
   353→   *  пересчитанная в локальные координаты корабля. Носовая балка —
   354→   *  к планширю НОСА, кормовая — к планширю кормы (±gunX) */
   355→  private fallsToBoat() {
   356→    if (!this.vis) return;
   357→    const b = this.vis.group;
   358→    const attr = this.vis.falls.geometry.getAttribute('position') as THREE.BufferAttribute;
   359→    this._qi.copy(this.ship.quaternion).invert();
   360→    // центр шлюпки в локалах корабля
   361→    const c = this._v2.copy(b.position).sub(this.ship.position).applyQuaternion(this._qi);
   362→    const rotY = this.vis.stowRotY;   // шлюпка выровнена вдоль борта
   363→    const cos = Math.cos(rotY), sin = Math.sin(rotY);
   364→    let i = 0;
   365→    for (let hi = 0; hi < this.vis.davitHeads.length; hi++) {
   366→      const head = this.vis.davitHeads[hi];
   367→      const zo = (hi === 0 ? 1 : -1) * this.vis.hangZ;
   368→      for (const xs of [this.vis.gunX, -this.vis.gunX]) {
   369→        // точка планширя шлюпки (xs, gunY, zo), поворот на rotY
   370→        const px = xs * cos + zo * sin;
   371→        const pz = -xs * sin + zo * cos;
   372→        attr.setXYZ(i, head.x, head.y, head.z);
   373→        attr.setXYZ(i + 1, c.x + px, c.y + this.vis.gunY, c.z + pz);
   374→        i += 2;
   375→      }
   376→    }
   377→    attr.needsUpdate = true;
   378→  }

   380→  /** отдача талов: шлюпка — самостоятельная сущность на воде */
   381→  private detachToWater(time: number) {
   382→    if (!this.vis) return;
   383→    const b = this.vis.group;
   384→    this.worldGroup.attach(b);
   385→    // шлюпка висела носом вдоль корпуса (к носу корабля) —
   386→    // курс на воде = курсу корабля
   387→    this.pos.set(b.position.x, b.position.y, b.position.z);
   388→    this.heading = this.ship.heading;
   389→    this.vel = 0;
   390→    this.yawRate = 0;
   391→    this.strokePhase = 0;
   392→    this.phase = 'afloat';
   393→    this.splashed = false;
   394→    this.setRowersVisible(true);
   395→    updateBoatFalls(this.vis, true);   // хвосты талов у шлюпбалок
   396→    this.msg?.('Шлюпка на воде! W/S — вёсла, A/D — поворот вёслами. B у борта — подняться на борт');
   397→  }
   398→
   399→  /** жизнь на воде: гребля, руль, качка, коллизии */
   400→  private floatUpdate(dt: number, time: number) {
   401→    if (!this.vis) return;
   402→    const b = this.vis.group;
   403→    // --- гребля: импульс только в рабочей фазе (sin φ > 0) ---
   404→    const rowing = this.row !== 0;
   405→    if (rowing) {
   406→      this.strokePhase += dt * (2.0 + Math.min(1, Math.abs(this.vel) / 1.4)) * Math.PI;
   407→    }
   408→    const power = rowing ? Math.max(0, Math.sin(this.strokePhase)) : 0;
   409→    this.vel += this.row * power * dt * 1.5;
   410→    this.vel -= this.vel * Math.abs(this.vel) * 0.34 * dt + this.vel * 0.14 * dt;
   411→    this.vel = THREE.MathUtils.clamp(this.vel, -MAX_REV, MAX_FWD);
   412→    // --- поворот дифференциальной греблёй (руля нет: на заднем
   413→    //     ходу руление зеркально) ---
   414→    const turn = this.steer * (0.3 + Math.min(1.0, Math.abs(this.vel)) * 0.34);
   415→    this.yawRate += (turn - this.yawRate) * Math.min(1, dt * 3.2);
   416→    this.heading += this.yawRate * dt * (this.vel < -0.05 ? -1 : 1);
   417→    this.pos.x += Math.sin(this.heading) * this.vel * dt;
   418→    this.pos.z += Math.cos(this.heading) * this.vel * dt;
   419→
   420→    // --- острова: не заплыть внутрь ---
   421→    this.groundCd = Math.max(0, this.groundCd - dt);
   422→    for (const isl of this.islands()) {
   423→      const dx = this.pos.x - isl.cx, dz = this.pos.z - isl.cz;
   424→      const d = Math.hypot(dx, dz);
   425→      const r = isl.r * 0.72 + 2;
   426→      if (d < r && d > 1e-3) {
   427→        this.pos.x = isl.cx + dx / d * r;
   428→        this.pos.z = isl.cz + dz / d * r;
   429→        this.vel *= 0.35;
   430→        if (this.groundCd <= 0) {
   431→          this.msg?.('Шлюпка ткнулась в отмель');
   432→          this.groundCd = 4;
   433→        }
   434→      }
   435→    }
   436→    // --- корабль: отталкивание от корпуса (по ОБВОДАМ — борт
   437→    //     следует кривизне корпуса: шлюпка скользит вдоль обшивки
   438→    //     «бортом к борту», не отскакивая коробкой) ---
   439→    {
   440→      this._qi.copy(this.ship.quaternion).invert();
   441→      this._v.copy(this.pos).sub(this.ship.position).applyQuaternion(this._qi);
   442→      const L = this.ship.preset.length;
   443→      if (Math.abs(this._v.z) < L * 0.55) {
   444→        const hw = this.ship.halfWidthAt(this._v.z) + 0.75;  // борт + полуширина яла + зазор
   445→        if (Math.abs(this._v.x) < hw) {
   446→          // вытолкнуть поперёк к ближайшему борту
   447→          const push = hw - Math.abs(this._v.x) + 0.05;
   448→          this._v.x += Math.sign(this._v.x || 1) * push;
   449→          this._v.applyQuaternion(this.ship.quaternion).add(this.ship.position);
   450→          this.pos.x = this._v.x; this.pos.z = this._v.z;
   451→          this.vel *= 0.4;
   452→        }
   453→      }
   454→    }
   455→
   456→    // --- ПЛАВУЧЕСТЬ ЯЛА ---
   457→    // ШТОРМОВОЙ МОДУЛЬ (Task 29): посадка/наклон — из ShipBuoyancy
   458→    // (shipPhysics.js игрока): 4+1 замер, нормаль, пружинный гейв.
   459→    // Отсчёт высоты — планширь яла ≈ КВЛ, поэтому depth=0, а
   460→    // immersion (настройка «посадка шлюпки») смещает её ±0.5.
   461→    const L = this.vis.len, B = this.vis.beam;
   462→    const draft = L * 0.155 * 0.15;            // осадка яла (визуальный офсет воротника)
   463→    const fx = Math.sin(this.heading), fz = Math.cos(this.heading);
   464→    if (!this.splashed) {
   465→      this.particles.splash(this.pos.x, waterHeightAt(this.pos.x, this.pos.z, time), this.pos.z, 1.5);
   466→      this.onSplash?.();
   467→      this.splashed = true;
   468→    }
   469→    if (SEA.model === 'storm') {
   470→      const hullDepth = L * 0.155;             // высота борта яла, м
   471→      this.buoy.setVessel({ type: 'boat', len: L, beam: B, depth: 0, deck: 0.12 });
   472→      this.buoy.update(dt, time, {
   473→        x: this.pos.x, z: this.pos.z, yaw: this.heading, speed: Math.abs(this.vel),
   474→      }, waterHeightAt);
   475→      this.pos.y = this.buoy.y + (this.buoy.vessel.immersion - 0.5) * hullDepth;
   476→      b.position.set(this.pos.x, this.pos.y + Math.sin(time * 1.7) * 0.015, this.pos.z);
   477→      // наклон — кватернион модуля + лёгкая качка от гребков
   478→      this._e.setFromQuaternion(this.buoy.q, 'YXZ');
   479→      this.pitch = THREE.MathUtils.clamp(this._e.x, -0.35, 0.35);
   480→      this.roll = THREE.MathUtils.clamp(this._e.z, -0.4, 0.4)
   481→        + (rowing ? Math.sin(this.strokePhase) * 0.035 : 0);
   482→      b.quaternion.setFromEuler(this._e.set(this.pitch, this.heading, this.roll, 'YXZ'));
   483→      if (this.buoy.events.floodPort || this.buoy.events.floodStar) this.stormFloods++;
   484→    } else {
   485→      // 7 ТОЧЕК НА ВОЛНЕ (Task 27→28): нос/корма/борта/мидель +
   486→      // ПОЛУНОС и ПОЛУКОРМА — между прежними 5 точками середина
   487→      // корпуса проваливалась ПОД гребень на коротких волнах
   488→      // («вода режет лодку»); посадка по средневзвешенной поверхности
   489→      // + БЫСТРОЕ сглаживание — ял ДЕРЖИТСЯ ПОВЕРХНОСТИ (нос всходит
   490→      // на гребень и скользит вниз), осадка мала — планширь всегда
   491→      // над водой
   492→      const sx = fz, sz = -fx;                    // штирборт яла
   493→      const hBow = waterHeightAt(this.pos.x + fx * L * 0.42, this.pos.z + fz * L * 0.42, time);
   494→      const hStern = waterHeightAt(this.pos.x - fx * L * 0.42, this.pos.z - fz * L * 0.42, time);
   495→      const hStar = waterHeightAt(this.pos.x + sx * B * 0.5, this.pos.z + sz * B * 0.5, time);
   496→      const hPort = waterHeightAt(this.pos.x - sx * B * 0.5, this.pos.z - sz * B * 0.5, time);
   497→      const hMid = waterHeightAt(this.pos.x, this.pos.z, time);
   498→      const hBowM = waterHeightAt(this.pos.x + fx * L * 0.21, this.pos.z + fz * L * 0.21, time);
   499→      const hSternM = waterHeightAt(this.pos.x - fx * L * 0.21, this.pos.z - fz * L * 0.21, time);
   500→      // посадка: прямое следование СРЕДНЕВЗВЕШЕННОЙ ПОВЕРХНОСТИ (пружина
   501→      // попадала в резонанс с волной и гонила ял в противофазе ±2.5 м —
   502→      // «лодка плавает под водой/над водой»); быстрое экспоненциальное
   503→      // сглаживание гасит дрожь замеров, но не отстаёт от волны.
   504→      // Осадка мала (0.15·полной высоты борта — Task 28: ял ПОВЫШЕ
   505→      // на воде, привальный брус не подрезается водой)
   506→      const targetY = hMid * 0.30
   507→        + (hBow + hStern) * 0.11 + (hPort + hStar) * 0.095
   508→        + (hBowM + hSternM) * 0.145 - draft;
   509→      this.pos.y += (targetY - this.pos.y) * Math.min(1, dt * 13.0);
   510→      b.position.set(this.pos.x, this.pos.y + Math.sin(time * 1.7) * 0.015, this.pos.z);
   511→      // наклон по УКЛОНАМ воды между точками (нос-корма, борт-борт) +
   512→      // лёгкая бортовая качка от гребков
   513→      const pitchSlope = (hBow - hStern) / (L * 0.84);
   514→      const rollSlope = (hStar - hPort) / (B * 1.0);
   515→      this.pitch = THREE.MathUtils.clamp(-pitchSlope, -0.35, 0.35);
   516→      this.roll = THREE.MathUtils.clamp(rollSlope, -0.4, 0.4)
   517→        + (rowing ? Math.sin(this.strokePhase) * 0.035 : 0);
   518→      b.quaternion.setFromEuler(this._e.set(this.pitch, this.heading, this.roll, 'YXZ'));
   519→    }
   520→
   521→    // --- вода ОБТЕКАЕТ ял: бурун у носа на ходу (как у корабля) ---
   522→    this.bowFoamCd = Math.max(0, this.bowFoamCd - dt);
   523→    if (Math.abs(this.vel) > 0.35 && this.bowFoamCd <= 0) {
   524→      const bx = this.pos.x + fx * L * 0.44, bz = this.pos.z + fz * L * 0.44;
   525→      this.particles.splash(bx, waterHeightAt(bx, bz, time) + 0.05, bz,
   526→        0.25 + Math.min(0.5, Math.abs(this.vel) * 0.3));
   527→      this.bowFoamCd = 0.5;
   528→    }
   529→
   530→    // --- вёсла: разворот на воду + гребок; гребцы в такт ---
   531→    this.oarDeploy += ((this.phase === 'afloat' ? 1 : 0) - this.oarDeploy) * Math.min(1, dt * 1.4);
   532→    this.animateOars(0, true);
   533→    for (let i = 0; i < this.rowers.length; i++) {
   534→      const r = this.rowers[i];
   535→      r.rotation.x = -0.22 * (rowing ? Math.sin(this.strokePhase) * Math.sign(this.row || 1) : 0)
   536→        + Math.sin(time * 1.2 + i * 2.1) * 0.02;
   537→      r.rotation.z = Math.sin(time * 1.6 + i) * 0.025;
   538→    }
   539→    // --- СЛЕД ОТ ШЛЮПКИ: пена за транцем при ходе (Task 26) ---
   540→    if (Math.abs(this.vel) > 0.12) {
   541→      const spdFrac = Math.min(1, Math.abs(this.vel) / MAX_FWD);
   542→      this._v2.set(
   543→        this.pos.x - Math.sin(this.heading) * this.vis.len * 0.48,
   544→        0,
   545→        this.pos.z - Math.cos(this.heading) * this.vis.len * 0.48);
   546→      this.wake.push(this._v2, spdFrac, dt, time);
   547→    }
   548→
   549→    // --- ПЕННЫЙ ВОРОТНИК ПО ВАТЕРЛИНИИ (Task 28): кольцо у самого
   550→    //     борта — вода «омывает» корпус вместо линии разреза;
   551→    //     высоту волны добавляет вершинный шейдер (как у корабля),
   552→    //     на CPU — привязка к лодке: позиция/курс/скорость/крен ---
   553→    if (this.foam) {
   554→      this.foam.mesh.visible = true;
   555→      this.foam.mesh.position.set(this.pos.x, this.pos.y + draft + 0.05, this.pos.z);
   556→      this.foam.mesh.rotation.y = this.heading;
   557→      const spdFrac = Math.min(1, Math.abs(this.vel) / MAX_FWD);
   558→      this.foam.update(time, spdFrac, this.roll, this._negShift);
   559→    }
   560→    // --- румпеля нет: курс держат вёсла ---
   561→  }

   563→  /** вёсла: стowed → на воде, гребок/погружение */
   564→  private animateOars(_dt: number, rowing: boolean) {
   565→    if (!this.vis) return;
   566→    const rowDir = this.row < 0 ? -1 : 1;
   567→    const sweep = rowing && this.row !== 0
   568→      ? -Math.cos(this.strokePhase) * SWEEP * rowDir : 0.1;
   569→    const dip = rowing && this.row !== 0
   570→      ? Math.max(0, Math.sin(this.strokePhase)) * DIP : 0;
   571→    for (const o of this.vis.oars) {
   572→      const side = o.userData.side as number;
   573→      const stowQ = o.userData.stowQ as THREE.Quaternion;
   574→      const yaw = new THREE.Quaternion().setFromAxisAngle(UP, side * sweep);
   575→      const d = new THREE.Quaternion().setFromAxisAngle(FWD, -side * (DIP0 + dip));
   576→      const rowQ = d.multiply(yaw);
   577→      o.quaternion.copy(stowQ).slerp(rowQ, this.oarDeploy);
   578→    }
   579→  }

   582→  state() {
   583→    return {
   584→      phase: this.phase, t: +this.t.toFixed(2),
   585→      x: +this.pos.x.toFixed(1), y: +this.pos.y.toFixed(1), z: +this.pos.z.toFixed(1),
   586→      heading: +this.heading.toFixed(2), vel: +this.vel.toFixed(2),
   587→      row: this.row, steer: this.steer,
   588→      dist: +this.distToHoist().toFixed(1),
   589→      oarDeploy: +this.oarDeploy.toFixed(2),
   590→      rowersVisible: this.rowers.length > 0 && this.rowers[0].visible,
   591→      wakePts: this.wake.count(),
   592→      foamCollar: !!this.foam && this.foam.mesh.visible,
   593→      draft: this.vis ? +(this.vis.len * 0.155 * 0.15).toFixed(3) : 0,
   594→    };
   595→  }
