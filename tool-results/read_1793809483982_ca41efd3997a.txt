    10→// ============================================================
    11→// СПУСК ШЛЮПКИ НА ВОДУ (по замечанию игрока: «можно спустить
    12→// шлюпку на воду и управлять ей, пока не вернёшься обратно на
    13→// борт корабля; добавь анимацию спуска и подъёма»).
    14→//
    15→// Фазы:
    16→//   stowed  — ял висит на БОРТОВЫХ шлюпбалках ВДОЛЬ борта —
    17→//             БОРТОМ К БОРТУ с корпусом, носом к носу корабля (Task 26)
    18→//   lowering— ФАЛЫ ТРАВЯТСЯ: шлюпка опускается за борт к воде
    19→//             (лёгкое покачивание на талях, дрейф от обшивки)
    20→//   afloat  — на воде: отдельная сущность в мире; W/S — вёсла
    21→//             (гребки синхронные, импульс в рабочей фазе),
    22→//             A/D — поворот вёслами (руля нет); качка на волнах
    23→//             Gerstner, вёсла развёрнуты на воду, гребцы качаются
    24→//             в такт; за транцем — пенный след
    25→//   raising — подъём: подтяжка под шлюпбалки (мир) → репарент →
    26→//             выборка талов до исходного положения. Доворот курса —
    27→//             ВСЕГДА по кратчайшей дуге с гашением крена/дифферента:
    28→//             старый slerp на ~180° ПЕРЕВОРАЧИВАЛ шлюпку через борт
    29→//             (замечание игрока Task 25).
    30→//
    31→// Управление шлюпкой — ПОКА игрок не вернётся на борт (B у борта
    32→// у шлюпбалок). Условие спуска: малый ход/якорь, шлюпка цела, есть
    33→// экипаж.
    34→// ============================================================
    35→
    36→export type BoatPhase = 'stowed' | 'lowering' | 'afloat' | 'raising';
    37→
    38→const LOWER_DUR = 7.0;     // спуск/подъём, с
    39→const PULL_DUR = 2.8;      // подтяжка к шлюпбалкам, с
    40→const MAX_FWD = 1.9;       // м/с под вёслами (~3.7 уз)
    41→const MAX_REV = 0.85;
    42→const SWEEP = 0.5;         // размах гребка, рад
    43→const DIP = 0.34;          // погружение лопасти, рад
    44→const DIP0 = 0.12;         // статичный наклон к воде
    45→
    46→const sm = (x: number) => { const c = Math.min(1, Math.max(0, x)); return c * c * (3 - 2 * c); };
    47→/** приведение угла к [−π, π] — повороты ТОЛЬКО по кратчайшей дуге */
    48→const wrapPi = (a: number) => {
    49→  a = (a + Math.PI) % (Math.PI * 2);
    50→  if (a < 0) a += Math.PI * 2;
    51→  return a - Math.PI;
    52→};

   134→  constructor(d: LaunchDeps) {

   206→  /** убрать за собой: след пены и воротник снимаются со сцены (смена корабля) */
   207→  dispose() {
   208→    this.wake.clear();
   209→    this.wake.mesh.parent?.remove(this.wake.mesh);
   210→    if (this.foam) this.foam.mesh.parent?.remove(this.foam.mesh);

   213→  /** расстояние от шлюпки до шлюпбалок борта (для HUD/подъёма) */
   214→  distToHoist(): number {
   215→    if (!this.active) return Infinity;
   216→    if (this.phase === 'afloat') {
   217→      this._v.copy(this.active.stowPos)
   218→        .applyQuaternion(this.ship.quaternion).add(this.ship.position);
   219→      return Math.hypot(this.pos.x - this._v.x, this.pos.z - this._v.z);
   220→    }
   221→    return 0;
   222→  }

   224→  /** ПОПЫТКА СПУСКА. Возвращает текст ошибки или null (успех).
   225→   *  Task 31: из двух ялов спускается первая ЦЕЛАЯ — штирборт,
   226→   *  за ним порт (запасная); разбитая пропускается */
   227→  tryLaunch(docked: boolean): string | null {
   228→    if (!this.boats.length) return 'На этом судне нет шлюпки';
   229→    if (this.phase !== 'stowed') return null;
   230→    const pick = this.boats.find(b => b.group.visible);
   231→    if (!pick) return 'Обе шлюпки разбиты — в море не починить';
   232→    if (this.ship.sinking || this.ship.sunk) return 'Корабль тонет — не до шлюпки!';
   233→    if (docked) return 'В порту шлюпка не нужна — причал рядом';
   234→    if (this.ship.crew < 1) return 'Некому грести — нужен хоть один человек команды';
   235→    if (!this.ship.anchored && this.ship.speedKnots() > 2.6) {
   236→      return 'Слишком большой ход — уберите паруса или отдайте якорь (F)';
   237→    }
   238→    this.active = pick;
   239→    this.phase = 'lowering';
   240→    this.t = 0;
   241→    this.splashed = false;
   242→    this.setRowersVisible(true);   // команда заняла места на банках
   243→    return null;
   244→  }
   245→
   246→  /** ПОПЫТКА ПОДЪЁМА (когда шлюпка на воде). Ошибка или null */
   247→  tryHoist(): string | null {
   248→    if (this.phase !== 'afloat' || !this.active) return null;
   249→    if (this.ship.sunk) return 'Корабль ушёл под воду — подниматься некуда…';
   250→    const d = this.distToHoist();
   251→    if (d > 15) return `Далеко от шлюпбалок (${Math.round(d)} м) — подойдите к борту`;
   252→    if (Math.abs(this.vel) > 1.25) return 'Замрите ход перед подъёмом';
   253→    // старт подтяжки
   254→    this.pullFrom.copy(this.active.group.position);
   255→    this.pullQuat.copy(this.active.group.quaternion);
   256→    this.phase = 'raising';
   257→    this.raiseStage = 0;
   258→    this.t = 0;
   259→    return null;
   260→  }
   261→
   262→  /** Немедленно вернуть шлюпку на шлюпбалки (смена корабля/смерть) */
   263→  abort() {
   264→    if (!this.active || this.phase === 'stowed') return;
   265→    const b = this.active.group;
   266→    if (b.parent !== this.active.falls.parent) {
   267→      // шлюпка была в мире — вернуть в группу мелких деталей
   268→      this.active.falls.parent?.add(b);
   269→    }
   270→    b.position.copy(this.active.stowPos);
   271→    b.rotation.set(0, this.active.stowRotY, 0);
   272→    b.quaternion.setFromEuler(new THREE.Euler(0, this.active.stowRotY, 0));
   273→    this.setRowersVisible(false);
   274→    this.oarDeploy = 0;
   275→    this.stowOars();
   276→    this.phase = 'stowed';
   277→    this.active = null;               // Task 31: активного больше нет
   278→    this.t = 0;
   279→    this.vel = 0;
   280→    this.yawRate = 0;
   281→    this.row = 0;
   282→    this.steer = 0;
   283→    const av = this.boats.find(x => x.group === b) ?? this.boats[0];
   284→    updateBoatFalls(av);
   285→  }

   287→  private setRowersVisible(v: boolean) {
   288→    // Task 31: гребцы — только АКТИВНОГО яла (у каждого свой экипаж)
   289→    const list = this.active ? this.rowersByBoat.get(this.active) : null;
   290→    if (list) for (const r of list) r.visible = v;
   291→    else if (!this.active) {
   292→      // убраны — спрятать всех (после подъёма/аварии)
   293→      for (const l of this.rowersByBoat.values()) for (const r of l) r.visible = false;
   294→    }
   295→  }

   297→  private stowOars() {
   298→    if (!this.active) return;
   299→    for (const o of this.active.oars) {
   300→      const sq = o.userData.stowQ as THREE.Quaternion;
   301→      if (sq) o.quaternion.copy(sq);
   302→    }
   303→  }

   305→  update(dt: number, time: number, negShift?: THREE.Vector3) {
   306→    this._negShift = negShift;
   307→    if (!this.boats.length) return;
   308→    // активный ял разбило прямо во время спуска/на воде — подобрать
   309→    // игрока на борт; следующий спуск возьмёт целую запасную (port)
   310→    if (this.active && !this.active.group.visible && this.phase !== 'stowed') {
   311→      const done = this.active;
   312→      this.abort();
   313→      this.msg?.('Шлюпку разнесло щепой! Вас подобрали на борт'
   314→        + (this.boats.some(b => b !== done && b.group.visible) ? ' (запасная на шлюпбалках)' : ''));
   315→      return;
   316→    }
   317→    // воротник показывается ТОЛЬКО на воде (не на талях/шлюпбалках)
   318→    if (this.foam && this.phase !== 'afloat') this.foam.mesh.visible = false;
   319→    // Task 31: активный ял (в фазах спуска/воды/подъёма он всегда
   320→    // есть — tryLaunch назначает, abort снимает вместе с фазой)
   321→    const vis = this.active;
   322→    if (vis && this.phase !== 'stowed') {
   323→      const b = vis.group;
   324→      switch (this.phase) {
   325→        case 'lowering': {
   326→          this.t += dt / LOWER_DUR;
   327→          const k = sm(this.t);
   328→          // цель: планширь шлюпки на уровень воды (в локалах корабля)
   329→          this._v.copy(b.position).applyQuaternion(this.ship.quaternion).add(this.ship.position);
   330→          const wy = waterHeightAt(this._v.x, this._v.z, time);
   331→          const targetY = vis.stowPos.y + (wy + 0.12 - this._v.y);
   332→          b.position.y = THREE.MathUtils.lerp(vis.stowPos.y, targetY, k);
   333→          // лёгкий дрейф ОТ обшивки — чистый спуск за борт
   334→          b.position.x = THREE.MathUtils.lerp(vis.stowPos.x, vis.stowPos.x + Math.sign(vis.stowPos.x || 1) * 0.35, k);
   335→          b.rotation.z = Math.sin(k * Math.PI) * 0.05;   // покачивание на талях
   336→          updateBoatFalls(vis);
   337→          if (this.t >= 1) this.detachToWater(time);
   338→          break;
   339→        }
   340→        case 'afloat': {
   341→          this.floatUpdate(dt, time);
   342→          break;
   343→        }
   344→        case 'raising': {
   345→          if (this.raiseStage === 0) {
   346→            // подтяжка: шлюпка в МИРЕ подтягивается под шлюпбалки борта
   347→            this.t += dt / PULL_DUR;
   348→            const k = sm(this.t);
   349→            this._v.set(vis.stowPos.x * 0.8, vis.stowPos.y * 0.55, vis.stowPos.z)
   350→              .applyQuaternion(this.ship.quaternion).add(this.ship.position);
   351→            this._v.y = Math.max(this._v.y, waterHeightAt(this._v.x, this._v.z, time) + 0.9);
   352→            b.position.lerpVectors(this.pullFrom, this._v, k);
   353→            // доворот к курсу корабля — по кратчайшей дуге, крен/дифферент
   354→            // гаснут (никаких переворотов через борт на ~180°)
   355→            this._e.setFromQuaternion(b.quaternion, 'YXZ');
   356→            const dy = wrapPi(this.ship.heading - this._e.y);
   357→            this._e.set(this._e.x * (1 - k), this._e.y + dy * k, this._e.z * (1 - k), 'YXZ');
   358→            b.quaternion.setFromEuler(this._e);
   359→            // фалы тянутся к поднимаемой шлюпке (мировые → локальные корабля)
   360→            this.fallsToBoat();
   361→            if (this.t >= 1) {
   362→              // репарент в группу мелких деталей (мировая поза сохранится)
   363→              vis.falls.parent?.attach(b);
   364→              this.raiseStage = 1;
   365→              this.t = 0;
   366→              this.pullY = b.position.y;
   367→              this.pullPos.copy(b.position);
   368→              this.setRowersVisible(false);
   369→            }
   370→          } else {
   371→            // выборка талов: подъём от борта до шлюпбалок
   372→            this.t += dt / LOWER_DUR;
   373→            const k = sm(this.t);
   374→            b.position.lerpVectors(this.pullPos, vis.stowPos, k);
   375→            b.rotation.z = Math.sin(k * Math.PI) * 0.05;
   376→            // курс — к stowRotY по кратчайшей дуге, качка гаснет
   377→            this._e.setFromQuaternion(b.quaternion, 'YXZ');
   378→            const k2 = Math.min(1, dt * 2.5);
   379→            const dy = wrapPi(vis.stowRotY - this._e.y);
   380→            this._e.set(this._e.x * (1 - k2), this._e.y + dy * k2, this._e.z * (1 - k2), 'YXZ');
   381→            b.quaternion.setFromEuler(this._e);
   382→            // вёсла убираются на планширь по мере подъёма
   383→            this.oarDeploy = Math.max(0, 1 - k * 1.6);
   384→            this.animateOars(0, false);
   385→            updateBoatFalls(vis);
   386→            if (this.t >= 1) {
   387→              b.position.copy(vis.stowPos);
   388→              this._q.setFromEuler(this._e.set(0, vis.stowRotY, 0));
   389→              b.quaternion.copy(this._q);
   390→              b.rotation.set(0, vis.stowRotY, 0);
   391→              this.stowOars();
   392→              this.phase = 'stowed';
   393→              this.active = null;           // Task 31: ял убран, активного нет
   394→              this.oarDeploy = 0;
   395→              updateBoatFalls(vis);
   396→            }
   397→          }
   398→          break;
   399→        }
   400→      }
   401→    } else {
   402→      // убраны: фалы слегка тянутся от качки корабля (у КАЖДОГО яла)
   403→      for (const bt of this.boats) {
   404→        if (bt.group.visible && Math.random() < dt * 4) updateBoatFalls(bt);
   405→      }
   406→    }
   407→    // лента следа живёт всегда: точки стареют и тают и после подъёма
   408→    this.wake.update(time);
   409→  }

   411→  /** фалы к шлюпке в МИРЕ (подтяжка): концы — позиция шлюпки,
   412→   *  пересчитанная в локальные координаты корабля. Носовая балка —
   413→   *  к планширю НОСА, кормовая — к планширю кормы (±gunX) */
   414→  private fallsToBoat() {
   415→    if (!this.active) return;
   416→    const b = this.active.group;
   417→    const attr = this.active.falls.geometry.getAttribute('position') as THREE.BufferAttribute;
   418→    this._qi.copy(this.ship.quaternion).invert();
   419→    // центр шлюпки в локалах корабля
   420→    const c = this._v2.copy(b.position).sub(this.ship.position).applyQuaternion(this._qi);
   421→    const rotY = this.active.stowRotY;   // шлюпка выровнена вдоль борта
   422→    const cos = Math.cos(rotY), sin = Math.sin(rotY);
   423→    let i = 0;
   424→    for (let hi = 0; hi < this.active.davitHeads.length; hi++) {
   425→      const head = this.active.davitHeads[hi];
   426→      const zo = (hi === 0 ? 1 : -1) * this.active.hangZ;
   427→      for (const xs of [this.active.gunX, -this.active.gunX]) {
   428→        // точка планширя шлюпки (xs, gunY, zo), поворот на rotY
   429→        const px = xs * cos + zo * sin;
   430→        const pz = -xs * sin + zo * cos;
   431→        attr.setXYZ(i, head.x, head.y, head.z);
   432→        attr.setXYZ(i + 1, c.x + px, c.y + this.active.gunY, c.z + pz);
   433→        i += 2;
   434→      }
   435→    }
   436→    attr.needsUpdate = true;
   437→  }

   439→  /** отдача талов: шлюпка — самостоятельная сущность на воде */
   440→  private detachToWater(time: number) {
   441→    if (!this.active) return;
   442→    const b = this.active.group;
   443→    this.worldGroup.attach(b);
   444→    // шлюпка висела носом вдоль корпуса (к носу корабля) —
   445→    // курс на воде = курсу корабля
   446→    this.pos.set(b.position.x, b.position.y, b.position.z);
   447→    this.heading = this.ship.heading;
   448→    this.vel = 0;
   449→    this.yawRate = 0;
   450→    this.strokePhase = 0;
   451→    this.phase = 'afloat';
   452→    this.splashed = false;
   453→    this.setRowersVisible(true);
   454→    updateBoatFalls(this.active, true);   // хвосты талов у шлюпбалок
   455→    this.msg?.('Шлюпка на воде! W/S — вёсла, A/D — поворот вёслами. B у борта — подняться на борт');
   456→  }

   458→  /** жизнь на воде: гребля, руль, качка, коллизии */
   459→  private floatUpdate(dt: number, time: number) {
   460→    if (!this.active) return;
   461→    const b = this.active.group;
   462→    // --- гребля: импульс только в рабочей фазе (sin φ > 0) ---
   463→    const rowing = this.row !== 0;
   464→    if (rowing) {
   465→      this.strokePhase += dt * (2.0 + Math.min(1, Math.abs(this.vel) / 1.4)) * Math.PI;
   466→    }
   467→    const power = rowing ? Math.max(0, Math.sin(this.strokePhase)) : 0;
   468→    this.vel += this.row * power * dt * 1.5;
   469→    this.vel -= this.vel * Math.abs(this.vel) * 0.34 * dt + this.vel * 0.14 * dt;
   470→    this.vel = THREE.MathUtils.clamp(this.vel, -MAX_REV, MAX_FWD);
   471→    // --- поворот дифференциальной греблёй (руля нет: на заднем
   472→    //     ходу руление зеркально) ---
   473→    const turn = this.steer * (0.3 + Math.min(1.0, Math.abs(this.vel)) * 0.34);
   474→    this.yawRate += (turn - this.yawRate) * Math.min(1, dt * 3.2);
   475→    this.heading += this.yawRate * dt * (this.vel < -0.05 ? -1 : 1);
   476→    this.pos.x += Math.sin(this.heading) * this.vel * dt;
   477→    this.pos.z += Math.cos(this.heading) * this.vel * dt;
   478→
   479→    // --- острова: не заплыть внутрь ---
   480→    this.groundCd = Math.max(0, this.groundCd - dt);
   481→    for (const isl of this.islands()) {
   482→      const dx = this.pos.x - isl.cx, dz = this.pos.z - isl.cz;
   483→      const d = Math.hypot(dx, dz);
   484→      const r = isl.r * 0.72 + 2;
   485→      if (d < r && d > 1e-3) {
   486→        this.pos.x = isl.cx + dx / d * r;
   487→        this.pos.z = isl.cz + dz / d * r;
   488→        this.vel *= 0.35;
   489→        if (this.groundCd <= 0) {
   490→          this.msg?.('Шлюпка ткнулась в отмель');
   491→          this.groundCd = 4;
   492→        }
   493→      }
   494→    }
   495→    // --- корабль: отталкивание от корпуса (по ОБВОДАМ — борт
   496→    //     следует кривизне корпуса: шлюпка скользит вдоль обшивки
   497→    //     «бортом к борту», не отскакивая коробкой) ---
   498→    {
   499→      this._qi.copy(this.ship.quaternion).invert();
   500→      this._v.copy(this.pos).sub(this.ship.position).applyQuaternion(this._qi);
   501→      const L = this.ship.preset.length;
   502→      if (Math.abs(this._v.z) < L * 0.55) {
   503→        const hw = this.ship.halfWidthAt(this._v.z) + 0.75;  // борт + полуширина яла + зазор
   504→        if (Math.abs(this._v.x) < hw) {
   505→          // вытолкнуть поперёк к ближайшему борту
   506→          const push = hw - Math.abs(this._v.x) + 0.05;
   507→          this._v.x += Math.sign(this._v.x || 1) * push;
   508→          this._v.applyQuaternion(this.ship.quaternion).add(this.ship.position);
   509→          this.pos.x = this._v.x; this.pos.z = this._v.z;
   510→          this.vel *= 0.4;
   511→        }
   512→      }
   513→    }
   514→
   515→    // --- ПЛАВУЧЕСТЬ ЯЛА ---
   516→    // ШТОРМОВОЙ МОДУЛЬ (Task 29): посадка/наклон — из ShipBuoyancy
   517→    // (shipPhysics.js игрока): 4+1 замер, нормаль, пружинный гейв.
   518→    // Отсчёт высоты — планширь яла ≈ КВЛ, поэтому depth=0, а
   519→    // immersion (настройка «посадка шлюпки») смещает её ±0.5.
   520→    // Task 31: осадка/пределы наклона/плавность — из настроек
   521→    // «Плавание и качка» (действуют во ВСЕХ моделях моря)
   522→    const L = this.active.len, B = this.active.beam;
   523→    const draft = L * 0.155 * 0.15 * BUOYANCY_CONFIG.boatDraftMul;  // осадка яла (настройка ×)
   524→    const fx = Math.sin(this.heading), fz = Math.cos(this.heading);
   525→    if (!this.splashed) {
   526→      this.particles.splash(this.pos.x, waterHeightAt(this.pos.x, this.pos.z, time), this.pos.z, 1.5);
   527→      this.onSplash?.();
   528→      this.splashed = true;
   529→    }
   530→    if (SEA.model === 'storm') {
   531→      const hullDepth = L * 0.155;             // высота борта яла, м
   532→      this.buoy.setVessel({ type: 'boat', len: L, beam: B, depth: 0, deck: 0.12 });
   533→      this.buoy.update(dt, time, {
   534→        x: this.pos.x, z: this.pos.z, yaw: this.heading, speed: Math.abs(this.vel),
   535→      }, waterHeightAt);
   536→      this.pos.y = this.buoy.y + (this.buoy.vessel.immersion - 0.5) * hullDepth;
   537→      b.position.set(this.pos.x, this.pos.y + Math.sin(time * 1.7) * 0.015, this.pos.z);
   538→      // наклон — кватернион модуля + лёгкая качка от гребков;
   539→      // пределы — настройки «углы наклона» (Task 31)
   540→      this._e.setFromQuaternion(this.buoy.q, 'YXZ');
   541→      this.pitch = THREE.MathUtils.clamp(this._e.x, -BUOYANCY_CONFIG.pitchMax, BUOYANCY_CONFIG.pitchMax);
   542→      this.roll = THREE.MathUtils.clamp(this._e.z, -BUOYANCY_CONFIG.heelMax, BUOYANCY_CONFIG.heelMax)
   543→        + (rowing ? Math.sin(this.strokePhase) * 0.035 : 0);
   544→      b.quaternion.setFromEuler(this._e.set(this.pitch, this.heading, this.roll, 'YXZ'));
   545→      if (this.buoy.events.floodPort || this.buoy.events.floodStar) this.stormFloods++;
   546→    } else {
   547→      // 7 ТОЧЕК НА ВОЛНЕ (Task 27→28): нос/корма/борта/мидель +
   548→      // ПОЛУНОС и ПОЛУКОРМА — между прежними 5 точками середина
   549→      // корпуса проваливалась ПОД гребень на коротких волнах
   550→      // («вода режет лодку»); посадка по средневзвешенной поверхности
   551→      // + БЫСТРОЕ сглаживание — ял ДЕРЖИТСЯ ПОВЕРХНОСТИ (нос всходит
   552→      // на гребень и скользит вниз), осадка мала — планширь всегда
   553→      // над водой
   554→      const sx = fz, sz = -fx;                    // штирборт яла
   555→      const hBow = waterHeightAt(this.pos.x + fx * L * 0.42, this.pos.z + fz * L * 0.42, time);
   556→      const hStern = waterHeightAt(this.pos.x - fx * L * 0.42, this.pos.z - fz * L * 0.42, time);
   557→      const hStar = waterHeightAt(this.pos.x + sx * B * 0.5, this.pos.z + sz * B * 0.5, time);
   558→      const hPort = waterHeightAt(this.pos.x - sx * B * 0.5, this.pos.z - sz * B * 0.5, time);
   559→      const hMid = waterHeightAt(this.pos.x, this.pos.z, time);
   560→      const hBowM = waterHeightAt(this.pos.x + fx * L * 0.21, this.pos.z + fz * L * 0.21, time);
   561→      const hSternM = waterHeightAt(this.pos.x - fx * L * 0.21, this.pos.z - fz * L * 0.21, time);
   562→      // посадка: прямое следование СРЕДНЕВЗВЕШЕННОЙ ПОВЕРХНОСТИ (пружина
   563→      // попадала в резонанс с волной и гонила ял в противофазе ±2.5 м —
   564→      // «лодка плавает под водой/над водой»); быстрое экспоненциальное
   565→      // сглаживание гасит дрожь замеров, но не отстаёт от волны.
   566→      // Осадка мала (0.15·полной высоты борта — Task 28: ял ПОВЫШЕ
   567→      // на воде, привальный брус не подрезается водой)
   568→      const targetY = hMid * 0.30
   569→        + (hBow + hStern) * 0.11 + (hPort + hStar) * 0.095
   570→        + (hBowM + hSternM) * 0.145 - draft;
   571→      this.pos.y += (targetY - this.pos.y) * Math.min(1, dt * BUOYANCY_CONFIG.boatSmooth);
   572→      b.position.set(this.pos.x, this.pos.y + Math.sin(time * 1.7) * 0.015, this.pos.z);
   573→      // наклон по УКЛОНАМ воды между точками (нос-корма, борт-борт) +
   574→      // лёгкая бортовая качка от гребков; пределы — настройки
   575→      // «углы наклона» (Task 31)
   576→      const pitchSlope = (hBow - hStern) / (L * 0.84);
   577→      const rollSlope = (hStar - hPort) / (B * 1.0);
   578→      this.pitch = THREE.MathUtils.clamp(-pitchSlope, -BUOYANCY_CONFIG.pitchMax, BUOYANCY_CONFIG.pitchMax);
   579→      this.roll = THREE.MathUtils.clamp(rollSlope, -BUOYANCY_CONFIG.heelMax, BUOYANCY_CONFIG.heelMax)
   580→        + (rowing ? Math.sin(this.strokePhase) * 0.035 : 0);
   581→      b.quaternion.setFromEuler(this._e.set(this.pitch, this.heading, this.roll, 'YXZ'));
   582→    }
   583→
   584→    // --- вода ОБТЕКАЕТ ял: бурун у носа на ходу (как у корабля) ---
   585→    this.bowFoamCd = Math.max(0, this.bowFoamCd - dt);
   586→    if (Math.abs(this.vel) > 0.35 && this.bowFoamCd <= 0) {
   587→      const bx = this.pos.x + fx * L * 0.44, bz = this.pos.z + fz * L * 0.44;
   588→      this.particles.splash(bx, waterHeightAt(bx, bz, time) + 0.05, bz,
   589→        0.25 + Math.min(0.5, Math.abs(this.vel) * 0.3));
   590→      this.bowFoamCd = 0.5;
   591→    }
   592→
   593→    // --- вёсла: разворот на воду + гребок; гребцы в такт ---
   594→    // (Task 31: гребцы — АКТИВНОГО яла, список из rowersByBoat)
   595→    this.oarDeploy += ((this.phase === 'afloat' ? 1 : 0) - this.oarDeploy) * Math.min(1, dt * 1.4);
   596→    this.animateOars(0, true);
   597→    const rowerList = this.rowersByBoat.get(this.active);
   598→    if (rowerList) for (let i = 0; i < rowerList.length; i++) {
   599→      const r = rowerList[i];
   600→      r.rotation.x = -0.22 * (rowing ? Math.sin(this.strokePhase) * Math.sign(this.row || 1) : 0)
   601→        + Math.sin(time * 1.2 + i * 2.1) * 0.02;
   602→      r.rotation.z = Math.sin(time * 1.6 + i) * 0.025;
   603→    }
   604→    // --- СЛЕД ОТ ШЛЮПКИ: пена за транцем при ходе (Task 26) ---
   605→    if (Math.abs(this.vel) > 0.12) {
   606→      const spdFrac = Math.min(1, Math.abs(this.vel) / MAX_FWD);
   607→      this._v2.set(
   608→        this.pos.x - Math.sin(this.heading) * this.active.len * 0.48,
   609→        0,
   610→        this.pos.z - Math.cos(this.heading) * this.active.len * 0.48);
   611→      this.wake.push(this._v2, spdFrac, dt, time);
   612→    }
   613→
   614→    // --- ПЕННЫЙ ВОРОТНИК ПО ВАТЕРЛИНИИ (Task 28): кольцо у самого
   615→    //     борта — вода «омывает» корпус вместо линии разреза;
   616→    //     высоту волны добавляет вершинный шейдер (как у корабля),
   617→    //     на CPU — привязка к лодке: позиция/курс/скорость/крен ---
   618→    if (this.foam) {
   619→      this.foam.mesh.visible = true;
   620→      this.foam.mesh.position.set(this.pos.x, this.pos.y + draft + 0.05, this.pos.z);
   621→      this.foam.mesh.rotation.y = this.heading;
   622→      const spdFrac = Math.min(1, Math.abs(this.vel) / MAX_FWD);
   623→      this.foam.update(time, spdFrac, this.roll, this._negShift);
   624→    }
   625→    // --- румпеля нет: курс держат вёсла ---
   626→  }
   627→
   628→  /** вёсла: стowed → на воде, гребок/погружение */
   629→  private animateOars(_dt: number, rowing: boolean) {
   630→    if (!this.active) return;
   631→    const rowDir = this.row < 0 ? -1 : 1;
   632→    const sweep = rowing && this.row !== 0
   633→      ? -Math.cos(this.strokePhase) * SWEEP * rowDir : 0.1;
   634→    const dip = rowing && this.row !== 0
   635→      ? Math.max(0, Math.sin(this.strokePhase)) * DIP : 0;
   636→    for (const o of this.active.oars) {
   637→      const side = o.userData.side as number;
   638→      const stowQ = o.userData.stowQ as THREE.Quaternion;
   639→      const yaw = new THREE.Quaternion().setFromAxisAngle(UP, side * sweep);
   640→      const d = new THREE.Quaternion().setFromAxisAngle(FWD, -side * (DIP0 + dip));
   641→      const rowQ = d.multiply(yaw);
   642→      o.quaternion.copy(stowQ).slerp(rowQ, this.oarDeploy);
   643→    }
   644→  }
   645→
   646→  /** для HUD/E2E */
   647→  state() {
   648→    const rowerList = this.active ? this.rowersByBoat.get(this.active) : null;
   649→    return {
   650→      phase: this.phase, t: +this.t.toFixed(2),
   651→      x: +this.pos.x.toFixed(1), y: +this.pos.y.toFixed(1), z: +this.pos.z.toFixed(1),
   652→      heading: +this.heading.toFixed(2), vel: +this.vel.toFixed(2),
   653→      row: this.row, steer: this.steer,
   654→      dist: +this.distToHoist().toFixed(1),
   655→      oarDeploy: +this.oarDeploy.toFixed(2),
   656→      rowersVisible: !!rowerList && rowerList.length > 0 && rowerList[0].visible,
   657→      wakePts: this.wake.count(),
   658→      foamCollar: !!this.foam && this.foam.mesh.visible,
   659→      draft: this.active ? +(this.active.len * 0.155 * 0.15 * BUOYANCY_CONFIG.boatDraftMul).toFixed(3) : 0,
   660→      // Task 31: два яла — какой на воде
   661→      boats: this.boats.length,
   662→      side: this.activeSide(),
   663→    };
   664→  }
   665→}
   666→
   667→const UP = new THREE.Vector3(0, 1, 0);
   668→const FWD = new THREE.Vector3(0, 0, 1);
   669→
