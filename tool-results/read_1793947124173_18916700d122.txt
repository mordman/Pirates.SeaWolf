    24→/** Роза ветров: подпись направления (откуда дует) */
    25→function windDirLabel(deg: number): string {
    26→  const names = ['С', 'ССВ', 'СВ', 'ВСВ', 'В', 'ВЮВ', 'ЮВ', 'ЮЮВ', 'Ю', 'ЮЮЗ', 'ЮЗ', 'ЗЮЗ', 'З', 'ЗСЗ', 'СЗ', 'ССЗ'];
    27→  return names[Math.round((((deg % 360) + 360) % 360) / 22.5) % 16];
    28→}

    55→function AmmoInput({ value, onChange }: { value: number; onChange: (v: number) => void }) {

    90→export default function SettingsPanel({ initial, onApply, onClose }:
    91→{ initial: GameSettings; onApply: (s: GameSettings) => void; onClose: () => void }) {
    92→  const [s, setS] = useState<GameSettings>(() => structuredClone(initial));
    93→  const [tab, setTab] = useState<Tab>('graphics');
    94→  const [binding, setBinding] = useState<{ key: keyof ControlsSettings; ammoIdx?: number } | null>(null);
    95→  // зеркало актуального состояния — обновление из обработчиков событий
    96→  // (НЕ из фазы рендера: вызов onApply внутри setS(prev=>...) —
    97→  // setState-в-рендере, React ругается «Cannot update a component…»)
    98→  const sRef = useRef(s);
    99→
   100→  // применяем сразу при каждом изменении
   101→  const update = (patch: (draft: GameSettings) => void) => {
   102→    const next = structuredClone(sRef.current);
   103→    patch(next);
   104→    sRef.current = next;
   105→    setS(next);
   106→    onApply(next);
   107→  };
   108→
   109→  // захват клавиши для переназначения
   110→  useEffect(() => {
   111→    if (!binding) return;
   112→    const onKey = (e: KeyboardEvent) => {
   113→      e.preventDefault();
   114→      e.stopPropagation();
   115→      if (e.code === 'Escape') { setBinding(null); return; }
   116→      update(d => {
   117→        if (binding.ammoIdx !== undefined) {
   118→          d.controls.ammo[binding.ammoIdx] = e.code;
   119→        } else {
   120→          const k = binding.key;
   121→          if (k !== 'ammo' && k !== 'invertRudder') d.controls[k] = e.code;
   122→        }
   123→      });
   124→      setBinding(null);
   125→    };
   126→    window.addEventListener('keydown', onKey, true);
   127→    return () => window.removeEventListener('keydown', onKey, true);
   128→  }, [binding]);
   129→
   130→  // все занятые коды (для подсветки конфликтов)
   131→  const usedCodes = new Map<string, string>();
   132→  (Object.keys(s.controls) as (keyof ControlsSettings)[]).forEach(k => {
   133→    if (k === 'invertRudder') return;
   134→    if (k === 'ammo') s.controls.ammo.forEach(c => usedCodes.set(c, 'Боеприпас'));
   135→    else usedCodes.set(s.controls[k] as string, CONTROL_LABELS[k]);
   136→  });
   137→
   138→  const bindBtn = (code: string, action: { key: keyof ControlsSettings; ammoIdx?: number }) => {
   139→    const conflict = binding === null && usedCodes.get(code) && usedCodes.get(code) !== CONTROL_LABELS[action.key];
   140→    const listening = binding && binding.key === action.key && binding.ammoIdx === action.ammoIdx;
   141→    return (
   142→      <button
   143→        onClick={() => setBinding(action)}
   144→        className={`min-w-[72px] rounded border px-2 py-1 text-[11px] font-bold transition ${
   145→          listening
   146→            ? 'animate-pulse border-amber-400 bg-amber-500/30 text-amber-100'
   147→            : conflict
   148→              ? 'border-red-700 bg-red-950/40 text-red-300'
   149→              : 'border-stone-600 bg-stone-900 text-amber-200 hover:border-amber-600'}`}
   150→      >
   151→        {listening ? 'Нажмите…' : keyLabel(code)}
   152→      </button>
   153→    );
   154→  };
   155→
   156→  const g = s.graphics;
   157→
   158→  return (
   159→    <div className="absolute inset-0 z-50 flex items-center justify-center bg-black/70 backdrop-blur-sm" onClick={onClose}>
   160→      <div className="w-[560px] max-h-[86vh] overflow-hidden rounded-xl border-2 border-amber-800 bg-[#1a120a] shadow-2xl"
   161→        onClick={e => e.stopPropagation()}>
   162→        <div className="flex items-center justify-between border-b border-amber-900/60 px-5 py-3">
   163→          <h2 className="font-serif text-xl font-bold text-amber-300">⚙ Настройки</h2>
   164→          <div className="flex gap-1">
   165→            {([['graphics', 'Графика'], ['weather', 'Погода'], ['gameplay', 'Игра'], ['controls', 'Управление'], ['audio', 'Звук']] as [Tab, string][]).map(([t, label]) => (
   166→              <button key={t} onClick={() => setTab(t)}
   167→                className={`rounded px-3 py-1 text-xs font-semibold transition ${
   168→                  tab === t ? 'bg-amber-700/60 text-amber-100' : 'text-stone-400 hover:text-amber-200'}`}>
   169→                {label}
   170→              </button>
   171→            ))}
   172→          </div>
   173→        </div>
   174→
   175→        <div className="max-h-[62vh] overflow-y-auto px-5 py-3">
   176→          {tab === 'graphics' && (
   177→            <div className="divide-y divide-stone-800/60">
   178→              <div className="pb-2">
   179→                <div className="mb-1.5 text-[10px] uppercase tracking-wide text-stone-500">Пресет качества</div>

   186→                          : 'border-stone-700 bg-stone-900/60 text-stone-300 hover:border-amber-700'}`}>
   187→                      {PRESET_LABELS[p]}
