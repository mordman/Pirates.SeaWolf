function woodTexture(base, seam, plankH){
  const S=512, c=document.createElement('canvas'); c.width=c.height=S;
  const x=c.getContext('2d');
  x.fillStyle=base; x.fillRect(0,0,S,S);
  const rows=Math.round(S/plankH);
  for(let i=0;i<rows;i++){
    const y0=i*plankH;
    x.fillStyle = Math.random()<0.5 ? `rgba(0,0,0,${Math.random()*0.10})` : `rgba(255,225,190,${Math.random()*0.07})`;
    x.fillRect(0,y0,S,plankH);
    for(let g=0; g<9; g++){                       // волокна
      x.strokeStyle=`rgba(30,15,5,${0.04+Math.random()*0.10})`;
      x.lineWidth=0.6+Math.random()*1.4;
      const yy=y0+3+Math.random()*(plankH-6), amp=0.8+Math.random()*2.2, f=0.015+Math.random()*0.03, ph=Math.random()*7;
      x.beginPath(); x.moveTo(-4,yy);
      for(let px=0;px<=S+4;px+=12) x.lineTo(px, yy+Math.sin(px*f+ph)*amp);
      x.stroke();
    }
    x.globalAlpha=0.85; x.fillStyle=seam; x.fillRect(0,y0,S,2.5); x.globalAlpha=1;   // шов
    x.fillStyle='rgba(255,240,210,0.07)'; x.fillRect(0,y0+3,S,1.2);
    const jx=40+Math.random()*(S-80);
    x.globalAlpha=0.8; x.fillStyle=seam; x.fillRect(jx,y0,2.5,plankH); x.globalAlpha=1;
    x.fillStyle='rgba(20,12,6,0.6)';                                 // гвозди
    for(let n=0;n<5;n++){
      const nx=20+n*(S-40)/4+(Math.random()-0.5)*6;
      x.beginPath(); x.arc(nx,y0+plankH-6,1.7,0,7); x.fill();
      x.beginPath(); x.arc(nx,y0+6,1.7,0,7); x.fill();
    }
  }
  const img=x.getImageData(0,0,S,S), d=img.data;                     // шум
  for(let i=0;i<d.length;i+=4){ const n=(Math.random()-0.5)*22; d[i]+=n; d[i+1]+=n; d[i+2]+=n; }
  x.putImageData(img,0,0);
  const t=new THREE.CanvasTexture(c);
  t.wrapS=t.wrapT=THREE.RepeatWrapping; t.colorSpace=THREE.SRGBColorSpace; t.anisotropy=MAXA;
  return t;
}

function sZ(z){ // сужение по длине
  let s=1;
  if(z>3){ const t=Math.min((z-3)/10.2,1); s*=1-0.86*Math.pow(t,1.6); }
  if(z>11.8) s*=1-0.6*Math.min((z-11.8)/1.5,1);
  if(z<-9.5){ const t=Math.min((-9.5-z)/3.5,1); s*=1-0.25*t*t; }
  return s;
}
function taper(z,y){ return sZ(z)*(0.42+0.58*clamp((y+3.4)/2.6,0,1)); }
function hwAt(z,y){ return 3.5*taper(z,y); }
function deckY(z){

    + 1.00*Math.pow(clamp((z-3)/10,0,1),1.7);
}

function deckGeom(z0,z1,inset,yOff,steps=42){
  const s=new THREE.Shape();
  const hw=z=>Math.max(3.5*sZ(z)-inset,0.05);
  s.moveTo(hw(z0),z0);
  for(let k=1;k<=steps;k++){ const z=z0+(z1-z0)*k/steps; s.lineTo(hw(z),z); }
  for(let k=steps;k>=0;k--){ const z=z0+(z1-z0)*k/steps; s.lineTo(-hw(z),z); }
  s.closePath();
  const g=new THREE.ShapeGeometry(s);
  const p=g.attributes.position;
  for(let i=0;i<p.count;i++){
    const x=p.getX(i), z=p.getY(i);
    p.setXYZ(i, x*0.99, deckY(z)+yOff-0.03, z);
  }
  g.computeVertexNormals();
  return g;
}

function wallStrip(z0,z1,xFn,ybFn,ytFn,steps,mat){
  const g=new THREE.BufferGeometry();
  const pos=[],uv=[],idx=[];
  for(let k=0;k<=steps;k++){
    const z=z0+(z1-z0)*k/steps, x=xFn(z), yb=ybFn(z), yt=ytFn(z), u=(z-z0)*0.5;
    pos.push(x,yb,z, x,yt,z);
    uv.push(0,yb*0.4, u,yt*0.4);
  }
  for(let k=0;k<steps;k++){ const a=k*2; idx.push(a,a+2,a+1, a+1,a+2,a+3); }
  g.setAttribute('position',new THREE.Float32BufferAttribute(pos,3));
  g.setAttribute('uv',new THREE.Float32BufferAttribute(uv,2));
  g.setIndex(idx); g.computeVertexNormals();
  const m=new THREE.Mesh(g,mat); m.castShadow=m.receiveShadow=true; return m;
}

function triSail(A,B,C,seg,belly){
  const g=new THREE.BufferGeometry();
  const pos=[],uv=[],idx=[], off=[];
  let acc=0;
  for(let i=0;i<=seg;i++){ off[i]=acc; acc+=seg-i+1; }
  for(let i=0;i<=seg;i++) for(let j=0;j<=seg-i;j++){
    const u=i/seg, v=j/seg, w=1-u-v;
    const x=A.x*w+B.x*u+C.x*v + belly*Math.sin(Math.PI*u)*Math.sin(Math.PI*v);
    const y=A.y*w+B.y*u+C.y*v;
    const z=A.z*w+B.z*u+C.z*v;
    pos.push(x,y,z); uv.push(u,v);
  }
  for(let i=0;i<seg;i++) for(let j=0;j<seg-i;j++){
    const a=off[i]+j, b=off[i+1]+j, c=off[i]+j+1;
    idx.push(a,b,c);
    if(j<seg-i-1){ const d=off[i+1]+j+1; idx.push(b,d,c); }
  }
  g.setAttribute('position',new THREE.Float32BufferAttribute(pos,3));
  g.setAttribute('uv',new THREE.Float32BufferAttribute(uv,2));
  g.setIndex(idx); g.computeVertexNormals(); return g;
}

    varying vec3 vDir;
    uniform float uTime; uniform vec3 uZenith,uHorizon,uSunCol,uGround,uSunDir;

    varying vec3 vWorld; varying vec3 vNrm; varying float vH;

    f.mesh.geometry.computeVertexNormals();

  wakeTex.offset.y=(tSim*0.04)%1;
  // чайки
  for(const b of birds){
    b.ang+=dt*b.sp;
    b.g.position.set(12+Math.cos(b.ang)*b.r, b.y+Math.sin(tSim*0.5+b.phase)*0.8, -25+Math.sin(b.ang)*b.r);
    b.g.rotation.y=-b.ang;
    const fl=Math.sin(tSim*7+b.phase)*0.6;
    b.wL.rotation.z=fl; b.wR.rotation.z=-fl;
  }
  controls.update();
  renderer.render(scene,camera);
}
