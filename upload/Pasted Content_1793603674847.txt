const G = 9.81;
const WAVES = [
  {dx: 1.00, dz: 0.25, len:118.0, st:0.180, phys:1, phase:0.0},
  {dx:-0.72, dz: 1.00, len: 64.0, st:0.150, phys:1, phase:1.7},
  {dx: 0.35, dz:-0.94, len: 38.0, st:0.120, phys:1, phase:4.1},
  {dx: 0.90, dz:-0.40, len: 15.0, st:0.110, phys:0, phase:0.5},
  {dx:-0.30, dz:-0.95, len:  8.5, st:0.090, phys:0, phase:2.6},
  {dx: 0.60, dz: 0.75, len:  4.8, st:0.070, phys:0, phase:5.2},
];
const NW = WAVES.length;
const PHYS = [];
WAVES.forEach(w=>{
  w.d = new THREE.Vector2(w.dx,w.dz).normalize();
  w.k = Math.PI*2/w.len;
  w.A0 = w.st/w.k;
  w.A  = w.A0;
  w.c  = Math.sqrt(G/w.k);
  if(w.phys) PHYS.push(w);
});
const WAVE_A = WAVES.map(w=>new THREE.Vector4(w.d.x,w.d.y,w.k,w.A));
const WAVE_B = WAVES.map(w=>new THREE.Vector4(w.c,w.A,w.phase,w.phys?0:1));
const windU = {value:SETTINGS.wind};

function applyWind(){
  const wind = SETTINGS.wind;
  WAVES.forEach((w,i)=>{
    const calm  = w.phys ? 0.10 : 0.04;
    const scale = calm + (1-calm)*Math.pow(wind, w.phys ? 1.25 : 1.7);
    w.A = w.A0*scale;
    WAVE_A[i].w = w.A;
    WAVE_B[i].y = w.A;
  });
  windU.value = wind;
}

const _v3 = new THREE.Vector3();
function evalDisp(x,z,t,list,out){
  let dX=0,dZ=0,h=0;
  for(let i=0;i<list.length;i++){
    const w=list[i], f=w.k*(w.d.x*x+w.d.y*z)-w.k*w.c*t+w.phase;
    const s=Math.sin(f),c=Math.cos(f);
    dX+=w.d.x*w.A*c; dZ+=w.d.y*w.A*c; h+=w.A*s;
  }
  return out.set(dX,h,dZ);
}
const _d = new THREE.Vector3();
function waterHeightAt(x,z,t){
  let px=x, pz=z;
  for(let i=0;i<3;i++){ evalDisp(px,pz,t,PHYS,_d); px=x-_d.x; pz=z-_d.z; }
  return evalDisp(px,pz,t,PHYS,_d).y;
}

const oceanVert = `
uniform float uTime;
uniform vec2  uOffset;
uniform vec4  uWaveA[${NW}];
uniform vec4  uWaveB[${NW}];
varying vec3  vW; varying vec3 vN; varying float vCrest;
#ifdef HAS_MICRO
float microFade(vec2 p){ return smoothstep(118.0, 82.0, length(p)); }
#endif
void main(){
  vec2 xz = position.xz + uOffset;
  vec2 d2 = vec2(0.0);
  float h=0.0, nx=0.0, nz=0.0, crest=0.0;
  for(int i=0;i<${NW};i++){
    vec2  dir = uWaveA[i].xy;
    float k   = uWaveA[i].z;
    float A   = uWaveA[i].w;
    float c   = uWaveB[i].x;
    float ph  = uWaveB[i].z;
    #ifdef HAS_MICRO
      A *= mix(1.0, microFade(position.xz), uWaveB[i].w);
    #endif
    float f  = k*dot(dir,xz) - k*c*uTime + ph;
    float s  = sin(f), co = cos(f);
    d2   += dir*(A*co);
    h    += A*s;
    nx   -= dir.x*k*A*co;
    nz   -= dir.y*k*A*co;
    crest+= k*A*s;
  }
  vec3 wp = vec3(xz.x+d2.x, h, xz.y+d2.y);
  vW = wp;
  vN = normalize(vec3(nx, 1.0-crest, nz));
  vCrest = crest;
  gl_Position = projectionMatrix * viewMatrix * vec4(wp,1.0);
}`;

const oceanFrag = `
uniform vec3  uSunDir, uDeep, uShallow, uSkyCol, uFogColor;
uniform float uFogDensity, uTime, uWind;
uniform vec4  uShip0;
uniform vec4  uShip1;
varying vec3  vW; varying vec3 vN; varying float vCrest;
float hash(vec2 p){ return fract(sin(dot(p, vec2(127.1,311.7)))*43758.5453); }
float noise(vec2 p){
  vec2 i=floor(p), f=fract(p); f=f*f*(3.0-2.0*f);
  return mix(mix(hash(i),hash(i+vec2(1,0)),f.x),
             mix(hash(i+vec2(0,1)),hash(i+vec2(1,1)),f.x), f.y);
}
void main(){
  vec3 N = normalize(vN);
  vec2 np = vW.xz*0.55 + uTime*vec2(0.35,-0.22);
  N.xz += (vec2(noise(np), noise(np+37.0))-0.5)*(0.08+0.14*uWind);
  N = normalize(N);
  vec3 V = normalize(cameraPosition - vW);
  float fres = pow(1.0-max(dot(N,V),0.0), 3.0);
  float hgt  = clamp(vW.y*0.10+0.45, 0.0, 1.0);
  vec3 col = mix(uDeep, uShallow, hgt*0.7 + vCrest*0.22);
  col = mix(col, uSkyCol, fres*0.75);
  vec3 H = normalize(V + uSunDir);
  col += pow(max(dot(N,H),0.0), 240.0)*1.2*vec3(1.0,0.93,0.82);
  col *= 0.72 + 0.38*max(dot(N,uSunDir),0.0);

  float n1 = noise(vW.xz*0.22 + uTime*0.35);
  float wf = smoothstep(0.50, 0.78, vCrest + (n1-0.5)*0.28);
  wf *= smoothstep(0.12, 0.55, uWind);

  vec2 rel = vW.xz - uShip0.xy;
  vec2 f = uShip0.zw; vec2 r = vec2(f.y,-f.x);
  float along = dot(rel,f), lat = dot(rel,r);
  float halfL = uShip1.z*0.55, beam = uShip1.y;
  float hull = (1.0-smoothstep(beam*0.85, beam*0.85+2.4, abs(lat)))
             *  smoothstep(halfL+3.5, halfL*0.55, abs(along));
  float behind = max(-along - halfL*0.35, 0.0);
  float spread = beam*0.9 + behind*0.22;
  float wake = (1.0-smoothstep(spread*0.5, spread, abs(lat)))
             * smoothstep(halfL+2.0, halfL*0.1, along)
             * exp(-behind*0.02)
             * (0.25 + 0.75*clamp(uShip1.x,0.0,1.0));
  float bowPush = (1.0-smoothstep(1.6,5.5,abs(lat)))
                * smoothstep(4.0,0.0,abs(along-halfL))
                * clamp(uShip1.x*1.7,0.0,1.0);
  float foam = clamp(wf + hull*0.9 + wake*0.95 + bowPush*0.85, 0.0, 1.0);
  float ft = noise(vW.xz*1.7 - uTime*1.2)*0.5 + noise(vW.xz*4.0 + uTime*0.8)*0.5;
  foam *= 0.5 + 0.65*ft;
  vec3 foamCol = vec3(0.82,0.90,0.92)*(0.6+0.5*max(dot(N,uSunDir),0.0));
  col = mix(col, foamCol, clamp(foam,0.0,1.0));

  float fogF = 1.0 - exp(-pow(distance(cameraPosition,vW)*uFogDensity, 2.0));
  col = mix(col, uFogColor, fogF);
  gl_FragColor = vec4(col, 1.0);
}`;

function profBeam(s){
  if(s>=0) return HULL.beam*Math.pow(1-Math.pow(s,1.7),0.78);
  const u=-s; return HULL.beam*Math.pow(1-Math.pow(u,2.6),0.62)*(1-0.35*u);
}
function deckYat(s){ return HULL.deckMid + 3.0*MU.smoothstep(s,0.35,1) + 2.0*MU.smoothstep(-s,0.40,1); }
function keelYat(s){
  if(s>=0) return -(2.6*(1-0.72*Math.pow(s,1.8)));
  const u=-s; return -(2.6*(1-0.25*u*u) + 0.9*MU.smoothstep(u,0.55,1));
}

function buildHull(){
  const ST=30, R=8, cols=2*R+1;
  const pos=[], uv=[], idx=[];
  for(let i=0;i<ST;i++){
    const s=i/(ST-1);
    for(let j=0;j<cols;j++){
      const u=Math.abs(j-R)/R;
      const side=(j-R)/R;
      const w=profBeam(s)*Math.pow(Math.sin(Math.pow(u,0.72)*Math.PI/2),0.9)*(1-0.05*u);
      pos.push(side*w, MU.lerp(keelYat(s),deckYat(s),u), s*HULL.length/2);
      uv.push(s*3, j/(cols-1)*1.6);
    }

function control(dt){
  if(keys.KeyW||keys.ArrowUp)   ctl.tTarget=Math.min(1, ctl.tTarget+dt*0.45);
  if(keys.KeyS||keys.ArrowDown) ctl.tTarget=Math.max(-0.35, ctl.tTarget-dt*0.45);
  ctl.rIn = ((keys.KeyD||keys.ArrowRight)?1:0) - ((keys.KeyA||keys.ArrowLeft)?1:0);
  ctl.throttle += (ctl.tTarget-ctl.throttle)*Math.min(1,dt*1.1);
  ctl.rudder   += (ctl.rIn-ctl.rudder)*Math.min(1,dt*2.6);

  const MAX=maxSpeed();
  const tSpeed = ctl.throttle>=0 ? ctl.throttle*MAX : ctl.throttle/0.35*ctl.REV;
  const rate   = tSpeed>ctl.speed ? 2.1 : 1.1;
  ctl.speed   += MU.clamp(tSpeed-ctl.speed, -dt*rate, dt*rate);
  ctl.x       += Math.sin(ctl.yaw)*ctl.speed*dt;
  ctl.z       += Math.cos(ctl.yaw)*ctl.speed*dt;
  const eff = ctl.rudder*(0.35+0.65*Math.min(Math.abs(ctl.speed)/MAX,1))*(ctl.speed<-0.1?-1:1);
  ctl.yaw   += eff*0.5*dt;
  rudderPivot.rotation.y = ctl.rudder*0.5;
  ship.userData.wheel.rotation.z = -ctl.rudder*0.9;
}

const buoy = {
  y:0, vy:0, prevBowH:0, floodCd:[0,0], slamCd:0, floodPulse:0,
  hBow:0,hStern:0,hPort:0,hStar:0,
  _f:new THREE.Vector3(), _v1:new THREE.Vector3(), _v2:new THREE.Vector3(),
  _u:new THREE.Vector3(), _r:new THREE.Vector3(), _m:new THREE.Matrix4(),
  _q:new THREE.Quaternion()
};
function buoyancy(dt,t){
  const L=HULL.length, B=HULL.beam;
  const sy=Math.sin(ctl.yaw), cy=Math.cos(ctl.yaw);
  const f0x=sy, f0z=cy, r0x=cy, r0z=-sy;
  const hBow  = waterHeightAt(ctl.x+f0x*L*0.42, ctl.z+f0z*L*0.42, t);
  const hStern= waterHeightAt(ctl.x-f0x*L*0.42, ctl.z-f0z*L*0.42, t);
  const hPort = waterHeightAt(ctl.x-r0x*B*0.78, ctl.z-r0z*B*0.78, t);
  const hStar = waterHeightAt(ctl.x+r0x*B*0.78, ctl.z+r0z*B*0.78, t);
  const hMid  = waterHeightAt(ctl.x, ctl.z, t);
  Object.assign(buoy,{hBow,hStern,hPort,hStar});

  const slopeZ=(hBow-hStern)/(L*0.84);
  buoy._f.set(f0x,slopeZ,f0z).normalize();
  buoy._v1.set(f0x*L*0.84, hBow-hStern, f0z*L*0.84);
  buoy._v2.set(r0x*B*1.56, hStar-hPort, r0z*B*1.56);
  buoy._u.copy(buoy._v1).cross(buoy._v2).normalize();
  buoy._r.copy(buoy._u).cross(buoy._f).normalize();
  buoy._m.makeBasis(buoy._r, buoy._u, buoy._f);
  buoy._q.setFromRotationMatrix(buoy._m);

  const sink = -(1-buoy._u.y)*3.2 - Math.min(Math.abs(ctl.speed)*0.02,0.2);
  const targetY = hMid*0.45+(hBow+hStern+hPort+hStar)*0.1375 + sink + 1.8;
  buoy.vy += ((targetY-buoy.y)*5.0 - buoy.vy*2.4)*dt;
  buoy.y  += buoy.vy*dt;
  ship.position.set(ctl.x, buoy.y, ctl.z);
  ship.quaternion.slerp(buoy._q, 1-Math.exp(-2.6*dt));

  const wvy=(hBow-buoy.prevBowH)/Math.max(dt,1e-4);
  const imm=hBow-(buoy.y-1.15);
  buoy.slamCd-=dt;
  if(imm>0.12 && Math.abs(ctl.speed)>2){
    const relVy=buoy.vy-wvy;
    if(imm>0.35 && relVy<-1.6 && buoy.slamCd<=0){
      spray.bowBurst(1+Math.min(Math.abs(relVy)*22,70));
      buoy.slamCd=0.25;
    } else sprayAcc += dt*(8+Math.abs(ctl.speed)*imm*9);
  }
  buoy.prevBowH=hBow;

  buoy.floodCd[0]-=dt; buoy.floodCd[1]-=dt;
  const deckEdge=buoy.y+2.0;
  if(hPort>deckEdge && buoy.floodCd[0]<=0){ spray.flood(-1); buoy.floodCd[0]=1.4; buoy.floodPulse=1; warnShow(); }
  if(hStar>deckEdge && buoy.floodCd[1]<=0){ spray.flood( 1); buoy.floodCd[1]=1.4; buoy.floodPulse=1; warnShow(); }
  buoy.floodPulse=Math.max(0,buoy.floodPulse-dt*1.4);

  if(Math.abs(ctl.speed)>1.2) wakeAcc += dt*(6+Math.abs(ctl.speed)*2.2);
}

class Spray{
  constructor(n){
    this.n=n; this.i=0;
    this.pos=new Float32Array(n*3); this.vel=new Float32Array(n*3);
    this.life=new Float32Array(n); this.max=new Float32Array(n);
    this.size=new Float32Array(n); this.type=new Float32Array(n);
    const aLife=new Float32Array(n), aSize=new Float32Array(n), aType=new Float32Array(n);
    for(let k=0;k<n;k++) this.pos[k*3+1]=-999;
    const g=new THREE.BufferGeometry();
    g.setAttribute('position',new THREE.BufferAttribute(this.pos,3).setUsage(THREE.DynamicDrawUsage));
    g.setAttribute('aLife',new THREE.BufferAttribute(aLife,1).setUsage(THREE.DynamicDrawUsage));
    g.setAttribute('aSize',new THREE.BufferAttribute(aSize,1).setUsage(THREE.DynamicDrawUsage));
    g.setAttribute('aType',new THREE.BufferAttribute(aType,1));
    this.aLife=aLife; this.aSize=aSize; this.aType=aType;
    const mat=new THREE.ShaderMaterial({
      transparent:true, depthWrite:false,
      vertexShader:`attribute float aLife,aSize,aType;
        varying float vL,vT;
        void main(){ vL=aLife; vT=aType;
          vec4 mv=modelViewMatrix*vec4(position,1.0);
          gl_PointSize=aSize*(240.0/-mv.z);
          gl_Position=projectionMatrix*mv; }`,
      fragmentShader:`varying float vL,vT;
        void main(){
          float d=length(gl_PointCoord-0.5);
          float a=smoothstep(0.5,0.12,d)*min(vL*2.2,1.0)*0.9;
          if(a<0.01) discard;
          vec3 col = vT<0.5 ? vec3(0.93,0.97,0.98)
                   : vT<1.5 ? vec3(0.80,0.91,0.88)
                   : vec3(0.16,0.40,0.32);
          gl_FragColor=vec4(col,a); }`
    });
    this.points=new THREE.Points(g,mat);
    this.points.frustumCulled=false; this.points.renderOrder=5;
    scene.add(this.points);
    this._p=new THREE.Vector3(); this._v=new THREE.Vector3();
  }
  spawn(px,py,pz,vx,vy,vz,life,size,type){
    const i=this.i; this.i=(this.i+1)%this.n;
    this.pos[i*3]=px; this.pos[i*3+1]=py; this.pos[i*3+2]=pz;
    this.vel[i*3]=vx; this.vel[i*3+1]=vy; this.vel[i*3+2]=vz;
    this.life[i]=life; this.max[i]=life; this.size[i]=size; this.type[i]=type;
  }
  bowBurst(count){
    for(let k=0;k<count;k++){
      const side=Math.random()<0.5?-1:1;
      this._p.set(side*(0.2+Math.random()*1.1), -0.4, HULL.length*0.46);
      this._p.applyQuaternion(ship.quaternion).add(ship.position);
      this._p.y=buoy.hBow+0.1;
      this._v.set(side*(1+Math.random()*3), 3.5+Math.random()*4.5, 1+Math.random()*2.5+ctl.speed*0.35);
      this._v.applyQuaternion(ship.quaternion);
      this.spawn(this._p.x,this._p.y,this._p.z,this._v.x,this._v.y,this._v.z,
        0.7+Math.random()*0.7, 1.4+Math.random()*2.2, 0);
    }
  }
  flood(side){
    for(let k=0;k<150;k++){
      const z=-HULL.length*0.35+Math.random()*HULL.length*0.75;
      this._p.set(side*HULL.beam*0.86, 2.0, z);
      this._p.applyQuaternion(ship.quaternion).add(ship.position);
      this._v.set(-side*(2+Math.random()*2.5), 1+Math.random()*2.5, ctl.speed*0.5+(Math.random()-0.5));
      this._v.applyQuaternion(ship.quaternion);
      this.spawn(this._p.x,this._p.y,this._p.z,this._v.x,this._v.y,this._v.z,
        1.1+Math.random()*0.9, 1.8+Math.random()*2.4, 2);
    }
  }
  wake(){
    const side=Math.random()<0.5?-1:1;
    this._p.set(side*(0.8+Math.random()*2.6), -0.5, -HULL.length*0.5);
    this._p.applyQuaternion(ship.quaternion).add(ship.position);
    this._p.y=buoy.hStern+0.05;
    this._v.set(side*0.6, 0.25, -2-Math.random()*2);
    this._v.applyQuaternion(ship.quaternion);
    this.spawn(this._p.x,this._p.y,this._p.z,this._v.x,this._v.y,this._v.z,
      2.2+Math.random()*2.2, 1.6+Math.random()*1.6, 1);
  }
  sideFoam(){
    const side=Math.random()<0.5?-1:1;
    const z=(Math.random()*2-1)*HULL.length*0.35;
    this._p.set(side*profBeam(z/(HULL.length/2)), -0.3, z);
    this._p.applyQuaternion(ship.quaternion).add(ship.position);
    this._p.y=waterHeightAt(this._p.x,this._p.z,clockT)+0.05;
    this._v.set(side*0.8,0.2,-1); this._v.applyQuaternion(ship.quaternion);
    this.spawn(this._p.x,this._p.y,this._p.z,this._v.x,this._v.y,this._v.z,
      1.4+Math.random()*1.2, 1.2+Math.random()*1.2, 1);
  }
  update(dt){
    for(let i=0;i<this.n;i++){
      if(this.life[i]<=0){ this.aLife[i]=0; continue; }
      this.life[i]-=dt;
      const t=this.type[i];
      this.vel[i*3+1]-=(t===1?1.6:8.5)*dt;
      const damp=1-(t===1?2.6:0.7)*dt;
      this.vel[i*3]*=damp; this.vel[i*3+1]*=damp; this.vel[i*3+2]*=damp;
      this.pos[i*3]+=this.vel[i*3]*dt;
      this.pos[i*3+1]+=this.vel[i*3+1]*dt;
      this.pos[i*3+2]+=this.vel[i*3+2]*dt;
      if(t===1) this.size[i]+=this.size[i]*0.55*dt;
      this.aLife[i]=Math.max(this.life[i]/this.max[i],0);
      this.aSize[i]=this.size[i]; this.aType[i]=t;
    }
    this.points.geometry.attributes.position.needsUpdate=true;
    this.points.geometry.attributes.aLife.needsUpdate=true;
    this.points.geometry.attributes.aSize.needsUpdate=true;
    this.points.geometry.attributes.aType.needsUpdate=true;
  }
}

function windEff(){
  return 0.12 + 0.62*SETTINGS.wind + 0.35*Math.min(Math.abs(ctl.speed)/maxSpeed(),1);
}
function animateCloth(t){
  const wind = windEff();
  for(const s of sqSails){
    const u=s.userData, pa=s.geometry.attributes.position, arr=pa.array, flat=u.flat;
    for(let i=0;i<pa.count;i++){
      const x=flat[i*3], y=flat[i*3+1];
      const uu=x/u.w+0.5, vv=y/u.h+0.5;
      const sinVal = Math.sin(Math.PI*uu);
      const bulge=(0.45+2.1*wind)*Math.pow(Math.max(0,sinVal),1.1)*(0.35+0.65*(1-vv));
      const fl=Math.sin(t*5.5+u.ph+uu*6+vv*3)*0.08*(1+wind*1.6)*(0.25+0.75*(1-vv));
      arr[i*3]=x; arr[i*3+1]=y; arr[i*3+2]=bulge+fl;
    }
    pa.needsUpdate=true;
  }
  for(const s of jibSails){
    const u=s.userData, pa=s.geometry.attributes.position, arr=pa.array, flat=u.flat;
    for(let i=0;i<pa.count;i++){
      arr[i*3]=flat[i*3]+Math.sin(t*6+u.ph+i)*0.06*(1+wind);
      arr[i*3+1]=flat[i*3+1]; arr[i*3+2]=flat[i*3+2];
    }
    pa.needsUpdate=true;
  }
  for(const f of flags){
    const u=f.userData, pa=f.geometry.attributes.position, arr=pa.array, flat=u.flat;
    const amp=0.04+0.06*wind+Math.abs(ctl.speed)*0.012;
    for(let i=0;i<pa.count;i++){
      const x=flat[i*3], ph=(x/u.w+0.5);
      arr[i*3]=x; arr[i*3+1]=flat[i*3+1];
      arr[i*3+2]=Math.sin(t*9+u.ph+ph*7)*amp*ph*2.2;
    }
    pa.needsUpdate=true;
  }
}

  else if(camMode===1){ _cp.copy(ship.position).addScaledVector(_fwd,6).add(new THREE.Vector3(0,10.5,0));
                   _cl.copy(ship.position).addScaledVector(_fwd,60).add(new THREE.Vector3(0,4,0)); }
  else { _cp.copy(ship.position).add(new THREE.Vector3(-34,46,-34));
                   _cl.copy(ship.position).add(new THREE.Vector3(0,3,0)); }

    camera.position.y+=(Math.random()-0.5)*shake;
    shake*=Math.exp(-4*dt);
  }
  camera.lookAt(camLook);
}
