class Pool {
  constructor(cap, additive, fade) {
    this.cap = cap; this.fade = fade; this.cur = 0;
    this.pos = new Float32Array(cap * 3); this.vel = new Float32Array(cap * 3);
    this.life = new Float32Array(cap); this.max = new Float32Array(cap);
    this.sz = new Float32Array(cap); this.gr = new Float32Array(cap);
    this.a0 = new Float32Array(cap); this.gv = new Float32Array(cap); this.dr = new Float32Array(cap);
    const col = new Float32Array(cap * 3), al = new Float32Array(cap), si = new Float32Array(cap);
    const g = new THREE.BufferGeometry();
    g.setAttribute('position', new THREE.BufferAttribute(this.pos, 3));
    g.setAttribute('aColor', new THREE.BufferAttribute(col, 3));
    g.setAttribute('aAlpha', new THREE.BufferAttribute(al, 1));
    g.setAttribute('aSize', new THREE.BufferAttribute(si, 1));
    this.geo = g;
    this.points = new THREE.Points(g, new THREE.ShaderMaterial({
      transparent: true, depthWrite: false,
      blending: additive ? THREE.AdditiveBlending : THREE.NormalBlending,
      vertexShader: 'attribute float aSize,aAlpha; attribute vec3 aColor; varying float vA; varying vec3 vC; void main(){ vA=aAlpha; vC=aColor; vec4 mv=modelViewMatrix*vec4(position,1.0); gl_PointSize=clamp(aSize*(300.0/-mv.z),1.0,180.0); gl_Position=projectionMatrix*mv; }',
      fragmentShader: 'varying float vA; varying vec3 vC; void main(){ float d=length(gl_PointCoord-0.5); float a=smoothstep(0.5,0.06,d)*vA; if(a<0.004) discard; gl_FragColor=vec4(vC,a); }'
    }));
    this.points.frustumCulled = false;
    this.points.renderOrder = additive ? 10 : 9;
    scene.add(this.points);
  }
  spawn(p, v, l, s, g, c, gv, dr, a) {
    const i = this.cur++ % this.cap, k = i * 3;
    this.pos[k] = p.x; this.pos[k + 1] = p.y; this.pos[k + 2] = p.z;
    this.vel[k] = v.x; this.vel[k + 1] = v.y; this.vel[k + 2] = v.z;
    this.life[i] = l; this.max[i] = l; this.sz[i] = s; this.gr[i] = g;
    this.a0[i] = a; this.gv[i] = gv; this.dr[i] = dr;
    const col = this.geo.attributes.aColor.array;
    col[k] = c.r; col[k + 1] = c.g; col[k + 2] = c.b;
  }
  update(dt) {
    const P = this.pos, V = this.vel;
    const al = this.geo.attributes.aAlpha.array, si = this.geo.attributes.aSize.array;
    for (let i = 0; i < this.cap; i++) {
      if (this.life[i] <= 0) { al[i] = 0; si[i] = 0; continue; }
      this.life[i] -= dt;
      const k = i * 3, f = 1 - Math.max(this.life[i], 0) / this.max[i];
      V[k + 1] += this.gv[i] * dt;
      const d = Math.pow(this.dr[i], dt * 60);
      V[k] *= d; V[k + 1] *= d; V[k + 2] *= d;
      P[k] += V[k] * dt; P[k + 1] += V[k + 1] * dt; P[k + 2] += V[k + 2] * dt;
      si[i] = this.sz[i] + this.gr[i] * f;
      al[i] = this.fade === 'smoke' ? this.a0[i] * Math.min(f * 6, 1) * (1 - f) : this.a0[i] * (1 - f);
    }
    this.geo.attributes.position.needsUpdate = true;
    this.geo.attributes.aAlpha.needsUpdate = true;
    this.geo.attributes.aSize.needsUpdate = true;
    this.geo.attributes.aColor.needsUpdate = true;
  }
}

function addDebris(obj, hx, hy, hz, mass, v, av, o = {}) {
  const b = new CANNON.Body({ mass, shape: new CANNON.Box(new CANNON.Vec3(Math.max(hx, 0.03), Math.max(hy, 0.03), Math.max(hz, 0.03))) });
  b.position.copy(obj.position); b.quaternion.copy(obj.quaternion);
  b.velocity.set(v.x, v.y, v.z); b.angularVelocity.set(av.x, av.y, av.z);
  b.linearDamping = o.linD !== undefined ? o.linD : 0.04;
  b.angularDamping = o.angD !== undefined ? o.angD : 0.1;
  world.addBody(b);
  const rec = {
    mesh: obj, body: b, halfH: hy,
    floatF: o.floatF !== undefined ? o.floatF : 1.3,
    type: o.type || 'wood', ttl: o.ttl !== undefined ? o.ttl : 30,
    born: simTime, phase: rand(0, 6), wet: false,
    s0: obj.scale.clone(), len: o.len || 0, fading: 0
  };
  debris.push(rec);
  while (debris.length > 470) removeDebris(debris[0]);
  return rec;
}
function removeDebris(r) {
  const i = debris.indexOf(r);
  if (i < 0) return;
  debris.splice(i, 1);
  scene.remove(r.mesh);
  world.removeBody(r.body);
}
function updateDebris(dt) {
  for (const r of debris.slice()) {
    const b = r.body;
    const surf = waveHeight(b.position.x, b.position.z, simTime);
    const sub = clamp((surf - (b.position.y - r.halfH)) / (2 * r.halfH), 0, 1);
    if (sub > 0) {
      if (!r.wet) {
        r.wet = true;
        if (b.velocity.length() > 2.5) splashAt(V3(b.position.x, 0, b.position.z), clamp(r.halfH * 2, 0.35, 1.4));
      }
      b.wakeUp();
      b.force.y += b.mass * 9.82 * sub * r.floatF;
      b.force.y += b.mass * 1.3 * Math.sin(simTime * 1.3 + r.phase) * sub;
      b.force.x += b.mass * 0.16 * S.wind * sub;
      const dk = r.type === 'cloth' ? 2.4 : 1;
      b.velocity.x *= 1 - Math.min(0.92, 0.11 * sub * dk);
      b.velocity.z *= 1 - Math.min(0.92, 0.11 * sub * dk);
      b.velocity.y *= 1 - Math.min(0.92, 0.14 * sub);
      b.angularVelocity.scale(1 - 0.07 * sub, b.angularVelocity);
      if (r.type === 'mast' && !r.broken) { r.broken = true; schedule(1, () => breakMastInWater(r)); }
    } else if (r.type === 'cloth' && b.velocity.y < -1.1) {
      b.force.y += b.mass * 9.82 * 0.58;
    }
  }
  world.step(1 / 60, dt, 3);
  for (const r of debris.slice()) {
    const b = r.body, m = r.mesh;
    m.position.copy(b.position); m.quaternion.copy(b.quaternion);
    if (r.type === 'mast') m.position.add(V3(0, -r.len / 2, 0).applyQuaternion(m.quaternion));
    const age = simTime - r.born;
    const kill = age > r.ttl || b.position.y < -39 || (r.type === 'cloth' && age > 22);
    if (kill && !r.fading) r.fading = simTime;
    if (r.fading) {
      const k = 1 - (simTime - r.fading) / 1.7;
      if (k <= 0) removeDebris(r);
      else m.scale.copy(r.s0).multiplyScalar(Math.max(k, 0.001));
    }
  }
}

const uOf = (x) => (x + 16) / 32;
function halfWidth(x) {
  const u = uOf(x);
  if (u < 0.45) return 2.55 + 1.65 * Math.sin((u / 0.45) * Math.PI / 2);
  return Math.max(0.24, 4.2 * Math.pow(Math.cos(((u - 0.45) / 0.55) * Math.PI / 2), 0.8));
}
function keelY(x) {
  const u = uOf(x);
  let y = -2.5;
  if (u < 0.15) y += 1.7 * Math.pow(1 - u / 0.15, 1.6);
  if (u > 0.8) y += 3.7 * Math.pow((u - 0.8) / 0.2, 1.8);
  return y;
}
function railY(x) {
  const u = uOf(x);
  return 3.25 + 2.05 * Math.pow(clamp((0.34 - u) / 0.34, 0, 1), 1.35) + 1.8 * Math.pow(clamp((u - 0.6) / 0.4, 0, 1), 1.5);
}
function sect(t) {
  const s = Math.pow(Math.sin(Math.min(t / 0.78, 1) * Math.PI / 2), 0.62);
  return t > 0.78 ? s * (1 - 0.14 * (t - 0.78) / 0.22) : s;
}
const QD = 4.34, POOP = 6.06, FC = 3.72;
function deckY(x) {
  const u = uOf(x);
  return 2.2 + 1.05 * Math.pow(clamp((0.28 - u) / 0.28, 0, 1), 1.6) + 0.95 * Math.pow(clamp((u - 0.62) / 0.38, 0, 1), 1.6);
}
const tAt = (x, y) => clamp((y - keelY(x)) / (railY(x) - keelY(x)), 0, 1);
const innerHalf = (x, y) => Math.max(0.3, halfWidth(x) * sect(tAt(x, y)) - 0.16);
const deckHalf = (x) => innerHalf(x, deckY(x));
function hullSurf(side) {
  return (u, v) => {
    const x = -15.2 + u * 30.8;
    return V3(x, keelY(x) + v * (railY(x) - keelY(x)), side * halfWidth(x) * sect(v));
  };
}
const hullOrient = (s) => (p) => V3(0, p.y - (keelY(p.x) + 0.45 * (railY(p.x) - keelY(p.x))), s);

function edgeSamples(n, e) {
  const a = [0, e];
  for (let i = 1; i < n; i++) a.push(e + (1 - 2 * e) * i / n);
  a.push(1 - e, 1);
  return a;
}
function buildPlankData(surf, u0, u1, v0, v1, orient) {
  const us = edgeSamples(3, 0.04), vs = edgeSamples(2, 0.07);
  const nu = us.length - 1, nv = vs.length - 1, rows = vs.length;
  const count = us.length * rows, vc = count * 2, th = 0.14;
  const base = new Float32Array(vc * 3), gn = new Float32Array(count * 3), oc = new Float32Array(count * 3);
  const P = V3(), du = V3(), dv = V3(), N = V3();
  const seed = rand(0, 20);
  for (let i = 0; i < us.length; i++) {
    for (let j = 0; j < rows; j++) {
      const uu = lerp(u0, u1, us[i]), vv = lerp(v0, v1, vs[j]);
      P.copy(surf(uu, vv));
      du.subVectors(surf(Math.min(uu + 0.002, 1), vv), surf(Math.max(uu - 0.002, 0), vv));
      dv.subVectors(surf(uu, Math.min(vv + 0.002, 1)), surf(uu, Math.max(vv - 0.002, 0)));
      N.crossVectors(du, dv).normalize();
      if (N.dot(orient(P)) < 0) N.negate();
      const k = i * rows + j, k3 = k * 3, b3 = (k + count) * 3;
      gn[k3] = N.x; gn[k3 + 1] = N.y; gn[k3 + 2] = N.z;
      base[k3] = P.x + N.x * th / 2; base[k3 + 1] = P.y + N.y * th / 2; base[k3 + 2] = P.z + N.z * th / 2;
      base[b3] = P.x - N.x * th / 2; base[b3 + 1] = P.y - N.y * th / 2; base[b3 + 2] = P.z - N.z * th / 2;
      let sh = (0.88 + 0.22 * hash(seed + i * 3.1 + j * 7.7)) * (0.95 + 0.05 * Math.sin(uu * 160 + seed * 9));
      if (i === 0 || i === us.length - 1 || j === 0 || j === rows - 1) sh *= 0.57;
      const wet = P.y < 0.05;
      oc[k3] = sh * (wet ? 0.55 : 1); oc[k3 + 1] = sh * (wet ? 0.63 : 1); oc[k3 + 2] = sh * (wet ? 0.55 : 1);
    }
  }
  const cells = [];
  for (let i = 0; i < nu; i++) {
    for (let j = 0; j < nv; j++) {
      cells.push({
        i, j, alive: true,
        center: surf(lerp(u0, u1, (us[i] + us[i + 1]) / 2), lerp(v0, v1, (vs[j] + vs[j + 1]) / 2)),
        seed: hash(i * 19.7 + j * 7.3 + seed)
      });
    }
  }
  const geo = new THREE.BufferGeometry();
  geo.setAttribute('position', new THREE.BufferAttribute(new Float32Array(vc * 3), 3));
  geo.setAttribute('color', new THREE.BufferAttribute(new Float32Array(vc * 3), 3));
  disposables.push(geo);
  return {
    geo, base, gn, oc, dent: new Float32Array(count), scorch: new Float32Array(count),
    jit: new Float32Array(count * 3), fresh: new Uint8Array(count), jdone: new Uint8Array(count),
    us, vs, nu, nv, rows, count, th, cells, total: cells.length, removed: 0
  };
}
function rebuildPlank(rec) {
  const d = rec.patch;
  const pos = d.geo.attributes.position.array, col = d.geo.attributes.color.array;
  for (let k = 0; k < d.count; k++) {
    const k3 = k * 3, b3 = (k + d.count) * 3;
    const dx = d.gn[k3] * d.dent[k] - d.jit[k3];
    const dy = d.gn[k3 + 1] * d.dent[k] - d.jit[k3 + 1];
    const dz = d.gn[k3 + 2] * d.dent[k] - d.jit[k3 + 2];
    pos[k3] = d.base[k3] - dx; pos[k3 + 1] = d.base[k3 + 1] - dy; pos[k3 + 2] = d.base[k3 + 2] - dz;
    pos[b3] = d.base[b3] - dx; pos[b3 + 1] = d.base[b3 + 1] - dy; pos[b3 + 2] = d.base[b3 + 2] - dz;
    let r = d.oc[k3] * (1 - d.scorch[k] * 0.72);
    let g = d.oc[k3 + 1] * (1 - d.scorch[k] * 0.75);
    let b = d.oc[k3 + 2] * (1 - d.scorch[k] * 0.78);
    if (d.fresh[k]) { r = Math.min(1, r * 1.4 + 0.06); g = Math.min(1, g * 1.32 + 0.04); b = Math.min(1, b * 1.2 + 0.02); }
    col[k3] = r; col[k3 + 1] = g; col[k3 + 2] = b;
    col[b3] = r; col[b3 + 1] = g; col[b3 + 2] = b;
  }
  const vi = (i, j) => i * d.rows + j, back = d.count, idx = [];
  const alive = (i, j) => i >= 0 && j >= 0 && i < d.nu && j < d.nv && d.cells[i * d.nv + j].alive;
  const wall = (a, b) => idx.push(a, b, b + back, a, b + back, a + back);
  for (const c of d.cells) {
    if (!c.alive) continue;
    const i = c.i, j = c.j;
    const a = vi(i, j), b = vi(i + 1, j), c1 = vi(i, j + 1), e = vi(i + 1, j + 1);
    idx.push(a, b, c1, b, e, c1, back + a, back + c1, back + b, back + b, back + c1, back + e);
    if (!alive(i - 1, j)) wall(a, c1);
    if (!alive(i + 1, j)) wall(b, e);
    if (!alive(i, j - 1)) wall(a, b);
    if (!alive(i, j + 1)) wall(c1, e);
  }
  d.geo.setIndex(idx);
  d.geo.attributes.position.needsUpdate = true;
  d.geo.attributes.color.needsUpdate = true;
  d.geo.computeVertexNormals();
  d.geo.computeBoundingSphere();
}

function createShip(id, name, z, hullMat, sailMat, crestTex, scrollTex) {
  const ship = {
    id, name, root: new THREE.Group(),
    parts: [], planks: [], cannons: [], masts: [], sails: [], flags: [],
    attachments: [], seamChecks: [], rayTargets: [],
    integrity: 1, totalWeight: 1, flood: 0, floodVol: 0, sink: 0, trim: 0, list: 0,
    holes: [], shots: 0, hits: 0, dead: false, deathTime: 0,
    hullMat, sailMat, crestTex, scrollTex, baseZ: z, bobPhase: id === 'A' ? 0 : 2.1,
    wl: id === 'A' ? wlA : wlB, gildParts: []
  };
  ship.root.position.set(0, 0, z);
  scene.add(ship.root);

  ship.reg = function (obj, rec) {
    rec.ship = ship; rec.obj = obj; rec.dead = false; rec.detached = false; rec.dependents = [];
    ship.parts.push(rec);
    obj.traverse((ch) => {
      if (ch.isMesh && !ch.userData.part) {
        ch.userData.part = rec;
        ship.rayTargets.push(ch);
        rayTargets.push(ch);
      }
    });
    obj.userData.part = rec;
    return rec;
  };
  ship.addPart = function (obj, nm, kind, hp, support) {
    ship.root.add(obj);
    const rec = ship.reg(obj, { name: nm, kind, hp, maxHp: hp });
    if (support && support.dependents) {
      rec.support = support;
      support.dependents.push(rec);
      ship.attachments.push({ child: rec, support });
    }
    return rec;
  };
  ship.link = function (child, support) {
    if (!support || !support.dependents) return child;
    child.support = support;
    support.dependents.push(child);
    ship.attachments.push({ child, support });
    return child;
  };
  ship.makePlank = function (surf, u0, u1, v0, v1, axis, nm, kind, side) {
    kind = kind || 'plank'; side = side || 0;
    const orient = axis === 'y' ? () => V3(-1, 0, 0)
      : (kind === 'deck' ? () => V3(0, 1, 0) : (axis === 'z' ? () => V3(-1, 0, 0) : hullOrient(side)));
    const patch = buildPlankData(surf, u0, u1, v0, v1, orient);
    const mo = new THREE.Mesh(patch.geo, kind === 'deck' ? M.deck : hullMat);
    mo.castShadow = true; mo.receiveShadow = true;
    ship.root.add(mo);
    const center = surf((u0 + u1) / 2, (v0 + v1) / 2);
    const normal = V3(patch.gn[0], patch.gn[1], patch.gn[2]);
    const p00 = surf(u0, v0), p10 = surf(u1, v0), p01 = surf(u0, v1);
    const area = V3().subVectors(p10, p00).cross(V3().subVectors(p01, p00)).length();
    const rec = ship.reg(mo, {
      name: nm, kind, side, stage: 0, damage: 0,
      surf, u0, u1, v0, v1, axis, center, normal, patch, area, hole: false
    });
    ship.planks.push(rec);
    rebuildPlank(rec);
    return rec;
  };

  buildHull(ship);
  buildDecks(ship);
  buildBulkheads(ship);
  buildFrame(ship);
  buildTrim(ship);
  buildGuns(ship);
  buildDeckFittings(ship);
  buildStairs(ship);
  buildStern(ship);
  buildMasts(ship);
  buildHold(ship);
  buildFloodComps(ship);

  ship.totalWeight = 0;
  for (const p of ship.parts) {
    ship.totalWeight += (p.kind === 'plank' || p.kind === 'deck') ? 1 : (p.kind === 'mast' ? 4 : 0.4);
  }
  return ship;
}

function buildHull(ship) {
  const ST = 8, T0 = 0.02;
  for (const side of [-1, 1]) {
    const surf = hullSurf(side);
    for (let s = 0; s < ST; s++) {
      const v0 = T0 + s / ST * (1 - T0), v1 = T0 + (s + 1) / ST * (1 - T0);
      const off = (s % 2) * 0.5, bounds = [0];
      for (let k = 0; k < 10; k++) {
        const val = (k + off) / 10;
        if (val > 0 && val < 1) bounds.push(val);
      }
      bounds.push(1); bounds.sort((a, b) => a - b);
      let cov = 0;
      for (let i = 0; i < bounds.length - 1; i++) {
        cov += bounds[i + 1] - bounds[i];
        ship.makePlank(surf, bounds[i], bounds[i + 1], v0, v1, 'x',
          'Доска обшивки · ' + (side < 0 ? 'левый' : 'правый') + ' борт', 'plank', side);
      }
      ship.seamChecks.push({ what: 'борт ' + (side < 0 ? 'ЛБ' : 'ПБ') + ' пояс ' + (s + 1), err: Math.abs(cov - 1) });
    }
  }
  const kS = keelY(-15.2), span = 6.9 - kS, rS = railY(-15.2), vR = (rS - kS) / span;
  const tw = (v) => {
    const hz = halfWidth(-15.2) * sect(Math.min(v / vR, 1));
    return v <= vR ? hz : lerp(hz, 2.05, (v - vR) / (1 - vR));
  };
  const ts = (u, v) => V3(-15.2 - v * 1.35, kS + v * span, (u - 0.5) * 2 * tw(v));
  for (let row = 0; row < 8; row++) {
    const v0 = row / 8, v1 = (row + 1) / 8;
    for (let u = 0; u < 2; u++) ship.makePlank(ts, u / 2, (u + 1) / 2, v0, v1, 'z', 'Доска транца', 'plank');
    ship.seamChecks.push({ what: 'транец ряд ' + (row + 1), err: 0 });
  }
  for (const side of [-1, 1]) {
    const qs = (u, v) => {
      const y = kS + v * (rS - kS), vt = (y - kS) / span;
      return V3(lerp(-15.2, -15.2 - vt * 1.35, u), y, side * lerp(halfWidth(-15.2) * sect(v), tw(vt), u));
    };
    for (let s = 0; s < 8; s++) ship.makePlank(qs, 0, 1, s / 8, (s + 1) / 8, 'x', 'Кормовая скула', 'plank', side);
  }
}

function buildDecks(ship) {
  function strips(x0, x1, yFn, hFn, n, segs, nm) {
    for (let s = 0; s < n; s++) {
      const f0 = s / n * 2 - 1, f1 = (s + 1) / n * 2 - 1;
      const off = (s % 2) * 0.5, bounds = [0];
      for (let k = 0; k < segs; k++) {
        const val = (k + off) / segs;
        if (val > 0 && val < 1) bounds.push(val);
      }
      bounds.push(1); bounds.sort((a, b) => a - b);
      const surf = (u, v) => {
        const x = x0 + u * (x1 - x0);
        return V3(x, yFn(x), lerp(f0, f1, v) * hFn(x));
      };
      let cov = 0;
      for (let i = 0; i < bounds.length - 1; i++) {
        cov += bounds[i + 1] - bounds[i];
        ship.makePlank(surf, bounds[i], bounds[i + 1], 0, 1, 'x', nm, 'deck');
      }
      ship.seamChecks.push({ what: nm + ' полоса ' + (s + 1), err: Math.abs(cov - 1) });
    }
  }
  strips(-15.0, 15.4, (x) => deckY(x) + 0.03, deckHalf, 10, 3, 'Главная палуба');
  strips(-15.0, -5.6, () => QD, (x) => innerHalf(x, QD), 9, 2, 'Квартердек');
  strips(-15.0, -10.5, () => POOP, (x) => Math.min(2.02, innerHalf(x, POOP)), 6, 1, 'Ют');
  strips(10.5, 15.3, () => FC, (x) => innerHalf(x, FC), 6, 1, 'Бак');
  strips(-13, 13, () => 0.52, () => 2.86, 6, 2, 'Орлоп-дек');
}

function buildBulkheads(ship) {
  function wall(x, y0, y1, hz, nm, door) {
    for (let i = 0; i < 5; i++) {
      const z0 = -hz + i / 5 * 2 * hz, z1 = -hz + (i + 1) / 5 * 2 * hz;
      const surf = (u, v) => V3(x, lerp(y0, y1, v), lerp(z0, z1, u));
      const rec = ship.makePlank(surf, 0, 1, 0, 1, 'y', nm, 'plank');
      if (door && i === 2) {
        const g = new THREE.Group();
        g.position.set(x - 0.1, (y0 + y1) / 2 - 0.15, 0);
        g.add(mk(G.box, M.dark, 0, 0, 0, 0.09, 1.35, 0.74));
        g.add(mk(G.box, ship.id === 'A' ? M.gold : M.crimson, -0.03, 0.42, 0, 0.04, 0.1, 0.6));
        g.add(mk(G.sph, M.brass, -0.08, 0, 0.24, 0.05, 0.05, 0.05));
        const dr = ship.addPart(g, 'Дверь каюты', 'door', 0.6, rec);
        rec.dependents.push(dr);
        ship.attachments.push({ child: dr, support: rec });
      }
    }
    ship.seamChecks.push({ what: nm, err: 0 });
  }
  wall(-5.6, deckY(-5.6), QD, innerHalf(-5.6, QD), 'Переборка квартердека', true);
  wall(-10.5, QD, POOP, 2.0, 'Переборка юта', true);
  wall(10.5, deckY(10.5), FC, innerHalf(10.5, FC), 'Переборка бака', false);
}

function buildFrame(ship) {
  const add = ship.addPart;
  add(mk(G.box, M.dark, 0, -2.56, 0, 31.6, 0.52, 0.64), 'Киль', 'keel', 6);
  const curve = new THREE.CatmullRomCurve3([V3(15.3, 0.9, 0), V3(16.1, 2.4, 0), V3(16.7, 3.9, 0), V3(17.3, 5.3, 0)]);
  const sg = new THREE.TubeGeometry(curve, 12, 0.23, 7);
  disposables.push(sg);
  const stem = new THREE.Mesh(sg, M.dark); stem.castShadow = true;
  ship.stemRec = add(stem, 'Форштевень', 'stem', 4);
  add(mk(G.box, M.dark, -15.6, 1.5, 0, 0.36, 7.6, 0.42), 'Ахтерштевень', 'stem', 4);
  add(mk(G.box, M.dark, -16.2, -0.6, 0, 0.26, 3.5, 1.15), 'Руль', 'rudder', 3);
  for (let x = -14; x <= 14; x += 2.4) {
    const pts = [], top = tAt(x, railY(x) - 0.15);
    for (let t = top; t >= 0.06; t -= 0.09) pts.push(V3(x, keelY(x) + t * (railY(x) - keelY(x)), -(halfWidth(x) * sect(t) - 0.21)));
    for (let t = 0.06; t <= top; t += 0.09) pts.push(V3(x, keelY(x) + t * (railY(x) - keelY(x)), halfWidth(x) * sect(t) - 0.21));
    const geo = new THREE.TubeGeometry(new THREE.CatmullRomCurve3(pts), 24, 0.09, 6);
    disposables.push(geo);
    const rib = new THREE.Mesh(geo, M.inner); rib.castShadow = true;
    add(rib, 'Шпангоут', 'rib', 1.6);

function supportAt(ship, x, y, z, kind) {
  let best = null, bd = Infinity;
  for (const p of ship.planks) {
    if (p.dead) continue;
    if (kind && p.kind !== kind) continue;
    const d = p.center.distanceToSquared(V3(x, y, z));
    if (d < bd) { bd = d; best = p; }
  }
  return best;
}

function buildStairs(ship) {
  const add = ship.addPart;
  function stairs(x, z, y0, y1, dir, width, nm) {
    const g = new THREE.Group();
    const rise = y1 - y0;
    const steps = Math.max(4, Math.round(rise / 0.26));
    const run = rise * 0.85;
    for (let i = 0; i < steps; i++) {
      const t = (i + 0.5) / steps;
      g.add(mk(G.box, M.light, x + dir * t * run, y0 + t * rise, z, run / steps * 1.25, 0.06, width));
    }
    for (const dz of [-width / 2 - 0.05, width / 2 + 0.05]) {
      const str = mk(G.box, M.dark, x + dir * run / 2, y0 + rise / 2, z + dz, run * 1.06, 0.1, 0.09);
      str.rotation.z = -dir * Math.atan2(rise, run); g.add(str);
      const hr = mk(G.box, M.dark, x + dir * run / 2, y0 + rise / 2 + 0.62, z + dz, run * 1.02, 0.06, 0.06);
      hr.rotation.z = -dir * Math.atan2(rise, run); g.add(hr);
      for (let i = 0; i <= 3; i++) {
        const t = i / 3;
        g.add(mk(G.cyl, M.dark, x + dir * t * run, y0 + t * rise + 0.31, z + dz, 0.028, 0.62, 0.028));
      }
    }
    return add(g, nm, 'stairs', 1.1, supportAt(ship, x, y0, z, 'deck'));
  }
  stairs(-4.3, 2.1, deckY(-4.3) + 0.03, QD, -1, 0.9, 'Трап на квартердек (ПБ)');
  stairs(-4.3, -2.1, deckY(-4.3) + 0.03, QD, -1, 0.9, 'Трап на квартердек (ЛБ)');
  stairs(-9.5, 1.35, QD, POOP, -1, 0.8, 'Трап на ют');
  stairs(9.4, 0, deckY(9.4) + 0.03, FC, 1, 1.0, 'Трап на бак');
  const comp = new THREE.Group();
  comp.position.set(3.7, deckY(3.7) + 0.03, 1.5);
  for (let i = 0; i < 6; i++) comp.add(mk(G.box, M.light, 0, -i * 0.3, -i * 0.22, 0.8, 0.06, 0.24));
  comp.add(mk(G.box, M.dark, -0.44, -0.75, -0.55, 0.08, 1.9, 0.08));
  comp.add(mk(G.box, M.dark, 0.44, -0.75, -0.55, 0.08, 1.9, 0.08));
  add(comp, 'Сход в трюм', 'stairs', 1, supportAt(ship, 3.7, deckY(3.7), 1.5, 'deck'));

  function railing(x0, x1, y, hz, nm) {
    const g = new THREE.Group();
    const n = Math.max(4, Math.round((x1 - x0) / 0.62));
    for (const side of [-1, 1]) {
      const z = side * hz;
      g.add(mk(G.box, M.dark, (x0 + x1) / 2, y + 0.62, z, x1 - x0, 0.08, 0.1));
      g.add(mk(G.box, ship.id === 'A' ? M.gold : M.crimson, (x0 + x1) / 2, y + 0.68, z, x1 - x0, 0.04, 0.12));
      for (let i = 0; i <= n; i++) {
        const x = lerp(x0, x1, i / n);
        g.add(mk(G.cyl, M.light, x, y + 0.31, z, 0.032, 0.62, 0.032));
        g.add(mk(G.sph, M.brass, x, y + 0.66, z, 0.04, 0.04, 0.04));
      }
    }
    g.add(mk(G.box, M.dark, x1, y + 0.62, 0, 0.1, 0.08, hz * 2));
    for (let i = 0; i <= 5; i++) g.add(mk(G.cyl, M.light, x1, y + 0.31, lerp(-hz, hz, i / 5), 0.032, 0.62, 0.032));
    return add(g, nm, 'rail', 1, supportAt(ship, (x0 + x1) / 2, y, 0, 'deck'));
  }
  railing(-14.7, -5.8, QD, innerHalf(-10, QD) - 0.05, 'Рейлинг квартердека');
  railing(-14.7, -10.7, POOP, 1.95, 'Рейлинг юта');
  railing(10.7, 15.1, FC, innerHalf(13, FC) - 0.05, 'Рейлинг бака');
}

function makeSail(ship, width, height, name, tier) {
  const nu = 14, nv = 9, rows = nv + 1;
  const pos = [], col = [], uvs = [];
  const base = new THREE.Color(ship.id === 'A' ? 0xefe6d0 : 0xe6d4c2);
  const seam = new THREE.Color(ship.id === 'A' ? 0xcbb894 : 0xc0a68e);
  const patch = new THREE.Color(ship.id === 'A' ? 0xc79f88 : 0xb28a86);
  const emblem = new THREE.Color(ship.id === 'A' ? 0xc9a05a : 0x9c4a44);
  for (let i = 0; i <= nu; i++) {
    for (let j = 0; j <= nv; j++) {
      const u = i / nu, v = j / nv;
      pos.push(0, -v * height, (u - 0.5) * width);
      uvs.push(u, v);
      let c = base.clone();
      if (Math.abs(v - 0.25) < 0.022 || Math.abs(v - 0.5) < 0.022 || Math.abs(v - 0.75) < 0.022) c.copy(seam);
      if (i === 0 || i === nu || j === 0 || j === nv) c.multiplyScalar(0.82);
      if (u > 0.66 && u < 0.8 && v > 0.52 && v < 0.68) c.copy(patch);
      if (u > 0.18 && u < 0.3 && v > 0.2 && v < 0.34) c.copy(patch);
      if (tier === 0 && Math.abs(u - 0.5) < 0.13 && Math.abs(v - 0.45) < 0.16) c.copy(emblem);
      col.push(c.r, c.g, c.b);
    }
  }
  const cells = [];
  for (let i = 0; i < nu; i++) {
    for (let j = 0; j < nv; j++) {
      const a = i * rows + j, b = (i + 1) * rows + j;
      cells.push({
        indices: [a, b, a + 1, b, b + 1, a + 1], alive: true,
        center: V3(0, -((j + 0.5) / nv) * height, (((i + 0.5) / nu) - 0.5) * width)
      });
    }
  }
  const geo = new THREE.BufferGeometry();
  geo.setAttribute('position', new THREE.Float32BufferAttribute(pos, 3));
  geo.setAttribute('color', new THREE.Float32BufferAttribute(col, 3));
  geo.setIndex(cells.flatMap((c) => c.indices));
  geo.computeVertexNormals();
  geo.computeBoundingSphere();
  geo.boundingSphere.radius += 2;
  disposables.push(geo);
  const mo = new THREE.Mesh(geo, ship.sailMat);
  mo.castShadow = true; mo.receiveShadow = true;
  const rec = ship.reg(mo, { name, kind: 'sail', cells, total: cells.length, removed: 0, geometry: geo, tier });
  ship.sails.push({
    record: rec, uvs, base: geo.attributes.position.array.slice(), geometry: geo, mesh: mo,
    width, height, phase: rand(0, 6.28)
  });
  return rec;
}
function makeFlag(ship, x, y, width, height, pattern, name, support) {
  const nu = 12, nv = 7, rows = nv + 1;
  const pos = [], col = [], idx = [], uvs = [];
  const cA = new THREE.Color(ship.id === 'A' ? 0xb3312f : 0x8d1f24);
  const cB = new THREE.Color(ship.id === 'A' ? 0xe8a33d : 0xdfa845);
  const cC = new THREE.Color(ship.id === 'A' ? 0x27578a : 0x2b8f83);
  const taper = pattern === 'pennant' ? 1 : 0;
  for (let i = 0; i <= nu; i++) {
    for (let j = 0; j <= nv; j++) {
      const u = i / nu, v = j / nv;
      pos.push(-u * width, (0.5 - v) * height * (1 - taper * u * 0.92), 0);
      uvs.push(u, v);
      let c = cA;
      if (pattern === 'ensign') {
        if (Math.abs(u - v * 0.9) < 0.14) c = cB;
        if (u < 0.24 && v < 0.36) c = cC;
        if (u > 0.42 && u < 0.58 && v > 0.36 && v < 0.64) c = cB;
      }
      if (pattern === 'pennant') c = u > 0.82 ? cB : (u > 0.5 ? cC : cA);
      if (pattern === 'jack') { c = cC; if (Math.abs(u - v) < 0.12) c = cB; }
      col.push(c.r, c.g, c.b);
    }
  }
  for (let i = 0; i < nu; i++) {
    for (let j = 0; j < nv; j++) {
      const a = i * rows + j, b = (i + 1) * rows + j;
      idx.push(a, b, a + 1, b, b + 1, a + 1);
    }
  }
  const geo = new THREE.BufferGeometry();
  geo.setAttribute('position', new THREE.Float32BufferAttribute(pos, 3));
  geo.setAttribute('color', new THREE.Float32BufferAttribute(col, 3));
  geo.setIndex(idx);
  geo.computeVertexNormals(); geo.computeBoundingSphere();
  geo.boundingSphere.radius += 1;
  disposables.push(geo);
  const mo = new THREE.Mesh(geo, M.flag);
  mo.position.set(x, y, 0); mo.castShadow = true;
  ship.root.add(mo);
  const rec = ship.reg(mo, { name, kind: 'flag', hp: 0.5, maxHp: 0.5 });
  if (support) ship.link(rec, support);
  ship.flags.push({ mesh: mo, uvs, geometry: geo, record: rec, phase: rand(0, 6.28) });
  return rec;
}

function partCenter(rec) {
  if (rec.center) return rec.obj.localToWorld(rec.center.clone());
  return rec.obj.getWorldPosition(V3());
}
function tangentBasis(n) {
  let t1 = V3().crossVectors(n, V3(0, 1, 0));
  if (t1.lengthSq() < 0.0001) t1 = V3().crossVectors(n, V3(1, 0, 0));
  t1.normalize();
  return { t1, t2: V3().crossVectors(n, t1).normalize() };
}
function spawnShard(mat, len, thick, p, v) {

  m.scale.set(thick, len, thick * rand(0.6, 1.15));

    V3(rand(-11, 11), rand(-11, 11), rand(-11, 11)),

}
function spawnChip(mat, sx, sy, sz, p, v, av, o = {}) {

}
function splinterBurst(p, dir, normal, energy, baseCount) {
  if (!S.splinters) return;
  const n = normal.clone();
  if (n.lengthSq() < 1e-6) n.copy(dir).negate();
  n.normalize();
  const inc = dir.clone().normalize();
  const { t1, t2 } = tangentBasis(n);
  const count = Math.round(baseCount * (0.75 + energy * 0.55));
  const spread = 0.4 + energy * 0.28;
  const mats = [M.hullA, M.hullB, M.wood, M.light, M.dark];
  for (let i = 0; i < count; i++) {
    schedule(i * 0.011, () => {
      const pos = p.clone().addScaledVector(n, rand(0.14, 0.5))
        .addScaledVector(t1, rand(-spread, spread)).addScaledVector(t2, rand(-spread, spread));
      const v = n.clone().multiplyScalar(rand(3.6, 9.8) * (0.7 + energy * 0.5))
        .addScaledVector(inc, rand(0.4, 2.6))
        .addScaledVector(t1, rand(-2.3, 2.3)).addScaledVector(t2, rand(-2.3, 2.3))
        .add(V3(0, rand(0.6, 3.4), 0));
      spawnShard(pick(mats), rand(0.22, 0.95) * (0.75 + energy * 0.3), rand(0.022, 0.062), pos, v);
    });
  }
  const chips = Math.max(2, Math.round(baseCount * 0.28));
  for (let i = 0; i < chips; i++) {
    schedule(0.02 + i * 0.018, () => {
      const pos = p.clone().addScaledVector(n, rand(0.16, 0.45))
        .addScaledVector(t1, rand(-spread, spread)).addScaledVector(t2, rand(-spread, spread));
      const v = n.clone().multiplyScalar(rand(3, 7) * (0.7 + energy * 0.4))
        .addScaledVector(t1, rand(-1.8, 1.8)).addScaledVector(t2, rand(-1.8, 1.8))
        .add(V3(0, rand(0.8, 2.8), 0));
      spawnChip(pick(mats), rand(0.28, 0.62), rand(0.05, 0.11), rand(0.14, 0.3), pos, v,
        V3(rand(-8, 8), rand(-8, 8), rand(-8, 8)), { ttl: rand(16, 26), floatF: rand(1.15, 1.4) });
    });
  }
}

}

  }

}
function dentPlank(rec, lp, depth, radius, scorch) {
  const d = rec.patch;
  for (let k = 0; k < d.count; k++) {
    const k3 = k * 3;
    const dx = d.base[k3] - lp.x, dy = d.base[k3 + 1] - lp.y, dz = d.base[k3 + 2] - lp.z;
    const dist = Math.sqrt(dx * dx + dy * dy + dz * dz);
    if (dist < radius) {
      let w = 1 - dist / radius; w *= w;
      d.dent[k] += depth * w;
      d.scorch[k] = Math.max(d.scorch[k], w * scorch);
    }
  }
  rebuildPlank(rec);
}
function usedByAlive(d, k) {
  const i = Math.floor(k / d.rows), j = k % d.rows;
  return [[i - 1, j - 1], [i - 1, j], [i, j - 1], [i, j]]
    .some(([a, b]) => a >= 0 && b >= 0 && a < d.nu && b < d.nv && d.cells[a * d.nv + b].alive);
}
function addHole(ship, rec) {
  if (rec.kind !== 'plank') return;
  const st = rec.center.x < -6 ? 0 : rec.center.x < 0 ? 1 : rec.center.x < 6 ? 2 : 3;
  ship.holes.push({ rec, comp: st * 2 + (rec.center.z >= 0 ? 1 : 0) });
  if (rec.center.y < 0.4) addLog('🌊 ' + ship.name + ': пробоина ниже ватерлинии', 'inf');
}
function tearPlank(rec, lp, energy, wp, dir, first) {
  const d = rec.patch;
  const radius = (first ? 0.3 : 0.24) + S.caliber * 0.011 + energy * 0.08;
  const removed = [];
  for (const c of d.cells) {
    if (!c.alive) continue;
    if (c.center.distanceTo(lp) < radius * (0.72 + 0.55 * c.seed)) { c.alive = false; d.removed++; removed.push(c); }
  }
  if (!removed.length) {
    let best = null, bd = Infinity;
    for (const c of d.cells) { if (!c.alive) continue; const q = c.center.distanceToSquared(lp); if (q < bd) { bd = q; best = c; } }
    if (best) { best.alive = false; d.removed++; removed.push(best); }
  }
  const touched = new Set();
  for (const c of removed) {
    for (const k of [c.i * d.rows + c.j, (c.i + 1) * d.rows + c.j, c.i * d.rows + c.j + 1, (c.i + 1) * d.rows + c.j + 1]) {
      if (touched.has(k)) continue;
      touched.add(k);
      if (usedByAlive(d, k)) {
        if (!d.jdone[k]) {
          d.jdone[k] = 1;
          d.jit[k * 3] += rand(-0.06, 0.06);
          d.jit[k * 3 + 1] += rand(-0.05, 0.05);
          d.jit[k * 3 + 2] += rand(-0.06, 0.06);
        }
        d.fresh[k] = 1;
      }
    }
  }
  rebuildPlank(rec);
  const nrm = rec.normal.clone().transformDirection(rec.obj.matrixWorld);
  splinterBurst(wp, dir, nrm, energy, first ? 16 : 10);
  if (!rec.hole) { rec.hole = true; addHole(rec.ship, rec); }
  puff(wp, 8, { col: CC.dust, size: 0.6, grow: 1.8, life: 1.4, spread: 1, up: 1, a: 0.4 });
  cameraShake(0.08 + energy * 0.03); sfx.breach();
  if (first) addLog('🕳 ' + rec.ship.name + ': пробита доска обшивки', rec.ship.id === 'A' ? 'a' : 'b');
  if (d.removed / d.total > 0.8) {
    rec.dead = true; rec.obj.removeFromParent();
    releaseDeps(rec, dir.clone().multiplyScalar(1.4));
  }
}
function plankHit(rec, wp, energy, explosion) {
  if (rec.dead) return;
  const lp = rec.obj.worldToLocal(wp.clone());
  const dir = wp.clone().sub(camera.position).normalize();
  if (rec.stage >= 3) { tearPlank(rec, lp, energy * 0.8, wp, dir, false); updateIntegrity(rec.ship); return; }
  rec.damage += energy * (explosion ? 2.4 : 1);
  const prev = rec.stage;
  rec.stage = rec.damage < 1 ? 1 : rec.damage < 2.1 ? 2 : 3;
  if (rec.stage >= 3) tearPlank(rec, lp, energy, wp, dir, prev < 3);
  else {
    dentPlank(rec, lp, Math.min((0.05 + 0.05 * rec.stage) * (0.5 + energy * 0.7), 0.32),
      (0.4 + 0.2 * rec.stage) * (0.6 + S.caliber / 45), 0.45 + 0.2 * rec.stage);
    splinterBurst(wp, dir, dir.clone().negate(), energy * 0.5, 4);
    sfx.wood(rec.stage);
  }
  updateIntegrity(rec.ship);
}
function tearSail(rec, wp, energy) {
  const lp = rec.obj.worldToLocal(wp.clone());
  const r = 0.6 + 0.75 * energy;
  let removed = 0;
  for (const c of rec.cells) {
    if (!c.alive) continue;
    if (c.center.distanceTo(lp) < r || (c.center.distanceTo(lp) < r * 1.5 && Math.random() < 0.25)) { c.alive = false; removed++; }
  }
  if (!removed) return;
  rec.removed += removed;
  rec.geometry.setIndex(rec.cells.filter((c) => c.alive).flatMap((c) => c.indices));
  for (let i = 0; i < 3; i++) {
    const m = mk(G.plane, rec.ship.sailMat, wp.x, wp.y, wp.z, rand(0.3, 0.6), rand(0.3, 0.6), 1);
    m.rotation.set(rand(0, 3), rand(0, 3), rand(0, 3)); scene.add(m);
    addDebris(m, 0.2, 0.02, 0.2, 0.1, V3(rand(-1, 1), 0.5, rand(-1, 1)), V3(3, 3, 3),
      { type: 'cloth', floatF: 1.12, linD: 0.55, angD: 0.65 });
  }
  sfx.wood(1);
  if (rec.removed / rec.total > 0.55) destroyPart(rec, wp, wp.clone().sub(camera.position).normalize(), energy);
}
function destroyPart(rec, point, dir, energy) {
  if (rec.dead || rec.detached) return;
  if (rec.kind === 'plank' || rec.kind === 'deck') {
    tearPlank(rec, rec.obj.worldToLocal(point.clone()), energy, point, dir, true);
    updateIntegrity(rec.ship); return;
  }
  const center = partCenter(rec);
  const v = dir.clone().multiplyScalar(2 + energy * 3);
  if (rec.kind === 'flag' || rec.kind === 'crew' || rec.kind === 'crest') {
    dropAttachment(rec, v.clone().multiplyScalar(0.35));
    updateIntegrity(rec.ship); return;
  }
  rec.dead = true;
  releaseDeps(rec, v.clone().multiplyScalar(0.4));
  if (rec.kind === 'powder') { rec.obj.removeFromParent(); explodeAt(rec.ship, center); updateIntegrity(rec.ship); return; }
  if ((rec.kind === 'mast' || rec.kind === 'topmast') && rec.mast) { fellMast(rec.mast, dir); updateIntegrity(rec.ship); return; }
  rec.obj.removeFromParent();
  if (rec.kind === 'cannon') {
    spawnChip(M.iron, 0.3, 0.3, 1.5, center.clone().add(V3(0, 0.6, 0)), v.clone().add(V3(0, 2, 0)),
      V3(rand(-3, 3), rand(-3, 3), rand(-3, 3)), { dense: 40, floatF: 0.12 });
    sfx.metal(); addLog('⚫ ' + rec.ship.name + ': пушка сорвана с лафета', rec.ship.id === 'A' ? 'a' : 'b');
  } else if (rec.kind === 'yard') {
    const g = rec.obj;
    const wp2 = g.getWorldPosition(V3()), wq = g.getWorldQuaternion(new THREE.Quaternion());
    scene.attach(g);
    const r = addDebris(g, (rec.length || 8) / 2, 0.12, 0.12, 20, dir.clone().multiplyScalar(2).add(V3(0, -0.5, 0)),
      V3(rand(-2, 2), rand(-2, 2), rand(-2, 2)), { type: 'spar', floatF: 1.15 });
    r.body.position.copy(wp2); r.body.quaternion.copy(wq);
    sfx.creak();
  } else {
    splinterBurst(center, dir.clone().negate(), V3(0, 1, 0), energy, rec.kind === 'figure' ? 6 : 10);
    sfx.breach();
  }
  updateIntegrity(rec.ship);
}
function applyHit(rec, wp, dir, energy, explosion = false) {
  if (!rec || rec.dead) return;
  if (rec.kind === 'plank' || rec.kind === 'deck') { plankHit(rec, wp, energy, explosion); return; }
  if (rec.kind === 'sail') { tearSail(rec, wp, energy); return; }
  rec.hp -= energy * 1.5;
  sparkBurst(wp, 4, CC.spark, 2, 0.3);
  if (rec.hp <= 0) destroyPart(rec, wp, dir, energy);
  else {
    if (['cannon', 'anchor', 'bell', 'chain'].includes(rec.kind)) sfx.metal(); else sfx.wood(1);
    splinterBurst(wp, dir, dir.clone().negate(), energy * 0.4, 3);
  }
  updateIntegrity(rec.ship);
}

function updateIntegrity(ship) {
  let loss = 0;
  for (const p of ship.parts) {
    if (p.kind === 'plank' || p.kind === 'deck') {
      if (p.dead) loss += 1;
      else if (p.stage >= 3) loss += 0.35 + 0.65 * (p.patch.removed / p.patch.total);
      else if (p.damage) loss += p.damage * 0.08;
    } else if (p.kind === 'mast') { if (p.mast && !p.mast.alive) loss += 4; }
    else if (p.dead) loss += 0.4;
  }
  ship.integrity = clamp(1 - loss / ship.totalWeight, 0, 1);
}
function updateFlooding(ship, dt) {
  if (ship.sink > 28) return;
  for (const h of ship.holes) {
    const r = h.rec;
    if (!r.patch) continue;
    const frac = r.dead ? 1 : r.patch.removed / r.patch.total;
    if (frac <= 0) continue;
    const wp = ship.root.localToWorld(r.center.clone());
    const head = waveHeight(wp.x, wp.z, simTime) - wp.y;
    if (head > 0.02) {
      const c = ship.floodComps[h.comp];
      const Q = 0.55 * (r.area * frac * 0.5) * Math.sqrt(19.64 * head) * S.floodRate;
      c.volume = Math.min(c.capacity, c.volume + Q * dt);
      if (Math.random() < dt * 3) {
        sparks.spawn(wp, V3(-wp.x * 0.02, -0.5, -Math.sign(wp.z - ship.baseZ || 1) * 1.4),
          0.7, 0.12, 0.04, CC.spray, -3, 0.99, 0.7);
      }
    }
  }
  for (let s = 0; s < 4; s++) {
    const A = ship.floodComps[s * 2], B = ship.floodComps[s * 2 + 1];
    const tr = (A.volume - B.volume) * 0.028 * dt;
    if (tr > 0) { const a = Math.min(tr, A.volume); A.volume -= a; B.volume = Math.min(B.capacity, B.volume + a); }
    else { const a = Math.min(-tr, B.volume); B.volume -= a; A.volume = Math.min(A.capacity, A.volume + a); }
  }
  for (let s = 0; s < 3; s++) {
    for (let side = 0; side < 2; side++) {
      const A = ship.floodComps[s * 2 + side], B = ship.floodComps[(s + 1) * 2 + side];
      const tr = (A.volume - B.volume) * 0.1 * dt;
      if (tr > 0) { const a = Math.min(tr, A.volume); A.volume -= a; B.volume = Math.min(B.capacity, B.volume + a); }
      else { const a = Math.min(-tr, B.volume); B.volume -= a; A.volume = Math.min(A.capacity, A.volume + a); }
    }
  }
  let vol = 0, mx = 0, mz = 0;
  for (const c of ship.floodComps) {
    vol += c.volume; mx += c.volume * c.cx; mz += c.volume * c.cz;
    const ratio = c.volume / c.capacity;
    if (ratio < 0.012 || !S.innerWater || ship.sink > 5.5) { c.mesh.visible = false; continue; }
    const bottomW = ship.root.localToWorld(V3(c.cx, -2.3, c.cz));
    const desiredTopW = ship.root.localToWorld(V3(c.cx, -2.3 + ratio * 4.3, c.cz));
    const topY = Math.min(desiredTopW.y, SEA - 0.05);
    if (topY <= bottomW.y + 0.02) { c.mesh.visible = false; continue; }
    const h = topY - bottomW.y;
    const cenW = ship.root.localToWorld(V3(c.cx, 0, c.cz));
    c.mesh.visible = true;
    c.mesh.position.set(cenW.x, bottomW.y + h / 2, cenW.z);
    c.mesh.scale.set(5.9, h, 2.6);
    c.mesh.rotation.set(0, 0, 0);
  }
  ship.floodVol = vol;
  ship.flood = clamp(vol / 112, 0, 1);
  const wpArea = 150 * (1 - 0.72 * ship.flood);
  let targetSink = vol / Math.max(wpArea, 28) * 1.15;
  const targetTrim = -clamp(mx / (vol * 8 + 34), -0.35, 0.35);
  const targetList = clamp(mz / (vol * 4 + 22) * 1.3, -0.6, 0.6);
  if (ship.flood > 0.8 && ship.sink > 1.8 && !ship.dead) {
    ship.dead = true; ship.deathTime = simTime;
    showBanner(ship.name + ' ТОНЕТ!', 'red');
    addLog('⚓ ' + ship.name + ' теряет плавучесть', 'red');
    sfx.sink(); checkEnd();
  }
  if (ship.dead) targetSink = Math.min(30, 6 + (simTime - ship.deathTime) * 0.42);
  ship.sink += (targetSink - ship.sink) * Math.min(1, dt * 1.1);
  ship.trim += (targetTrim - ship.trim) * Math.min(1, dt * 0.9);
  ship.list += (targetList - ship.list) * Math.min(1, dt * 0.85);
}
function updateWaterline(ship) {
  const wl = ship.wl;
  const inv = ship.id === 'A' ? waterU.uInvA : waterU.uInvB;
  const mask = ship.id === 'A' ? waterU.uMaskA : waterU.uMaskB;
  inv.value.copy(ship.root.matrixWorld).invert();
  if (ship.sink > 7.5) { mask.value = 0; return; }
  mask.value = 1;
  const m = ship.root.matrixWorld;
  const sP = hullSurf(-1), sS = hullSurf(1);
  for (let i = 0; i < WL_N; i++) {
    const u = i / (WL_N - 1);
    for (let s = 0; s < 2; s++) {
      const surf = s === 0 ? sP : sS;
      const keelW = surf(u, 0.02).applyMatrix4(m).y;
      const railW = surf(u, 1).applyMatrix4(m).y;
      let hw = 0;
      if (keelW < SEA && railW > SEA) {
        let lo = 0.02, hi = 1;
        for (let k = 0; k < 12; k++) {
          const t = (lo + hi) / 2;
          if (surf(u, t).applyMatrix4(m).y < SEA) lo = t; else hi = t;
        }
        hw = Math.abs(surf(u, (lo + hi) / 2).z);
      }
      wl.data[i * 4 + s] = Math.round(clamp(hw / WL_MAX, 0, 1) * 255);
    }
    wl.data[i * 4 + 2] = 0;
    wl.data[i * 4 + 3] = 255;
  }
  wl.tex.needsUpdate = true;
}

function livingCannons(ship) {

  return ship.cannons.filter((c) => !c.dead && c.side === side);

function chooseTarget(def, willHit) {
  if (willHit) {
    const cands = def.parts.filter((p) => !p.dead &&
      ['plank', 'deck', 'mast', 'sail', 'cannon', 'barrel', 'powder', 'crate', 'flag', 'crew', 'crest', 'trim', 'rail', 'stairs'].includes(p.kind));
    if (!cands.length) return { part: null, point: def.root.position.clone() };
    const rec = pick(cands);
    const c = partCenter(rec);
    c.add(V3(rand(-0.5, 0.5), rand(-0.5, 0.5), rand(-0.5, 0.5)));
    return { part: rec, point: c };
  }
  const base = def.root.position;
  const ms = Math.random() < 0.5 ? -1 : 1;
  return { part: null, point: V3(base.x + rand(-18, 18), 0, base.z + ms * rand(7, 17)) };
}
function fireBroadside(att, def, forced = false) {
  if (!ready || S.over || (S.paused && !forced)) return;
  if (att.dead || def.dead) return;
  const cannons = livingCannons(att);
  if (!cannons.length) return;
  const n = Math.min(S.guns, cannons.length);
  const acc = att.id === 'A' ? S.accA : S.accB;
  att.shots++;
  addLog('🔥 ' + att.name + ': залп ' + n + ' орудий (' + Math.round(acc * 100) + '%)', att.id === 'A' ? 'a' : 'b');
  const pool = cannons.slice(), chosen = [];
  for (let i = 0; i < n; i++) chosen.push(pool.splice((Math.random() * pool.length) | 0, 1)[0]);
  chosen.forEach((cr, i) => {
    schedule(i * 0.07, () => {
      if (cr.dead || att.dead || def.dead) return;
      const hit = Math.random() < acc;
      spawnBall(att, cr, chooseTarget(def, hit), hit, def);
    });
  });
}
function spawnBall(att, cr, target, willHit, def) {
  const muzzle = cr.obj.localToWorld(cr.muzzleLocal.clone());
  const ft = clamp(muzzle.distanceTo(target.point) / 62, 0.9, 2.4);
  const grav = 9.82 * 0.42;
  const vel = target.point.clone().sub(muzzle).divideScalar(ft);
  vel.y += 0.5 * grav * ft;
  const radius = 0.13 + S.caliber * 0.005;
  const core = new THREE.Mesh(G.ball, M.ball);
  core.scale.setScalar(radius);
  core.castShadow = true;
  core.position.copy(muzzle);
  scene.add(core);
  let lp = null;
  if (target.part && !target.part.dead) lp = target.part.obj.worldToLocal(target.point.clone());
  projectiles.push({
    mesh: core, start: muzzle.clone(), velocity: vel, time: 0, ft, grav,
    attacker: att, defender: def, part: target.part, lp, point: target.point.clone(), willHit
  });
  flash(muzzle, 0xffb066, 6, 18, 0.13);
  puff(muzzle, 7, { size: 0.5, grow: 1.7, life: 1.5, spread: 0.9, up: 0.6, a: 0.42 });
  sparkBurst(muzzle, 6, CC.spark, 3.5, 0.25, -2);
  sfx.shot(S.caliber); cameraShake(0.05);
  const home = cr.obj.position.clone();
  cr.obj.position.add(V3(0, 0, -cr.side * 0.14));
  schedule(0.2, () => { if (!cr.dead) cr.obj.position.copy(home); });
}
function updateProjectiles(dt) {
  for (const p of projectiles.slice()) {
    p.time += dt;
    const pos = p.start.clone().addScaledVector(p.velocity, p.time);
    pos.y -= 0.5 * p.grav * p.time * p.time;
    p.mesh.position.copy(pos);
    p.mesh.rotation.x += dt * 2.2;
    p.mesh.rotation.z += dt * 1.4;
    if (p.time >= p.ft) {
      scene.remove(p.mesh);
      projectiles.splice(projectiles.indexOf(p), 1);
      const energy = Math.pow(S.caliber / 24, 1.5);
      if (p.willHit && p.part && !p.part.dead && p.lp) {
        const wp = p.part.obj.localToWorld(p.lp.clone());
        p.attacker.hits++;
        applyHit(p.part, wp, p.velocity.clone().normalize(), energy);
        sparkBurst(wp, 6, CC.spark, 3, 0.35);
        addLog('💥 ' + p.defender.name + ': попадание — ' + p.part.name.toLowerCase(), p.attacker.id === 'A' ? 'a' : 'b');
      } else splashAt(p.point, 0.7 + S.caliber / 40);
    }
  }
}
function updateBattle() {
  if (S.paused || S.over || !ready) return;
  if (simTime >= nextA) { fireBroadside(ships[0], ships[1]); nextA = simTime + S.reload * rand(0.85, 1.2); }
  if (simTime >= nextB) { fireBroadside(ships[1], ships[0]); nextB = simTime + S.reload * rand(0.85, 1.2) + 0.4; }
}

  updateBState();
}

  disposables = []; rayTargets = []; ships = [];
  S.over = false; S.paused = false;
  simTime = 0; nextA = 2.2; nextB = 3.6;
  buildFleet(); updateBState(); updateHUD();

}

function validateGeometry(silent = false) {
  let gaps = 0, degen = 0, orphans = 0, stairs = 0;
  for (const ship of ships) {
    for (const c of ship.seamChecks) if (c.err > 1e-6) gaps++;
    for (const p of ship.planks) if (!p.patch || p.patch.total === 0 || p.area < 1e-4) degen++;
    for (const a of ship.attachments) if (!a.child.dead && a.support && a.support.dead) orphans++;
    stairs += ship.parts.filter((p) => p.kind === 'stairs' && !p.dead).length;
  }
  checkAttachments();
  const ok = gaps === 0 && degen === 0 && orphans === 0;
  const el = $('geoStatus');
  el.className = 'geo ' + (ok ? 'ok' : 'warn');
  el.textContent = ok
    ? '✔ Квартердек, ют и бак сомкнуты: щелей 0 · вырожденных 0 · висящих 0 · трапов ' + stairs
    : '⚠ щелей ' + gaps + ' · вырожденных ' + degen + ' · висящих ' + orphans + ' (сброшены)';
  if (!silent) { addLog(ok ? '🔍 Палубы проверены: щелей нет, трапы на месте' : '🔍 Обнаружены дефекты — устранены', ok ? 'grn' : 'red'); sfx.tick(); }
  return ok;
}

const raycaster = new THREE.Raycaster();
const hlCache = new Map();
let hovered = null;
function hlMat(base) {
  if (!hlCache.has(base.uuid)) {
    const m = base.clone();
    m.emissive = new THREE.Color(0xf2b544);
    m.emissiveIntensity = 0.4;
    hlCache.set(base.uuid, m);
  }
  return hlCache.get(base.uuid);
}
function clearHover() {
  if (hovered) { hovered.mesh.material = hovered.material; hovered = null; }
  $('tooltip').style.display = 'none';
}
let hoverT = 0;
function updateHover(now) {
  if (now - hoverT < 100) return;
  hoverT = now;
  if (!S.labels || !pointer.inside || dragging || !ready) { clearHover(); return; }
  raycaster.setFromCamera({ x: (pointer.x / innerWidth) * 2 - 1, y: -(pointer.y / innerHeight) * 2 + 1 }, camera);
  let rec = null, target = null;
  for (const h of raycaster.intersectObjects(rayTargets, false)) {
    const p = h.object.userData.part;
    if (p && !p.dead) { rec = p; target = h.object; break; }
  }
  if (!rec) { clearHover(); return; }
  if (!hovered || hovered.mesh !== target) {
    clearHover();
    hovered = { mesh: target, material: target.material };
    target.material = hlMat(target.material);
  }
  let state;
  if (rec.kind === 'plank' || rec.kind === 'deck') {
    state = rec.stage >= 3 ? 'пробита · ' + Math.round(rec.patch.removed / rec.patch.total * 100) + '%'
      : (rec.stage ? ['', 'вмятина', 'глубокая вмятина'][rec.stage] : 'цела, шов сомкнут');
  } else if (rec.kind === 'sail') state = 'полотнище ' + Math.round((1 - rec.removed / rec.total) * 100) + '%';
  else if (rec.maxHp) state = 'прочность ' + Math.max(0, Math.round(rec.hp / rec.maxHp * 100)) + '%';
  else state = 'деталь';
  const tt = $('tooltip');
  tt.innerHTML = '<b>' + rec.ship.name + '</b><span>' + rec.name + ' · ' + state + '</span>';
  tt.style.display = 'block';
  tt.style.left = Math.min(pointer.x + 18, innerWidth - 290) + 'px';
  tt.style.top = Math.min(pointer.y + 14, innerHeight - 80) + 'px';
}

let shakeAmt = 0;
function cameraShake(a) { shakeAmt = Math.min(shakeAmt + a, 0.7); }
const clock = new THREE.Clock();
let hudT = 0, attachT = 0, wlT = 0;

function animate() {
  requestAnimationFrame(animate);
  const realDt = Math.min(clock.getDelta(), 0.05);
  const dt = S.paused ? 0 : realDt * S.timeScale;
  simTime += dt;

  for (const task of scheduled.slice()) {
    if (simTime >= task.t) { scheduled.splice(scheduled.indexOf(task), 1); task.fn(); }
  }
  if (camTween) {
    camTween.time += realDt;
    const k = easeOut(clamp(camTween.time / camTween.d, 0, 1));
    camera.position.lerpVectors(camTween.fp, camTween.tp, k);
    controls.target.lerpVectors(camTween.ft, camTween.tt, k);
    if (camTween.time >= camTween.d) { camTween = null; controls.enabled = true; }
  }
  controls.update();

  if (ready) {
    updateBattle();
    for (const ship of ships) updateFlooding(ship, dt);
  }

  const gust = 1 + Math.sin(simTime * 0.7) * 0.18 * S.gust + Math.sin(simTime * 1.9 + 1.3) * 0.1 * S.gust;
  for (const ship of ships) {
    const damp = 1 - ship.flood * 0.6;
    ship.root.position.y = waveHeight(0, ship.baseZ, simTime + ship.bobPhase) * 0.55 * damp - ship.sink;
    ship.root.rotation.z = (waveHeight(5, ship.baseZ, simTime) - waveHeight(-5, ship.baseZ, simTime)) / 10 * 0.45 * damp + ship.trim;
    ship.root.rotation.x = (waveHeight(0, ship.baseZ + 3, simTime) - waveHeight(0, ship.baseZ - 3, simTime)) / 6 * 0.35 * damp + ship.list;
    ship.root.updateMatrixWorld(true);

    for (const sail of ship.sails) {
      if (sail.record.dead) continue;
      const pos = sail.geometry.attributes.position.array;
      const n = pos.length / 3;
      const reef = 1 - S.sailSet * (sail.record.tier === 1 ? 1.15 : 0.7);
      const fill = clamp(S.billow * S.wind * gust, 0, 3);
      for (let i = 0; i < n; i++) {
        const u = sail.uvs[i * 2], v = sail.uvs[i * 2 + 1];
        const belly = Math.sin(Math.PI * u) * Math.sin(Math.PI * Math.min(v * reef, 1));
        const leech = Math.pow(v, 1.6) * 0.28 * fill;
        const flutter = Math.sin(simTime * (3.4 + S.gust * 2) + u * 8 + v * 4 + sail.phase) * 0.055 * S.wind * (0.3 + v);
        const ripple = Math.sin(simTime * 6 + u * 16) * 0.018 * S.gust * v;
        pos[i * 3] = sail.base[i * 3] + belly * (0.34 + 0.62 * fill) + leech + flutter + ripple;
        pos[i * 3 + 1] = sail.base[i * 3 + 1] * clamp(reef, 0.16, 1) + belly * 0.06 * fill;
        pos[i * 3 + 2] = sail.base[i * 3 + 2] * (1 - 0.06 * (1 - reef));
      }
      sail.geometry.attributes.position.needsUpdate = true;
      sail.geometry.computeVertexNormals();
    }
    for (const flag of ship.flags) {
      if (flag.record.dead) continue;
      const pos = flag.geometry.attributes.position.array;
      const n = pos.length / 3;
      for (let i = 0; i < n; i++) {
        const u = flag.uvs[i * 2], v = flag.uvs[i * 2 + 1];
        const w = Math.sin(u * 7.5 - simTime * (3.4 + 3.4 * S.wind) + flag.phase) * 0.16 * u * (0.35 + S.wind);
        pos[i * 3 + 2] = w + Math.sin(v * 4 + simTime * 2.2) * 0.03 * u;
      }
      flag.geometry.attributes.position.needsUpdate = true;
      flag.geometry.computeVertexNormals();
    }
  }

  wlT += realDt;
  if (wlT > 0.032) {
    wlT = 0;
    for (const ship of ships) updateWaterline(ship);
  }
  waterU.uTime.value = simTime;

  for (const c of clouds) { c.position.x += realDt * 0.55; if (c.position.x > 470) c.position.x = -470; }
  for (const r of godRays) r.quaternion.copy(camera.quaternion);
  if (AU.sea) AU.sea.gain.value = 0.035 + 0.045 * S.wave;
  if (AU.wind) AU.wind.gain.value = 0.006 + 0.02 * S.wind;

  if (ready && !S.paused) { updateDebris(dt); updateProjectiles(dt); }
  smoke.update(dt); sparks.update(dt);
  for (const r of rings) {
    if (r.t > 0) {
      r.t -= dt;
      const f = 1 - r.t / r.d;
      r.m.scale.setScalar(1 + f * r.r);
      r.m.position.y = waveHeight(r.m.position.x, r.m.position.z, simTime) + 0.06;
      r.m.material.opacity = 0.55 * (1 - f);
    } else r.m.material.opacity = 0;
  }
  for (const f of flashes) {
    if (f.t > 0) { f.t -= realDt; f.light.intensity = f.i * Math.max(f.t / f.d, 0); }
    else f.light.intensity = 0;
  }
  attachT += realDt;
  if (attachT > 0.5) { attachT = 0; checkAttachments(); }
  updateHover(performance.now());
  hudT += realDt;
  if (hudT > 0.2) { hudT = 0; updateHUD(); }

  let sv = null;
  if (shakeAmt > 0.001) {
    sv = V3(rand(-1, 1), rand(-1, 1), rand(-1, 1)).multiplyScalar(shakeAmt * 0.22);
    camera.position.add(sv);
    shakeAmt *= Math.pow(0.001, realDt);
  }
  composer.render();
  if (sv) camera.position.sub(sv);
  if (shotPending) {
    shotPending = false;
    try {
      const a = document.createElement('a');
      a.download = 'galleon-duel.png';
      a.href = canvas.toDataURL('image/png');
      a.click();
      addLog('📷 Скриншот сохранён', 'inf');
    } catch (e) { addLog('Скриншот недоступен'); }
  }
}
